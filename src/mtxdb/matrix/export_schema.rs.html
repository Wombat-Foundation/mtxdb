1//! Wire schema of the `mtxdb-export/v1` interchange.
2//!
3//! A homeserver writes a room (or its signing keys) as JSONL; `mtxdb import`
4//! reads it. Both sides must spell every record tag and field the same way and
5//! agree on what each verdict code number means, so those live here once. The
6//! module is pure: names, numbers and tag conversions, no JSON library and no
7//! I/O. The `kind` of a `server_keys` record is
8//! [`crate::interop::KeyRecordKind::as_str`].
9//!
10//! Each line is one JSON object tagged by [`field::T`]:
11//!
12//! 1. [`tag::HEADER`]: [`field::FORMAT`], [`field::SERVER`], [`field::ROOM_ID`].
13//! 2. [`tag::EVENT`]: the event's identity, edges, flags, verdict codes and PDU.
14//! 3. [`tag::END`]: [`field::EVENTS`], the number of event records. A file
15//!    without it was truncated.
16//!
17//! A keys export is [`tag::SERVER_KEYS`] records only.
18
19/// Format tag written to the header line. Bump it when the schema below
20/// changes incompatibly; the importer rejects any other value.
21pub const EXPORT_FORMAT: &str = "mtxdb-export/v1";
22
23/// Record tags: the value of [`field::T`].
24pub mod tag {
25    /// First line of a room export.
26    pub const HEADER: &str = "header";
27    /// One event of a room export.
28    pub const EVENT: &str = "event";
29    /// Last line of a room export.
30    pub const END: &str = "end";
31    /// One signing-key record of a keys export.
32    pub const SERVER_KEYS: &str = "server_keys";
33}
34
35/// Field names, by record.
36pub mod field {
37    /// The record tag, on every record.
38    pub const T: &str = "t";
39
40    /// `header`: the format tag, [`super::EXPORT_FORMAT`].
41    pub const FORMAT: &str = "format";
42    /// `header`: the exporting server's name.
43    pub const SERVER: &str = "server";
44    /// `header`: the room every event record belongs to.
45    pub const ROOM_ID: &str = "room_id";
46
47    /// `event`: the event's id.
48    pub const EVENT_ID: &str = "event_id";
49    /// `event`: ids of the event's `prev_events`.
50    pub const PREV_EVENTS: &str = "prev_events";
51    /// `event`: ids of the event's `auth_events`.
52    pub const AUTH_EVENTS: &str = "auth_events";
53    /// `event`: whether the event is an outlier.
54    pub const OUTLIER: &str = "outlier";
55    /// `event`: whether the event soft-failed.
56    pub const SOFT_FAILED: &str = "soft_failed";
57    /// `event`, optional: why it soft-failed, a [`super::SoftFailCode`] number.
58    pub const SOFT_FAIL_CODE: &str = "soft_fail_code";
59    /// `event`: whether the event was rejected.
60    pub const REJECTED: &str = "rejected";
61    /// `event`, optional: why it was rejected, a [`super::RejectionCode`] number.
62    pub const REJECTION_CODE: &str = "rejection_code";
63    /// `event`, optional: id of the event that redacted this one.
64    pub const REDACTED_BY: &str = "redacted_by";
65    /// `event`: the PDU as the source stored it.
66    pub const PDU: &str = "pdu";
67
68    /// `end`: how many `event` records the file holds.
69    pub const EVENTS: &str = "events";
70
71    /// `server_keys`: the server the keys belong to.
72    pub const ORIGIN: &str = "origin";
73    /// `server_keys`: the record kind, a `KeyRecordKind` tag.
74    pub const KIND: &str = "kind";
75    /// `server_keys`: base64 of the record's payload bytes.
76    pub const PAYLOAD: &str = "payload";
77}

79/// Why an event was rejected: the server refused it outright.
80///
81/// The numbers are part of the export format and of every stored verdict.
82/// They are written out explicitly so renumbering a variant is a visible,
83/// breaking edit; never reuse or reorder one. A number this build does not
84/// know reads back as [`Self::Unknown`].
85#[derive(Clone, Copy, Debug, Eq, PartialEq)]
86#[repr(u8)]
87pub enum RejectionCode {
88    /// The event's signatures did not verify.
89    SignatureVerificationFailed = 0,
90    /// The PDU is malformed.
91    InvalidPduFormat = 1,
92    /// An auth event it depends on was itself rejected.
93    DependsOnRejectedAuthEvent = 2,
94    /// Every prev event it builds on was rejected.
95    PrevEventsRejected = 3,
96    /// An auth event is missing and could not be fetched.
97    MissingAuthEvent = 4,
98    /// Two auth events claim the same (type, state key).
99    DuplicateAuthEventKey = 5,
100    /// The room's create event is missing.
101    MissingCreateEvent = 6,
102    /// The auth check against its resolved auth events failed.
103    AuthCheckFailed = 7,
104    /// A `/get_missing_events` response carried a structurally invalid prev
105    /// event.
106    StructurallyInvalidInGetMissingEvents = 8,
107    /// A prev event stayed unknown after `/state_ids` recovery failed.
108    PrevEventUnknownStateIdsFailed = 9,
109    /// Every prev event was present but state recovery still failed.
110    StateResolutionFailedWithPrevsPresent = 10,
111    /// A reason this build cannot name.
112    Unknown = 11,
113}
114
115impl RejectionCode {
116    /// Every code, in number order.
117    pub const ALL: [Self; 12] = [
118        Self::SignatureVerificationFailed,
119        Self::InvalidPduFormat,
120        Self::DependsOnRejectedAuthEvent,
121        Self::PrevEventsRejected,
122        Self::MissingAuthEvent,
123        Self::DuplicateAuthEventKey,
124        Self::MissingCreateEvent,
125        Self::AuthCheckFailed,
126        Self::StructurallyInvalidInGetMissingEvents,
127        Self::PrevEventUnknownStateIdsFailed,
128        Self::StateResolutionFailedWithPrevsPresent,
129        Self::Unknown,
130    ];
131
132    /// The code's number on the wire and on disk.
133    #[must_use]
134    pub const fn to_u8(self) -> u8 {
135        self as u8
136    }
137
138    /// The code for a number; any number this build does not define is
139    /// [`Self::Unknown`].
140    #[must_use]
141    pub const fn from_u8(value: u8) -> Self {
142        match value {
143            0 => Self::SignatureVerificationFailed,
144            1 => Self::InvalidPduFormat,
145            2 => Self::DependsOnRejectedAuthEvent,
146            3 => Self::PrevEventsRejected,
147            4 => Self::MissingAuthEvent,
148            5 => Self::DuplicateAuthEventKey,
149            6 => Self::MissingCreateEvent,
150            7 => Self::AuthCheckFailed,
151            8 => Self::StructurallyInvalidInGetMissingEvents,
152            9 => Self::PrevEventUnknownStateIdsFailed,
153            10 => Self::StateResolutionFailedWithPrevsPresent,
154            _ => Self::Unknown,
155        }
156    }
157
158    /// The stable `snake_case` name stored in reason strings.
159    #[must_use]
160    pub const fn tag(self) -> &'static str {
161        match self {
162            Self::SignatureVerificationFailed => "signature_verification_failed",
163            Self::InvalidPduFormat => "invalid_pdu_format",
164            Self::DependsOnRejectedAuthEvent => "depends_on_rejected_auth_event",
165            Self::PrevEventsRejected => "prev_events_rejected",
166            Self::MissingAuthEvent => "missing_auth_event",
167            Self::DuplicateAuthEventKey => "duplicate_auth_event_key",
168            Self::MissingCreateEvent => "missing_create_event",
169            Self::AuthCheckFailed => "auth_check_failed",
170            Self::StructurallyInvalidInGetMissingEvents => {
171                "structurally_invalid_in_get_missing_events"
172            }
173            Self::PrevEventUnknownStateIdsFailed => "prev_event_unknown_state_ids_failed",
174            Self::StateResolutionFailedWithPrevsPresent => {
175                "state_resolution_failed_with_prevs_present"
176            }
177            Self::Unknown => "unknown",
178        }
179    }
180
181    /// Read the code from a reason string: `tag` or `tag: detail`. `None` for
182    /// a tag this build does not know (including `unknown` itself).
183    #[must_use]
184    pub fn parse(reason: &str) -> Option<Self> {
185        let tag = reason.split(':').next().unwrap_or(reason).trim();
186        Self::ALL
187            .into_iter()
188            .filter(|code| *code != Self::Unknown)
189            .find(|code| code.tag() == tag)
190    }
191}

193/// Why an event soft-failed: accepted into the DAG but not into current state.
194///
195/// Numbered explicitly for the same reason as [`RejectionCode`]; a number this
196/// build does not know reads back as [`Self::Unknown`].
197#[derive(Clone, Copy, Debug, Eq, PartialEq)]
198#[repr(u8)]
199pub enum SoftFailCode {
200    /// It failed auth against the room's current state.
201    AuthCheckFailed = 0,
202    /// It arrived soft-failed from an import.
203    Imported = 1,
204    /// An operator marked it.
205    Manual = 2,
206    /// A reason this build cannot name.
207    Unknown = 3,
208}
209
210impl SoftFailCode {
211    /// Every code, in number order.
212    pub const ALL: [Self; 4] = [
213        Self::AuthCheckFailed,
214        Self::Imported,
215        Self::Manual,
216        Self::Unknown,
217    ];
218
219    /// The code's number on the wire and on disk.
220    #[must_use]
221    pub const fn to_u8(self) -> u8 {
222        self as u8
223    }
224
225    /// The code for a number; any number this build does not define is
226    /// [`Self::Unknown`].
227    #[must_use]
228    pub const fn from_u8(value: u8) -> Self {
229        match value {
230            0 => Self::AuthCheckFailed,
231            1 => Self::Imported,
232            2 => Self::Manual,
233            _ => Self::Unknown,
234        }
235    }
236
237    /// The stable `snake_case` name stored in reason strings.
238    #[must_use]
239    pub const fn tag(self) -> &'static str {
240        match self {
241            Self::AuthCheckFailed => "auth_check_failed",
242            Self::Imported => "imported",
243            Self::Manual => "manual",
244            Self::Unknown => "unknown",
245        }
246    }
247
248    /// Read the code from a reason string: `tag` or `tag: detail`. `None` for
249    /// a tag this build does not know (including `unknown` itself).
250    #[must_use]
251    pub fn parse(reason: &str) -> Option<Self> {
252        let tag = reason.split(':').next().unwrap_or(reason).trim();
253        Self::ALL
254            .into_iter()
255            .filter(|code| *code != Self::Unknown)
256            .find(|code| code.tag() == tag)
257    }
258}
