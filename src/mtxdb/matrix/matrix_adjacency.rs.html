1//! Matrix event adjacency over the generic short-id primitives.
2//!
3//! One room maps to two [`ShortIdIndex`](crate::short_id::ShortIdIndex) scopes:
4//!
5//! - **events**: event id -> room-local `u32` short id, with three immutable
6//!   adjacency families keyed by the event: `prev` (plain), `auth` (plain) and
7//!   `relations` (typed, at most one edge per event, as in Synapse's unique
8//!   `event_relations(event_id)` row);
9//! - **relation kinds**: relation type string (`m.annotation`, ...) -> `u16`
10//!   kind id.
11//!
12//! Kind ids are persisted in the dictionary and never regenerated: once a type
13//! has an id it keeps it for the life of the room's data, so stored relation
14//! edges stay meaningful. The four relation types defined by the spec get fixed
15//! ids ([`KNOWN_RELATION_TYPES`](crate::matrix_adjacency::KNOWN_RELATION_TYPES)); any other type string is preserved (never
16//! dropped) and gets the next free id on first sight. Ids above `u16::MAX` are a
17//! hard error.
18//!
19//! What is *not* stored here: whether an event is still visible. A relation
20//! disappears when its source event is redacted, rejected or soft-failed, but
21//! those are facts about the source event held elsewhere (and rejection is
22//! reversible). Reads therefore take an [`EventVisibility`](crate::matrix_adjacency::EventVisibility) filter and apply it
23//! to the relation at query time; `prev` and `auth` are signed event-core fields
24//! that redaction never changes, so they are not filtered. The transitive
25//! auth-closure layer must read only [`MatrixAdjacency::auth_of`](crate::matrix_adjacency::MatrixAdjacency::auth_of), never
26//! relations.
27//!
28//! Reverse lookups (who relates to this event) are not provided: fan-in is
29//! unbounded and belongs in an ordered index, not a per-event edge record.
30//!
31//! JSON handling is the caller's: this module takes already-extracted event
32//! ids and relation type strings. Built on the transaction layer in
33//! [`crate::database`].
34

42/// Family of an event's `prev_events`.
43pub const PREV: EdgeFamily = EdgeFamily::plain(1);
44/// Family of an event's `auth_events`.
45pub const AUTH: EdgeFamily = EdgeFamily::plain(2);
46/// Family of an event's single relation (`m.relates_to`), typed by relation kind.
47pub const RELATIONS: EdgeFamily = EdgeFamily::typed(3);

49/// Relation types defined by the Matrix spec, with fixed kind ids (1-based, in
50/// this order). They are seeded into every room's dictionary before any other
51/// type, so their ids are identical across rooms and stores.
52pub const KNOWN_RELATION_TYPES: [&str; 4] =
53    ["m.annotation", "m.reference", "m.replace", "m.thread"];

55/// Decides whether an event's relation is currently visible.
56///
57/// Implemented by the caller against its authoritative redaction and
58/// disposition state (redacted, rejected, soft-failed). Must be cheap: it is
59/// consulted once per relation read.
60pub trait EventVisibility {
61    /// Whether `event_id`'s relation should be shown.
62    fn is_visible(&self, event_id: &str) -> bool;
63}

65/// A filter that shows everything.
66#[derive(Debug, Clone, Copy, Default)]
67pub struct AlwaysVisible;
68
69impl EventVisibility for AlwaysVisible {
70    fn is_visible(&self, _event_id: &str) -> bool {
71        true
72    }
73}

75impl<F: Fn(&str) -> bool> EventVisibility for F {
76    fn is_visible(&self, event_id: &str) -> bool {
77        self(event_id)
78    }
79}

135fn collection(prefix: &str, room_id: &str) -> [u8; 16] {
136    let mut canonical = prefix.as_bytes().to_vec();
137    canonical.extend_from_slice(room_id.as_bytes());
138    derive_collection_id(Some(MEMBER_NAMESPACE_INTL), &canonical)
139}

141fn utf8(bytes: Vec<u8>, what: &str) -> Result<String, StorageError> {
142    String::from_utf8(bytes).map_err(|_| StorageError::Corrupt(format!("{what} is not UTF-8")))
143}

163    /// Record an event's `prev`, `auth` and (optional) relation in one
164    /// transaction, allocating short ids for the event and every referenced
165    /// event. Returns the event's short id.
166    ///
167    /// Re-recording identical data is a no-op; different `prev` or `auth` for an
168    /// already-recorded event is a collision error (an event's core fields never
169    /// change).
170    ///
171    /// A relation can be added but never changed or removed. An event recorded
172    /// without one may be recorded again with one, which stores it. An event
173    /// recorded with one may be recorded again without (a redacted copy has no
174    /// `m.relates_to`), which leaves the stored relation as it was; recording a
175    /// *different* relation is a collision. Passing `None` therefore means "no
176    /// relation known", not "no relation", so redaction never has to be stored.
177    ///
178    /// The relation kind is allocated first in its own transaction. That step
179    /// is idempotent and permanent, so a failure after it leaves only an unused
180    /// dictionary entry.
181    ///
182    /// # Errors
183    /// Storage errors; a collision as described; or an error if the relation
184    /// kind dictionary is out of `u16` ids.
185    pub fn record_event(
186        &self,
187        db: &Database,
188        event_id: &str,
189        prev: &[&str],
190        auth: &[&str],
191        relation: Option<RelationRef<'_>>,
192    ) -> Result<u32, StorageError> {
193        let kind = match relation {
194            Some(relation) => Some(self.kind_id(db, relation.rel_type)?),
195            None => None,
196        };
197        let prev_edges: Vec<EdgeKey<'_>> = prev
198            .iter()
199            .map(|id| EdgeKey::plain(id.as_bytes()))
200            .collect();
201        let auth_edges: Vec<EdgeKey<'_>> = auth
202            .iter()
203            .map(|id| EdgeKey::plain(id.as_bytes()))
204            .collect();
205        let relation_edges: Vec<EdgeKey<'_>> = match (relation, kind) {
206            (Some(relation), Some(kind)) => vec![EdgeKey {
207                target: relation.target.as_bytes(),
208                kind,
209            }],
210            _ => Vec::new(),
211        };
212        let mut families = vec![
213            FamilyEdges {
214                family: PREV,
215                edges: &prev_edges,
216            },
217            FamilyEdges {
218                family: AUTH,
219                edges: &auth_edges,
220            },
221        ];
222        // No relation known means no relations record at all, so a relation can
223        // be added later and a stored one is never contradicted by its absence.
224        if !relation_edges.is_empty() {
225            families.push(FamilyEdges {
226                family: RELATIONS,
227                edges: &relation_edges,
228            });
229        }
230        let recorded = self
231            .events
232            .record_event(db, event_id.as_bytes(), &families)?;
233        Ok(recorded.id)
234    }

236    /// Record several events in **one** transaction, returning each event's
237    /// short id in order.
238    ///
239    /// Either every event is recorded or none is. Events may reference each
240    /// other, and the same event twice with the same data is a no-op. Relation
241    /// kinds are allocated first, each in its own idempotent and permanent
242    /// transaction, so a failure leaves at most unused dictionary entries.
243    ///
244    /// # Errors
245    /// As [`Self::record_event`], applied to the whole batch.
246    pub fn record_events(
247        &self,
248        db: &Database,
249        events: &[EventRecord<'_>],
250    ) -> Result<Vec<u32>, StorageError> {
251        struct Prepared<'a> {
252            owner: &'a [u8],
253            prev: Vec<EdgeKey<'a>>,
254            auth: Vec<EdgeKey<'a>>,
255            relation: Vec<EdgeKey<'a>>,
256        }
257        let mut kinds: HashMap<&str, u16> = HashMap::new();
258        let mut prepared: Vec<Prepared<'_>> = Vec::with_capacity(events.len());
259        for event in events {
260            let relation = match event.relation {
261                Some(relation) => {
262                    let kind = if let Some(kind) = kinds.get(relation.rel_type) {
263                        *kind
264                    } else {
265                        let kind = self.kind_id(db, relation.rel_type)?;
266                        kinds.insert(relation.rel_type, kind);
267                        kind
268                    };
269                    vec![EdgeKey {
270                        target: relation.target.as_bytes(),
271                        kind,
272                    }]
273                }
274                None => Vec::new(),
275            };
276            prepared.push(Prepared {
277                owner: event.event_id.as_bytes(),
278                prev: event
279                    .prev
280                    .iter()
281                    .map(|id| EdgeKey::plain(id.as_bytes()))
282                    .collect(),
283                auth: event
284                    .auth
285                    .iter()
286                    .map(|id| EdgeKey::plain(id.as_bytes()))
287                    .collect(),
288                relation,
289            });
290        }
291        let families: Vec<Vec<FamilyEdges<'_>>> = prepared
292            .iter()
293            .map(|event| {
294                let mut list = vec![
295                    FamilyEdges {
296                        family: PREV,
297                        edges: &event.prev,
298                    },
299                    FamilyEdges {
300                        family: AUTH,
301                        edges: &event.auth,
302                    },
303                ];
304                // As in `record_event`: no relation known, no relations record.
305                if !event.relation.is_empty() {
306                    list.push(FamilyEdges {
307                        family: RELATIONS,
308                        edges: &event.relation,
309                    });
310                }
311                list
312            })
313            .collect();
314        let batch: Vec<BatchEvent<'_>> = prepared
315            .iter()
316            .zip(&families)
317            .map(|(event, families)| BatchEvent {
318                owner_payload: None,
319                owner: event.owner,
320                families,
321            })
322            .collect();
323        Ok(self
324            .events
325            .record_events(db, &batch)?
326            .into_iter()
327            .map(|recorded| recorded.id)
328            .collect())
329    }

366    /// The event's relation if it has one and `visibility` shows it.
367    ///
368    /// # Errors
369    /// Returns an error on a read failure, a corrupt record, or a relation kind
370    /// missing from the dictionary.
371    pub fn relation_of(
372        &self,
373        db: &Database,
374        event_id: &str,
375        visibility: &dyn EventVisibility,
376    ) -> Result<Option<Relation>, StorageError> {
377        if !visibility.is_visible(event_id) {
378            return Ok(None);
379        }
380        let Some(id) = self.short_id(db, event_id)? else {
381            return Ok(None);
382        };
383        let Some(stored) = self.events.edges(db, id, RELATIONS)? else {
384            return Ok(None);
385        };
386        let Some(edge) = stored.first() else {
387            return Ok(None);
388        };
389        let target = self
390            .events
391            .resolve(db, &[edge.target])?
392            .into_iter()
393            .next()
394            .flatten()
395            .ok_or_else(|| StorageError::Corrupt("relation target does not resolve".to_owned()))?;
396        let rel_type = self
397            .relation_type(db, edge.kind)?
398            .ok_or_else(|| StorageError::Corrupt("relation kind not in dictionary".to_owned()))?;
399        Ok(Some(Relation {
400            target: utf8(target, "relation target")?,
401            rel_type,
402        }))
403    }

405    /// The persisted kind id of `rel_type`, allocating one on first sight.
406    ///
407    /// The spec-defined types are always part of the same allocation batch,
408    /// ahead of `rel_type`, so on a fresh dictionary they take ids 1..=4 in one
409    /// transaction and no other type can be allocated before them, however calls
410    /// race.
411    ///
412    /// # Errors
413    /// An error (publishing nothing) once the dictionary would exceed `u16::MAX`
414    /// ids; or a storage error.
415    pub fn kind_id(&self, db: &Database, rel_type: &str) -> Result<u16, StorageError> {
416        let mut keys: Vec<&[u8]> = KNOWN_RELATION_TYPES.iter().map(|t| t.as_bytes()).collect();
417        keys.push(rel_type.as_bytes());
418        let ids = self.kinds.get_or_create(db, &keys)?;
419        // The index refuses ids above `u16::MAX` inside the allocation
420        // transaction, so this conversion cannot fail and nothing is published
421        // past the limit.
422        ids.last()
423            .and_then(|id| u16::try_from(*id).ok())
424            .ok_or_else(|| StorageError::Corrupt("relation kind id exceeds u16".to_owned()))
425    }

427    /// The relation type string of a kind id.
428    ///
429    /// # Errors
430    /// Returns an error on a read failure or a corrupt record.
431    pub fn relation_type(&self, db: &Database, kind: u16) -> Result<Option<String>, StorageError> {
432        self.kinds
433            .resolve(db, &[u32::from(kind)])?
434            .into_iter()
435            .next()
436            .flatten()
437            .map(|bytes| utf8(bytes, "relation type"))
438            .transpose()
439    }

457    /// Check the event and kind scopes and that every stored relation kind
458    /// resolves.
459    ///
460    /// # Errors
461    /// Returns an error only when a record cannot be read.
462    pub fn verify(&self, db: &Database) -> Result<AdjacencyVerifyReport, StorageError> {
463        let mut report = AdjacencyVerifyReport::default();
464        report.problems.extend(
465            self.events
466                .verify(db, &[PREV, AUTH, RELATIONS])?
467                .problems
468                .into_iter()
469                .map(|problem| format!("events: {problem}")),
470        );
471        report.problems.extend(
472            self.kinds
473                .verify(db, &[])?
474                .problems
475                .into_iter()
476                .map(|problem| format!("relation kinds: {problem}")),
477        );
478        Ok(report)
479    }

481    /// Remove the room's adjacency and relation-kind dictionary in one commit:
482    /// the kind ids live and die with the edges that use them.
483    ///
484    /// # Errors
485    /// Returns an error if the delete cannot be staged or committed.
486    pub fn purge(&self, db: &Database) -> Result<(), StorageError> {
487        let txn = db.begin_transaction();
488        self.stage_purge(&txn)?;
489        txn.commit()
490    }

492    /// Stage the room's removal into `txn`.
493    ///
494    /// # Errors
495    /// Returns an error if a delete cannot be staged.
496    pub fn stage_purge(&self, txn: &DatabaseTransaction<'_>) -> Result<(), StorageError> {
497        self.events.stage_purge(txn)?;
498        self.kinds.stage_purge(txn)
499    }

501    fn targets_of(
502        &self,
503        db: &Database,
504        event_id: &str,
505        family: EdgeFamily,
506    ) -> Result<Option<Vec<String>>, StorageError> {
507        let Some(id) = self.short_id(db, event_id)? else {
508            return Ok(None);
509        };
510        let Some(stored) = self.events.edges(db, id, family)? else {
511            return Ok(None);
512        };
513        let ids: Vec<u32> = stored.iter().map(|edge| edge.target).collect();
514        self.events
515            .resolve(db, &ids)?
516            .into_iter()
517            .map(|key| {
518                key.ok_or_else(|| StorageError::Corrupt("edge target does not resolve".to_owned()))
519                    .and_then(|bytes| utf8(bytes, "event id"))
520            })
521            .collect::<Result<Vec<_>, _>>()
522            .map(Some)
523    }
