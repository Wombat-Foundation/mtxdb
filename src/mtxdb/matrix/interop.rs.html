1//! Shared on-disk layouts for the server/importer boundary.
2//!
3//! mtxdb is the only store for events, verdicts, signing keys and state. A
4//! server writes those records through its adapters; an `mtxdb import`
5//! batch writes the same records from an export file. Both must agree on
6//! every byte, or a room that imported cleanly reads as empty after a
7//! restart. This module owns that agreement: the encodings, the collection
8//! and node derivations, and the data-format version the import marker
9//! vouches for.
10//!
11//! Identity note: room collection ids use mtxdb's BLAKE3 group derivation
12//! plus a namespace bias. Other interop ids use BLAKE3-128; where multiple
13//! logical ids share a collection, their derivation uses a NUL-separated
14//! domain tag. Event record ids deliberately match the Matrix template's
15//! BLAKE3-128 of the complete event id.
16//!
17//! State HAMT construction and state derivation are deliberately not
18//! implemented here. The server and offline importer must use the same
19//! `rezzy` HAMT builder and encoding; this module owns only stable mtxdb
20//! collection/node identities and bounded publication helpers.
21

28/// Serialized `rezzy::hamt::RootHandle` length used by the server's state
29/// adapter. Field order is codec version, routing version, routing parameters,
30/// structural hash, then state-group id.
31pub const STATE_ROOT_HANDLE_V1_LEN: usize = 70;
32
33/// The byte-level root handle shared by the importer and the Matrix server.
34#[derive(Clone, Copy, Debug, Eq, PartialEq)]
35pub struct StateRootHandleV1 {
36    /// HAMT node codec version.
37    pub codec_version: u8,
38    /// HAMT routing version.
39    pub routing_version: u8,
40    /// Parameters for the selected routing version.
41    pub routing_params: [u8; 4],
42    /// Full content-addressed structural hash of the root node.
43    pub structural_hash: [u8; 32],
44    /// Cross-server state-group identity derived from the state's `LtHash`.
45    pub state_group_id: [u8; 32],
46}
47
48/// Encode the server's fixed-width state root handle as 70 bytes, ordered as
49/// codec version, routing version, routing parameters, structural hash, and
50/// state-group ID. Field values are copied without validation.
51#[must_use]
52pub fn encode_state_root_handle_v1(handle: StateRootHandleV1) -> [u8; STATE_ROOT_HANDLE_V1_LEN] {
53    let mut out = [0; STATE_ROOT_HANDLE_V1_LEN];
54    out[0] = handle.codec_version;
55    out[1] = handle.routing_version;
56    out[2..6].copy_from_slice(&handle.routing_params);
57    out[6..38].copy_from_slice(&handle.structural_hash);
58    out[38..70].copy_from_slice(&handle.state_group_id);
59    out
60}

62/// Decode the server's fixed-width state root handle.
63///
64/// Version and routing-parameter values are preserved without validation.
65///
66/// # Errors
67/// Returns [`StorageError::Corrupt`] unless `bytes` has the exact v1 length.
68pub fn decode_state_root_handle_v1(bytes: &[u8]) -> Result<StateRootHandleV1, StorageError> {
69    if bytes.len() != STATE_ROOT_HANDLE_V1_LEN {
70        return Err(StorageError::Corrupt(format!(
71            "malformed state root handle: expected {STATE_ROOT_HANDLE_V1_LEN} bytes, got {}",
72            bytes.len()
73        )));
74    }
75    let malformed = || StorageError::Corrupt("malformed state root handle".to_owned());
76    let (codec_version, fields) = bytes.split_first().ok_or_else(malformed)?;
77    let (routing_version, fields) = fields.split_first().ok_or_else(malformed)?;
78    let (routing_params, fields) = fields
79        .split_first_chunk::<4>()
80        .ok_or_else(|| StorageError::Corrupt("malformed state root handle".to_owned()))?;
81    let (structural_hash, fields) = fields
82        .split_first_chunk::<32>()
83        .ok_or_else(|| StorageError::Corrupt("malformed state root handle".to_owned()))?;
84    let (state_group_id, _) = fields.split_first_chunk::<32>().ok_or_else(malformed)?;
85    Ok(StateRootHandleV1 {
86        codec_version: *codec_version,
87        routing_version: *routing_version,
88        routing_params: *routing_params,
89        structural_hash: *structural_hash,
90        state_group_id: *state_group_id,
91    })
92}

141/// Derive a 128-bit BLAKE3 identifier from a domain and logical identifier.
142///
143/// Hashes `domain`, a NUL separator, then `identifier`, taking the first
144/// 16 digest bytes. Domains must exclude NUL bytes for unambiguous separation.
145/// Callers that need collision detection must retain the original identifier
146/// beside the derived id — 128 bits is not enough to rule out a collision by
147/// itself.
148#[must_use]
149pub fn derive_id(domain: &[u8], identifier: &[u8]) -> [u8; 16] {
150    let mut hasher = blake3::Hasher::new();
151    hasher.update(domain);
152    hasher.update(&[0]);
153    hasher.update(identifier);
154
155    let mut id = [0_u8; 16];
156    id.copy_from_slice(&hasher.finalize().as_bytes()[..16]);
157    id
158}

160/// Number of bucketed verdict collections. Fixed for the life of a database:
161/// the bucket is part of where a verdict lives, so changing it orphans every
162/// existing entry.
163pub const VERDICT_BUCKETS: u32 = 256;
164
165/// Version byte of the verdict value encoding.
166pub const VERDICT_VERSION: u8 = 1;
167const HAS_REJECTION: u8 = 1;
168const HAS_SOFT_FAIL: u8 = 1 << 1;
169const HAS_REDACTION: u8 = 1 << 2;
170const IS_OUTLIER: u8 = 1 << 3;
171
172/// Everything recorded about an event beyond its body and edges.
173#[derive(Clone, Debug, Default, Eq, PartialEq)]
174pub struct Verdict {
175    /// Why the event was rejected, as the source numbered it.
176    pub rejection: Option<u8>,
177    /// Why the event soft-failed, as the source numbered it.
178    pub soft_fail: Option<u8>,
179    /// Event id of the redaction that redacted this event.
180    pub redacted_by: Option<Vec<u8>>,
181    /// Whether the event is stored as an outlier. This is independent of its
182    /// rejection and soft-fail verdicts.
183    pub outlier: bool,
184}
185
186impl Verdict {
187    /// Whether there is nothing worth recording.
188    #[must_use]
189    pub fn is_empty(&self) -> bool {
190        self.rejection.is_none()
191            && self.soft_fail.is_none()
192            && self.redacted_by.is_none()
193            && !self.outlier
194    }
195}

197/// Derive an event record id as the first 16 bytes of the BLAKE3 hash of the

207/// Derive the per-event verdict collection holding `node`.
208///
209/// Bucketed so one write's cost stays proportional to its bucket rather
210/// than to the server's total event count: mtxdb clones a collection's
211/// index on every write, so a single shared collection would make every
212/// verdict write pay for every other.
213#[must_use]
214pub fn event_verdict_collection_id(node: &[u8; 16]) -> [u8; 16] {
215    let bucket = u32::from_le_bytes([node[0], node[1], node[2], node[3]]) % VERDICT_BUCKETS;
216    derive_id(b"sys", format!("event-verdict:{bucket}").as_bytes())
217}

219/// Encode one event's verdict for storage.
220///
221/// # Errors
222/// Returns [`StorageError::Corrupt`] if a field does not fit its length prefix.
223pub fn encode_verdict(event_id: &[u8], verdict: &Verdict) -> Result<Vec<u8>, StorageError> {
224    let too_large = || StorageError::Corrupt("verdict field is too large".to_owned());
225    let mut flags = 0_u8;
226    if verdict.outlier {
227        flags |= IS_OUTLIER;
228    }
229    if verdict.rejection.is_some() {
230        flags |= HAS_REJECTION;
231    }
232    if verdict.soft_fail.is_some() {
233        flags |= HAS_SOFT_FAIL;
234    }
235    if verdict.redacted_by.is_some() {
236        flags |= HAS_REDACTION;
237    }
238
239    let mut out = vec![VERDICT_VERSION, flags];
240    out.extend_from_slice(
241        &u32::try_from(event_id.len())
242            .map_err(|_| too_large())?
243            .to_le_bytes(),
244    );
245    out.extend_from_slice(event_id);
246    out.extend(verdict.rejection);
247    out.extend(verdict.soft_fail);
248    if let Some(redactor) = &verdict.redacted_by {
249        out.extend_from_slice(
250            &u32::try_from(redactor.len())
251                .map_err(|_| too_large())?
252                .to_le_bytes(),
253        );
254        out.extend_from_slice(redactor);
255    }
256    Ok(out)
257}

259/// Decode a stored verdict, returning the event id it was written for.
260///
261/// # Errors
262/// Returns [`StorageError::Corrupt`] for a truncated record or an unknown
263/// version.
264pub fn decode_verdict(bytes: &[u8]) -> Result<(&[u8], Verdict), StorageError> {
265    let malformed = || StorageError::Corrupt("malformed event verdict".to_owned());
266    let take_len = |rest: &[u8]| -> Result<(usize, usize), StorageError> {
267        let (length, _) = rest.split_first_chunk::<4>().ok_or_else(malformed)?;
268        Ok((
269            usize::try_from(u32::from_le_bytes(*length)).map_err(|_| malformed())?,
270            4,
271        ))
272    };
273
274    let (&version, rest) = bytes.split_first().ok_or_else(malformed)?;
275    if version != VERDICT_VERSION {
276        return Err(StorageError::Corrupt(
277            "unknown event verdict version".to_owned(),
278        ));
279    }
280    let (&flags, rest) = rest.split_first().ok_or_else(malformed)?;
281    let (length, header) = take_len(rest)?;
282    let rest = rest.get(header..).ok_or_else(malformed)?;
283    if rest.len() < length {
284        return Err(malformed());
285    }
286    let (event_id, mut rest) = rest.split_at(length);
287
288    let mut verdict = Verdict {
289        outlier: flags & IS_OUTLIER != 0,
290        ..Verdict::default()
291    };
292    if flags & HAS_REJECTION != 0 {
293        let (&code, tail) = rest.split_first().ok_or_else(malformed)?;
294        verdict.rejection = Some(code);
295        rest = tail;
296    }
297    if flags & HAS_SOFT_FAIL != 0 {
298        let (&code, tail) = rest.split_first().ok_or_else(malformed)?;
299        verdict.soft_fail = Some(code);
300        rest = tail;
301    }
302    if flags & HAS_REDACTION != 0 {
303        let (length, header) = take_len(rest)?;
304        let tail = rest.get(header..).ok_or_else(malformed)?;
305        verdict.redacted_by = Some(tail.get(..length).ok_or_else(malformed)?.to_vec());
306    }
307    Ok((event_id, verdict))
308}

311pub const KEY_RECORD_VERSION: u8 = 1;

313/// Which of an origin's signing-key records.
314#[derive(Clone, Copy, Debug, Eq, PartialEq)]
315pub enum KeyRecordKind {
316    /// Cumulative `verify_keys` and `old_verify_keys` ever seen.
317    Historical,
318    /// Key ids bound only through a notary so far.
319    Provisional,
320    /// Key ids observed as currently active by a direct fetch.
321    Corroborated,
322    /// The last verbatim self-signed payload, re-served by the notary.
323    Notary,
324}
325
326impl KeyRecordKind {
327    /// Every kind, for scan and summary paths.
328    pub const ALL: [Self; 4] = [
329        Self::Historical,
330        Self::Provisional,
331        Self::Corroborated,
332        Self::Notary,
333    ];
334
335    /// The kind tag written into the record-id pre-image.
336    #[must_use]
337    pub const fn as_str(self) -> &'static str {
338        match self {
339            Self::Historical => "historical",
340            Self::Provisional => "provisional",
341            Self::Corroborated => "corroborated",
342            Self::Notary => "notary",
343        }
344    }
345
346    /// Parse a kind tag, or `None` for an unknown tag.
347    #[must_use]
348    pub fn parse(tag: &str) -> Option<Self> {
349        match tag {
350            "historical" => Some(Self::Historical),
351            "provisional" => Some(Self::Provisional),
352            "corroborated" => Some(Self::Corroborated),
353            "notary" => Some(Self::Notary),
354            _ => None,
355        }
356    }
357
358    fn as_bytes(self) -> &'static [u8] {
359        self.as_str().as_bytes()
360    }
361}

369/// Derive the record id of one signing-key record of `origin`.
370#[must_use]
371pub fn server_key_record_id(kind: KeyRecordKind, origin: &[u8]) -> [u8; 16] {
372    let mut identifier = Vec::with_capacity(
373        kind.as_bytes()
374            .len()
375            .saturating_add(origin.len())
376            .saturating_add(1),
377    );
378    identifier.extend_from_slice(kind.as_bytes());
379    identifier.push(0);
380    identifier.extend_from_slice(origin);
381    derive_id(b"server-key", &identifier)
382}

384/// Encode one signing-key record for storage.
385///
386/// # Errors
387/// Returns [`StorageError::Corrupt`] if the origin does not fit its length
388/// prefix.
389pub fn encode_server_key(origin: &[u8], payload: &[u8]) -> Result<Vec<u8>, StorageError> {
390    let length = u32::try_from(origin.len())
391        .map_err(|_| StorageError::Corrupt("origin is too large".to_owned()))?;
392    let mut out = Vec::with_capacity(
393        5_usize
394            .saturating_add(origin.len())
395            .saturating_add(payload.len()),
396    );
397    out.push(KEY_RECORD_VERSION);
398    out.extend_from_slice(&length.to_le_bytes());
399    out.extend_from_slice(origin);
400    out.extend_from_slice(payload);
401    Ok(out)
402}

404/// Decode a stored signing-key record into its origin and payload.
405///
406/// # Errors
407/// Returns [`StorageError::Corrupt`] for a truncated record or an unknown
408/// version.
409pub fn decode_server_key(bytes: &[u8]) -> Result<(&[u8], &[u8]), StorageError> {
410    let malformed = || StorageError::Corrupt("malformed signing-key record".to_owned());
411    let (&version, rest) = bytes.split_first().ok_or_else(malformed)?;
412    if version != KEY_RECORD_VERSION {
413        return Err(StorageError::Corrupt(
414            "unknown signing-key record version".to_owned(),
415        ));
416    }
417    let (length, rest) = rest.split_first_chunk::<4>().ok_or_else(malformed)?;
418    let length = usize::try_from(u32::from_le_bytes(*length)).map_err(|_| malformed())?;
419    if rest.len() < length {
420        return Err(malformed());
421    }
422    Ok(rest.split_at(length))
423}

478/// Encode an event locator: `[version][u32 LE length][event id][room id]`.
479///
480/// The full event id rides along because the record id is only 128 bits.
481///
482/// # Errors
483/// Returns [`StorageError::Corrupt`] if the event id does not fit its length
484/// prefix.
485pub fn encode_event_locator(event_id: &[u8], room_id: &[u8]) -> Result<Vec<u8>, StorageError> {
486    let length = u32::try_from(event_id.len())
487        .map_err(|_| StorageError::Corrupt("event id is too large".to_owned()))?;
488    let mut out = Vec::with_capacity(
489        1_usize
490            .saturating_add(4)
491            .saturating_add(event_id.len())
492            .saturating_add(room_id.len()),
493    );
494    out.push(LOCATOR_VERSION);
495    out.extend_from_slice(&length.to_le_bytes());
496    out.extend_from_slice(event_id);
497    out.extend_from_slice(room_id);
498    Ok(out)
499}

522/// Encode an edge list (prev or auth) as repeated `[u32 LE length][event id]`.

524/// # Errors

527    let too_large = || StorageError::Corrupt("event edge is too large".to_owned());
528    let mut out = Vec::new();
529    for edge in edges {
530        let length = u32::try_from(edge.len()).map_err(|_| too_large())?;

532        out.extend_from_slice(edge);
533    }
534    Ok(out)
535}

584/// # Errors
585/// Returns [`StorageError::Corrupt`] unless the record is exactly four bytes.
586pub fn decode_auth_counter(bytes: &[u8]) -> Result<u32, StorageError> {
587    <[u8; 4]>::try_from(bytes)
588        .map(u32::from_be_bytes)
589        .map_err(|_| StorageError::Corrupt("malformed auth counter".to_owned()))
590}

592/// Encode the per-room short-id counter as big-endian `u32`.
593#[must_use]
594pub const fn encode_auth_counter(value: u32) -> [u8; 4] {
595    value.to_be_bytes()
596}

598/// Decode an edge list written by [`encode_edges`].
599///
600/// # Errors
601/// Returns [`StorageError::Corrupt`] for a truncated record.
602pub fn decode_edges(bytes: &[u8]) -> Result<Vec<Vec<u8>>, StorageError> {
603    let malformed = || StorageError::Corrupt("malformed event edge record".to_owned());
604    let mut edges = Vec::new();
605    let mut rest = bytes;
606    while !rest.is_empty() {
607        let (length, tail) = rest.split_first_chunk::<4>().ok_or_else(malformed)?;
608        let length = usize::try_from(u32::from_le_bytes(*length)).map_err(|_| malformed())?;
609        let (edge, tail) = tail.split_at_checked(length).ok_or_else(malformed)?;
610        edges.push(edge.to_vec());
611        rest = tail;
612    }
613    Ok(edges)
614}

642/// Encode the import marker for storage.
643#[must_use]
644pub fn encode_import_marker(marker: ImportMarker) -> Vec<u8> {
645    let mut out = vec![MARKER_VERSION, marker.data_format];
646    out.extend_from_slice(&marker.events.to_le_bytes());
647    out.extend_from_slice(&marker.closures.to_le_bytes());
648    out
649}

651/// Decode a stored import marker.
652///
653/// # Errors
654/// Returns [`StorageError::Corrupt`] for a truncated record, an unknown
655/// version, or a data format this build does not read.
656pub fn decode_import_marker(bytes: &[u8]) -> Result<ImportMarker, StorageError> {
657    let malformed = || StorageError::Corrupt("malformed import marker".to_owned());
658    let (&version, rest) = bytes.split_first().ok_or_else(malformed)?;
659    if version != MARKER_VERSION {
660        return Err(StorageError::Corrupt(
661            "unknown import marker version".to_owned(),
662        ));
663    }
664    let (&data_format, rest) = rest.split_first().ok_or_else(malformed)?;
665    let (events, rest) = rest.split_first_chunk::<8>().ok_or_else(malformed)?;
666    let (closures, _) = rest.split_first_chunk::<8>().ok_or_else(malformed)?;
667    if data_format != DATA_FORMAT {
668        return Err(StorageError::Corrupt(
669            "unsupported import data format".to_owned(),
670        ));
671    }
672    Ok(ImportMarker {
673        data_format,
674        events: u64::from_le_bytes(*events),
675        closures: u64::from_le_bytes(*closures),
676    })
677}

683/// Pool holding HAMT nodes and state roots.
684pub const STATE_POOL: ShardType = ShardType::State;
685
686/// Default charged staging budget per state batch, including per-put framing.
687pub const STATE_BATCH_BYTES: usize = 48 << 20;
688
689/// Stage HAMT nodes, then roots, into bounded transactions of at most
690/// `batch_bytes` charged bytes, committing each before the next begins.
691///
692/// All supplied nodes are committed before any root. The caller must supply
693/// every node needed by the roots that is not already stored; references and
694/// content hashes are not validated. An interrupted run can leave unreferenced
695/// nodes and already-committed roots. `nodes` and `roots` are
696/// `(collection, node id, bytes)` stored in [`STATE_POOL`]; the caller derives
697/// ids with the functions above. The budget includes payload bytes plus
698/// [`crate::journal::TXN_STAGE_PUT_OVERHEAD_BYTES`] per record. Returns the
699/// number of transactions committed, or zero if both inputs are empty.
700///
701/// # Errors
702/// Returns the first staging or commit error; earlier batches stay committed.
703/// Returns [`StorageError::Io`] with [`std::io::ErrorKind::InvalidInput`] for
704/// a zero or over-engine-limit budget, staging-size overflow, or a single
705/// record that cannot fit in the requested budget.
706///
707/// This is not an atomic migration across batches. The caller must publish its
708/// import-complete marker only after this function succeeds; on failure,
709/// already-committed nodes/roots remain visible, with nodes always committed
710/// before roots.
711pub fn commit_state_batches<'a>(
712    db: &crate::Database,
713    nodes: impl IntoIterator<Item = ([u8; 16], [u8; 16], &'a [u8])>,
714    roots: impl IntoIterator<Item = ([u8; 16], [u8; 16], &'a [u8])>,
715    batch_bytes: usize,
716) -> Result<usize, StorageError> {
717    fn commit_records<'a>(
718        db: &crate::Database,
719        records: impl IntoIterator<Item = ([u8; 16], [u8; 16], &'a [u8])>,
720        batch_bytes: usize,
721        committed: &mut usize,
722    ) -> Result<(), StorageError> {
723        let mut txn = db.begin_transaction();
724        let mut staged = 0_usize;
725        for (collection, node, bytes) in records {
726            let charge = bytes
727                .len()
728                .checked_add(crate::journal::TXN_STAGE_PUT_OVERHEAD_BYTES)
729                .ok_or_else(|| {
730                    StorageError::Io(std::io::Error::new(
731                        std::io::ErrorKind::InvalidInput,
732                        "state record staging size overflow",
733                    ))
734                })?;
735            if charge > batch_bytes {
736                return Err(StorageError::Io(std::io::Error::new(
737                    std::io::ErrorKind::InvalidInput,
738                    "one state record exceeds the transaction staging budget",
739                )));
740            }
741            if staged > 0 && staged.saturating_add(charge) > batch_bytes {
742                txn.commit()?;
743                *committed = committed.saturating_add(1);
744                txn = db.begin_transaction();
745                staged = 0;
746            }
747            txn.put(
748                STATE_POOL,
749                collection,
750                node,
751                &crate::storage::NodeData::new(bytes::Bytes::copy_from_slice(bytes)),
752            )?;
753            staged = staged.saturating_add(charge);
754        }
755        if staged > 0 {
756            txn.commit()?;
757            *committed = committed.saturating_add(1);
758        }
759        Ok(())
760    }
761    if batch_bytes == 0 || batch_bytes > crate::journal::MAX_TXN_STAGE_BYTES {
762        return Err(StorageError::Io(std::io::Error::new(
763            std::io::ErrorKind::InvalidInput,
764            format!(
765                "state batch budget must be in 1..={} bytes",
766                crate::journal::MAX_TXN_STAGE_BYTES
767            ),
768        )));
769    }
770    let mut committed = 0_usize;
771    commit_records(db, nodes, batch_bytes, &mut committed)?;
772    commit_records(db, roots, batch_bytes, &mut committed)?;
773    Ok(committed)
774}
