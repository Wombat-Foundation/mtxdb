28/// On-disk journal format version. The single place that decides frame
29/// dialect: callers match on this rather than comparing raw version numbers,
30/// so a future bump only has to add a variant here.
31#[derive(Clone, Copy, Debug, PartialEq, Eq)]
32enum JournalVersion {
33    /// Per-pool segment. Mutation frames are untagged and the flag byte must be
34    /// zero. Version 4 adds the embedded durability-mark header fields.
35    V4,
36    /// Shared multi-pool segment. Every mutation frame carries a mandatory pool
37    /// tag in the flag byte, so one physical WAL can carry the state,
38    /// event-DAG, and edges pools and recovery can route each frame back
39    /// to its pool.
40    V5PoolTagged,
41}
42
43impl JournalVersion {
44    /// Version a freshly created per-pool segment uses.
45    const fn per_pool() -> Self {
46        Self::V4
47    }
48
49    /// Version a freshly created shared (multi-pool) segment uses.
50    #[cfg(feature = "multi-reader")]
51    const fn shared() -> Self {
52        Self::V5PoolTagged
53    }
54
55    const fn as_u32(self) -> u32 {
56        match self {
57            Self::V4 => 4,
58            Self::V5PoolTagged => 5,
59        }
60    }
61
62    fn from_u32(value: u32) -> io::Result<Self> {
63        match value {
64            4 => Ok(Self::V4),
65            5 => Ok(Self::V5PoolTagged),
66            _ => Err(invalid_data("unsupported journal version")),
67        }
68    }
69
70    /// Whether every mutation frame in this segment must carry a pool tag.
71    const fn is_pool_tagged(self) -> bool {
72        matches!(self, Self::V5PoolTagged)
73    }
74}

85/// Size of the unit a torn write can damage. 4096 matches common 4 KiB
86/// physical sectors (advanced-format disks), which is the usual atomic write unit, though no filesystem or device guarantees it: with 512-byte
87/// spacing both slots and the base header could share one such unit and a
88/// single torn write could reach all three. It costs 12 KiB per segment.
89const MARK_SECTOR_LEN: usize = 4096;
90/// The durability mark lives in two alternating slots, each alone in its own
91/// [`MARK_SECTOR_LEN`] unit after the one that holds the immutable base header. A torn write
92/// of one slot can therefore damage only that slot: never the other slot, and
93/// never the base header, which nothing rewrites after the file is created.
94/// Each slot is `generation(8) | durable_len(8) | crc(4)`; an
95/// all-zero slot has never been written and claims nothing.
96const MARK_SLOT_OFFSETS: [usize; 2] = [MARK_SECTOR_LEN, 2 * MARK_SECTOR_LEN];
97const MARK_SLOT_LEN: usize = 20;
98/// Where journal groups begin: after the base header's sector and both mark
99/// sectors. Everything that needs "the end of the header" uses this.
100const FILE_HEADER_LEN: usize = 3 * MARK_SECTOR_LEN;

138/// Segment size above which a sync forces a full index checkpoint, because only
139/// a full checkpoint records journal coverage and lets the segment be
140/// reclaimed. A quarter of [`MAX_SEGMENT_LEN`] leaves headroom for writes that
141/// land while the checkpoint runs. It is a margin, not a guarantee: a writer
142/// that appends more than the remaining space during one checkpoint still fills
143/// the segment.
144/// A coordinator starts with this and can be given another with
145/// [`JournalCoordinator::set_reclaim_trigger_len`].
146pub(crate) const RECLAIM_TRIGGER_LEN: u64 = MAX_SEGMENT_LEN / 4;

174/// Lifecycle of a transaction's staged journal mutations.
175#[cfg(feature = "multi-reader")]
176#[derive(Debug, Clone, Copy, PartialEq, Eq)]
177pub enum TxnStageState {
178    /// The SQL transaction attempt is active and may add mutations.
179    Active,
180    /// The published journal group is being materialized into pack/index
181    /// storage. Per-mutation progress makes retries resumable.
182    Materializing,
183    /// The journal group is published; pack/index application is still
184    /// pending or retrying.
185    JournalPublished,
186    /// The SQL attempt failed; retained callbacks must not publish its data.
187    Discarded,
188    /// Every per-pool group was appended successfully.
189    Published,
190}

297    /// Look several records up in the mutations staged for `pool`, in the order
298    /// of `node_ids`, taking the stage lock once and scanning the staged
299    /// mutations once however many records are asked for.
300    ///
301    /// This is what lets a transaction read its own uncommitted writes. The
302    /// staged mutations are in append order, so the newest one that names a
303    /// record wins. A staged `DeleteCollection` for the record's collection
304    /// hides everything older, both earlier staged puts and whatever the live
305    /// pool holds, while a put staged after it is newer and is seen. Only an
306    /// active stage answers: once the stage is discarded, or publication has
307    /// begun, every record is `Absent` and the caller reads the live store,
308    /// which by then serves the group through the transaction overlay.
309    ///
310    /// The single pass keeps the newest put position for each wanted record
311    /// and the position of the last delete for the collection, so the cost is
312    /// the staged mutations plus the records asked for, not their product.
313    /// Payloads are cloned only for records that resolve to a put.
314    #[must_use]
315    pub fn lookup_many(
316        &self,
317        pool: ShardType,
318        collection_id: &[u8; 16],
319        node_ids: &[[u8; 16]],
320    ) -> Vec<StagedLookup> {
321        #[cfg(test)]
322        self.lookup_many_calls
323            .fetch_add(1, std::sync::atomic::Ordering::Relaxed);
324        let data = self.data.lock();
325        if self.state.load(Ordering::Acquire) != Self::ACTIVE || node_ids.is_empty() {
326            return vec![StagedLookup::Absent; node_ids.len()];
327        }
328        let wanted: HashSet<[u8; 16]> = node_ids.iter().copied().collect();
329        let mut newest_put: HashMap<[u8; 16], (usize, &Vec<u8>)> = HashMap::new();
330        let mut last_delete: Option<usize> = None;
331        for (position, mutation) in data.pools[pool_index(pool)].iter().enumerate() {
332            match mutation {
333                Mutation::Put {
334                    collection_id: staged_collection,
335                    node_id,
336                    payload,
337                } if staged_collection == collection_id && wanted.contains(node_id) => {
338                    newest_put.insert(*node_id, (position, payload));
339                }
340                Mutation::DeleteCollection {
341                    collection_id: staged_collection,
342                } if staged_collection == collection_id => last_delete = Some(position),
343                _ => {}
344            }
345        }
346        node_ids
347            .iter()
348            .map(|node_id| match (newest_put.get(node_id), last_delete) {
349                // A put older than the delete is hidden by it.
350                (Some((position, _)), Some(delete)) if *position < delete => StagedLookup::Deleted,
351                (Some((_, payload)), _) => StagedLookup::Put((*payload).clone()),
352                (None, Some(_)) => StagedLookup::Deleted,
353                (None, None) => StagedLookup::Absent,
354            })
355            .collect()
356    }

366    /// Discard an active attempt. Safe to call more than once.
367    ///
368    /// See the "Not a rollback" section on [`TxnStage`]: this drops only the
369    /// buffered journal mutations and never touches packs or the live index.
370    pub fn discard(&self) {
371        let mut data = self.data.lock();
372        if self.state.load(Ordering::Acquire) == Self::ACTIVE {
373            data.pools.iter_mut().for_each(Vec::clear);
374            data.applied.iter_mut().for_each(Vec::clear);
375            data.bytes = 0;
376            self.state.store(Self::DISCARDED, Ordering::Release);
377        }
378    }

495    /// Ensure an estimated batch fits before buffering its journal mutations.
496    ///
497    /// # Errors
498    /// Returns `InvalidInput` if the stage is not active or the estimated
499    /// total exceeds [`MAX_TXN_STAGE_BYTES`].
500    pub fn ensure_capacity(&self, additional: usize) -> io::Result<()> {
501        if self.state.load(Ordering::Acquire) != Self::ACTIVE {
502            return Err(io::Error::other("transaction stage is not active"));
503        }
504        let data = self.data.lock();
505        let total = data.bytes.checked_add(additional).ok_or_else(|| {
506            io::Error::new(
507                io::ErrorKind::InvalidInput,
508                "transaction stage size overflow",
509            )
510        })?;
511        if total > MAX_TXN_STAGE_BYTES {
512            return Err(io::Error::new(
513                io::ErrorKind::InvalidInput,
514                "transaction journal stage exceeds its 64 MiB limit",
515            ));
516        }
517        Ok(())
518    }

520    /// Add an immutable put to this attempt's pool-ordered journal batch.
521    ///
522    /// # Errors
523    /// Returns `InvalidInput` if adding the payload would exceed the stage
524    /// limit, or another error if the stage has already been discarded or
525    /// published.
526    pub fn stage_put(
527        &self,
528        pool: ShardType,
529        collection_id: [u8; 16],
530        node_id: [u8; 16],
531        payload: Vec<u8>,
532    ) -> io::Result<()> {
533        let charge = payload.len().saturating_add(64);
534        let mut data = self.data.lock();
535        if self.state.load(Ordering::Acquire) != Self::ACTIVE {
536            return Err(io::Error::other("transaction stage is not active"));
537        }
538        let total = data.bytes.checked_add(charge).ok_or_else(|| {
539            io::Error::new(
540                io::ErrorKind::InvalidInput,
541                "transaction stage size overflow",
542            )
543        })?;
544        if total > MAX_TXN_STAGE_BYTES {
545            return Err(io::Error::new(
546                io::ErrorKind::InvalidInput,
547                "transaction journal stage exceeds its 64 MiB limit",
548            ));
549        }
550        data.pools[pool_index(pool)].push(Mutation::Put {
551            collection_id,
552            node_id,
553            payload,
554        });
555        data.applied[pool_index(pool)].push(false);
556        data.bytes = total;
557        Ok(())
558    }

560    /// Add a batch of immutable puts atomically.
561    ///
562    /// # Errors
563    /// Returns `InvalidInput` if the batch would exceed the stage limit, or
564    /// another error if the stage has already been discarded or published.
565    pub fn stage_puts(
566        &self,
567        pool: ShardType,
568        collection_id: [u8; 16],
569        entries: &[([u8; 16], Vec<u8>)],
570    ) -> io::Result<()> {
571        let charge = entries
572            .iter()
573            .try_fold(0usize, |total, (_, payload)| {
574                total.checked_add(payload.len().saturating_add(64))
575            })
576            .ok_or_else(|| {
577                io::Error::new(
578                    io::ErrorKind::InvalidInput,
579                    "transaction stage size overflow",
580                )
581            })?;
582        let mut data = self.data.lock();
583        if self.state.load(Ordering::Acquire) != Self::ACTIVE {
584            return Err(io::Error::other("transaction stage is not active"));
585        }
586        let total = data.bytes.checked_add(charge).ok_or_else(|| {
587            io::Error::new(
588                io::ErrorKind::InvalidInput,
589                "transaction stage size overflow",
590            )
591        })?;
592        if total > MAX_TXN_STAGE_BYTES {
593            return Err(io::Error::new(
594                io::ErrorKind::InvalidInput,
595                "transaction journal stage exceeds its 64 MiB limit",
596            ));
597        }
598        let pool_mutations = &mut data.pools[pool_index(pool)];
599        pool_mutations.extend(entries.iter().map(|(node_id, payload)| Mutation::Put {
600            collection_id,
601            node_id: *node_id,
602            payload: payload.clone(),
603        }));
604        data.applied[pool_index(pool)].extend(vec![false; entries.len()]);
605        data.bytes = total;
606        Ok(())
607    }
608
609    /// Add a collection delete to this attempt. Callers must not eagerly
610    /// mutate the live index for a delete that can still roll back.
611    ///
612    /// # Errors
613    /// Returns `InvalidInput` if adding the mutation would exceed the stage
614    /// limit, or another error if the stage has already been discarded or
615    /// published.
616    pub fn stage_delete_collection(
617        &self,
618        pool: ShardType,
619        collection_id: [u8; 16],
620    ) -> io::Result<()> {
621        let mut data = self.data.lock();
622        if self.state.load(Ordering::Acquire) != Self::ACTIVE {
623            return Err(io::Error::other("transaction stage is not active"));
624        }
625        let total = data.bytes.checked_add(64).ok_or_else(|| {
626            io::Error::new(
627                io::ErrorKind::InvalidInput,
628                "transaction stage size overflow",
629            )
630        })?;
631        if total > MAX_TXN_STAGE_BYTES {
632            return Err(io::Error::new(
633                io::ErrorKind::InvalidInput,
634                "transaction journal stage exceeds its 64 MiB limit",
635            ));
636        }
637        data.pools[pool_index(pool)].push(Mutation::DeleteCollection { collection_id });
638        data.applied[pool_index(pool)].push(false);
639        data.bytes = total;
640        Ok(())
641    }
642
643    /// Append staged pool groups in dependency order: edges, event-DAG,
644    /// then state. Does not fsync; the ordinary coalesced sync remains the
645    /// durability boundary. Repeated calls are safe, including after a
646    /// partial error.
647    ///
648    /// # Errors
649    /// Returns an error if a staged pool has no coordinator or its journal
650    /// cannot append the group's complete framing and trailer.
651    pub fn publish(
652        &self,
653        coordinators: [Option<&JournalCoordinator>; ShardType::ALL.len()],
654    ) -> io::Result<()> {
655        let mut data = self.data.lock();
656        match self.state.load(Ordering::Acquire) {
657            Self::DISCARDED | Self::JOURNAL_PUBLISHED | Self::MATERIALIZING | Self::PUBLISHED => {
658                return Ok(())
659            }
660            _ => {}
661        }
662        if coordinators.iter().all(Option::is_none) {
663            // Journaling is disabled process-wide, so there is no journal to
664            // publish into.
665            self.state.store(Self::PUBLISHED, Ordering::Release);
666            return Ok(());
667        }
668        // Preserve the historical per-pool append order for the existing
669        // pools; ServerInfo is appended after them without changing the WAL
670        // ordering of legacy frames.
671        let pools = [
672            ShardType::Edges,
673            ShardType::EventDag,
674            ShardType::State,
675            ShardType::ServerInfo,
676        ];
677
678        // A shared-WAL database gives every pool the same coordinator. Keep
679        // the transaction as one journal group in that case, so readers never
680        // observe only a prefix of a cross-pool commit. Standalone/per-pool
681        // journals cannot provide that boundary and retain the ordered,
682        // retry-safe fallback below.
683        let active_pool_count = pools
684            .iter()
685            .filter(|pool| {
686                let index = pool_index(**pool);
687                !data.appended[index] && !data.pools[index].is_empty()
688            })
689            .count();
690        if active_pool_count == 0 {
691            self.state.store(Self::PUBLISHED, Ordering::Release);
692            return Ok(());
693        }
694        let active_coordinators = pools
695            .iter()
696            .filter_map(|pool| {
697                let index = pool_index(*pool);
698                (!data.appended[index] && !data.pools[index].is_empty())
699                    .then(|| coordinators[index])
700            })
701            .flatten()
702            .collect::<Vec<_>>();
703        if active_coordinators.len() == active_pool_count {
704            if let Some(coordinator) = active_coordinators.first().copied() {
705                if active_coordinators
706                    .iter()
707                    .all(|candidate| std::ptr::eq(*candidate, coordinator))
708                {
709                    let batches = pools
710                        .iter()
711                        .filter_map(|pool| {
712                            let index = pool_index(*pool);
713                            (!data.appended[index] && !data.pools[index].is_empty())
714                                .then_some((*pool, data.pools[index].as_slice()))
715                        })
716                        .collect::<Vec<_>>();
717                    let receipt = coordinator.publish_tagged_groups(&batches)?;
718                    data.appended.fill(true);
719                    data.receipt = Some(receipt);
720                    self.mark_journal_published()?;
721                    return Ok(());
722                }
723            }
724        }
725
726        if active_pool_count > 1 {
727            return Err(io::Error::new(
728                io::ErrorKind::Unsupported,
729                "cross-pool transactions require one shared journal coordinator",
730            ));
731        }
732
733        for pool in pools {
734            let index = pool_index(pool);
735            if data.appended[index] || data.pools[index].is_empty() {
736                data.appended[index] = true;
737                continue;
738            }
739            let coordinator = coordinators[pool_index(pool)].ok_or_else(|| {
740                io::Error::new(io::ErrorKind::NotConnected, "staged pool has no journal")
741            })?;
742            let receipt = coordinator.publish_group_tagged(pool, &data.pools[index])?;
743            data.appended[index] = true;
744            data.receipt = Some(receipt);
745        }
746        self.mark_journal_published()?;
747        Ok(())
748    }
749}

932#[derive(Default)]
933struct DurableWaitTotals {
934    calls: AtomicU64,
935    total_ns: AtomicU64,
936    max_ns: AtomicU64,
937    buckets: [AtomicU64; 5],
938}
939
940impl DurableWaitTotals {
941    fn observe(&self, duration: std::time::Duration) {
942        let nanos = u64::try_from(duration.as_nanos()).unwrap_or(u64::MAX);
943        self.calls.fetch_add(1, Ordering::Relaxed);
944        self.total_ns.fetch_add(nanos, Ordering::Relaxed);
945        self.max_ns.fetch_max(nanos, Ordering::Relaxed);
946        let micros = duration.as_micros();
947        let bucket = if micros < 1_000 {
948            0
949        } else if micros < 10_000 {
950            1
951        } else if micros < 100_000 {
952            2
953        } else if micros < 1_000_000 {
954            3
955        } else {
956            4
957        };
958        self.buckets[bucket].fetch_add(1, Ordering::Relaxed);
959    }
960
961    fn snapshot(&self) -> DurableWaitLatency {
962        DurableWaitLatency {
963            calls: self.calls.load(Ordering::Relaxed),
964            total: std::time::Duration::from_nanos(self.total_ns.load(Ordering::Relaxed)),
965            max: std::time::Duration::from_nanos(self.max_ns.load(Ordering::Relaxed)),
966            buckets: std::array::from_fn(|i| self.buckets[i].load(Ordering::Relaxed)),
967        }
968    }

1134/// The last group of the longest prefix of `groups` that a shared segment may
1135/// drop: every pool with a frame in each group has reported coverage through the
1136/// group's `last_lsn`, and the group holds no untagged frame (which no pool's
1137/// coverage can account for). `None` when even the first group is still needed.
1138///
1139/// The one place this rule lives: the directory and the scan fallback both call
1140/// it, so they cannot disagree.
1141#[cfg(feature = "multi-reader")]
1142fn boundary_through(
1143    groups: &[GroupMark],
1144    covered: &HashMap<ShardType, u64>,
1145) -> Option<(u64, Option<ShardType>)> {
1146    let mut boundary = None;
1147    for group in groups {
1148        for (index, pool) in ShardType::ALL.into_iter().enumerate() {
1149            if group.pools & (1_u8 << index) != 0
1150                && !covered.get(&pool).is_some_and(|lsn| *lsn >= group.last_lsn)
1151            {
1152                // This pool's coverage stops the cut here; report it as the
1153                // blocker so reclaim can say why the suffix was retained.
1154                return boundary.map(|lsn| (lsn, Some(pool)));
1155            }
1156        }
1157        if group.pools & UNATTRIBUTED_POOL_BIT != 0 {
1158            // No pool's coverage can account for an untagged frame: a real
1159            // blocker with no pool to name.
1160            return boundary.map(|lsn| (lsn, None));
1161        }
1162        boundary = Some(group.last_lsn);
1163    }
1164    boundary.map(|lsn| (lsn, None))
1165}

1195/// How long the remembered window is for a segment whose last complete group
1196/// ends at `valid_len`: the last [`CONSUMED_TAIL_LEN`] bytes, but never
1197/// reaching into the file header.
1198#[must_use]
1199pub(crate) fn consumed_tail_len(valid_len: u64) -> usize {
1200    let header_len = u64::try_from(FILE_HEADER_LEN).unwrap_or(u64::MAX);
1201    let available = valid_len.saturating_sub(header_len);
1202    usize::try_from(available).map_or(CONSUMED_TAIL_LEN, |len| len.min(CONSUMED_TAIL_LEN))
1203}

1272    base_sequence: u64,
1273    base_lsn: u64,

1685    /// Highest committed group `last_lsn` that carried at least one frame for
1686    /// `pool` and is fully in the packs, or 0 when there is none.

1688    /// A pool checkpoint must record this, not [`Self::committed_lsn`]. One
1689    /// shared group can interleave several pools' frames, so the global
1690    /// committed LSN can run ahead of what this pool's index has materialized;
1691    /// recording it would claim coverage the checkpoint does not have and let
1692    /// reclaim drop another pool's uncovered frames.
1693    #[must_use]

1695    pub fn committed_lsn_for_pool(&self, pool: ShardType) -> u64 {
1696        let committed = self.pool_committed.lock().get(&pool).copied().unwrap_or(0);
1697        // A published transaction whose writes are not all in the packs yet
1698        // holds coverage below its first frame.
1699        match self.unmaterialized.lock().first() {
1700            Some(first_lsn) => committed.min(first_lsn.saturating_sub(1)),
1701            None => committed,
1702        }

1716        let previous = self.committed_lsn.fetch_max(through_lsn, Ordering::AcqRel);
1717        if through_lsn > previous {
1718            let covered = through_lsn.saturating_sub(previous);
1719            self.commits.fetch_add(1, Ordering::Relaxed);
1720            self.commit_records.fetch_add(covered, Ordering::Relaxed);
1721            self.max_commit_records
1722                .fetch_max(covered, Ordering::Relaxed);
1723            self.commit_epoch.fetch_add(1, Ordering::AcqRel);
1724            self.wake_committer_if_backlogged();
1725        }
1726    }

1728    /// Advance per-pool committed watermarks for every appended group whose
1729    /// `last_lsn` is at or below `through_lsn`.
1730    fn promote_pool_committed(&self, through_lsn: u64) {
1731        let mut promotions = self.pending_promotions.lock();
1732        let mut watermarks = self.pool_committed.lock();
1733        let (ready, pending): (Vec<PendingPromotion>, Vec<PendingPromotion>) = promotions
1734            .drain(..)
1735            .partition(|(last_lsn, _)| *last_lsn <= through_lsn);

1737        for (_, extents) in ready {
1738            for (pool, lsn) in extents {

1740                *entry = (*entry).max(lsn);
1741            }
1742        }

1745    /// Reclaim the longest prefix of the shared segment whose every group is
1746    /// durably covered by all of that group's own pools, returning the reclaim
1747    /// result or `None` when nothing is reclaimable yet.
1748    ///
1749    /// A group is reclaimed atomically, so it may only be dropped once every
1750    /// pool that contributed a frame to it has reported coverage through the
1751    /// group's `last_lsn`. The pools in one group are independent of the pools
1752    /// in another, so the reclaimable prefix is **not** a single minimum across
1753    /// every pool: an idle pool whose last frame sits in an early group must not
1754    /// pin later groups that its frames never reached. Walk the live groups in
1755    /// order and stop at the first one any contributing pool has not covered.
1756    ///
1757    /// # Errors
1758    /// Propagates a scan or segment-rewrite failure from [`Self::reclaim_through`].
1759    #[cfg(feature = "multi-reader")]
1760    pub fn reclaim_shared(&self) -> io::Result<Option<Reclaim>> {
1761        // The group directory answers this without reading the segment. The
1762        // lock order matches `reclaim_through`: the sync lock, then the journal,
1763        // and coverage last.
1764        {
1765            let _sync = self.sync_lock.lock();
1766            let mut journal = self.journal.lock();
1767            let boundary = {
1768                let coverage = self.coverage.lock();
1769                journal.shared_reclaim_boundary(&coverage.covered)
1770            };
1771            match boundary {
1772                SharedBoundary::Through(covered_lsn, blocked_by) => {
1773                    let reclaimed = journal
1774                        .reclaim_through_with_blocker(covered_lsn, blocked_by)
1775                        .map(Some);
1776                    let coverage = self.coverage.lock();
1777                    self.record_reclaim_outcome(&journal, &coverage.covered);
1778                    return reclaimed;
1779                }
1780                SharedBoundary::NothingCovered => {
1781                    let coverage = self.coverage.lock();
1782                    self.record_reclaim_outcome(&journal, &coverage.covered);
1783                    return Ok(None);
1784                }
1785                SharedBoundary::Untrusted => {}
1786            }
1787        }
1788        // The directory did not match the file: decide from a scan of it.
1789        let scan = Journal::scan_read_only(&self.path)?;
1790        let marks = marks_from_scan(&scan);
1791        let boundary = {
1792            let coverage = self.coverage.lock();
1793            boundary_through(&marks, &coverage.covered)
1794        };
1795        match boundary {
1796            Some((covered_lsn, blocked_by)) => self
1797                .reclaim_through_with_blocker(covered_lsn, blocked_by)
1798                .map(Some),
1799            None => Ok(None),
1800        }
1801    }

1836    /// Note how a reclaim ended, with the journal and sync locks held. A
1837    /// segment still over the trigger means the reclaim is stalled: remember
1838    /// it, and say so once, naming the pools it is waiting on.
1839    #[cfg(feature = "multi-reader")]
1840    fn record_reclaim_outcome(&self, journal: &Journal, covered: &HashMap<ShardType, u64>) {
1841        let len = journal.file_len;
1842        if len <= self.reclaim_trigger_len() {
1843            *self.reclaim_stall.lock() = None;
1844            *self.last_remediation.lock() = None;
1845            return;
1846        }
1847        let epoch = self.coverage_epoch.load(Ordering::Acquire);
1848        let mut stall = self.reclaim_stall.lock();
1849        let previous = *stall;
1850        let emergency = len >= journal.segment_cap.saturating_sub(journal.segment_cap / 4);
1851        // The first reclaim to leave the segment over the trigger is normal:
1852        // pools sync one after another, so the first to cross it cannot yet
1853        // reclaim what the others have not reported. It is only worth saying
1854        // once the segment has grown by an eighth of the trigger without a
1855        // reclaim getting it back under the trigger, or it reaches the
1856        // emergency zone. The
1857        // back-off and remediation below act on the stall from the start.
1858        let grace_from = previous.map_or(len, |old| old.grace_from);
1859        let already_reported = previous.is_some_and(|old| old.reported);
1860        let report = !already_reported
1861            && (emergency || len >= grace_from.saturating_add(self.reclaim_retry_growth()));
1862        *stall = Some(ReclaimStall {
1863            at_len: len,
1864            coverage_epoch: epoch,
1865            grace_from,
1866            reported: already_reported || report,
1867        });
1868        let entered_emergency = emergency
1869            && previous.is_some_and(|old| {
1870                old.at_len < journal.segment_cap.saturating_sub(journal.segment_cap / 4)
1871            });
1872        if report {
1873            self.reclaim_stalls.fetch_add(1, Ordering::Relaxed);
1874        }
1875        if report || (already_reported && entered_emergency) {
1876            eprintln!(
1877                "warning: shared WAL reclaim is stalled at {len} of {} bytes (trigger {}); \
1878                 waiting on pools {:?} to report coverage",
1879                journal.segment_cap,
1880                self.reclaim_trigger_len(),
1881                journal.blocking_pools(covered)
1882            );
1883        }
1884    }

1886    /// If a stalled reclaim has reached the emergency zone, make the pools
1887    /// holding it back checkpoint, except `caller`, which is mid-sync. Runs
1888    /// with no journal or persistence lock held, at most once per further
1889    /// eighth of the trigger of growth unless coverage moved, and never from
1890    /// inside another remediation.
1891    #[cfg(feature = "multi-reader")]
1892    pub fn remediate_blockers(&self, caller: Option<ShardType>) {
1893        if REMEDIATING.with(std::cell::Cell::get) {
1894            return;
1895        }
1896        let len = self.segment_len();
1897        if len < self.emergency_len() || !self.is_reclaim_stalled() {
1898            return;
1899        }
1900        let epoch = self.coverage_epoch.load(Ordering::Acquire);
1901        {
1902            let mut last = self.last_remediation.lock();
1903            if let Some((last_len, last_epoch)) = *last {
1904                if last_epoch == epoch && len < last_len.saturating_add(self.reclaim_retry_growth())
1905                {
1906                    return;
1907                }
1908            }
1909            *last = Some((len, epoch));
1910        }
1911        let Some(remediate) = self.blocker_remediation.lock().clone() else {
1912            return;
1913        };
1914        REMEDIATING.with(|flag| flag.set(true));
1915        for pool in self.reclaim_blockers() {
1916            if Some(pool) != caller {
1917                remediate(pool);
1918            }
1919        }
1920        REMEDIATING.with(|flag| flag.set(false));
1921    }

2010    /// Durably commit pending mutations and report where the time went.
2011    ///
2012    /// # Errors
2013    /// Returns an I/O error if the target is invalid, the journal is poisoned,
2014    /// or appending/fsyncing the group fails.
2015    pub fn sync_through_timed(
2016        &self,
2017        target_lsn: u64,
2018    ) -> io::Result<(Option<CommitReceipt>, JournalSyncTimings)> {
2019        self.sync_calls.fetch_add(1, Ordering::Relaxed);
2020        let mut timings = JournalSyncTimings::default();
2021        let in_flight = self
2022            .sync_in_flight
2023            .fetch_add(1, Ordering::AcqRel)
2024            .saturating_add(1);
2025        timings.journal_in_flight = in_flight;
2026        let _in_flight_guard = SyncInFlightGuard(&self.sync_in_flight);
2027        if target_lsn > self.published_lsn.load(Ordering::Acquire) {
2028            return Err(io::Error::new(
2029                io::ErrorKind::InvalidInput,
2030                "sync target has not been published",
2031            ));
2032        }
2033        if self.covered_by_prior_commit(target_lsn, &mut timings) {
2034            return Ok((None, timings));
2035        }
2036        self.reject_if_poisoned()?;
2037
2038        let _sync = self.acquire_sync_lock(&mut timings);
2039        self.reject_if_poisoned()?;
2040        // A concurrent fsync that finished while we waited usually covered us.
2041        if self.covered_by_prior_commit(target_lsn, &mut timings) {
2042            return Ok((None, timings));
2043        }
2044        let previously_committed = self.committed_lsn.load(Ordering::Acquire);
2045
2046        let capture = self.capture_sync_range(target_lsn)?;
2047        self.flush_captured_file(&capture, &mut timings)?;
2048        let receipt = self.record_durable_range(previously_committed, &capture, &mut timings);
2049        Ok((Some(receipt), timings))
2050    }

2074    /// Take [`Self::sync_lock`], counting this caller as a waiter when another
2075    /// fsync is in flight, and record how long the wait took.
2076    fn acquire_sync_lock(&self, timings: &mut JournalSyncTimings) -> MutexGuard<'_, ()> {
2077        let lock_started = std::time::Instant::now();
2078        let guard = if let Some(guard) = self.sync_lock.try_lock() {
2079            guard
2080        } else {
2081            timings.journal_waiter = true;
2082            self.journal_waiters.fetch_add(1, Ordering::Relaxed);

2089    /// Capture what a sync needs, under the journal lock.
2090    ///
2091    /// Every published group is already complete in the segment, so the
2092    /// sync only needs to make its bytes durable. Under the journal lock,
2093    /// capture the newest visible LSN (everything at or below it has been
2094    /// fully written), the group sequence, and a handle to the current
2095    /// file; then release the lock so publishers run during the fsync.
2096    /// Syncing through the newest visible LSN, not just `target_lsn`, lets
2097    /// one fsync cover every caller that queued behind the previous one.
2098    fn capture_sync_range(&self, target_lsn: u64) -> io::Result<SyncCapture> {
2099        let mut journal = self.journal.lock();
2100        if journal.poisoned {
2101            return Err(io::Error::other(
2102                "journal handle is poisoned after an earlier failed commit",
2103            ));
2104        }
2105        let through_lsn = self.visible_lsn.load(Ordering::Acquire);
2106        if target_lsn > through_lsn {
2107            return Err(io::Error::other(
2108                "published sync target is not yet visible in the journal",
2109            ));
2110        }
2111        let file = match journal.file.try_clone() {
2112            Ok(file) => file,
2113            Err(error) => {
2114                self.poisoned.store(true, Ordering::Release);
2115                journal.poisoned = true;
2116                return Err(error);
2117            }
2118        };
2119        Ok(SyncCapture {
2120            file,
2121            through_lsn,
2122            sequence: journal.next_sequence.saturating_sub(1),
2123            path: journal.path.clone(),
2124            file_len: journal.file_len,
2125        })
2126    }

2128    /// Fsync the captured handle with no journal lock held, recording the
2129    /// fsync time in `timings`.

2132        capture: &SyncCapture,
2133        timings: &mut JournalSyncTimings,
2134    ) -> io::Result<()> {
2135        let fsync_started = std::time::Instant::now();

2139        let hooked: io::Result<()> = Ok(());

2270    /// Single publication path. Holds `publication` for the whole append so
2271    /// LSN assignment, the append and the visibility advance are one step.
2272    ///
2273    /// A transaction group (`transaction` set) is published before its writes
2274    /// are materialized into the packs. Until [`Self::transaction_materialized`]
2275    /// says otherwise, no pool may claim coverage at or past its first frame:
2276    /// the packs do not hold those frames yet, so a claim (and the reclaim it
2277    /// would license) could lose an acknowledged commit to a crash. It is
2278    /// registered here, under the journal lock, so no sync can promote a
2279    /// watermark past the group before the group is on the floor list.
2280    fn publish_groups(
2281        &self,
2282        staged: &[(Option<ShardType>, &[Mutation])],
2283        transaction: bool,
2284    ) -> io::Result<Option<CommitReceipt>> {
2285        let _publication = self.publication.lock();
2286        if self.poisoned.load(Ordering::Acquire) {
2287            return Err(io::Error::other(
2288                "journal is poisoned after a failed append",
2289            ));
2290        }
2291        let mut journal = self.journal.lock();
2292        if self.poisoned.load(Ordering::Acquire) {
2293            return Err(io::Error::other(
2294                "journal is poisoned after a failed append",
2295            ));
2296        }
2297        let group_mutations: Vec<(Option<ShardType>, Mutation)> = staged
2298            .iter()
2299            .flat_map(|(pool, mutations)| mutations.iter().cloned().map(|m| (*pool, m)))
2300            .collect();
2301        if group_mutations.is_empty() {
2302            return Ok(None);
2303        }
2304        let expected_first_lsn = journal.next_lsn;
2305        let expected_count = u64::try_from(group_mutations.len()).unwrap_or(u64::MAX);
2306        let sequence = self
2307            .sequence
2308            .as_ref()
2309            .map(|counter| counter.fetch_add(1, Ordering::Relaxed));
2310        let receipt = match journal.append_group_for_current_mode(&group_mutations, sequence) {
2311            Ok(receipt) => receipt,
2312            Err(error) => {
2313                if journal.poisoned {
2314                    self.poisoned.store(true, Ordering::Release);
2315                }
2316                return Err(self.explain_full_segment(error, &journal));
2317            }
2318        };
2319        if transaction {
2320            self.unmaterialized.lock().insert(receipt.first_lsn);
2321        }
2322        debug_assert_eq!(
2323            receipt.first_lsn, expected_first_lsn,
2324            "appended group must start at the journal tail"
2325        );
2326        debug_assert_eq!(
2327            receipt
2328                .last_lsn
2329                .saturating_sub(receipt.first_lsn)
2330                .saturating_add(1),
2331            expected_count,
2332            "appended group must cover every mutation"
2333        );
2334        // Queue this group's per-pool extents before it becomes visible, still
2335        // under the journal lock. A sync that captures the new visible LSN
2336        // then always finds the extents and promotes each pool's watermark;
2337        // queuing after the visibility advance would let a sync race past them.
2338        let extents = pool_extents(
2339            receipt.last_lsn,
2340            group_mutations.iter().map(|(pool, _)| *pool),
2341        );
2342        if !extents.is_empty() {
2343            self.pending_promotions
2344                .lock()
2345                .push((receipt.last_lsn, extents));
2346        }
2347        self.published_lsn
2348            .fetch_max(receipt.last_lsn, Ordering::Release);
2349        self.visible_lsn
2350            .fetch_max(receipt.last_lsn, Ordering::Release);
2351        drop(journal);
2352        // Wake the committer only when a burst is at or above its early-flush
2353        // bound; this is the hot path, so ordinary publishes must not pay a
2354        // futex wake. `committer_wake_threshold` is zero when no committer is
2355        // running. Only the first publish at or above the bound wakes in a
2356        // commit epoch; any commit advances the epoch and re-arms the next
2357        // crossing.
2358        let threshold = self.committer_wake_threshold.load(Ordering::Relaxed);
2359        if threshold > 0 && self.pending_count() >= threshold && self.claim_threshold_wake() {
2360            #[cfg(test)]
2361            self.threshold_wakes.fetch_add(1, Ordering::Relaxed);
2362            self.wake_durable_waiters();
2363        }
2364        Ok(Some(receipt))
2365    }

2622    /// Block until the group covering `token` is durable.
2623    ///
2624    /// With a background committer running this waits for it to flush; the
2625    /// bounded poll interval lets the waiter notice a committer that stopped or
2626    /// a poisoned journal. A committer that failed terminally is reported
2627    /// rather than waited on. Without a committer, this is exactly
2628    /// [`Self::sync_through`] on the token's LSN — the historical blocking
2629    /// behavior, so a caller that never opts into background commits is
2630    /// unaffected.
2631    ///
2632    /// # Errors
2633    /// Returns an error if the target was never published, the journal is
2634    /// poisoned, the committer failed, or the committing sync fails.
2635    pub fn wait_durable(&self, token: DurabilityToken) -> io::Result<Option<CommitReceipt>> {
2636        let target = token.lsn;
2637        if target == 0 || target <= self.committed_lsn.load(Ordering::Acquire) {
2638            self.durable_waits_already_durable
2639                .fetch_add(1, Ordering::Relaxed);
2640            return Ok(None);
2641        }
2642        let started = std::time::Instant::now();
2643        let result = self.wait_durable_blocking(target);
2644        self.durable_wait.observe(started.elapsed());
2645        result
2646    }

2648    fn wait_durable_blocking(&self, target: u64) -> io::Result<Option<CommitReceipt>> {
2649        loop {
2650            if target <= self.committed_lsn.load(Ordering::Acquire) {
2651                return Ok(None);
2652            }
2653            // Recheck every predicate under `durable_lock`, the same lock the
2654            // notifiers take, so a state change cannot slip between check and
2655            // park. The poll interval remains a backstop, not a requirement.
2656            let mut guard = self.durable_lock.lock();
2657            if target <= self.committed_lsn.load(Ordering::Acquire) {
2658                return Ok(None);
2659            }
2660            if self.poisoned.load(Ordering::Acquire) {
2661                return Err(io::Error::other(
2662                    "journal is poisoned after a failed commit",
2663                ));
2664            }
2665            // A stopped committer falls through to a direct commit, preserving
2666            // the historical no-committer behavior and error text, but a
2667            // committer that failed must not be silently papered over.
2668            if let Some(failure) = self.background_failure_detail() {
2669                return Err(failure.into_io());
2670            }
2671            if !self.has_background_committer() {
2672                drop(guard);
2673                return self.sync_through(target);
2674            }
2675            // Only with a live committer is an unpublished target validated
2676            // here; the no-committer path above reaches `sync_through` first.
2677            if target > self.published_lsn.load(Ordering::Acquire) {
2678                return Err(io::Error::new(
2679                    io::ErrorKind::InvalidInput,
2680                    "sync target has not been published",
2681                ));
2682            }
2683            self.durable_cv.wait_for(&mut guard, DURABILITY_WAIT_POLL);
2684        }
2685    }

2688    ///
2689    /// Used by the background committer's timer and by
2690    /// [`Self::stop_background_committer`]. A no-op returning `None` when
2691    /// nothing is pending (so an idle tick neither fsyncs nor inflates the
2692    /// coalescing counter). This is the journal-level durability boundary; a
2693    /// caller that also needs its packfile checkpoint rewritten should use the

2700        if target == 0 || target <= self.committed_lsn.load(Ordering::Acquire) {
2701            return Ok(None);
2702        }
2703        let receipt = self.sync_through(target)?;
2704        if receipt.is_some() {

2715    /// Start a background thread that fsyncs the pending WAL group on a bounded
2716    /// interval, coalescing every mutation published in the window into one
2717    /// group. Flushes early once `max_pending` records are outstanding.
2718    /// Idempotent: a call while one is running (or stopping) is a no-op.
2719    ///
2720    /// The thread holds only a [`Weak`] reference between iterations, so
2721    /// dropping the last `Arc<JournalCoordinator>` lets it exit within one
2722    /// interval without an `Arc` cycle. That is a bounded delay, not immediate
2723    /// teardown — call [`Self::stop_background_committer`] for deterministic
2724    /// cleanup and a final flush.
2725    ///
2726    /// # Errors
2727    /// Returns [`io::ErrorKind::InvalidInput`] if `interval` is zero or
2728    /// `max_pending` is zero (both would busy-loop or flush continuously), or
2729    /// another error if the committer thread cannot be spawned.
2730    pub fn start_background_committer(
2731        self: &Arc<Self>,
2732        config: GroupCommitConfig,
2733    ) -> io::Result<()> {
2734        if config.interval.is_zero() {
2735            return Err(io::Error::new(
2736                io::ErrorKind::InvalidInput,
2737                "background committer interval must be greater than zero",
2738            ));
2739        }
2740        if config.max_pending == 0 {
2741            return Err(io::Error::new(
2742                io::ErrorKind::InvalidInput,
2743                "background committer max_pending must be greater than zero",
2744            ));
2745        }
2746        let mut slot = self.background.lock();
2747        // Running or Stopping: never spawn a second committer.
2748        if !matches!(*slot, BackgroundState::Stopped) {
2749            return Ok(());
2750        }
2751        *self.background_failure.lock() = None;
2752        // A failed run can leave the notification epoch equal to the current
2753        // commit epoch, which would suppress the first threshold wake after a
2754        // restart. Reset it so the first crossing always wakes.
2755        self.threshold_notified_epoch
2756            .store(u64::MAX, Ordering::Release);
2757        let stop = Arc::new(AtomicBool::new(false));
2758        let weak = Arc::downgrade(self);
2759        let stop_clone = Arc::clone(&stop);
2760        let handle = std::thread::Builder::new()
2761            .name("mtxdb-journal-committer".to_owned())
2762            .spawn(move || {
2763                // Catch a panic so the terminal failure is recorded and waiters
2764                // are released instead of polling forever on a dead worker.
2765                let outcome = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
2766                    Self::background_committer_loop(&weak, config, &stop_clone)
2767                }));
2768                if let Some(coordinator) = weak.upgrade() {
2769                    let failure = match outcome {
2770                        Ok(Ok(())) => None,
2771                        Ok(Err(failure)) => Some(failure),
2772                        Err(_) => Some(BackgroundFailure {
2773                            kind: io::ErrorKind::Other,
2774                            message: "background committer panicked".to_owned(),
2775                        }),
2776                    };
2777                    coordinator.finish_background(failure);
2778                }
2779            })
2780            .map_err(|error| {
2781                io::Error::other(format!("failed to spawn journal committer: {error}"))
2782            })?;
2783        self.committer_wake_threshold
2784            .store(config.max_pending, Ordering::Release);
2785        *slot = BackgroundState::Running(BackgroundCommitter {
2786            stop,
2787            handle: Some(handle),
2788        });
2789        Ok(())
2790    }

2792    /// Signal the background committer to stop, join it, then flush any final
2793    /// pending group so a quiet stream's last write is durable before this
2794    /// returns. Safe to call when no committer is running (it still flushes).
2795    ///
2796    /// The committer slot is held in a `Stopping` state across the join, so a
2797    /// concurrent [`Self::start_background_committer`] cannot spawn a second
2798    /// worker mid-teardown.
2799    ///
2800    /// # Errors
2801    /// Propagates a failure from the final [`Self::flush_durable`], and surfaces
2802    /// a terminal committer failure recorded by the worker.
2803    pub fn stop_background_committer(&self) -> io::Result<()> {
2804        let handle = {
2805            let mut slot = self.background.lock();
2806            match std::mem::replace(&mut *slot, BackgroundState::Stopping) {
2807                BackgroundState::Running(mut committer) => {
2808                    committer.stop.store(true, Ordering::Release);
2809                    committer.handle.take()
2810                }
2811                other => {
2812                    *slot = other;
2813                    None
2814                }
2815            }
2816        };
2817        self.committer_wake_threshold.store(0, Ordering::Release);
2818        self.wake_durable_waiters();
2819        if let Some(handle) = handle {
2820            // A panic in the committer is caught and recorded by the worker's
2821            // wrapper; joining still succeeds and the failure is surfaced below.
2822            let _ = handle.join();
2823        }
2824        // Transition `Stopping -> Stopped` and take the recorded failure under
2825        // one `background` lock. Otherwise a `start_background_committer` that
2826        // sees `Stopped` can clear `background_failure` in the window between
2827        // the transition and the take, silently losing the worker's failure.
2828        let failure = {
2829            let mut slot = self.background.lock();
2830            if matches!(*slot, BackgroundState::Stopping) {
2831                *slot = BackgroundState::Stopped;
2832            }
2833            self.background_failure.lock().take()
2834        };
2835        self.flush_durable()?;
2836        if let Some(failure) = failure {
2837            return Err(failure.into_io());
2838        }
2839        Ok(())
2840    }

2842    /// Mark the committer slot stopped, publish any terminal failure, and wake
2843    /// waiters. Called by the worker itself as it exits.
2844    fn finish_background(&self, failure: Option<BackgroundFailure>) {
2845        // Transition, publish the failure, and clear the wake threshold under a
2846        // single `background` lock. A concurrent `start_background_committer`
2847        // (which holds that lock to clear `background_failure` and set its own
2848        // threshold) therefore either sees this worker still `Running` and does
2849        // nothing, or runs after the failure is already recorded; it cannot
2850        // erase the failure or have its threshold zeroed by this exiting worker.
2851        {
2852            let mut slot = self.background.lock();
2853            // Leave `Stopping` alone: `stop_background_committer` owns that
2854            // transition and takes the failure after joining this thread.
2855            if matches!(*slot, BackgroundState::Running(_)) {
2856                *slot = BackgroundState::Stopped;
2857                self.committer_wake_threshold.store(0, Ordering::Release);
2858            }
2859            if let Some(failure) = failure {
2860                *self.background_failure.lock() = Some(failure);
2861            }
2862        }
2863        self.wake_durable_waiters();
2864    }

2866    fn background_committer_loop(
2867        weak: &Weak<Self>,
2868        config: GroupCommitConfig,
2869        stop: &Arc<AtomicBool>,
2870    ) -> Result<(), BackgroundFailure> {
2871        let interval = config.interval;
2872        loop {
2873            let Some(coordinator) = weak.upgrade() else {
2874                return Ok(());
2875            };
2876            let now = std::time::Instant::now();
2877            let deadline = now.checked_add(interval).unwrap_or(now);
2878            {
2879                let mut guard = coordinator.durable_lock.lock();
2880                loop {
2881                    if stop.load(Ordering::Acquire) || coordinator.poisoned.load(Ordering::Acquire)
2882                    {
2883                        break;
2884                    }
2885                    let now = std::time::Instant::now();
2886                    if now >= deadline {
2887                        break;
2888                    }
2889                    if coordinator.pending_count() >= config.max_pending {
2890                        break;
2891                    }
2892                    let timeout = deadline.saturating_duration_since(now);
2893                    #[cfg(test)]
2894                    coordinator.committer_parks.fetch_add(1, Ordering::Release);
2895                    coordinator.durable_cv.wait_for(&mut guard, timeout);
2896                }
2897            }
2898            if stop.load(Ordering::Acquire) || coordinator.poisoned.load(Ordering::Acquire) {
2899                return Ok(());
2900            }
2901            // The final flush on stop is done by `stop_background_committer`
2902            // after join, so a stop that lands here does not double-fsync.
2903            if let Err(error) = coordinator.flush_durable() {
2904                if coordinator.poisoned.load(Ordering::Acquire) {
2905                    // Poison is already reported to waiters via the poison bit.
2906                    return Ok(());
2907                }
2908                return Err(BackgroundFailure::from_io(&error));
2909            }
2910        }
2911    }

3037    /// Read up to `max_len` bytes of the segment ending at `end_offset` from an
3038    /// already-open `file` into `window`, never reaching back into the file
3039    /// header. Returns `false` when the file is shorter than `end_offset` or
3040    /// has no group bytes before it, leaving `window` unspecified.
3041    ///
3042    /// A read-only overlay keeps this window from the end of the last group it
3043    /// consumed and compares it on later refreshes: file length alone cannot
3044    /// show that the bytes already consumed are still the same ones. The
3045    /// window is compared byte for byte instead of relying on the group
3046    /// trailer, because the trailer is not a content fingerprint: the group
3047    /// checksum covers `header || records` and every record already ends with
3048    /// its own CRC32, so for groups of the same shape it comes out identical
3049    /// whatever the payload. The window includes the last record's own CRC,
3050    /// which does depend on the record's full content.
3051    ///
3052    /// Reads through the caller's descriptor with a positioned read, so the
3053    /// refresh path pays one syscall and reuses `window`'s allocation instead
3054    /// of opening the file each time. The caller must notice a replaced file
3055    /// itself (its descriptor keeps the old inode). Never repairs or creates
3056    /// the file.
3057    ///
3058    /// # Errors
3059    /// Returns `io::Error` if the read fails.
3060    pub(crate) fn read_tail_ending_at(
3061        file: &File,
3062        end_offset: u64,
3063        max_len: u64,
3064        window: &mut Vec<u8>,
3065    ) -> io::Result<bool> {
3066        let header_len = u64::try_from(FILE_HEADER_LEN).unwrap_or(u64::MAX);
3067        if end_offset <= header_len {
3068            return Ok(false);
3069        }
3070        let start = end_offset.saturating_sub(max_len).max(header_len);
3071        let window_len = usize::try_from(end_offset.saturating_sub(start))
3072            .map_err(|_| invalid_data("tail window exceeds the address space"))?;
3073        window.clear();
3074        window.resize(window_len, 0);
3075        #[cfg(unix)]
3076        let read = std::os::unix::fs::FileExt::read_exact_at(file, window, start);
3077        #[cfg(not(unix))]
3078        let read = {
3079            let mut handle = file;
3080            handle
3081                .seek(SeekFrom::Start(start))
3082                .and_then(|_| handle.read_exact(window))
3083        };
3084        match read {
3085            Ok(()) => Ok(true),
3086            // The file is shorter than the window's end: the consumed prefix is gone.
3087            Err(error) if error.kind() == io::ErrorKind::UnexpectedEof => Ok(false),
3088            Err(error) => Err(error),
3089        }
3090    }

3141    /// Open a journal whose on-disk version must be `version`, creating it with

3144    ///

3146    /// Same as [`Self::open`], plus `InvalidData` if an existing complete
3147    /// segment was written by a different version.
3148    fn open_versioned(
3149        path: impl AsRef<Path>,
3150        version: JournalVersion,
3151        base_lsn: u64,
3152    ) -> io::Result<(Self, Scan)> {
3153        let path = path.as_ref().to_path_buf();
3154        if let Some(parent) = path.parent() {
3155            if !parent.as_os_str().is_empty() {
3156                fs::create_dir_all(parent)?;
3157            }
3158        }
3159        let mut file = OpenOptions::new()
3160            .create(true)
3161            .read(true)
3162            .write(true)
3163            .truncate(false)
3164            .open(&path)?;
3165
3166        let len = file.metadata()?.len();
3167        if len < FILE_HEADER_LEN as u64 {
3168            // Empty (fresh) or a crash while writing the header: no committed
3169            // group can exist yet, so reset and re-initialise rather than
3170            // failing open forever on a partial header.
3171            file.set_len(0)?;
3172            file.seek(SeekFrom::Start(0))?;
3173            write_file_header(&mut file, version, 1, base_lsn)?;
3174            file.sync_all()?;
3175            sync_parent_dir(&path)?;
3176        }
3177
3178        let bytes = fs::read(&path)?;
3179        let (found_version, base_sequence, base_lsn) = validate_file_header(&bytes)?;
3180        if found_version != version {
3181            return Err(invalid_data(
3182                "journal segment version does not match open mode",
3183            ));
3184        }
3185        if bytes.len() as u64 > MAX_SEGMENT_LEN {
3186            return Err(invalid_data("journal segment exceeds the 256 MiB limit"));
3187        }
3188        let file_len = u64::try_from(bytes.len()).unwrap_or(u64::MAX);
3189        let scan = scan_bytes(&bytes, base_sequence, base_lsn, version, || {
3190            read_durable_len_from_header(&bytes, version, file_len)
3191        })?;
3192        if scan.truncated_tail {
3193            file.set_len(scan.valid_len)?;
3194        }
3195        file.seek(SeekFrom::Start(scan.valid_len))?;
3196        if scan.truncated_tail || !scan.groups.is_empty() {
3197            // What recovery keeps may exist only in the page cache (a process
3198            // crash leaves it there), yet the coordinator will report it
3199            // committed. Make it durable now, and record that in the hint, so a
3200            // later sync never claims bytes that were not fsynced. The same
3201            // fsync makes a truncation durable, including one that dropped
3202            // every group, so a dropped tail cannot reappear.
3203            file.sync_all()?;
3204        }
3205        let next_sequence = scan
3206            .groups
3207            .last()
3208            .map_or(base_sequence, |group| group.sequence.saturating_add(1));
3209        let next_lsn = scan
3210            .groups
3211            .last()
3212            .map_or(base_lsn, |group| group.last_lsn.saturating_add(1));
3213
3214        let mark_generation = if version.is_pool_tagged() {
3215            newest_mark(&bytes).map_or(0, |slot| slot.generation)
3216        } else {
3217            0
3218        };
3219        let mut journal = Self {
3220            path,
3221            file,
3222            file_len: scan.valid_len,
3223            base_sequence,
3224            base_lsn,
3225            mark_generation,
3226            version,
3227            next_sequence,
3228            next_lsn,
3229            poisoned: false,
3230            segment_cap: MAX_SEGMENT_LEN,
3231            groups: marks_from_scan(&scan),
3232        };
3233        if !scan.groups.is_empty() {
3234            // Recovery fsynced what it kept above, so record that in the mark.
3235            journal.write_mark(scan.valid_len);
3236        }
3237        Ok((journal, scan))
3238    }

3316    /// Shared implementation for a group of mutations, each paired with its
3317    /// optional pool tag. The tag must match the segment's dialect (see
3318    /// [`JournalVersion::is_pool_tagged`]).
3319    fn append_group_inner(
3320        &mut self,
3321        mutations: &[(Option<ShardType>, Mutation)],
3322        sequence: Option<u64>,
3323    ) -> io::Result<CommitReceipt> {
3324        if self.poisoned {
3325            return Err(io::Error::other(
3326                "journal handle is poisoned after an earlier failed commit",
3327            ));
3328        }
3329        if mutations.is_empty() {
3330            return Err(io::Error::new(
3331                io::ErrorKind::InvalidInput,
3332                "cannot commit an empty journal group",
3333            ));
3334        }
3335
3336        let record_count = u32::try_from(mutations.len()).map_err(|_| {
3337            io::Error::new(io::ErrorKind::InvalidInput, "too many journal mutations")
3338        })?;
3339        let first_lsn = self.next_lsn;
3340        let last_lsn = first_lsn
3341            .checked_add(u64::from(record_count).saturating_sub(1))
3342            .ok_or_else(|| io::Error::other("journal LSN exhausted"))?;
3343        let sequence = sequence.unwrap_or(self.next_sequence);
3344        if sequence < self.next_sequence {
3345            return Err(io::Error::new(
3346                io::ErrorKind::InvalidInput,
3347                "journal group sequence would move backwards",
3348            ));
3349        }
3350
3351        let mut payload = Vec::new();
3352        for (index, (pool, mutation)) in mutations.iter().enumerate() {
3353            let lsn = first_lsn
3354                .checked_add(u64::try_from(index).map_err(|_| {
3355                    io::Error::new(io::ErrorKind::InvalidInput, "mutation index exceeds u64")
3356                })?)
3357                .ok_or_else(|| io::Error::other("journal LSN exhausted"))?;
3358            encode_mutation(lsn, *pool, self.version, mutation, &mut payload)?;
3359            if u64::try_from(payload.len()).unwrap_or(u64::MAX) > MAX_GROUP_LEN {
3360                return Err(io::Error::new(
3361                    io::ErrorKind::InvalidInput,
3362                    "journal group exceeds the 256 MiB format limit",
3363                ));
3364            }
3365        }
3366        let payload_len = u64::try_from(payload.len())
3367            .map_err(|_| io::Error::new(io::ErrorKind::InvalidInput, "group too large"))?;
3368        if payload_len > MAX_GROUP_LEN {
3369            return Err(io::Error::new(
3370                io::ErrorKind::InvalidInput,
3371                "journal group exceeds the 256 MiB segment limit",
3372            ));
3373        }
3374
3375        let header = encode_group_header(sequence, first_lsn, last_lsn, payload_len, record_count);
3376        let mut group_checksum = Hasher::new();
3377        group_checksum.update(&header);
3378        group_checksum.update(&payload);
3379        let trailer = encode_group_trailer(sequence, group_checksum.finalize());
3380
3381        let group_size = u64::try_from(
3382            header
3383                .len()
3384                .saturating_add(payload.len())
3385                .saturating_add(trailer.len()),
3386        )
3387        .map_err(|_| io::Error::new(io::ErrorKind::InvalidInput, "group too large"))?;
3388        if self.file_len.saturating_add(group_size) > self.segment_cap {
3389            return Err(io::Error::new(
3390                io::ErrorKind::WouldBlock,
3391                "journal segment is full; drain/rotate before accepting more writes",
3392            ));
3393        }
3394        let next_sequence = sequence
3395            .checked_add(1)
3396            .ok_or_else(|| io::Error::other("journal group sequence exhausted"))?;
3397        let next_lsn = last_lsn
3398            .checked_add(1)
3399            .ok_or_else(|| io::Error::other("journal LSN exhausted"))?;
3400
3401        // One write per group: every autocommit publish lands here, so the
3402        // group goes out as a single contiguous buffer instead of three
3403        // syscalls. A torn write still leaves an incomplete tail that recovery
3404        // truncates, exactly as before.
3405        let mut frame = Vec::with_capacity(
3406            header
3407                .len()
3408                .saturating_add(payload.len())
3409                .saturating_add(trailer.len()),
3410        );
3411        frame.extend_from_slice(&header);
3412        frame.extend_from_slice(&payload);
3413        frame.extend_from_slice(&trailer);
3414        let write_result = self.file.write_all(&frame);
3415        if let Err(error) = write_result {
3416            self.poisoned = true;
3417            return Err(error);
3418        }
3419        self.file_len = self.file_len.saturating_add(group_size);
3420        self.note_appended_group(sequence, first_lsn, last_lsn, mutations);
3421
3422        self.next_sequence = next_sequence;
3423        self.next_lsn = next_lsn;
3424
3425        Ok(CommitReceipt {
3426            sequence,
3427            first_lsn,
3428            last_lsn,
3429            bytes_written: group_size,
3430        })
3431    }

3433    /// Fsync the bytes [`Self::append_group`] wrote, making the group durable.
3434    ///
3435    /// Separate from [`Self::append_group`] so a caller can publish a
3436    /// visibility boundary after the group is complete but before the fsync.
3437    /// On failure this handle is poisoned: the appended group's on-disk state
3438    /// is uncertain, so the caller must reopen and rescan.
3439    ///
3440    /// # Errors
3441    /// Returns `io::Error` if the handle is poisoned or the sync fails.
3442    pub fn make_durable(&mut self) -> io::Result<()> {
3443        if self.poisoned {
3444            return Err(io::Error::other(
3445                "journal handle is poisoned after an earlier failed commit",
3446            ));
3447        }
3448        if let Err(error) = self.file.sync_all() {
3449            self.poisoned = true;
3450            return Err(error);
3451        }
3452        // The fsync returned, so everything appended so far is durable: record
3453        // that in the durability mark, after the fact.
3454        self.write_mark(self.file_len);
3455        Ok(())
3456    }

3458    /// Record, in the slot not holding the newest mark, that `durable_len`
3459    /// bytes are durable. Call it only after an fsync
3460    /// that covered them, so the mark can only under-claim.
3461    ///
3462    /// Only pool-tagged segments keep a mark. The slot is written with a
3463    /// positioned write, so it never moves the append cursor. A failed write
3464    /// is ignored: the previous mark stays in the other slot, or recovery sees
3465    /// no mark and truncates conservatively.
3466    fn write_mark(&mut self, durable_len: u64) {
3467        if !self.version.is_pool_tagged() {
3468            return;
3469        }
3470        let generation = self.mark_generation.saturating_add(1);
3471        let bytes = encode_mark_slot(MarkSlot {
3472            generation,
3473            durable_len,
3474        });
3475        let slot = usize::try_from(generation % 2).unwrap_or(0);
3476        let offset = u64::try_from(MARK_SLOT_OFFSETS[slot]).unwrap_or(u64::MAX);
3477        #[cfg(unix)]
3478        let written = std::os::unix::fs::FileExt::write_all_at(&self.file, &bytes, offset);
3479        // Elsewhere a positioned write moves the shared cursor, so restore it
3480        // to the end of the segment, where appends continue. Appends take the
3481        // journal lock too, so nothing writes in between.
3482        #[cfg(not(unix))]
3483        let written = self
3484            .file
3485            .seek(SeekFrom::Start(offset))
3486            .and_then(|_| self.file.write_all(&bytes))
3487            .and_then(|()| self.file.seek(SeekFrom::Start(self.file_len)).map(drop));
3488        if written.is_ok() {
3489            self.mark_generation = generation;
3490        }
3491    }

3661    /// Reclaim using the group directory: pick the cut from it and read only
3662    /// the retained suffix from disk. The dropped prefix is never read.
3663    fn reclaim_through_directory(&mut self, covered_lsn: u64) -> io::Result<Reclaim> {
3664        let cut = self
3665            .groups
3666            .partition_point(|group| group.last_lsn <= covered_lsn);
3667        let Some(last_dropped) = cut.checked_sub(1).and_then(|index| self.groups.get(index)) else {
3668            return Ok(Reclaim {
3669                retained_groups: u64::try_from(self.groups.len()).unwrap_or(u64::MAX),
3670                reclaimed_bytes: 0,
3671                ..Default::default()
3672            });
3673        };
3674        let copy_started = std::time::Instant::now();
3675        let header_len = FILE_HEADER_LEN as u64;
3676        let cut_offset = last_dropped.end_offset;
3677        let retained = self.groups.get(cut..).unwrap_or_default();
3678        let (new_base_sequence, new_base_lsn) = retained
3679            .first()
3680            .map_or((self.next_sequence, self.next_lsn), |group| {
3681                (group.sequence, group.first_lsn)
3682            });
3683        let mut rebuilt = file_header_bytes(self.version, new_base_sequence, new_base_lsn);
3684        if let Some(first) = retained.first() {
3685            // Read the suffix straight into the buffer that becomes the new
3686            // segment, after its header, so it is allocated once.
3687            let header_len = rebuilt.len();
3688            read_range_into(
3689                &self.path,
3690                cut_offset,
3691                self.file_len.saturating_sub(cut_offset),
3692                &mut rebuilt,
3693            )?;
3694            // The suffix is copied byte for byte, so check it the way the full
3695            // scan would have: it must hold exactly the groups the directory
3696            // says, each valid. Anything else means the directory is stale, and
3697            // the scan path decides instead.
3698            let Ok(checked) = scan_groups_from(
3699                rebuilt.get(header_len..).unwrap_or_default(),
3700                cut_offset,
3701                first.sequence,
3702                first.first_lsn,
3703                new_base_lsn,
3704                self.version,
3705                || u64::MAX,
3706            ) else {
3707                return self.reclaim_through_scan(covered_lsn);
3708            };
3709            if checked.groups.len() != retained.len()
3710                || checked.valid_len != self.file_len
3711                || checked.truncated_tail
3712            {
3713                return self.reclaim_through_scan(covered_lsn);
3714            }
3715        }
3716        let reclaimed_bytes = self
3717            .file_len
3718            .saturating_sub(u64::try_from(rebuilt.len()).unwrap_or(u64::MAX));
3719        let retained_groups = u64::try_from(retained.len()).unwrap_or(u64::MAX);
3720        let shift = cut_offset.saturating_sub(header_len);
3721        let survivors = retained
3722            .iter()
3723            .map(|group| GroupMark {
3724                end_offset: group.end_offset.saturating_sub(shift),
3725                ..*group
3726            })
3727            .collect::<Vec<_>>();
3728        let copy = copy_started.elapsed();
3729        let rebuilt_len = rebuilt.len();
3730        let fsync_started = std::time::Instant::now();
3731        self.install_rebuilt(rebuilt, new_base_sequence, new_base_lsn)?;
3732        let fsync = fsync_started.elapsed();
3733        self.groups = survivors;
3734        Ok(Reclaim {
3735            retained_groups,
3736            reclaimed_bytes,
3737            retained_bytes: u64::try_from(rebuilt_len.saturating_sub(FILE_HEADER_LEN))
3738                .unwrap_or(u64::MAX),
3739            copy,
3740            fsync,
3741            ..Default::default()
3742        })
3743    }

3745    /// Reclaim by reading and decoding the whole segment. Used when the group
3746    /// directory cannot be trusted, and it rebuilds the directory from what it
3747    /// finds.
3748    fn reclaim_through_scan(&mut self, covered_lsn: u64) -> io::Result<Reclaim> {
3749        let scan_started = std::time::Instant::now();
3750        let bytes = fs::read(&self.path)?;
3751        let (version, base_sequence, base_lsn) = validate_file_header(&bytes)?;
3752        // Reclaim rewrites the segment from its own live file, so any invalid
3753        // group there is corruption, not a crash tail: treat every byte as
3754        // durable and let it fail.
3755        let scan = scan_bytes(&bytes, base_sequence, base_lsn, version, || u64::MAX)?;
3756        let boundary = scan_started.elapsed();
3757        let retained = scan
3758            .groups
3759            .iter()
3760            .filter(|group| group.last_lsn > covered_lsn)
3761            .collect::<Vec<_>>();

3764            return Ok(Reclaim {

3768                ..Default::default()

3770        }

3772        let copy_started = std::time::Instant::now();

3774            .first()
3775            .map_or((self.next_sequence, self.next_lsn), |group| {
3776                (group.sequence, group.first_lsn)
3777            });

3779        let mut survivors = Vec::with_capacity(retained.len());

3791            });
3792        }
3793        let copy = copy_started.elapsed();

3795        let rebuilt_len = rebuilt.len();
3796        let reclaimed_bytes =
3797            u64::try_from(bytes.len().saturating_sub(rebuilt.len())).unwrap_or(u64::MAX);

3803            retained_groups,
3804            reclaimed_bytes,
3805            retained_bytes: u64::try_from(rebuilt_len.saturating_sub(FILE_HEADER_LEN))

3807            boundary,
3808            copy,
3809            fsync,
3810            ..Default::default()
3811        })

3814    /// Durably replace the segment with `rebuilt` (a new header plus the
3815    /// retained groups) and point this handle at it.
3816    fn install_rebuilt(
3817        &mut self,
3818        mut rebuilt: Vec<u8>,
3819        new_base_sequence: u64,
3820        new_base_lsn: u64,
3821    ) -> io::Result<()> {
3822        // The rebuilt file is fsynced before it replaces the segment, so all of
3823        // it is durable. Put its mark inside it, first generation, so the mark
3824        // is durable with the data and the old segment's marks, which described
3825        // offsets that no longer exist, are simply gone.
3826        let rebuilt_mark = self.version.is_pool_tagged().then(|| MarkSlot {
3827            generation: 1,
3828            durable_len: u64::try_from(rebuilt.len()).unwrap_or(u64::MAX),
3829        });
3830        if let Some(mark) = rebuilt_mark {
3831            let offset = MARK_SLOT_OFFSETS[1];
3832            let end = offset.saturating_add(MARK_SLOT_LEN);
3833            rebuilt[offset..end].copy_from_slice(&encode_mark_slot(mark));
3834        }
3835
3836        let temp_path = self.path.with_extension("rotate");
3837        let write_result = (|| -> io::Result<()> {
3838            let mut temp = OpenOptions::new()
3839                .create(true)
3840                .write(true)
3841                .truncate(true)
3842                .open(&temp_path)?;
3843            temp.write_all(&rebuilt)?;
3844            temp.sync_all()?;
3845            drop(temp);
3846            fs::rename(&temp_path, &self.path)
3847        })();
3848        if let Err(error) = write_result {
3849            let _ = fs::remove_file(&temp_path);
3850            return Err(error);
3851        }
3852        // The rename has replaced the path, so from here onward any failure
3853        // must poison the handle: continuing to append through the old file
3854        // descriptor would write an unlinked inode and could acknowledge data
3855        // that is absent from the journal path.
3856        let replacement = match OpenOptions::new().read(true).write(true).open(&self.path) {
3857            Ok(file) => file,
3858            Err(error) => {
3859                self.poisoned = true;
3860                return Err(error);
3861            }
3862        };
3863        self.file = replacement;
3864        self.file_len = u64::try_from(rebuilt.len()).unwrap_or(u64::MAX);
3865        self.base_sequence = new_base_sequence;
3866        self.base_lsn = new_base_lsn;
3867        self.mark_generation = u64::from(rebuilt_mark.is_some());
3868        if let Err(error) = sync_parent_dir(&self.path) {
3869            self.poisoned = true;
3870            return Err(error);
3871        }
3872        if let Err(error) = self.file.seek(SeekFrom::End(0)) {
3873            self.poisoned = true;
3874            return Err(error);
3875        }
3876        Ok(())
3877    }
3878}

3880/// Append `len` bytes of the file at `path`, starting at `start`, to `out`.

3882    let mut file = File::open(path)?;
3883    file.seek(SeekFrom::Start(start))?;
3884    let len = usize::try_from(len).map_err(|_| invalid_data("range too large"))?;
3885    let filled = out.len();
3886    out.resize(filled.saturating_add(len), 0);
3887    if let Err(error) = file.read_exact(out.get_mut(filled..).unwrap_or_default()) {
3888        out.truncate(filled);
3889        return Err(error);
3890    }
3891    Ok(())
3892}

3915/// Validate the file header, returning the base `(sequence, lsn)` this segment
3916/// starts numbering at. A rotated segment records the first surviving group's
3917/// numbers here so scanning never has to assume the sequence begins at 1.
3918fn validate_file_header(bytes: &[u8]) -> io::Result<(JournalVersion, u64, u64)> {
3919    let Some(header) = bytes.get(..FILE_HEADER_LEN) else {
3920        return Err(invalid_data("truncated journal file header"));
3921    };
3922    if &header[FH_MAGIC] != FILE_MAGIC {
3923        return Err(invalid_data("invalid journal magic"));
3924    }
3925    let version = JournalVersion::from_u32(u32::from_le_bytes(
3926        header[FH_VERSION].try_into().expect("fixed slice"),
3927    ))?;
3928    let mut crc = Hasher::new();
3929    crc.update(&header[FH_BASE_CRC_COVERED]);
3930    if u32::from_le_bytes(header[FH_BASE_CRC].try_into().expect("fixed slice")) != crc.finalize() {
3931        return Err(invalid_data("journal header checksum mismatch"));
3932    }
3933    Ok((
3934        version,
3935        u64::from_le_bytes(header[FH_BASE_SEQUENCE].try_into().expect("fixed slice")),
3936        u64::from_le_bytes(header[FH_BASE_LSN].try_into().expect("fixed slice")),
3937    ))
3938}

3940/// One durability-mark slot: which generation it is, and what it claims.
3941#[derive(Clone, Copy)]

3944    durable_len: u64,

3947fn encode_mark_slot(slot: MarkSlot) -> [u8; MARK_SLOT_LEN] {
3948    let mut bytes = [0_u8; MARK_SLOT_LEN];
3949    bytes[..8].copy_from_slice(&slot.generation.to_le_bytes());
3950    bytes[8..16].copy_from_slice(&slot.durable_len.to_le_bytes());
3951    let mut crc = Hasher::new();
3952    crc.update(&bytes[..16]);
3953    bytes[16..20].copy_from_slice(&crc.finalize().to_le_bytes());
3954    bytes
3955}

3957/// Decode one slot: `None` if it is unwritten (all zero) or fails its checksum.
3958fn decode_mark_slot(bytes: &[u8]) -> Option<MarkSlot> {
3959    let bytes = bytes.get(..MARK_SLOT_LEN)?;
3960    let mut crc = Hasher::new();
3961    crc.update(&bytes[..16]);
3962    if u32::from_le_bytes(bytes[16..20].try_into().ok()?) != crc.finalize() {
3963        return None;
3964    }
3965    let slot = MarkSlot {
3966        generation: u64::from_le_bytes(bytes[..8].try_into().ok()?),
3967        durable_len: u64::from_le_bytes(bytes[8..16].try_into().ok()?),
3968    };
3969    // Generation zero is never written, so a checksum-valid slot claiming it
3970    // is not a mark.
3971    (slot.generation != 0).then_some(slot)
3972}

3974/// The newest valid mark in a segment's header region, or `None` when neither
3975/// slot holds one: unwritten, torn, or corrupt. `None` means "durability
3976/// unknown", which only ever widens what recovery is willing to truncate.
3977fn newest_mark(bytes: &[u8]) -> Option<MarkSlot> {
3978    MARK_SLOT_OFFSETS
3979        .iter()
3980        .filter_map(|&offset| bytes.get(offset..).and_then(decode_mark_slot))
3981        .max_by_key(|slot| slot.generation)
3982}

3984/// How many leading bytes of the segment are known durable. Per-pool segments
3985/// keep no mark and count every byte durable, so an invalid group fails closed.
3986/// A missing or unreadable mark, or one claiming more than the file holds,
3987/// yields `0`.
3988fn read_durable_len_from_header(bytes: &[u8], version: JournalVersion, file_len: u64) -> u64 {
3989    if !version.is_pool_tagged() {
3990        return u64::MAX;
3991    }
3992    newest_mark(bytes)
3993        .map(|slot| slot.durable_len)
3994        .filter(|&durable_len| durable_len <= file_len)
3995        .unwrap_or(0)
3996}

4028/// Re-encode a committed group with its original sequence and LSNs, for a
4029/// segment rewrite. Mirrors exactly the framing `commit_group` writes.
4030fn encode_group(
4031    group: &CommittedGroup,
4032    version: JournalVersion,
4033    into: &mut Vec<u8>,
4034) -> io::Result<()> {
4035    let mut payload = Vec::new();
4036    for entry in &group.entries {
4037        encode_mutation(
4038            entry.lsn,
4039            entry.pool,
4040            version,
4041            &entry.mutation,
4042            &mut payload,
4043        )?;
4044    }
4045    let record_count = u32::try_from(group.entries.len())
4046        .map_err(|_| io::Error::new(io::ErrorKind::InvalidData, "too many journal mutations"))?;
4047    let payload_len = u64::try_from(payload.len())
4048        .map_err(|_| io::Error::new(io::ErrorKind::InvalidData, "journal group too large"))?;
4049    let header = encode_group_header(
4050        group.sequence,
4051        group.first_lsn,
4052        group.last_lsn,
4053        payload_len,
4054        record_count,
4055    );
4056    let mut group_crc = Hasher::new();
4057    group_crc.update(&header);
4058    group_crc.update(&payload);
4059    let trailer = encode_group_trailer(group.sequence, group_crc.finalize());
4060    into.extend_from_slice(&header);
4061    into.extend_from_slice(&payload);
4062    into.extend_from_slice(&trailer);
4063    Ok(())
4064}

4066fn encode_mutation(
4067    lsn: u64,
4068    pool: Option<ShardType>,
4069    version: JournalVersion,
4070    mutation: &Mutation,
4071    into: &mut Vec<u8>,
4072) -> io::Result<()> {
4073    // The flag byte is the pool tag in a v3 segment and must be zero in a v2
4074    // segment; enforce the pairing rather than silently writing an ambiguous
4075    // frame.
4076    let pool_byte = match (version.is_pool_tagged(), pool) {
4077        (true, Some(pool)) => pool_tag(pool),
4078        (true, None) => {
4079            return Err(io::Error::new(
4080                io::ErrorKind::InvalidInput,
4081                "a pool-tagged journal frame requires a pool tag",
4082            ))
4083        }
4084        (false, Some(_)) => {
4085            return Err(io::Error::new(
4086                io::ErrorKind::InvalidInput,
4087                "a per-pool journal frame must not carry a pool tag",
4088            ))
4089        }
4090        (false, None) => 0,
4091    };
4092
4093    let start = into.len();
4094    // Zeroed fixed area; `MF_FLAGS` and the DeleteCollection-only fields stay
4095    // zero by construction. Index the fixed area by its named field ranges so
4096    // no raw offset arithmetic is needed.
4097    into.resize(start.saturating_add(FRAME_FIXED_LEN), 0);
4098    let payload = {
4099        let frame = &mut into[start..];
4100        frame[MF_KIND].copy_from_slice(&match mutation {
4101            Mutation::Put { .. } => [1],
4102            Mutation::DeleteCollection { .. } => [2],
4103        });
4104        frame[MF_POOL].copy_from_slice(&[pool_byte]);
4105        frame[MF_LSN].copy_from_slice(&lsn.to_le_bytes());
4106        match mutation {
4107            Mutation::Put {
4108                collection_id,
4109                node_id,
4110                payload,
4111            } => {
4112                frame[MF_COLLECTION].copy_from_slice(collection_id);
4113                frame[MF_NODE].copy_from_slice(node_id);
4114                let payload_len = u32::try_from(payload.len()).map_err(|_| {
4115                    io::Error::new(io::ErrorKind::InvalidInput, "mutation payload exceeds u32")
4116                })?;
4117                frame[MF_PAYLOAD_LEN].copy_from_slice(&payload_len.to_le_bytes());
4118                payload.as_slice()
4119            }
4120            Mutation::DeleteCollection { collection_id } => {
4121                frame[MF_COLLECTION].copy_from_slice(collection_id);
4122                &[]
4123            }
4124        }
4125    };
4126    into.extend_from_slice(payload);
4127    let mut crc = Hasher::new();
4128    crc.update(&into[start..]);
4129    into.extend_from_slice(&crc.finalize().to_le_bytes());
4130    Ok(())
4131}

4142/// Validate a group header's magic, length, and CRC, returning its fields.
4143fn parse_group_header(header: &[u8]) -> io::Result<GroupHeader> {
4144    if &header[..4] != GROUP_MAGIC
4145        || u32::from_le_bytes(header[4..8].try_into().expect("fixed slice"))
4146            != u32::try_from(GROUP_HEADER_LEN).expect("GROUP_HEADER_LEN fits in u32")
4147    {
4148        return Err(invalid_data("invalid journal group header"));
4149    }
4150    let mut header_crc = Hasher::new();
4151    header_crc.update(&header[..44]);
4152    if u32::from_le_bytes(header[44..48].try_into().expect("fixed slice")) != header_crc.finalize()
4153    {
4154        return Err(invalid_data("journal group header checksum mismatch"));
4155    }
4156    Ok(GroupHeader {
4157        sequence: u64::from_le_bytes(header[8..16].try_into().expect("fixed slice")),
4158        first_lsn: u64::from_le_bytes(header[16..24].try_into().expect("fixed slice")),
4159        last_lsn: u64::from_le_bytes(header[24..32].try_into().expect("fixed slice")),
4160        payload_len: u64::from_le_bytes(header[32..40].try_into().expect("fixed slice")),
4161        record_count: u32::from_le_bytes(header[40..44].try_into().expect("fixed slice")),
4162    })
4163}

4165/// Verify a group's commit trailer and whole-group CRC, returning its payload.
4166fn verify_group_payload<'a>(
4167    bytes: &'a [u8],
4168    header: &[u8],
4169    group: &GroupHeader,
4170    payload_start: usize,
4171    payload_len: usize,
4172) -> io::Result<&'a [u8]> {
4173    let payload_end = payload_start
4174        .checked_add(payload_len)
4175        .ok_or_else(|| invalid_data("journal group length overflow"))?;
4176    let trailer_end = payload_end
4177        .checked_add(GROUP_TRAILER_LEN)
4178        .ok_or_else(|| invalid_data("journal group length overflow"))?;
4179    let trailer = bytes
4180        .get(payload_end..trailer_end)
4181        .ok_or_else(|| invalid_data("truncated journal commit trailer"))?;
4182    if &trailer[..4] != GROUP_COMMIT_MAGIC
4183        || u64::from_le_bytes(trailer[4..12].try_into().expect("fixed slice")) != group.sequence
4184    {
4185        return Err(invalid_data("invalid journal commit trailer"));
4186    }
4187    let payload = bytes
4188        .get(payload_start..payload_end)
4189        .ok_or_else(|| invalid_data("truncated journal group payload"))?;
4190    let mut group_crc = Hasher::new();
4191    group_crc.update(header);
4192    group_crc.update(payload);
4193    if u32::from_le_bytes(trailer[12..16].try_into().expect("fixed slice")) != group_crc.finalize()
4194    {
4195        return Err(invalid_data("journal committed group checksum mismatch"));
4196    }
4197    Ok(payload)
4198}

4221/// What validating one group at a cursor found.
4222enum GroupStep {
4223    /// A complete, valid group that occupies `total_len` bytes.
4224    Group {
4225        group: GroupHeader,
4226        entries: Vec<JournalEntry>,
4227        total_len: usize,
4228    },
4229    /// Not enough bytes remain for a complete group.
4230    Incomplete,
4231}
4232
4233/// Why a group failed validation, split by whether a crash can explain it.
4234enum GroupFault {
4235    /// Damage that a torn or missing page can produce: a bad group header or a
4236    /// bad commit trailer or checksum. Above the durable mark this is an
4237    /// unacknowledged crash tail; at or below it, it is corruption.
4238    Crash(io::Error),
4239    /// A group whose header checksum verified but whose contents are wrong
4240    /// (sequence or LSN regression, impossible bounds, undecodable records).
4241    /// A crash cannot do that to an append-only file, so this is a bug or real
4242    /// corruption wherever it appears and is never truncated.
4243    Fatal(io::Error),
4244}

4246/// Validate the group starting at `cursor` in `bytes`.
4247fn scan_one_group(
4248    bytes: &[u8],
4249    cursor: usize,
4250    base_offset: u64,
4251    expected_sequence: u64,
4252    expected_lsn: u64,
4253    version: JournalVersion,
4254) -> Result<GroupStep, GroupFault> {
4255    let remaining = bytes.len().saturating_sub(cursor);
4256    if remaining < GROUP_HEADER_LEN {
4257        return Ok(GroupStep::Incomplete);
4258    }
4259    let header = bytes
4260        .get(cursor..cursor.saturating_add(GROUP_HEADER_LEN))
4261        .ok_or_else(|| GroupFault::Crash(invalid_data("truncated journal group header")))?;
4262    let group = parse_group_header(header).map_err(GroupFault::Crash)?;
4263    if group.sequence < expected_sequence
4264        || group.first_lsn != expected_lsn
4265        || group.last_lsn < group.first_lsn
4266        || group
4267            .last_lsn
4268            .saturating_sub(group.first_lsn)
4269            .saturating_add(1)
4270            != u64::from(group.record_count)
4271        || group.record_count == 0
4272        || group.payload_len > MAX_GROUP_LEN
4273        || u64::from(group.record_count).saturating_mul(MIN_FRAME_LEN as u64) > group.payload_len
4274    {
4275        return Err(GroupFault::Fatal(invalid_data(
4276            "invalid journal sequence or group bounds",
4277        )));
4278    }
4279    let payload_len = usize::try_from(group.payload_len).map_err(|_| {
4280        GroupFault::Fatal(invalid_data("journal group length exceeds address space"))
4281    })?;
4282    let total_len = GROUP_HEADER_LEN
4283        .checked_add(payload_len)
4284        .and_then(|len| len.checked_add(GROUP_TRAILER_LEN))
4285        .ok_or_else(|| GroupFault::Fatal(invalid_data("journal group length overflow")))?;
4286    if remaining < total_len {
4287        return Ok(GroupStep::Incomplete);
4288    }
4289    let payload_index = cursor.saturating_add(GROUP_HEADER_LEN);
4290    let payload = verify_group_payload(bytes, header, &group, payload_index, payload_len)
4291        .map_err(GroupFault::Crash)?;
4292    let payload_offset = base_offset
4293        .saturating_add(u64::try_from(payload_index).unwrap_or(u64::MAX))
4294        .try_into()
4295        .unwrap_or(usize::MAX);
4296    let entries = decode_mutations(
4297        payload,
4298        group.first_lsn,
4299        group.record_count,
4300        payload_offset,
4301        version,
4302    )
4303    .map_err(GroupFault::Fatal)?;
4304    Ok(GroupStep::Group {
4305        group,
4306        entries,
4307        total_len,
4308    })
4309}

4311/// Scan complete groups from `bytes`, which begins at absolute file offset
4312/// `base_offset`.
4313///
4314/// `expected_sequence` is a floor, not an exact match: a segment sharing a
4315/// global sequence with other pools may skip values. `expected_lsn` must match
4316/// the first group's `first_lsn` exactly, since LSNs stay contiguous within a
4317/// segment. `Scan::valid_len` is absolute in the file, not relative to `bytes`.
4318///
4319/// `durable_len` yields the embedded durability mark: how many
4320/// leading bytes of the file a completed fsync covered. It is called at most
4321/// once, and only when a group fails validation in a way a crash can explain. A group that fails validation
4322/// in a way a crash can explain ([`GroupFault::Crash`]) at or above it was never
4323/// acknowledged, so the scan stops there and reports a truncated tail, dropping
4324/// everything after it, even later groups that happen to be intact. The same
4325/// failure below it is corruption and is returned as an error. Return `u64::MAX`
4326/// to treat every byte as durable, so any invalid group fails; return `0` when
4327/// nothing is known, so the first invalid group ends the scan.
4328fn scan_groups_from(
4329    bytes: &[u8],
4330    base_offset: u64,
4331    mut expected_sequence: u64,
4332    mut expected_lsn: u64,
4333    segment_base_lsn: u64,
4334    version: JournalVersion,
4335    durable_len: impl Fn() -> u64,
4336) -> io::Result<Scan> {
4337    let mut cursor = 0usize;
4338    let mut valid_len = base_offset;
4339    let mut groups = Vec::new();
4340    let mut group_ends = Vec::new();
4341    let mut truncated_tail = false;
4342
4343    while cursor < bytes.len() {
4344        let (group, entries, total_len) = match scan_one_group(
4345            bytes,
4346            cursor,
4347            base_offset,
4348            expected_sequence,
4349            expected_lsn,
4350            version,
4351        ) {
4352            Ok(GroupStep::Group {
4353                group,
4354                entries,
4355                total_len,
4356            }) => (group, entries, total_len),
4357            Ok(GroupStep::Incomplete) => {
4358                truncated_tail = true;
4359                break;
4360            }
4361            Err(GroupFault::Fatal(error)) => return Err(error),
4362            Err(GroupFault::Crash(error)) => {
4363                let at = base_offset.saturating_add(u64::try_from(cursor).unwrap_or(u64::MAX));
4364                // The durability mark matters only here, on a failure a crash
4365                // can explain, so it is read only now. Reading it on every scan
4366                // would cost a file open and read on the hot incremental path
4367                // for something a healthy segment never consults.
4368                if at >= durable_len() {
4369                    truncated_tail = true;
4370                    break;
4371                }
4372                return Err(error);
4373            }
4374        };
4375        groups.push(CommittedGroup {
4376            sequence: group.sequence,
4377            first_lsn: group.first_lsn,
4378            last_lsn: group.last_lsn,
4379            entries,
4380        });
4381        cursor = cursor.saturating_add(total_len);
4382        valid_len = base_offset.saturating_add(u64::try_from(cursor).unwrap_or(u64::MAX));
4383        group_ends.push(valid_len);
4384        expected_sequence = group
4385            .sequence
4386            .checked_add(1)
4387            .ok_or_else(|| invalid_data("journal group sequence overflow"))?;
4388        expected_lsn = group
4389            .last_lsn
4390            .checked_add(1)
4391            .ok_or_else(|| invalid_data("journal LSN overflow"))?;
4392    }
4393
4394    let consumed_tail = bytes[cursor.saturating_sub(CONSUMED_TAIL_LEN)..cursor].to_vec();
4395    Ok(Scan {
4396        groups,
4397        valid_len,
4398        truncated_tail,
4399        base_lsn: segment_base_lsn,
4400        consumed_tail,
4401        group_ends,
4402    })
4403}

4405fn decode_mutations(
4406    payload: &[u8],
4407    first_lsn: u64,
4408    record_count: u32,
4409    payload_offset: usize,
4410    version: JournalVersion,
4411) -> io::Result<Vec<JournalEntry>> {
4412    let mut cursor = 0_usize;
4413    let mut entries = Vec::with_capacity(usize::try_from(record_count).unwrap_or(0));
4414    for index in 0..record_count {
4415        let start = cursor;
4416        if payload.len().saturating_sub(cursor) < MIN_FRAME_LEN {
4417            return Err(invalid_data("truncated journal mutation frame"));
4418        }
4419        let fixed_end = cursor
4420            .checked_add(FRAME_FIXED_LEN)
4421            .ok_or_else(|| invalid_data("journal mutation length overflow"))?;
4422        let frame = payload
4423            .get(cursor..fixed_end)
4424            .ok_or_else(|| invalid_data("truncated journal mutation frame"))?;
4425        let kind = frame[MF_KIND.start];
4426        let tag = frame[MF_POOL.start];
4427        if frame[MF_FLAGS] != [0, 0] {
4428            return Err(invalid_data("unsupported journal mutation flags"));
4429        }
4430        let pool = if version.is_pool_tagged() {
4431            if !matches!(tag, 1..=4) {
4432                return Err(invalid_data("pool-tagged frame has no valid pool tag"));
4433            }
4434            pool_from_tag(tag)
4435        } else {
4436            if tag != 0 {
4437                return Err(invalid_data("untagged frame has a nonzero flag byte"));
4438            }
4439            None
4440        };
4441        let lsn = u64::from_le_bytes(frame[MF_LSN].try_into().expect("fixed slice"));
4442        let expected = first_lsn
4443            .checked_add(u64::from(index))
4444            .ok_or_else(|| invalid_data("journal mutation LSN overflow"))?;
4445        if lsn != expected {
4446            return Err(invalid_data("journal mutation LSN out of order"));
4447        }
4448        let collection_id: [u8; 16] = frame[MF_COLLECTION].try_into().expect("fixed slice");
4449        let node_id: [u8; 16] = frame[MF_NODE].try_into().expect("fixed slice");
4450        let payload_len = usize::try_from(u32::from_le_bytes(
4451            frame[MF_PAYLOAD_LEN].try_into().expect("fixed slice"),
4452        ))
4453        .map_err(|_| invalid_data("journal mutation payload length exceeds address space"))?;
4454        let frame_end = fixed_end
4455            .checked_add(payload_len)
4456            .ok_or_else(|| invalid_data("journal mutation length overflow"))?;
4457        let crc_end = frame_end
4458            .checked_add(FRAME_TRAILER_LEN)
4459            .ok_or_else(|| invalid_data("journal mutation length overflow"))?;
4460        let data = payload
4461            .get(fixed_end..frame_end)
4462            .ok_or_else(|| invalid_data("journal mutation payload exceeds group"))?;
4463        let crc_bytes = payload
4464            .get(frame_end..crc_end)
4465            .ok_or_else(|| invalid_data("journal mutation checksum exceeds group"))?;
4466        let mut crc = Hasher::new();
4467        crc.update(
4468            payload
4469                .get(start..frame_end)
4470                .ok_or_else(|| invalid_data("journal mutation frame exceeds group"))?,
4471        );
4472        if u32::from_le_bytes(crc_bytes.try_into().expect("fixed slice")) != crc.finalize() {
4473            return Err(invalid_data("journal mutation checksum mismatch"));
4474        }
4475        let mutation = match kind {
4476            1 => Mutation::Put {
4477                collection_id,
4478                node_id,
4479                payload: data.to_vec(),
4480            },
4481            2 if payload_len == 0 && node_id == [0; 16] => {
4482                Mutation::DeleteCollection { collection_id }
4483            }
4484            _ => return Err(invalid_data("unknown or malformed journal mutation kind")),
4485        };
4486        entries.push(JournalEntry {
4487            lsn,
4488            offset: u64::try_from(payload_offset.saturating_add(start)).unwrap_or(u64::MAX),
4489            frame_len: u64::try_from(crc_end.saturating_sub(start)).unwrap_or(u64::MAX),
4490            pool,
4491            mutation,
4492        });
4493        cursor = crc_end;
4494    }
4495    if cursor != payload.len() {
4496        return Err(invalid_data("extra bytes after journal mutation frames"));
4497    }
4498    Ok(entries)
4499}

4513/// Exclusive writer claim on a database root's shared WAL.
4514///
4515/// One shared segment serves every pool, so a single writer process must own
4516/// it. This takes the same crash-safe `{pid, starttime}` lock the per-pool
4517/// writer lock uses, at `<db_root>/.mtxdb.wal.lock`, and releases it on drop
4518/// (see `ShardPool::acquire_writer_lock` for the staleness contract). The
4519/// holder opens the segment with [`Journal::open_shared`], builds one
4520/// [`JournalCoordinator`], and attaches each pool with
4521/// `PackfileStorage::enable_shared_journal`.
4522#[cfg(feature = "multi-reader")]
4523pub struct SharedWalLock {
4524    _lock: crate::shard::WriterLock,
4525}
4526
4527#[cfg(feature = "multi-reader")]
4528impl SharedWalLock {
4529    /// Acquire the root shared-WAL writer lock.
4530    ///
4531    /// # Errors
4532    /// Returns `WouldBlock` if another live writer holds it, or an I/O error
4533    /// if the lock file cannot be created.
4534    pub fn acquire(db_root: impl AsRef<Path>) -> io::Result<Self> {
4535        let db_root = db_root.as_ref();
4536        // Require a database root.
4537        if !crate::layout::is_database_root(db_root)? {
4538            return Err(io::Error::new(
4539                io::ErrorKind::InvalidData,
4540                format!(
4541                    "{} is not an mtxdb database root (missing {}); initialize it first",
4542                    db_root.display(),
4543                    crate::layout::DB_META_FILENAME
4544                ),
4545            ));
4546        }
4547        let path = db_root.join(".mtxdb.wal.lock");
4548        Ok(Self {
4549            _lock: crate::shard::ShardPool::acquire_lock_path(&path)?,
4550        })
4551    }
4552}
