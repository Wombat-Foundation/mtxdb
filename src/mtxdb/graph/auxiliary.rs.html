1//! Named auxiliary indexes for application-owned lookup tables.
2//!
3//! Auxiliary indexes are logical stores, not additional packfile pools. They
4//! use the normal [`StorageEngine`] so callers do not create one directory or
5//! file per room, namespace, or index name.
6//!
7//! Identity note: auxiliary key digests and collection ids are intentionally
8//! BLAKE3 (via [`DigestAlgorithm::Blake3`] and [`derive_collection_id`]). These
9//! indexes are core-internal and unreleased, so there is no persisted SHA-256
10//! data to remain compatible with and no versioned dual-read or migration is
11//! required. Reverting these derivations to SHA-256 would be a regression, not
12//! a compatibility fix.
13

24const VALUE_MAGIC: &[u8; 4] = b"AUX1";
25const DIGEST_LEN: usize = 32;
26
27/// The canonical identity of an auxiliary-index key.
28pub type AuxiliaryKeyDigest = [u8; DIGEST_LEN];
29
30/// Derive the canonical full digest for an auxiliary-index key.
31#[must_use]
32pub fn auxiliary_key_digest(key: &[u8]) -> AuxiliaryKeyDigest {
33    DigestAlgorithm::Blake3.digest(key)
34}
35
36/// Derive the logical collection identity for a named auxiliary index.
37///
38/// Auxiliary indexes are core-internal collections, so they use the shared
39/// [`derive_collection_id`] with the core-internal pool namespace rather than a
40/// private domain string. The returned 16-byte value is the routing id; key
41/// identities remain the full 32-byte digests stored in each envelope.
42#[must_use]
43pub fn auxiliary_collection_id(name: &str) -> [u8; 16] {
44    derive_collection_id(Some(MEMBER_NAMESPACE_INTL), name.as_bytes())
45}
46
47fn physical_id(digest: &AuxiliaryKeyDigest) -> NodeId {
48    let mut id = [0u8; 16];
49    id.copy_from_slice(&digest[..16]);
50    id
51}

53/// A named logical auxiliary index backed by an existing storage engine.
54pub struct AuxiliaryIndex<'a, S: StorageEngine + ?Sized> {
55    engine: &'a S,
56    name: String,
57    collection_id: [u8; 16],
58}

94    /// Ensure that collection metadata is established for this auxiliary index.

109    /// Read a value by its logical key.
110    ///
111    /// Verifies the full 32-byte digest before returning the application
112    /// payload.
113    ///
114    /// # Errors
115    /// Returns [`StorageError::Corrupt`] for a malformed or mismatched
116    /// envelope, or propagates a backend error.
117    pub fn get(&self, key: &[u8]) -> Result<Option<Vec<u8>>, StorageError> {
118        let digest = auxiliary_key_digest(key);
119        let node_id = physical_id(&digest);
120        let data = self.engine.get(&self.collection_id, &node_id)?;
121        decode_stored_value(&digest, data.as_ref())
122    }
123
124    /// Read several logical keys with one backend lookup.
125    ///
126    /// The returned values are in the same order as `keys`. Missing keys are
127    /// represented by `None`; stored envelopes are validated just like
128    /// [`Self::get`].
129    ///
130    /// # Errors
131    /// Returns a storage or auxiliary-envelope error.
132    pub fn get_many(&self, keys: &[&[u8]]) -> Result<Vec<Option<Vec<u8>>>, StorageError> {
133        let digests: Vec<AuxiliaryKeyDigest> =
134            keys.iter().map(|key| auxiliary_key_digest(key)).collect();
135        let ids: Vec<NodeId> = digests.iter().map(physical_id).collect();
136        let existing = self.engine.get_many(&self.collection_id, &ids)?;
137        existing
138            .iter()
139            .zip(digests.iter())
140            .map(|(data, digest)| decode_stored_value(digest, data.as_ref()))
141            .collect()
142    }

144    /// Insert or update a value by its logical key.
145    ///
146    /// Atomically establishes the collection metadata on the first write
147    /// and appends the key-value envelope. Retains idempotency (same key and value succeeds),
148    /// permits same-key replacement with updated values, and rejects truncated-ID collisions
149    /// (distinct key with the same 16-byte physical node ID) with [`StorageError::Collision`].
150    ///
151    /// # Errors
152    /// Returns [`StorageError::Collision`] on physical ID key collisions,
153    /// [`StorageError::Corrupt`] on corrupted existing records, or propagates backend error.
154    #[allow(clippy::arithmetic_side_effects)]
155    pub fn put(&self, key: &[u8], value: &[u8]) -> Result<(), StorageError> {
156        let digest = auxiliary_key_digest(key);
157        let node_id = physical_id(&digest);
158        if let Some(existing) = self.engine.get(&self.collection_id, &node_id)? {
159            if !existing.bytes.is_empty() {
160                decode_value(&digest, &existing.bytes)?;
161            }
162        }
163        let capacity = VALUE_MAGIC
164            .len()
165            .checked_add(DIGEST_LEN)
166            .and_then(|length| length.checked_add(value.len()))
167            .ok_or_else(|| StorageError::Corrupt("auxiliary value length overflow".to_owned()))?;
168        let mut encoded = Vec::with_capacity(capacity);
169        encoded.extend_from_slice(VALUE_MAGIC);
170        encoded.extend_from_slice(&digest);
171        encoded.extend_from_slice(value);
172        let data = NodeData::new(bytes::Bytes::from(encoded));
173
174        let mut validate = |existing: Option<&NodeData>| -> Result<(), StorageError> {
175            if let Some(existing) = existing {
176                if !existing.bytes.is_empty() {
177                    decode_value(&digest, &existing.bytes)?;
178                }
179            }
180            Ok(())
181        };
182
183        self.engine.create_or_upsert_established_validated(
184            &self.collection_id,
185            &self.metadata(),
186            &node_id,
187            &data,
188            &mut validate,
189        )
190    }

192    /// Insert or replace several values with one read and one write batch.
193    ///
194    /// Existing records are still checked for full-digest collisions, but
195    /// callers avoid one storage round trip per auxiliary entry.
196    ///
197    /// Writes are last-writer-wins per record and are not serialized against
198    /// other writers: if two `put_many` calls race on the same key, the value
199    /// from the batch the backing store commits last is retained. Callers must
200    /// not rely on any ordering between concurrent writers.
201    ///
202    /// # Errors
203    /// Returns [`StorageError::Collision`] for distinct keys with the same
204    /// physical ID, [`StorageError::Corrupt`] for malformed existing envelopes,
205    /// and propagates storage errors.
206    pub fn put_many(&self, entries: &[(&[u8], &[u8])]) -> Result<usize, StorageError> {
207        if entries.is_empty() {
208            self.ensure_metadata()?;
209            return Ok(0);
210        }
211        let mut physical: Vec<(NodeId, NodeData)> = Vec::with_capacity(entries.len());
212        let mut seen: HashMap<NodeId, AuxiliaryKeyDigest> = HashMap::with_capacity(entries.len());
213        for (key, value) in entries {
214            let digest = auxiliary_key_digest(key);
215            let node_id = physical_id(&digest);
216            if let Some(previous) = seen.insert(node_id, digest) {
217                if previous != digest {
218                    return Err(StorageError::Collision(
219                        "auxiliary-index key digest collision".to_owned(),
220                    ));
221                }
222            }
223            let capacity = VALUE_MAGIC
224                .len()
225                .checked_add(DIGEST_LEN)
226                .and_then(|length| length.checked_add(value.len()))
227                .ok_or_else(|| {
228                    StorageError::Corrupt("auxiliary value length overflow".to_owned())
229                })?;
230            let mut encoded = Vec::with_capacity(capacity);
231            encoded.extend_from_slice(VALUE_MAGIC);
232            encoded.extend_from_slice(&digest);
233            encoded.extend_from_slice(value);
234            physical.push((node_id, NodeData::new(bytes::Bytes::from(encoded))));
235        }
236
237        self.ensure_metadata()?;
238        let ids: Vec<NodeId> = physical.iter().map(|(id, _)| *id).collect();
239        let existing = self.engine.get_many(&self.collection_id, &ids)?;
240        for ((_, data), existing) in physical.iter().zip(existing) {
241            if let Some(existing) = existing {
242                // The full digest is retained in the envelope. Validate it
243                // before allowing the batch to replace the record.
244                let digest_start = VALUE_MAGIC.len();
245                let digest_end = digest_start.checked_add(DIGEST_LEN).ok_or_else(|| {
246                    StorageError::Corrupt("auxiliary digest length overflow".to_owned())
247                })?;
248                let digest: AuxiliaryKeyDigest = data
249                    .bytes
250                    .get(digest_start..digest_end)
251                    .and_then(|bytes| bytes.try_into().ok())
252                    .ok_or_else(|| {
253                        StorageError::Corrupt(
254                            "new auxiliary value has an incomplete digest envelope".to_owned(),
255                        )
256                    })?;
257                decode_value(&digest, &existing.bytes)?;
258            }
259        }
260        self.engine.put_many(&self.collection_id, &physical)
261    }

263    /// Stage removal of one value by logical key. Returns `true` if a value
264    /// was present in the transaction's view, including earlier staged writes.
265    /// A missing key returns `false`. The caller must commit the transaction to
266    /// publish the removal. `pool` must select the engine backing this index.
267    ///
268    /// # Errors
269    /// Returns [`StorageError::Collision`] for a digest collision,
270    /// [`StorageError::Corrupt`] for a malformed nonempty envelope,
271    /// [`StorageError::Internal`] if the transaction's pool differs from this
272    /// index's engine, or propagates a storage or staging error. A concurrent
273    /// change is reported as [`StorageError::StaleRead`] on commit.
274    pub fn remove(
275        &self,
276        transaction: &DatabaseTransaction<'_>,
277        pool: ShardType,
278        key: &[u8],
279    ) -> Result<bool, StorageError> {
280        Ok(self.stage_remove_many(transaction, pool, &[key])? != 0)
281    }
282
283    /// Stage several removals in one database transaction, returning the
284    /// number of values present in the transaction's view, including earlier
285    /// staged writes. Missing keys do not count. `pool` must select the engine
286    /// backing this index. The caller must commit `transaction` to publish the
287    /// removals. Duplicate keys are removed once. Record-version
288    /// preconditions for records read from the live store come from the same
289    /// transaction-view read that validates their envelopes, so a concurrent
290    /// replacement after validation surfaces as [`StorageError::StaleRead`]
291    /// on commit. A record already staged in this transaction uses its live
292    /// version because its staged read has only a synthetic version token.
293    ///
294    /// # Errors
295    /// Returns [`StorageError::Collision`] for a digest collision,
296    /// [`StorageError::Corrupt`] for a malformed nonempty envelope,
297    /// [`StorageError::Internal`] if the transaction's pool differs from this
298    /// index's engine, or propagates a storage or staging error. A concurrent
299    /// change is reported as [`StorageError::StaleRead`] on commit.
300    pub fn stage_remove_many(
301        &self,
302        transaction: &DatabaseTransaction<'_>,
303        pool: ShardType,
304        keys: &[&[u8]],
305    ) -> Result<usize, StorageError> {
306        let engine_identity = std::ptr::from_ref(self.engine).cast::<()>() as usize;
307        if engine_identity != transaction.pool_identity(pool) {
308            return Err(StorageError::Internal(
309                "auxiliary index and transaction use different storage pools".to_owned(),
310            ));
311        }
312
313        let mut ids = Vec::with_capacity(keys.len());
314        let mut digests_by_id: HashMap<NodeId, AuxiliaryKeyDigest> =
315            HashMap::with_capacity(keys.len());
316        for key in keys {
317            let digest = auxiliary_key_digest(key);
318            let id = physical_id(&digest);
319            if let Some(previous) = digests_by_id.insert(id, digest) {
320                if previous != digest {
321                    return Err(StorageError::Collision(
322                        "auxiliary-index key digest collision".to_owned(),
323                    ));
324                }
325            } else {
326                ids.push(id);
327            }
328        }
329        if ids.is_empty() {
330            return Ok(0);
331        }
332
333        // Validate the transaction's view and retain the matching versions
334        // for the removal preconditions.
335        let (existing, versions) =
336            transaction.get_with_record_versions(pool, &self.collection_id, &ids)?;
337        let mut present = 0usize;
338        for (id, record) in ids.iter().zip(&existing) {
339            if let Some(record) = record {
340                let digest = digests_by_id.get(id).ok_or_else(|| {
341                    StorageError::Internal("auxiliary digest lookup failed".to_owned())
342                })?;
343                if !record.bytes.is_empty() {
344                    decode_value(digest, &record.bytes)?;
345                }
346                present = present.checked_add(1).ok_or_else(|| {
347                    StorageError::Internal("auxiliary removal count overflow".to_owned())
348                })?;
349            }
350        }
351        transaction
352            .remove_many_with_record_versions(pool, self.collection_id, &ids, &versions)
353            .map_err(StorageError::Io)?;
354        Ok(present)
355    }
356}

358fn decode_value(
359    expected_digest: &AuxiliaryKeyDigest,
360    encoded: &[u8],
361) -> Result<Vec<u8>, StorageError> {
362    if !encoded.starts_with(VALUE_MAGIC) {
363        return Err(StorageError::Corrupt(
364            "auxiliary-index value is missing its envelope".to_owned(),
365        ));
366    }
367    let digest_start = VALUE_MAGIC.len();
368    let value_start = digest_start.checked_add(DIGEST_LEN).ok_or_else(|| {
369        StorageError::Corrupt("auxiliary-index envelope length overflow".to_owned())
370    })?;
371    if encoded.len() < value_start {
372        return Err(StorageError::Corrupt(
373            "truncated auxiliary-index value envelope".to_owned(),
374        ));
375    }
376    if encoded[digest_start..value_start] != expected_digest[..] {
377        return Err(StorageError::Collision(
378            "auxiliary-index key digest collision".to_owned(),
379        ));
380    }
381    Ok(encoded[value_start..].to_vec())
382}
383
384fn decode_stored_value(
385    digest: &AuxiliaryKeyDigest,
386    data: Option<&NodeData>,
387) -> Result<Option<Vec<u8>>, StorageError> {
388    let Some(data) = data else {
389        return Ok(None);
390    };
391    if data.bytes.is_empty() {
392        return Ok(None);
393    }
394    decode_value(digest, &data.bytes).map(Some)
395}
