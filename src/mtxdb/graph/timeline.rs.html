1//! Persisted, ordered timeline index with copy-on-write roots.
2//!
3//! The packfile index maps node id -> frame and has no key order, so it cannot
4//! answer a range read over a room's timeline. This module adds a **derived**
5//! ordered index built from ordinary records, in the same spirit as
6//! [`crate::auxiliary::AuxiliaryIndex`]: it is rebuildable, packfiles stay
7//! authoritative, and event payloads are never duplicated — a leaf stores only
8//! the ordering key plus the event's node id; the caller fetches the payload.
9//!
10//! The persisted key is the total order
11//!
12//! ```text
13//! (room_id, topological_ordering, stream_ordering, node_id, event_ref)
14//! ```
15//!
16//! This mirrors Synapse's `events_order_room (room_id, topological_ordering,
17//! stream_ordering)` history order.
18//!
19//! - `room_id` scopes the index; a timeline read is a room-prefix range.
20//! - `topological_ordering` (Matrix: depth) and `stream_ordering` are
21//!   caller-supplied, signed 64-bit and **persisted**; the index never
22//!   recomputes them. `stream_ordering` is signed because Synapse gives
23//!   backfilled events negative stream positions. The caller must define a
24//!   deterministic fallback for events without a computable position (missing
25//!   parents, cycles) before building, because the in-memory DAG order cannot
26//!   be regenerated reliably after reopening.
27//! - `node_id` is the operational 128-bit id of the stored event and the
28//!   tie-break after the two orderings, so equal positions never tie.
29//! - `event_ref` is a caller-supplied 32-byte event identity (reference hash),
30//!   kept as the final disambiguator so two ids for the same content stay
31//!   distinct and ordered deterministically.
32//!
33//! **Prototype.** This is a copy-on-write leaf index, not the production
34//! ordered-run format (memtable, sorted runs, fences, blooms, compaction); see
35//! `docs/docs/2026-09-30-ordered-index-reconciliation.md`. The `TML*` formats
36//! are not stable.
37//!
38//! Layout (all content-addressed, immutable except the head):
39//! - **leaf** (`TMLF`): up to [`TIMELINE_LEAF_CAP`] sorted entries;
40//! - **root** (`TMLR`): a manifest of `(first_key, leaf_id)` in order plus the
41//!   total entry count;
42//! - **head** (`TMLH`): a fixed-id record holding the current root id, the only
43//!   mutable id (last write wins).
44//!
45//! A rebuild writes new leaves and a new root and then swaps the head; old
46//! records survive until GC, so a [`TimelineCursor`] that pins the root it
47//! started on keeps paginating that snapshot even after an update. Forward
48//! paging returns keys strictly greater than the cursor, reverse strictly less,
49//! so resuming is exact and idempotent in both directions.
50

59/// Version byte of every timeline record and the cursor encoding.
60pub const TIMELINE_FORMAT_VERSION: u8 = 0x01;
61/// Maximum entries per leaf record.
62pub const TIMELINE_LEAF_CAP: usize = 128;
63
64/// Magic prefix of a leaf record (`b"TMLF"`).
65pub const TIMELINE_LEAF_MAGIC: [u8; 4] = *b"TMLF";
66/// Magic prefix of a root manifest record (`b"TMLR"`).
67pub const TIMELINE_ROOT_MAGIC: [u8; 4] = *b"TMLR";
68/// Magic prefix of the head record (`b"TMLH"`).
69pub const TIMELINE_HEAD_MAGIC: [u8; 4] = *b"TMLH";
70/// Magic prefix of an opaque cursor (`b"TMLC"`).
71pub const TIMELINE_CURSOR_MAGIC: [u8; 4] = *b"TMLC";
72
73/// Reserved node id of the head record (the only mutable id in the index).
74pub const TIMELINE_HEAD_ID: NodeId = *b"MTXD-TML-HEAD-v1";
75
76/// A caller-supplied event identity (reference hash).
77pub type EventRef = [u8; 32];
78
79const ENTRY_LEN: usize = 16 + 8 + 8 + 16 + 32;
80const LEAF_HEADER_LEN: usize = 7;
81const ROOT_HEADER_LEN: usize = 17;
82const ROOT_LEAF_LEN: usize = ENTRY_LEN + 16;
83const HEAD_LEN: usize = 5 + 16 + 8;
84const CURSOR_LEN: usize = 5 + 16 + ENTRY_LEN;
85const MAX_NODE_ID: NodeId = [0xff; 16];
86const MAX_EVENT_REF: EventRef = [0xff; 32];

88/// Order-preserving big-endian encoding of a signed position: flipping the
89/// sign bit makes byte order equal numeric order, so the wire key is
90/// byte-comparable.
91fn encode_position(value: i64) -> [u8; 8] {
92    let mut bytes = value.to_be_bytes();
93    bytes[0] ^= 0x80;
94    bytes
95}
96
97fn decode_position(mut bytes: [u8; 8]) -> i64 {
98    bytes[0] ^= 0x80;
99    i64::from_be_bytes(bytes)
100}

102/// One ordered entry: the full
103/// `(room, topological_ordering, stream_ordering, node, event_ref)` key.
104///
105/// Field order is the sort order (`derive(Ord)` compares in declaration order).
106#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
107pub struct TimelineEntry {
108    /// Room scope; the leading key component.
109    pub room_id: [u8; 16],
110    /// Caller-supplied topological ordering (Matrix: depth).
111    pub topological_ordering: i64,
112    /// Caller-supplied stream ordering; negative for backfilled events.
113    pub stream_ordering: i64,
114    /// The event's operational 128-bit node id; tie-break after the orderings.
115    pub node_id: NodeId,
116    /// Caller-supplied 32-byte event identity; final disambiguator.
117    pub event_ref: EventRef,
118}
119
120impl TimelineEntry {
121    /// Construct an entry.
122    #[must_use]
123    pub const fn new(
124        room_id: [u8; 16],
125        topological_ordering: i64,
126        stream_ordering: i64,
127        node_id: NodeId,
128        event_ref: EventRef,
129    ) -> Self {
130        Self {
131            room_id,
132            topological_ordering,
133            stream_ordering,
134            node_id,
135            event_ref,
136        }
137    }
138
139    fn encode_into(self, out: &mut Vec<u8>) {
140        out.extend_from_slice(&self.room_id);
141        out.extend_from_slice(&encode_position(self.topological_ordering));
142        out.extend_from_slice(&encode_position(self.stream_ordering));
143        out.extend_from_slice(&self.node_id);
144        out.extend_from_slice(&self.event_ref);
145    }
146
147    fn decode(bytes: &[u8]) -> Result<Self, StorageError> {
148        if bytes.len() != ENTRY_LEN {
149            return Err(StorageError::Corrupt("timeline entry length".to_owned()));
150        }
151        let position = |range: std::ops::Range<usize>, what: &str| {
152            <[u8; 8]>::try_from(&bytes[range])
153                .map(decode_position)
154                .map_err(|_| StorageError::Corrupt(format!("timeline entry {what}")))
155        };
156        let mut room_id = [0u8; 16];
157        room_id.copy_from_slice(&bytes[..16]);
158        let topological_ordering = position(16..24, "topological ordering")?;
159        let stream_ordering = position(24..32, "stream ordering")?;
160        let mut node_id = [0u8; 16];
161        node_id.copy_from_slice(&bytes[32..48]);
162        let mut event_ref = [0u8; 32];
163        event_ref.copy_from_slice(&bytes[48..ENTRY_LEN]);
164        Ok(Self {
165            room_id,
166            topological_ordering,
167            stream_ordering,
168            node_id,
169            event_ref,
170        })
171    }
172}

174/// An opaque, stable cursor naming the snapshot root plus the last entry read.
175///
176/// It carries the full ordering key, so both directions resume exactly.
177#[derive(Debug, Clone, Copy, PartialEq, Eq)]
178pub struct TimelineCursor {
179    root_id: NodeId,
180    entry: TimelineEntry,
181}
182
183impl TimelineCursor {
184    /// Build a cursor for `root_id` at `entry`.
185    #[must_use]
186    pub const fn new(root_id: NodeId, entry: TimelineEntry) -> Self {
187        Self { root_id, entry }
188    }
189
190    /// The snapshot root this cursor pins.
191    #[must_use]
192    pub const fn root_id(&self) -> NodeId {
193        self.root_id
194    }
195
196    /// The entry this cursor names.
197    #[must_use]
198    pub const fn entry(&self) -> TimelineEntry {
199        self.entry
200    }
201
202    /// Encode the cursor as opaque, versioned bytes.
203    #[must_use]
204    pub fn encode(&self) -> Vec<u8> {
205        let mut out = Vec::with_capacity(CURSOR_LEN);
206        out.extend_from_slice(&TIMELINE_CURSOR_MAGIC);
207        out.push(TIMELINE_FORMAT_VERSION);
208        out.extend_from_slice(&self.root_id);
209        self.entry.encode_into(&mut out);
210        out
211    }
212
213    /// Decode cursor bytes produced by [`Self::encode`].
214    ///
215    /// Returns `None` for a wrong magic, version, or length.
216    #[must_use]
217    pub fn decode(bytes: &[u8]) -> Option<Self> {
218        if bytes.len() != CURSOR_LEN
219            || bytes[..4] != TIMELINE_CURSOR_MAGIC
220            || bytes[4] != TIMELINE_FORMAT_VERSION
221        {
222            return None;
223        }
224        let mut root_id = [0u8; 16];
225        root_id.copy_from_slice(&bytes[5..21]);
226        let entry = TimelineEntry::decode(&bytes[21..CURSOR_LEN]).ok()?;
227        Some(Self { root_id, entry })
228    }
229}

240/// The timeline index collection id for a caller-chosen index name.
241#[must_use]
242pub fn timeline_collection_id(name: &str) -> [u8; 16] {
243    let mut canonical = Vec::with_capacity(name.len().saturating_add(9));
244    canonical.extend_from_slice(b"timeline:");
245    canonical.extend_from_slice(name.as_bytes());
246    derive_collection_id(Some(MEMBER_NAMESPACE_INTL), &canonical)
247}

249fn content_id(payload: &[u8]) -> NodeId {
250    let digest = DigestAlgorithm::Blake3.digest(payload);
251    let mut id = [0u8; 16];
252    id.copy_from_slice(&digest[..16]);
253    id
254}

305    /// Ensure the index collection's metadata is established.
306    ///

313    }

315    /// Build (or rebuild) the index from `entries`, returning the new root id.
316    ///
317    /// Entries are sorted and de-duplicated by the full key. Identical inputs
318    /// produce identical content-addressed records, so a replay is idempotent;
319    /// only the head pointer changes between builds. Old roots stay readable,
320    /// so cursors pinned to them keep working.
321    ///
322    /// # Errors
323    /// Propagates the backend error, or [`StorageError::Internal`] if a count
324    /// exceeds the record layout.
325    pub fn build(&self, entries: &[TimelineEntry]) -> Result<NodeId, StorageError> {
326        self.ensure_metadata()?;
327        let mut sorted = entries.to_vec();
328        sorted.sort_unstable();
329        sorted.dedup();
330
331        let mut manifest: Vec<(TimelineEntry, NodeId)> = Vec::new();
332        for chunk in sorted.chunks(TIMELINE_LEAF_CAP) {
333            let payload = encode_leaf(chunk)?;
334            let id = content_id(&payload);
335            self.write_record(&id, payload)?;
336            let first = *chunk
337                .first()
338                .ok_or_else(|| StorageError::Internal("empty timeline leaf".to_owned()))?;
339            manifest.push((first, id));
340        }
341
342        let count = u64::try_from(sorted.len()).unwrap_or(u64::MAX);
343        let root_payload = encode_root(&manifest, count)?;
344        let root_id = content_id(&root_payload);
345        self.write_record(&root_id, root_payload)?;
346        self.write_record(&TIMELINE_HEAD_ID, encode_head(&root_id, count))?;
347        Ok(root_id)
348    }

350    /// The current head's `(root_id, entry_count)`, if the index was built.
351    ///
352    /// # Errors
353    /// [`StorageError::Corrupt`] if the head record does not decode.
354    pub fn head(&self) -> Result<Option<(NodeId, u64)>, StorageError> {
355        match self.engine.get(&self.collection_id, &TIMELINE_HEAD_ID)? {
356            None => Ok(None),
357            Some(data) => decode_head(&data.bytes).map(Some),
358        }
359    }

361    /// Read one page of `room_id` in `forward` (ascending) or reverse order.
362    ///
363    /// Forward returns keys strictly greater than `cursor`; reverse strictly
364    /// less. `None` starts at the room's first (forward) or last (reverse)
365    /// entry and reads the current head; a `Some` cursor reads the root it
366    /// pins, so pagination is stable across rebuilds. `next_cursor` is `Some`
367    /// whenever the page filled `limit` (a following call may return empty).
368    ///
369    /// # Errors
370    /// [`StorageError::Corrupt`] if the cursor names a different room or a
371    /// record fails to decode; [`StorageError::NotFound`] if a referenced
372    /// record is missing.
373    #[allow(clippy::similar_names, reason = "room_id/root_id are the domain names")]
374    pub fn page(
375        &self,
376        room_id: &[u8; 16],
377        cursor: Option<&TimelineCursor>,
378        forward: bool,
379        limit: usize,
380    ) -> Result<TimelinePage, StorageError> {
381        let empty = TimelinePage {
382            entries: Vec::new(),
383            next_cursor: None,
384        };
385        if limit == 0 {
386            return Ok(empty);
387        }
388
389        let (root_id, key) = match cursor {
390            Some(cursor) => {
391                if cursor.entry.room_id != *room_id {
392                    return Err(StorageError::Corrupt(
393                        "timeline cursor names a different room".to_owned(),
394                    ));
395                }
396                (cursor.root_id, Some(cursor.entry()))
397            }
398            None => match self.head()? {
399                Some((root_id, _count)) => {
400                    let key = if forward {
401                        room_start(room_id)
402                    } else {
403                        room_end(room_id)
404                    };
405                    (root_id, Some(key))
406                }
407                None => return Ok(empty),
408            },
409        };
410
411        let root_bytes = self
412            .engine
413            .get(&self.collection_id, &root_id)?
414            .ok_or(StorageError::NotFound(root_id))?;
415        let manifest = decode_root(&root_bytes.bytes)?;
416        if manifest.is_empty() {
417            return Ok(empty);
418        }
419
420        let mut out: Vec<TimelineEntry> = Vec::with_capacity(limit);
421        if forward {
422            let start = forward_start(&manifest, key);
423            for (_, leaf_id) in manifest.iter().skip(start) {
424                let leaf = self.load_leaf(leaf_id)?;
425                for entry in &leaf {
426                    if entry.room_id < *room_id {
427                        continue;
428                    }
429                    if entry.room_id > *room_id {
430                        return Ok(TimelinePage {
431                            entries: out,
432                            next_cursor: None,
433                        });
434                    }
435                    if key.is_some_and(|k| *entry <= k) {
436                        continue;
437                    }
438                    out.push(*entry);
439                    if out.len() == limit {
440                        return Ok(TimelinePage {
441                            next_cursor: Some(TimelineCursor::new(root_id, *entry)),
442                            entries: out,
443                        });
444                    }
445                }
446            }
447        } else {
448            let Some(start) = reverse_start(&manifest, key) else {
449                return Ok(empty);
450            };
451            for index in (0..=start).rev() {
452                let leaf = self.load_leaf(&manifest[index].1)?;
453                for entry in leaf.iter().rev() {
454                    if entry.room_id > *room_id {
455                        continue;
456                    }
457                    if entry.room_id < *room_id {
458                        return Ok(TimelinePage {
459                            entries: out,
460                            next_cursor: None,
461                        });
462                    }
463                    if key.is_some_and(|k| *entry >= k) {
464                        continue;
465                    }
466                    out.push(*entry);
467                    if out.len() == limit {
468                        return Ok(TimelinePage {
469                            next_cursor: Some(TimelineCursor::new(root_id, *entry)),
470                            entries: out,
471                        });
472                    }
473                }
474            }
475        }
476        Ok(TimelinePage {
477            entries: out,
478            next_cursor: None,
479        })
480    }

482    fn write_record(&self, id: &NodeId, payload: Vec<u8>) -> Result<(), StorageError> {
483        let data = NodeData::new(Bytes::from(payload));
484        let mut validate = |_existing: Option<&NodeData>| Ok(());
485        self.engine.create_or_upsert_established_validated(
486            &self.collection_id,
487            &self.metadata(),
488            id,
489            &data,
490            &mut validate,
491        )
492    }
493
494    fn load_leaf(&self, id: &NodeId) -> Result<Vec<TimelineEntry>, StorageError> {
495        let bytes = self
496            .engine
497            .get(&self.collection_id, id)?
498            .ok_or(StorageError::NotFound(*id))?;
499        decode_leaf(&bytes.bytes)
500    }
501}

503fn forward_start(manifest: &[(TimelineEntry, NodeId)], key: Option<TimelineEntry>) -> usize {
504    let Some(key) = key else {
505        return 0;
506    };
507    // The first leaf that can hold a key greater than `key` is the last leaf
508    // whose first key is <= `key`; if every leaf starts after `key`, leaf 0.
509    manifest
510        .partition_point(|entry| entry.0 <= key)
511        .saturating_sub(1)
512}
513
514fn reverse_start(
515    manifest: &[(TimelineEntry, NodeId)],
516    key: Option<TimelineEntry>,
517) -> Option<usize> {
518    let bound = key?;
519    // The last leaf that can hold a key less than `bound` starts strictly
520    // before it; when none does, there is nothing to return.
521    manifest
522        .partition_point(|entry| entry.0 < bound)
523        .checked_sub(1)
524}

526fn encode_leaf(entries: &[TimelineEntry]) -> Result<Vec<u8>, StorageError> {
527    let count = u16::try_from(entries.len())
528        .map_err(|_| StorageError::Internal("timeline leaf exceeds u16 entries".to_owned()))?;
529    let mut out = Vec::with_capacity(
530        entries
531            .len()
532            .saturating_mul(ENTRY_LEN)
533            .saturating_add(LEAF_HEADER_LEN),
534    );
535    out.extend_from_slice(&TIMELINE_LEAF_MAGIC);
536    out.push(TIMELINE_FORMAT_VERSION);
537    out.extend_from_slice(&count.to_be_bytes());
538    for entry in entries {
539        entry.encode_into(&mut out);
540    }
541    Ok(out)
542}
543
544fn decode_leaf(bytes: &[u8]) -> Result<Vec<TimelineEntry>, StorageError> {
545    if bytes.len() < LEAF_HEADER_LEN
546        || bytes[..4] != TIMELINE_LEAF_MAGIC
547        || bytes[4] != TIMELINE_FORMAT_VERSION
548    {
549        return Err(StorageError::Corrupt("timeline leaf header".to_owned()));
550    }
551    let count = u16::from_be_bytes([bytes[5], bytes[6]]) as usize;
552    let expected = count
553        .checked_mul(ENTRY_LEN)
554        .and_then(|body| body.checked_add(LEAF_HEADER_LEN))
555        .ok_or_else(|| StorageError::Corrupt("timeline leaf length".to_owned()))?;
556    if bytes.len() != expected {
557        return Err(StorageError::Corrupt("timeline leaf length".to_owned()));
558    }
559    let mut entries = Vec::with_capacity(count);
560    for index in 0..count {
561        let start = index
562            .checked_mul(ENTRY_LEN)
563            .and_then(|offset| offset.checked_add(LEAF_HEADER_LEN))
564            .ok_or_else(|| StorageError::Corrupt("timeline leaf offset".to_owned()))?;
565        let end = start
566            .checked_add(ENTRY_LEN)
567            .ok_or_else(|| StorageError::Corrupt("timeline leaf offset".to_owned()))?;
568        entries.push(TimelineEntry::decode(&bytes[start..end])?);
569    }
570    Ok(entries)
571}
572
573fn encode_root(manifest: &[(TimelineEntry, NodeId)], count: u64) -> Result<Vec<u8>, StorageError> {
574    let leaf_count = u32::try_from(manifest.len())
575        .map_err(|_| StorageError::Internal("timeline root exceeds u32 leaves".to_owned()))?;
576    let mut out = Vec::with_capacity(
577        manifest
578            .len()
579            .saturating_mul(ROOT_LEAF_LEN)
580            .saturating_add(ROOT_HEADER_LEN),
581    );
582    out.extend_from_slice(&TIMELINE_ROOT_MAGIC);
583    out.push(TIMELINE_FORMAT_VERSION);
584    out.extend_from_slice(&count.to_be_bytes());
585    out.extend_from_slice(&leaf_count.to_be_bytes());
586    for (first, leaf_id) in manifest {
587        first.encode_into(&mut out);
588        out.extend_from_slice(leaf_id);
589    }
590    Ok(out)
591}

593fn decode_root(bytes: &[u8]) -> Result<Vec<(TimelineEntry, NodeId)>, StorageError> {
594    if bytes.len() < ROOT_HEADER_LEN
595        || bytes[..4] != TIMELINE_ROOT_MAGIC
596        || bytes[4] != TIMELINE_FORMAT_VERSION
597    {
598        return Err(StorageError::Corrupt("timeline root header".to_owned()));
599    }
600    let leaf_count = u32::from_be_bytes([bytes[13], bytes[14], bytes[15], bytes[16]]) as usize;
601    let expected = leaf_count
602        .checked_mul(ROOT_LEAF_LEN)
603        .and_then(|body| body.checked_add(ROOT_HEADER_LEN))
604        .ok_or_else(|| StorageError::Corrupt("timeline root length".to_owned()))?;
605    if bytes.len() != expected {
606        return Err(StorageError::Corrupt("timeline root length".to_owned()));
607    }
608    let mut manifest = Vec::with_capacity(leaf_count);
609    for index in 0..leaf_count {
610        let start = index
611            .checked_mul(ROOT_LEAF_LEN)
612            .and_then(|offset| offset.checked_add(ROOT_HEADER_LEN))
613            .ok_or_else(|| StorageError::Corrupt("timeline root offset".to_owned()))?;
614        let entry_end = start
615            .checked_add(ENTRY_LEN)
616            .ok_or_else(|| StorageError::Corrupt("timeline root offset".to_owned()))?;
617        let next_leaf = start
618            .checked_add(ROOT_LEAF_LEN)
619            .ok_or_else(|| StorageError::Corrupt("timeline root offset".to_owned()))?;
620        let first = TimelineEntry::decode(&bytes[start..entry_end])?;
621        let mut leaf_id = [0u8; 16];
622        leaf_id.copy_from_slice(&bytes[entry_end..next_leaf]);
623        manifest.push((first, leaf_id));
624    }
625    Ok(manifest)
626}
627
628fn encode_head(root_id: &NodeId, count: u64) -> Vec<u8> {
629    let mut out = Vec::with_capacity(HEAD_LEN);
630    out.extend_from_slice(&TIMELINE_HEAD_MAGIC);
631    out.push(TIMELINE_FORMAT_VERSION);
632    out.extend_from_slice(root_id);
633    out.extend_from_slice(&count.to_be_bytes());
634    out
635}
636
637fn decode_head(bytes: &[u8]) -> Result<(NodeId, u64), StorageError> {
638    if bytes.len() != HEAD_LEN
639        || bytes[..4] != TIMELINE_HEAD_MAGIC
640        || bytes[4] != TIMELINE_FORMAT_VERSION
641    {
642        return Err(StorageError::Corrupt("timeline head".to_owned()));
643    }
644    let mut root_id = [0u8; 16];
645    root_id.copy_from_slice(&bytes[5..21]);
646    let count = u64::from_be_bytes(
647        bytes[21..HEAD_LEN]
648            .try_into()
649            .map_err(|_| StorageError::Corrupt("timeline head count".to_owned()))?,
650    );
651    Ok((root_id, count))
652}
