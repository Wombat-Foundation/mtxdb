14/// Per-database-instance tuning for [`LossyIndex`] construction.
15///
16/// `seed` is mixed into bucket *and* tag derivation to prevent adversarial
17/// precomputation against a fixed, unsalted content hash. Every real
18/// `PackfileStorage`-sourced config must carry the pool's actual persisted
19/// seed; the `Default` impl uses `seed: 0` only for tests and transient
20/// construction that never touches disk.
21#[derive(Debug, Clone, Copy)]
22pub struct IndexConfig {
23    /// Mixed into bucket and tag derivation to prevent adversarial
24    /// precomputation. See `index/mod.rs` seed docs.
25    pub seed: u64,
26    /// Minimum initial capacity for a newly created collection's index.
27    /// Composes with the hard 16-entry minimum: `effective_min = floor.max(16)`.
28    pub floor: u32,
29    /// Load factor (percent, 1..=90) at which the index grows.
30    pub load_factor_percent: u8,
31}
32
33impl Default for IndexConfig {
34    fn default() -> Self {
35        Self {
36            seed: 0,
37            floor: 64,
38            load_factor_percent: 75,
39        }
40    }
41}
42
43impl IndexConfig {
44    /// Validate the configuration.
45    ///
46    /// # Errors
47    /// Returns `Err(message)` if `load_factor_percent` is 0 or greater than
48    /// 90, or if `floor` is 0.
49    pub fn validate(&self) -> Result<(), &'static str> {
50        if self.load_factor_percent == 0 || self.load_factor_percent > 90 {
51            return Err("load_factor_percent must be in 1..=90");
52        }
53        if self.floor == 0 {
54            return Err("floor must be > 0");
55        }
56        Ok(())
57    }
58}

60/// Splitmix64-style avalanche finalizer. Cheap, well-distributed, no
61/// dependencies. Used to mix the per-pool seed into bucket and tag
62/// derivation so an attacker cannot precompute hash → bucket/tag mappings
63/// without knowing the seed.
64#[inline]
65fn mix(x: u64) -> u64 {
66    let mut v = x;
67    v = v.wrapping_add(0x9E37_79B9_7F4A_7C15);
68    v = (v ^ (v >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
69    v = (v ^ (v >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
70    v ^ (v >> 31)
71}

73/// Per-entry entry in the lossy fanout index.
74///
75/// Layout: `[16-bit tag | 16-bit slot | 32-bit offset]` packed into a `u64`.
76///
77/// - **tag** (high 16 bits): truncated fingerprint for fast rejection.
78/// - **`slot`** (next 16 bits): which shard file this record lives in. The
79///   field is a full 16 bits, but live shard IDs only ever reach
80///   `crate::shard::MAX_SHARDS` (4096), so the top 4 bits of this field are
81///   reserved — `IndexEntry::new` rejects a slot that is not a valid live ID.
82///   They are *not* reclaimed as extra offset bits: `slot()` still decodes
83///   all 16, and widening the offset would change the on-disk encoding.
84/// - **offset** (low 32 bits): byte offset within the shard, stored as
85///   `offset + 1` so that the all-zeros encoding is reserved as the empty
86///   sentinel. Actual offset 0 is stored as 1, and `offset()` subtracts 1
87///   to recover the real value.
88///
89/// Empty slots are all-zeros. This is safe regardless of tag value: the
90/// `offset + 1` encoding guarantees a live entry's low 32 bits are never zero,
91/// so the all-zeros pattern can never be produced by a real insert — the tag
92/// bits play no role in reserving the sentinel.
93#[derive(Debug, Clone, Copy, PartialEq, Eq)]
94pub struct IndexEntry(u64);
95
96impl IndexEntry {
97    const EMPTY: Self = Self(0);
98
99    /// Largest byte offset representable by the 32-bit offset field while
100    /// reserving the all-zeros encoding for the empty-entry sentinel.
101    pub const MAX_OFFSET: u64 = (1u64 << 32) - 2;
102
103    const TAG_SHIFT: u64 = 48; // SHARD_BITS + OFFSET_BITS
104    const SHARD_SHIFT: u64 = 32; // OFFSET_BITS
105    const OFFSET_MASK: u64 = 0xFFFF_FFFF;
106
107    /// Create a new entry from its components.
108    ///
109    /// The offset is stored as `offset + 1` so that the all-zeros `u64`
110    /// encoding is reserved as the empty sentinel. Actual offset 0 is
111    /// stored as 1 in the entry.
112    ///
113    /// # Panics
114    /// Panics if `tag` exceeds 16 bits, `slot` is not a valid live shard ID
115    /// (`>= crate::shard::MAX_SHARDS`), or `offset` exceeds
116    /// [`Self::MAX_OFFSET`].
117    #[must_use]
118    pub fn new(tag: u32, slot: u16, offset: u64) -> Self {
119        assert!(tag <= 0xFFFF, "tag must fit in 16 bits");
120        assert!(
121            usize::from(slot) < crate::shard::MAX_SHARDS,
122            "slot {slot} is not a valid live shard ID (must be < MAX_SHARDS)"
123        );
124        assert!(
125            offset <= Self::MAX_OFFSET,
126            "offset must fit in 32 bits minus 1 (reserved for empty sentinel)"
127        );
128        Self(
129            (u64::from(tag) << Self::TAG_SHIFT)
130                | (u64::from(slot) << Self::SHARD_SHIFT)
131                | ((offset.wrapping_add(1)) & Self::OFFSET_MASK),
132        )
133    }
134
135    /// The sentinel value representing an empty entry.
136    #[must_use]
137    pub fn empty() -> Self {
138        Self::EMPTY
139    }
140
141    /// Returns `true` if this entry is the empty sentinel.
142    #[must_use]
143    pub fn is_empty(self) -> bool {
144        self.0 == 0
145    }
146
147    /// The 16-bit tag stored in this entry.
148    #[must_use]
149    pub fn tag(self) -> u32 {
150        ((self.0 >> Self::TAG_SHIFT) & 0xFFFF) as u32
151    }
152
153    /// The 16-bit shard ID stored in this entry.
154    #[must_use]
155    pub fn slot(self) -> u16 {
156        ((self.0 >> Self::SHARD_SHIFT) & 0xFFFF) as u16
157    }
158
159    /// The byte offset within the shard stored in this entry.
160    #[must_use]
161    pub fn offset(self) -> u64 {
162        (self.0 & Self::OFFSET_MASK).wrapping_sub(1)
163    }
164}

166/// A per-collection lossy fanout index for content-addressed records.
167///
168/// Uses open addressing with linear probing on a power-of-two sized table.
169/// The index is mmap-able (flat `u64` array) and small enough to stay
170/// resident in RAM for active collections.
171///
172/// Design decisions (from docs):
173/// - Partition by collection: ~8KB per 1000-node collection, 100 active collections < 1MB.
174/// - Empty entry terminates probe (write-once, no tombstones needed).
175/// - Tag collisions coexist in the probe chain; a tag is only a candidate
176///   filter and never an overwrite/equality proof.
177/// - Owned indexes keep `homes` and `tails` identity side tables: two
178///   anonymous `u64`s (16 bytes) per entry beyond the packed entry array.
179///   Checkpoints retain only packed slots; identities are hydrated lazily
180///   from authoritative frame headers when a writer needs them.
181/// - Power-of-two capacity: shift-and-mask bucket selection, cache-aligned probes.
182#[derive(Debug)]
183pub struct LossyIndex {
184    /// Power-of-two capacity.
185    capacity: u32,
186    /// Bitmask for bucket selection: capacity - 1.
187    mask: u32,
188    /// Shift to extract top bits from hash for bucket index.
189    shift: u32,
190    /// Per-instance config: seed, floor, load factor.
191    config: IndexConfig,
192    /// The flat entry array. Checkpoint-loaded indexes borrow the immutable
193    /// raw slots from their mmap until their first write, which clones them
194    /// into the owned atomic representation.
195    slots: EntryStorage,
196    /// The first 64 bits of each entry's hash. Index slots intentionally pack
197    /// only a tag, shard, and offset; retaining the home hash separately lets
198    /// a live writer grow the table without rescanning the packfiles.
199    homes: Mutex<Vec<u64>>,
200    /// The remaining 40 bits of each live entry's hash, plus a high-bit
201    /// marker that says the identity is known. Together with `homes` and the
202    /// packed tag this makes overwrite equality exact without putting
203    /// full hashes in the checkpoint.
204    tails: Mutex<Vec<u64>>,
205    /// An index with incomplete identity tables remains readable, but must be
206    /// rebuilt from packfiles if it ever needs to grow. Hydrated checkpoint
207    /// indexes are growable without that pack scan.
208    can_grow: bool,
209    /// Number of occupied slots.
210    len: AtomicU32,
211    /// Longest linear-probe chain walked by any insert or lookup against
212    /// this index instance so far (never reset, monotonically
213    /// non-decreasing). Pure observability — insert has no probe-length
214    /// cap (see the doc on `InsertError::TableFull`: growth is what keeps
215    /// chains short, not a cap), so this exists to let an operator notice
216    /// a probe chain growing unexpectedly (e.g. an unseeded/misconfigured
217    /// deployment under adversarial content) without changing behavior.
218    max_probe_len: AtomicU32,
219}
220
221#[derive(Debug)]
222enum EntryStorage {
223    Owned(Vec<AtomicU64>),
224    Mmap { mmap: Arc<Mmap>, offset: usize },
225}
226
227impl EntryStorage {
228    #[inline]
229    fn get(&self, index: usize) -> u64 {
230        match self {
231            Self::Owned(slots) => slots[index].load(Ordering::Acquire),
232            Self::Mmap { mmap, offset } => {
233                let start = offset.saturating_add(index.saturating_mul(8));
234                let bytes: [u8; 8] = mmap[start..start.saturating_add(8)]
235                    .try_into()
236                    .expect("validated checkpoint entry range");
237                u64::from_le_bytes(bytes)
238            }
239        }
240    }
241
242    fn materialize(&self, capacity: usize) -> Vec<AtomicU64> {
243        (0..capacity)
244            .map(|index| AtomicU64::new(self.get(index)))
245            .collect()
246    }
247}

249impl LossyIndex {
250    /// Whether the persisted identity tables are sufficient to rehash every
251    /// occupied slot without consulting the packs. The probe-chain check is
252    /// intentional: a corrupted home value could otherwise make `grow()` lose
253    /// an entry while still looking superficially hydrated.
254    fn identity_tables_complete(
255        capacity: u32,
256        shift: u32,
257        slots: &EntryStorage,
258        homes: &[u64],
259        tails: &[u64],
260    ) -> bool {
261        let cap = capacity as usize;
262        if homes.len() != cap || tails.len() != cap {
263            return false;
264        }
265        let mask = usize::try_from(capacity.wrapping_sub(1)).unwrap_or(usize::MAX);
266        (0..cap).all(|bucket| {
267            if IndexEntry(slots.get(bucket)).is_empty() {
268                return true;
269            }
270            if tails[bucket] >> 63 == 0 {
271                return false;
272            }
273            let mut probe =
274                usize::try_from((homes[bucket] >> shift) & u64::from(capacity.wrapping_sub(1)))
275                    .unwrap_or(usize::MAX);
276            while probe != bucket {
277                if IndexEntry(slots.get(probe)).is_empty() {
278                    return false;
279                }
280                probe = probe.wrapping_add(1) & mask;
281            }
282            true
283        })
284    }
285
286    /// [`Self::identity_tables_complete`] over this index's current tables.
287    fn identities_now_complete(&self) -> bool {
288        let homes = self.homes.lock();
289        let tails = self.tails.lock();
290        Self::identity_tables_complete(self.capacity, self.shift, &self.slots, &homes, &tails)
291    }
292
293    /// A copy of an identity side table, or an all-zero (unhydrated) one of
294    /// `capacity` entries if the source has none.
295    fn identity_table(table: &Mutex<Vec<u64>>, capacity: u32) -> Vec<u64> {
296        let table = table.lock();
297        if table.len() == capacity as usize {
298            table.clone()
299        } else {
300            vec![0; capacity as usize]
301        }
302    }
303}
304
305impl Clone for LossyIndex {
306    fn clone(&self) -> Self {
307        Self {
308            capacity: self.capacity,
309            mask: self.mask,
310            shift: self.shift,
311            config: self.config,
312            slots: EntryStorage::Owned(self.slots.materialize(self.capacity as usize)),
313            // Keep whatever identity tables the source holds. A mapping loaded
314            // with its persisted `homes`/`tails` carries hydrated identities, and
315            // dropping them here made every entry lose its identity on the first
316            // write after a reopen (the next checkpoint then persisted zeros). A
317            // mapping loaded without tables has none, so it gets zeroed ones.
318            // Preserve the source's completeness gate: a hydrated checkpoint
319            // remains growable after materialization, while an incomplete one
320            // still falls back to a pack rebuild at a capacity boundary.
321            homes: Mutex::new(Self::identity_table(&self.homes, self.capacity)),
322            tails: Mutex::new(Self::identity_table(&self.tails, self.capacity)),
323            can_grow: self.can_grow,
324            len: AtomicU32::new(self.len.load(Ordering::Acquire)),
325            max_probe_len: AtomicU32::new(self.max_probe_len.load(Ordering::Relaxed)),
326        }
327    }
328}

333impl LossyIndex {
334    /// Create a new index with at least `min_capacity` slots using default config.
335    /// Capacity is rounded up to the next power of two.
336    /// The floor is set to 0 so `new()` preserves backward compatibility
337    /// (real callers should use `with_config` with an explicit floor).
338    ///
339    /// # Panics
340    /// Panics if `min_capacity` exceeds `u32::MAX` when rounded to a power of two.
341    #[must_use]
342    pub fn new(min_capacity: usize) -> Self {
343        Self::with_config(
344            min_capacity,
345            IndexConfig {
346                floor: 1,
347                ..IndexConfig::default()
348            },
349        )
350    }

352    /// Create a new index with at least `min_capacity` slots using the given config.
353    /// Capacity is rounded up to the next power of two. The configured `floor`
354    /// composes with the hard 16-entry minimum: `effective_min = floor.max(16)`.
355    ///
356    /// # Panics
357    /// Panics if `min_capacity` exceeds `u32::MAX` when rounded to a power of two,
358    /// or if `config` fails validation.
359    #[must_use]
360    pub fn with_config(min_capacity: usize, config: IndexConfig) -> Self {
361        config.validate().expect("invalid IndexConfig");
362        let effective_min = (config.floor.max(16)) as usize;
363        let capacity_usize = min_capacity.max(effective_min).next_power_of_two();
364        let capacity = u32::try_from(capacity_usize).expect("index capacity exceeds u32::MAX");
365        let shift = 64_u32.wrapping_sub(capacity.trailing_zeros());
366        Self {
367            mask: capacity.wrapping_sub(1),
368            capacity,
369            shift,
370            config,
371            slots: EntryStorage::Owned((0..capacity_usize).map(|_| AtomicU64::new(0)).collect()),
372            homes: Mutex::new(vec![0; capacity_usize]),
373            tails: Mutex::new(vec![0; capacity_usize]),
374            can_grow: true,
375            len: AtomicU32::new(0),
376            max_probe_len: AtomicU32::new(0),
377        }
378    }
379
380    /// This index's entry-table capacity (always a power of two).
381    #[must_use]
382    pub(crate) fn capacity(&self) -> u32 {
383        self.capacity
384    }
385
386    /// Longest linear-probe chain observed so far by any insert or lookup
387    /// against this index instance. Pure observability, never reset, never
388    /// influences control flow — see the field doc on `max_probe_len`.
389    #[must_use]
390    pub fn max_probe_len(&self) -> u32 {
391        self.max_probe_len.load(Ordering::Relaxed)
392    }
393
394    /// Record a probe chain of `len` steps as the new high-water mark if it
395    /// exceeds the current one. `Relaxed` is enough: this is an
396    /// observability counter with no ordering dependency on any other
397    /// field, and a lost update under concurrent writers only means a
398    /// transient undercount, never a wrong value that persists.
399    #[inline]
400    fn bump_max_probe_len(&self, len: u32) {
401        self.max_probe_len.fetch_max(len, Ordering::Relaxed);
402    }
403
404    /// Extract the bucket index from a 16-byte hash.
405    ///
406    /// The seed is `XORed` into the raw home value before the avalanche mix,
407    /// so the same content hash lands in different buckets across pools
408    /// with different seeds. The mixed value is what gets stored in `homes`,
409    /// so `grow()` (which rehashes from stored homes) sees the already-mixed
410    /// value and doesn't need to know the seed at all.
411    #[inline]
412    fn bucket(&self, hash: &[u8; 16]) -> usize {
413        let raw = u64::from_be_bytes(hash[..8].try_into().unwrap());
414        let home = mix(raw ^ self.config.seed);
415        self.bucket_for_home(home)
416    }
417
418    /// Mix a raw home value with this index's seed. Used by insert to store
419    /// the post-mix value in `homes`, and by `hydrate_slot_identity` to
420    /// reproduce the same home from a recovered hash.
421    #[inline]
422    fn mix_home(&self, raw_home: u64) -> u64 {
423        mix(raw_home ^ self.config.seed)
424    }
425
426    #[inline]
427    fn bucket_for_home(&self, home: u64) -> usize {
428        let masked = (home >> self.shift) & u64::from(self.mask);
429        // Safety: masked is always <= mask < capacity which fits in usize on all platforms
430        usize::try_from(masked).unwrap_or(usize::MAX)
431    }
432
433    /// Extract the 16-bit tag from a 16-byte hash, seeded with this index's seed.
434    ///
435    /// Uses bytes 8..12, disjoint from the bytes `bucket()` reads (0..8).
436    /// Seeding the tag prevents an attacker from grinding offline for
437    /// tag collisions against a known target bucket — without the seed,
438    /// a narrower 16-bit tag would make I/O amplification attacks feasible.
439    #[inline]
440    fn tag(&self, hash: &[u8; 16]) -> u32 {
441        let raw = u32::from_be_bytes(hash[8..12].try_into().unwrap());
442        let mixed = mix(u64::from(raw) ^ self.config.seed);
443        (mixed >> 48) as u32
444    }
445
446    /// The packed-entry tag for `hash`, exposed to crate-local recovery code
447    /// that must validate a persisted entry against its authoritative frame.
448    ///
449    /// Must use this index's own seed: every stored entry's tag was computed
450    /// via `self.tag(hash)` at insert time, so a recovery comparison against
451    /// an unseeded (or differently-seeded) tag would fail closed on every
452    /// occupied entry, not just colliding ones.
453    #[inline]
454    pub(crate) fn tag_for_hash(&self, hash: &[u8; 16]) -> u32 {
455        self.tag(hash)
456    }
457
458    #[inline]
459    fn tail(hash: &[u8; 16]) -> u64 {
460        const KNOWN: u64 = 1_u64 << 63;
461        let mut bytes = [0; 8];
462        bytes[3..].copy_from_slice(&hash[11..]);
463        KNOWN | u64::from_be_bytes(bytes)
464    }

466    /// Insert a (hash → `slot`, offset) mapping.
467    ///
468    /// # Errors
469    /// Returns `InsertError::TableFull` if the table has less than 25% free slots
470    /// and the hash is not already present (overwrites are always allowed).
471    pub fn insert(&self, hash: &[u8; 16], slot: u16, offset: u64) -> Result<(), InsertError> {
472        self.insert_tracked(hash, slot, offset).map(|_| ())
473    }
474
475    /// Like [`Self::insert`], but also reports the write's exact landing spot —
476    /// the affected bucket and the packed entry value stored there — so the
477    /// delta persistence layer can append a replay frame without re-probing.
478    ///
479    /// # Errors
480    /// Returns `InsertError::TableFull` under the same conditions as
481    /// [`Self::insert`] (the table is left unmodified).
482    // `bucket` is always below `self.capacity`, which is capped at `u32::MAX`
483    // by construction, so the checked `u32::try_from` in `insert_undoable`
484    // below can never fail.
485    pub fn insert_tracked(
486        &self,
487        hash: &[u8; 16],
488        slot: u16,
489        offset: u64,
490    ) -> Result<(u32, u64), InsertError> {
491        self.insert_undoable(hash, slot, offset)
492            .map(|(bucket, value, _undo)| (bucket, value))
493    }
494
495    /// Like [`Self::insert_tracked`], but also returns a [`EntryUndo`]
496    /// capturing the entry's prior contents, so a caller that needs to
497    /// mutate the live (not cloned) index for a multi-record batch can
498    /// undo this one write if a later entry in the batch fails.
499    ///
500    /// # Errors
501    /// Returns the same errors as [`Self::insert_tracked`] and leaves the
502    /// table unmodified in every error case.
503    ///
504    /// # Panics
505    /// Never panics in practice: `hash[..8]` is always exactly 8 bytes for a
506    /// `&[u8; 16]` input, so the `try_into` this performs cannot fail.
507    pub fn insert_undoable(
508        &self,
509        hash: &[u8; 16],
510        slot: u16,
511        offset: u64,
512    ) -> Result<(u32, u64, EntryUndo), InsertError> {
513        let tag = self.tag(hash);
514        let raw_home = u64::from_be_bytes(hash[..8].try_into().unwrap());
515        let home = self.mix_home(raw_home);
516        let mut bucket = self.bucket_for_home(home);
517        let mut probe_len: u32 = 0;
518
519        loop {
520            // `bucket` is masked by `self.mask` (capacity - 1) and capacity is
521            // a u32, so the conversion is lossless; it is done once here so the
522            // return/undo structs below don't each need a checked cast.
523            let bucket_wire = u32::try_from(bucket)
524                .expect("bucket is masked by the u32 capacity, so it always fits");
525            let EntryStorage::Owned(slots) = &self.slots else {
526                self.bump_max_probe_len(probe_len);
527                return Err(InsertError::TableFull);
528            };
529            let entry = IndexEntry(slots[bucket].load(Ordering::Acquire));
530            if entry.is_empty() {
531                // Only reject when actually inserting into a new entry
532                let threshold = self
533                    .capacity
534                    .wrapping_mul(u32::from(self.config.load_factor_percent))
535                    / 100;
536                if self.len.load(Ordering::Relaxed) >= threshold {
537                    self.bump_max_probe_len(probe_len);
538                    return Err(InsertError::TableFull);
539                }
540                let old_home = self.homes.lock()[bucket];
541                let old_tail = self.tails.lock()[bucket];
542                // Writers are serialized by the collection put lock. Publish
543                // the entry last so concurrent readers see either the prior
544                // empty entry or a complete record location.
545                self.homes.lock()[bucket] = home;
546                self.tails.lock()[bucket] = Self::tail(hash);
547                let value = IndexEntry::new(tag, slot, offset).0;
548                slots[bucket].store(value, Ordering::Release);
549                self.len.fetch_add(1, Ordering::Relaxed);
550                let undo = EntryUndo {
551                    bucket: bucket_wire,
552                    old_entry: 0,
553                    old_home,
554                    old_tail,
555                    was_empty: true,
556                };
557                self.bump_max_probe_len(probe_len);
558                return Ok((bucket_wire, value, undo));
559            }
560            // A tag only filters candidates; it never proves key equality.
561            // Checkpoint-derived slots initially lack their 40-bit tail and
562            // ask storage to hydrate it from the frame before deciding.
563            if entry.tag() == tag {
564                let tail = self.tails.lock()[bucket];
565                if tail == 0 {
566                    self.bump_max_probe_len(probe_len);
567                    return Err(InsertError::NeedsIdentity {
568                        bucket: bucket_wire,
569                        slot: entry.slot(),
570                        offset: entry.offset(),
571                    });
572                }
573                if self.homes.lock()[bucket] == home && tail == Self::tail(hash) {
574                    let old_entry = entry.0;
575                    let old_home = self.homes.lock()[bucket];
576                    let value = IndexEntry::new(tag, slot, offset).0;
577                    slots[bucket].store(value, Ordering::Release);
578                    let undo = EntryUndo {
579                        bucket: bucket_wire,
580                        old_entry,
581                        old_home,
582                        old_tail: tail,
583                        was_empty: false,
584                    };
585                    self.bump_max_probe_len(probe_len);
586                    return Ok((bucket_wire, value, undo));
587                }
588            }
589            probe_len = probe_len.saturating_add(1);
590            bucket = bucket.wrapping_add(1) & self.mask as usize;
591        }
592    }

594    /// Reverse a write reported by [`Self::insert_undoable`]. Callers must
595    /// replay a batch's undo entries in the reverse order they were
596    /// produced, and only against the same live index that produced them
597    /// (never a cloned/owned index materialized after the fact).
598    ///
599    /// # Panics
600    /// Never panics; a no-op if called against a checkpoint-mmap-backed
601    /// index, since [`Self::insert_undoable`] cannot have produced an undo
602    /// entry against one (it errors with `TableFull` first).
603    pub fn rollback_slot(&self, undo: &EntryUndo) {
604        let EntryStorage::Owned(slots) = &self.slots else {
605            return;
606        };
607        let bucket = undo.bucket as usize;
608        self.homes.lock()[bucket] = undo.old_home;
609        self.tails.lock()[bucket] = undo.old_tail;
610        slots[bucket].store(undo.old_entry, Ordering::Release);
611        if undo.was_empty {
612            self.len.fetch_sub(1, Ordering::Relaxed);
613        }
614    }

616    /// Mark a checkpoint-derived entry with the full identity recovered from
617    /// its authoritative frame. Returns `false` if the entry changed or the
618    /// supplied hash cannot describe that entry.
619    pub(crate) fn hydrate_slot_identity(&self, bucket: u32, hash: &[u8; 16]) -> bool {
620        let bucket = bucket as usize;
621        if bucket >= self.capacity as usize {
622            return false;
623        }
624        let entry = IndexEntry(self.entry_at(bucket));
625        if entry.is_empty() || entry.tag() != self.tag(hash) {
626            return false;
627        }
628        let raw_home = u64::from_be_bytes(hash[..8].try_into().expect("16-byte hash"));
629        let home = self.mix_home(raw_home);
630        self.homes.lock()[bucket] = home;
631        self.tails.lock()[bucket] = Self::tail(hash);
632        true
633    }

635    /// Apply recorded delta frames on top of a checkpoint-backed index,
636    /// replicating the live insert/overwrite sequence that produced them.
637    ///
638    /// Only valid on an owned (materialized) index — callers must clone an
639    /// mmap-backed index first. Frames are applied exactly as the live
640    /// session wrote them: an overwrite of a non-empty entry leaves the length
641    /// unchanged, a write into an empty entry increments it. The index's `len`
642    /// after replay therefore equals the checkpoint's occupancy plus the
643    /// number of fresh-entry writes, which is what the sidecar integrity sum
644    /// checks against.
645    ///
646    /// # Errors
647    /// Returns `DeltaReplayError` if a frame targets a bucket outside this
648    /// index's capacity, or carries the empty-entry sentinel (either is a
649    /// structurally inconsistent log, rejected wholesale), or the index is
650    /// still mmap-backed.
651    pub fn replay_frames(&self, frames: &[DeltaFrame]) -> Result<(), DeltaReplayError> {
652        let EntryStorage::Owned(slots) = &self.slots else {
653            return Err(DeltaReplayError::RequiresOwnedIndex);
654        };
655        // Validate every frame before mutating any entry: the contract is
656        // wholesale rejection of a structurally inconsistent log, so a
657        // caller must never observe a partially replayed index.
658        for frame in frames {
659            let bucket = frame.bucket as usize;
660            if bucket >= self.capacity as usize {
661                return Err(DeltaReplayError::FrameOutOfBounds {
662                    bucket: frame.bucket,
663                    capacity: self.capacity,
664                });
665            }
666            if frame.slot == 0 {
667                // A legitimate insert never logs the empty sentinel (see
668                // `record_delta`'s only caller). Storing it here would erase
669                // whatever live entry currently occupies this bucket and cut
670                // the probe chain short for anything beyond it — reject the
671                // whole log instead of silently losing data.
672                return Err(DeltaReplayError::EmptySlot {
673                    bucket: frame.bucket,
674                });
675            }
676            // The packed entry's slot field must name a live shard. A writer
677            // only ever encodes a slot from its own `ShardPool` (bounded by
678            // `MAX_SHARDS`; see `IndexEntry::new`), so a value at or above the
679            // cap can only be log corruption or a structurally invalid frame.
680            // Accepting it would install an entry no shard scan can resolve —
681            // `referenced_slot_ids` and friends skip `id >= MAX_SHARDS` — so
682            // the record would silently vanish from recovery.
683            let decoded = IndexEntry(frame.slot);
684            if usize::from(decoded.slot()) >= crate::shard::MAX_SHARDS {
685                return Err(DeltaReplayError::SlotOutOfRange {
686                    bucket: frame.bucket,
687                    slot: decoded.slot(),
688                });
689            }
690        }
691        for frame in frames {
692            let bucket = frame.bucket as usize;
693            let previous = slots[bucket].load(Ordering::Acquire);
694            if previous == 0 {
695                self.len.fetch_add(1, Ordering::Relaxed);
696            }
697            slots[bucket].store(frame.slot, Ordering::Release);
698        }
699        Ok(())
700    }

702    /// Double this live index's capacity without reading packfiles.
703    ///
704    /// Returns `false` when the identity side tables are incomplete or the
705    /// index cannot grow further. Complete checkpoint identities can rehash
706    /// without reading packfiles; older/incomplete checkpoints retain the
707    /// packfile rebuild fallback.
708    pub fn grow(&self) -> Option<Self> {
709        // The gate is checked when growth is used, not only when the index was
710        // loaded: delta replay stores slot values straight into the table without
711        // identities, so a table that was complete at load can be incomplete by
712        // the time it has to grow.
713        if !self.can_grow || self.capacity > u32::MAX / 2 || !self.identities_now_complete() {
714            return None;
715        }
716        let grown = Self::with_config((self.capacity as usize).saturating_mul(2), self.config);
717        let homes = self.homes.lock();
718        let tails = self.tails.lock();
719        for (index, home) in homes.iter().copied().enumerate() {
720            let entry = IndexEntry(self.slots.get(index));
721            if entry.is_empty() {
722                continue;
723            }
724            let mut bucket = grown.bucket_for_home(home);
725            while !IndexEntry(grown.entry_at(bucket)).is_empty() {
726                bucket = bucket.wrapping_add(1) & grown.mask as usize;
727            }
728            grown.homes.lock()[bucket] = home;
729            grown.tails.lock()[bucket] = tails[index];
730            let EntryStorage::Owned(slots) = &grown.slots else {
731                unreachable!("new index is owned")
732            };
733            slots[bucket].store(entry.0, Ordering::Relaxed);
734            grown.len.fetch_add(1, Ordering::Relaxed);
735        }
736        Some(grown)
737    }

739    /// Double the table by recovering each occupied entry's full hash from
740    /// its stored location.
741    ///
742    /// Checkpoint-backed indexes deliberately omit `homes`, so their first
743    /// post-restart resize cannot use [`Self::grow`].  The packed slots still
744    /// retain every `(slot, offset)`, however.  A caller can therefore
745    /// recover only the `len` source hashes from the authoritative records
746    /// and rehash without rescanning every pack in the store. Locations are
747    /// visited in `(shard, offset)` order rather than hash-table order, so a
748    /// cold mmap walk follows pack order and lets the kernel coalesce page
749    /// faults/read-ahead. The callback must return the hash for precisely the
750    /// supplied location.
751    ///
752    /// `Ok(None)` means the table cannot grow further; callers retain their
753    /// full-rebuild fallback for that terminal case.
754    pub(crate) fn grow_by_recovering_hashes<E>(
755        &self,
756        mut hash_at: impl FnMut(u16, u64, u32) -> Result<[u8; 16], E>,
757    ) -> Result<Option<Self>, E> {
758        if self.capacity > u32::MAX / 2 {
759            return Ok(None);
760        }
761
762        let mut locations = Vec::with_capacity(self.len());
763        for index in 0..self.capacity as usize {
764            let entry = IndexEntry(self.entry_at(index));
765            if entry.is_empty() {
766                continue;
767            }
768            locations.push((entry.slot(), entry.offset(), entry.tag()));
769        }
770        locations.sort_unstable_by_key(|(slot, offset, _)| (*slot, *offset));
771
772        let grown = Self::with_config((self.capacity as usize).saturating_mul(2), self.config);
773        for (slot, offset, slot_tag) in locations {
774            let hash = hash_at(slot, offset, slot_tag)?;
775            // A doubled table is at most 37.5% full because insertion only
776            // requests growth at 75%, so this cannot hit TableFull.
777            let _ = grown.insert(&hash, slot, offset);
778        }
779        Ok(Some(grown))
780    }

783    /// Returns `(slot, offset)` if found, `None` if absent.
784    ///
785    /// An empty entry terminates the probe — this is correct because:
786    /// 1. Class A tables are write-once with no deletes (no tombstones needed).
787    /// 2. An empty entry means the key was never inserted.
788    #[inline]
789    #[must_use]
790    pub fn lookup(&self, hash: &[u8; 16]) -> Option<(u16, u64)> {

792    }

794    /// Look up all candidate offsets for a hash, yielding tag collisions.
795    ///
796    /// Candidates with hydrated in-memory identity (home + tail) are verified
797    /// before yielding, so tag collisions in a live/warm index are filtered
798    /// out here rather than forcing the caller to disk. For a entry whose
799    /// identity is not yet hydrated (e.g. a fresh checkpoint restore), the
800    /// caller **must** verify the candidate against the caller-requested
801    /// hash — a raw tag collision (~1/65536 at 16-bit tags) can still surface
802    /// unverified.
803    ///
804    /// Yields `(slot, offset)` for each entry whose tag matches and isn't
805    /// ruled out, then terminates at the first empty entry or after
806    /// `capacity` probes.
807    ///
808    /// # Panics
809    /// Never panics in practice: `hash[..8]` is always exactly 8 bytes for a
810    /// `&[u8; 16]` input, so the `try_into` this performs cannot fail.
811    #[inline]
812    #[must_use]
813    pub fn lookup_all(&self, hash: &[u8; 16]) -> LookupIter<'_> {
814        let tag = self.tag(hash);
815        let bucket = self.bucket(hash);
816        let raw_home = u64::from_be_bytes(hash[..8].try_into().unwrap());
817        let mixed_home = self.mix_home(raw_home);
818        let expected_tail = Self::tail(hash);
819        LookupIter {
820            index: self,
821            tag,
822            mixed_home,
823            expected_tail,
824            bucket,
825            mask: self.mask as usize,
826            remaining: self.capacity as usize,
827            steps: 0,
828        }
829    }

936    /// Rewrites every occupied entry's `slot` through `remap`, in place.
937    ///
938    /// A checkpoint-loaded index's `slot`s were encoded by the writer's
939    /// own local `ShardPool` slots at checkpoint-write time — a process-local
940    /// handle, not a stable identity. `remap` (built by the caller from the
941    /// checkpoint's persisted pack table, translated through this reader's
942    /// own currently-open packs) reconciles that against this reader's own
943    /// entry numbering, which can differ once any shard has ever been retired
944    /// (see `CHECKPOINT_VERSION`'s v6 doc comment).
945    ///
946    /// Checks first without touching storage: the common case (no shard has
947    /// ever been retired, so a fresh reader's entry numbering already agrees
948    /// with the checkpoint's) needs no rewrite at all, and mmap-backed
949    /// indexes must stay mmap-backed when nothing actually changes — callers
950    /// and tests depend on an unmodified checkpoint-loaded index staying
951    /// zero-copy until its first real write. Only materializes into an owned
952    /// copy (writes are never applied to the mmap) when at least one
953    /// occupied entry's `slot` actually needs to change.
954    ///
955    /// Returns `false`, leaving the index unmodified, if any occupied entry's
956    /// `slot` has no entry in `remap` — the checkpoint's pack table
957    /// should cover every `slot` any of its own index entries use, so a
958    /// miss means the checkpoint is internally inconsistent and the caller
959    /// should fall back to a full rescan rather than serve unresolvable
960    /// slots.
961    #[must_use]
962    pub fn remap_slots(&mut self, remap: &std::collections::HashMap<u16, u16>) -> bool {
963        let capacity = self.capacity as usize;
964        let mut needs_rewrite = false;
965        for index in 0..capacity {
966            let entry = IndexEntry(self.slots.get(index));
967            if entry.is_empty() {
968                continue;
969            }
970            let Some(&new_slot) = remap.get(&entry.slot()) else {
971                return false;
972            };
973            if new_slot != entry.slot() {
974                needs_rewrite = true;
975            }
976        }
977        if !needs_rewrite {
978            return true;
979        }
980        if !matches!(self.slots, EntryStorage::Owned(_)) {
981            self.slots = EntryStorage::Owned(self.slots.materialize(capacity));
982        }
983        let EntryStorage::Owned(slots) = &self.slots else {
984            unreachable!("just materialized to Owned above");
985        };
986        for cell in slots {
987            let raw = cell.load(Ordering::Acquire);
988            let entry = IndexEntry(raw);
989            if entry.is_empty() {
990                continue;
991            }
992            let Some(&new_slot) = remap.get(&entry.slot()) else {
993                return false;
994            };
995            if new_slot == entry.slot() {
996                continue;
997            }
998            let remapped = IndexEntry::new(entry.tag(), new_slot, entry.offset()).0;
999            cell.store(remapped, Ordering::Relaxed);
1000        }
1001        true
1002    }

1013    /// Serialize the index to bytes for persistence.
1014    ///
1015    /// Format (all little-endian):
1016    /// ```text
1017    ///   [capacity: u64]
1018    ///   [slot_0: u64] ... [slot_{cap-1}: u64]     ← capacity × 8 B
1019    ///   [home_0: u64] ... [home_{cap-1}: u64]     ← capacity × 8 B (0 if unhydrated)
1020    ///   [tail_0: u64] ... [tail_{cap-1}: u64]     ← capacity × 8 B (0 if unhydrated)
1021    /// ```
1022    /// Total: `8 + capacity * 24` bytes.
1023    #[must_use]
1024    pub fn serialize(&self) -> Vec<u8> {
1025        let cap = self.capacity as usize;
1026        let byte_len = 8_usize.wrapping_add(cap.wrapping_mul(24));
1027        let mut buf = Vec::with_capacity(byte_len);
1028        buf.extend_from_slice(&u64::from(self.capacity).to_le_bytes());
1029        for index in 0..cap {
1030            let entry = IndexEntry(self.entry_at(index));
1031            buf.extend_from_slice(&entry.0.to_le_bytes());
1032        }
1033        let homes = self.homes.lock();
1034        let tails = self.tails.lock();
1035        for index in 0..cap {
1036            buf.extend_from_slice(&homes[index].to_le_bytes());
1037        }
1038        for index in 0..cap {
1039            buf.extend_from_slice(&tails[index].to_le_bytes());
1040        }
1041        buf
1042    }

1057    /// Deserialize an index from bytes with a given config (providing seed).
1058    ///
1059    /// Accepts both the legacy slots-only format (`8 + cap*8` bytes) and the
1060    /// v4 slots+homes+tails format (`8 + cap*24` bytes). Homes and tails are
1061    /// populated from the blob when present; zeroed otherwise.
1062    ///
1063    /// # Errors
1064    /// Returns `DeserializationError::TooShort` if the data is too short,
1065    /// or `DeserializationError::InvalidCapacity` if the capacity is not
1066    /// a power of two or is less than 16.
1067    ///
1068    /// # Panics
1069    /// Panics if the 8-byte capacity header cannot be read (guaranteed by the length check).
1070    pub fn deserialize_with_config(
1071        data: &[u8],
1072        config: IndexConfig,
1073    ) -> Result<Self, DeserializationError> {
1074        if data.len() < 8 {
1075            return Err(DeserializationError::TooShort);
1076        }
1077        let capacity_wire = u64::from_le_bytes(data[..8].try_into().unwrap());
1078        let capacity = u32::try_from(capacity_wire).map_err(|_| DeserializationError::TooShort)?;
1079        let capacity_usize = capacity as usize;
1080
1081        // Reject unsupported capacities: must be power of two and >= 16
1082        if capacity < 16 || !capacity.is_power_of_two() {
1083            return Err(DeserializationError::InvalidCapacity);
1084        }
1085
1086        let slots_len = 8_usize.wrapping_add(capacity_usize.wrapping_mul(8));
1087        let full_len = 8_usize.wrapping_add(capacity_usize.wrapping_mul(24));
1088        if data.len() < slots_len {
1089            return Err(DeserializationError::TooShort);
1090        }
1091        let has_homes_tails = data.len() >= full_len;
1092
1093        let mut slots = Vec::with_capacity(capacity_usize);
1094        let mut len: u32 = 0;
1095        for i in 0..capacity_usize {
1096            let offset = 8_usize.wrapping_add(i.wrapping_mul(8));
1097            let val = u64::from_le_bytes(data[offset..offset.wrapping_add(8)].try_into().unwrap());
1098            let entry = IndexEntry(val);
1099            if !entry.is_empty() {
1100                len = len.wrapping_add(1);
1101            }
1102            slots.push(AtomicU64::new(entry.0));
1103        }
1104
1105        let shift = 64_u32.wrapping_sub(capacity.trailing_zeros());
1106
1107        let homes = if has_homes_tails {
1108            let base = slots_len;
1109            let mut h = Vec::with_capacity(capacity_usize);
1110            for i in 0..capacity_usize {
1111                let offset = base.wrapping_add(i.wrapping_mul(8));
1112                h.push(u64::from_le_bytes(
1113                    data[offset..offset.wrapping_add(8)].try_into().unwrap(),
1114                ));
1115            }
1116            h
1117        } else {
1118            vec![0; capacity_usize]
1119        };
1120
1121        let tails = if has_homes_tails {
1122            let base = slots_len.wrapping_add(capacity_usize.wrapping_mul(8));
1123            let mut t = Vec::with_capacity(capacity_usize);
1124            for i in 0..capacity_usize {
1125                let offset = base.wrapping_add(i.wrapping_mul(8));
1126                t.push(u64::from_le_bytes(
1127                    data[offset..offset.wrapping_add(8)].try_into().unwrap(),
1128                ));
1129            }
1130            t
1131        } else {
1132            vec![0; capacity_usize]
1133        };
1134
1135        let slots = EntryStorage::Owned(slots);
1136        let can_grow = Self::identity_tables_complete(capacity, shift, &slots, &homes, &tails);
1137        Ok(Self {
1138            mask: capacity.wrapping_sub(1),
1139            capacity,
1140            shift,
1141            config,
1142            slots,
1143            homes: Mutex::new(homes),
1144            tails: Mutex::new(tails),
1145            can_grow,
1146            len: AtomicU32::new(len),
1147            max_probe_len: AtomicU32::new(0),

1184    /// Build a read-only index directly over a validated checkpoint's raw
1185    /// slots with a given config, loading pre-hydrated homes and tails from
1186    /// the checkpoint mmap. This eliminates cold-start packfile reads for
1187    /// tag-collision verification when the checkpoint carries identity data.
1188    ///
1189    /// The slots remain mmap-backed; homes and tails are copied into owned
1190    /// `Vec<u64>` so they can be updated by `hydrate_slot_identity` if needed.
1191    ///
1192    /// # Panics
1193    /// Panics if `[homes_offset, homes_offset + capacity*8)` or
1194    /// `[tails_offset, tails_offset + capacity*8)` fall outside `mmap` —
1195    /// callers must only pass offsets validated against a checkpoint's own
1196    /// declared body length (see `checkpoint::read_checkpoint`).
1197    #[must_use]
1198    pub fn from_mmap_slots_with_homes_tails(
1199        mmap: Arc<Mmap>,
1200        slots_offset: usize,
1201        homes_offset: usize,
1202        tails_offset: usize,
1203        capacity: u32,
1204        len: u32,
1205        config: IndexConfig,
1206    ) -> Self {
1207        let cap = usize::try_from(capacity).unwrap_or(usize::MAX);
1208        let mut homes = Vec::with_capacity(cap);
1209        let mut tails = Vec::with_capacity(cap);
1210        let buf: &[u8] = &mmap;
1211        for i in 0..cap {
1212            let h_off = homes_offset.wrapping_add(i.wrapping_mul(8));
1213            let t_off = tails_offset.wrapping_add(i.wrapping_mul(8));
1214            homes.push(u64::from_le_bytes(
1215                buf[h_off..h_off.wrapping_add(8)].try_into().unwrap(),
1216            ));
1217            tails.push(u64::from_le_bytes(
1218                buf[t_off..t_off.wrapping_add(8)].try_into().unwrap(),
1219            ));
1220        }
1221        let slots = EntryStorage::Mmap {
1222            mmap,
1223            offset: slots_offset,
1224        };
1225        let can_grow = Self::identity_tables_complete(
1226            capacity,
1227            64_u32.wrapping_sub(capacity.trailing_zeros()),
1228            &slots,
1229            &homes,
1230            &tails,
1231        );
1232        Self {
1233            mask: capacity.wrapping_sub(1),
1234            capacity,
1235            shift: 64_u32.wrapping_sub(capacity.trailing_zeros()),
1236            config,
1237            slots,
1238            homes: Mutex::new(homes),
1239            tails: Mutex::new(tails),
1240            can_grow,
1241            len: AtomicU32::new(len),
1242            max_probe_len: AtomicU32::new(0),
1243        }
1244    }

1246    /// True while the index borrows raw slots from a checkpoint mapping.
1247    #[must_use]
1248    pub fn is_mmap_backed(&self) -> bool {
1249        matches!(self.slots, EntryStorage::Mmap { .. })
1250    }

1254        self.slots.get(index)

1330/// Iterator over candidate offsets for a hash lookup.
1331///
1332/// Yields `(slot, offset)` for each entry whose tag matches and whose
1333/// in-memory identity (home + tail) matches the query hash, terminating at
1334/// the first empty entry or after the full capacity is probed.
1335///
1336/// When the index has hydrated identity (live/warm index), candidates are
1337/// verified in-memory before yielding — this eliminates disk I/O for tag
1338/// collisions on the normal path. For checkpoint-backed indexes without
1339/// hydrated identity (tail == 0), all tag-matching candidates are yielded
1340/// and the caller must verify against the packfile.
1341pub struct LookupIter<'a> {
1342    index: &'a LossyIndex,
1343    tag: u32,
1344    /// The mixed home for the query hash, for in-memory verification.
1345    mixed_home: u64,
1346    /// The expected tail for the query hash.
1347    expected_tail: u64,
1348    bucket: usize,
1349    mask: usize,
1350    remaining: usize,
1351    /// Steps walked so far, for `LossyIndex::max_probe_len` observability.
1352    steps: u32,
1353}
1354
1355impl Iterator for LookupIter<'_> {
1356    type Item = (u16, u64);
1357
1358    #[inline]
1359    fn next(&mut self) -> Option<Self::Item> {
1360        while self.remaining > 0 {
1361            self.remaining = self.remaining.wrapping_sub(1);
1362            let entry = IndexEntry(self.index.entry_at(self.bucket));
1363            if entry.is_empty() {
1364                self.index.bump_max_probe_len(self.steps);
1365                return None;
1366            }
1367            self.steps = self.steps.saturating_add(1);
1368            let current = self.bucket;
1369            self.bucket = self.bucket.wrapping_add(1) & self.mask;
1370            if entry.tag() != self.tag {
1371                continue;
1372            }
1373            // Tag matches. For a live index with hydrated identity, verify
1374            // home + tail in-memory to avoid yielding false positives that
1375            // would force the caller into an expensive packfile read.
1376            let tails = self.index.tails.lock();
1377            let tail = if tails.len() > current {
1378                tails[current]
1379            } else {
1380                0
1381            };
1382            drop(tails);
1383            if tail != 0 {
1384                // Identity is hydrated — check home and tail.
1385                let homes = self.index.homes.lock();
1386                let home = if homes.len() > current {
1387                    homes[current]
1388                } else {
1389                    0
1390                };
1391                drop(homes);
1392                if home == self.mixed_home && tail == self.expected_tail {
1393                    self.index.bump_max_probe_len(self.steps);
1394                    return Some((entry.slot(), entry.offset()));
1395                }
1396                // Tag matched but home/tail didn't — false positive, skip.
1397                continue;
1398            }
1399            // Checkpoint-backed without hydrated identity — yield and let
1400            // the caller verify against the packfile.
1401            self.index.bump_max_probe_len(self.steps);
1402            return Some((entry.slot(), entry.offset()));
1403        }
1404        self.index.bump_max_probe_len(self.steps);
1405        None
1406    }
1407}
