1//! On-disk serialization of the per-collection indexes (`index.checkpoint`).
2//!
3//! Packfiles stay authoritative: this file is a rebuildable acceleration
4//! structure. A full `rewrite` (see `PackfileStorage::persist_index_checkpoint`)
5//! is done atomically (temp file + rename) whenever the delta log can't
6//! continue — a structural change, a broken continuation, or the log's size
7//! cap. Between rewrites, incremental index mutations are persisted as
8//! batches in the `index.delta` log ([`crate::index::delta`]); the checkpoint
9//! pins the exact pack set those frames continue, and a crash at any point
10//! leaves a fingerprint mismatch the next open resolves with a rescan — never
11//! a wrong replay.
12//!
13//! Layout (all little-endian, fixed width — see [`crate::index::format`]):
14//!
15//! ```text
16//!   [CheckpointHeader 64B]
17//!   [CollectionDirEntry * count, 40B each]
18//!   [collection 0 raw slots: capacity * 8B]
19//!   [collection 1 raw slots: capacity * 8B]
20//!   ...
21//! ```
22//!
23//! The header's `pack_fingerprint` is a deterministic hash of the
24//! `(pack_id, length)` set the checkpoint describes. An opener only trusts
25//! the file when that fingerprint matches the packs currently on disk (optionally
26//! with a gated delta replay continuing it); any append, rotation, or repack
27//! changes it and forces a rescan.
28

43/// Magic identifying the persisted-index checkpoint format.
44pub const CHECKPOINT_MAGIC: [u8; 8] = *b"MTXI0001";
45/// Current wire version (see [`CheckpointHeader::version`]).
46///
47/// Bumped to 6: the header carries a pack table (`slot -> pack_id`
48/// bindings for every pack live at checkpoint-write time). `slot` in a
49/// checkpoint's raw index slots (and in `DeltaFrame.slot`) is the writer's
50/// local, process-scoped `ShardPool` slot, not a stable identity — a fresh
51/// reader's `discover_shards` reassigns slots by first-free-in-`pack_id`-order,
52/// which does not reproduce the writer's numbering once any shard has ever
53/// been retired (retirement leaves a permanent hole in the writer's table).
54/// The pack table lets a reader translate every checkpoint-encoded slot to
55/// its own local slot for the same `pack_id` (stable: `ShardPool::next_pack_id`
56/// is persisted and monotonic-only within a pool, never reused) instead of
57/// trusting the writer's raw slot number. A v5 checkpoint has no pack table
58/// and is rejected outright by the strict magic/version check below,
59/// forcing the normal full-rescan fallback — never a partial/best-effort
60/// read of a v5 file under the v6 reader.
61///
62/// Bumped to 7: the header carries `base_delta_seq`, the highest redo `delta_seq`
63/// the checkpoint incorporates (see `CheckpointHeader::base_delta_seq`). A v6
64/// checkpoint has no such field and is rebuilt.
65///
66/// Bumped to 5: the header now carries the journal `covered_lsn` the index
67/// snapshot incorporates, so a read-committed reader binds its overlay
68/// coverage to the exact index it loaded instead of a separately-read
69/// `journal.lsn`. A v4 checkpoint has no such field and is rebuilt.
70///
71/// Bumped to 4: homes and tails are now persisted alongside packed slots.
72/// A pre-v4 checkpoint has `homes_bytes`/`tails_bytes` of 0 and is loaded
73/// with empty identity side tables (cold-start tag-collision verification
74/// cost). A v4 checkpoint carries hydrated identity, eliminating packfile
75/// reads for tag collisions on cold start.
76pub const CHECKPOINT_VERSION: u32 = 7;
77/// File name of the persisted index checkpoint inside a store's base dir.
78pub const INDEX_CHECKPOINT_FILE: &str = "index.checkpoint";

80/// Whether an opener verifies the checkpoint body's CRC32.
81///
82/// This policy is intentionally independent from [`crate::packfile::ChecksumPolicy`]:
83/// frame checksums, delta-log checksums, and checkpoint integrity protect
84/// different on-disk structures and must not be disabled together by accident.
85/// `WriteOnly` retains the CRC when writing a checkpoint for offline/recovery
86/// tooling, but trusts it during normal opens.
87#[derive(Debug, Clone, Copy, PartialEq, Eq)]
88pub enum CheckpointChecksumPolicy {
89    /// Verify the persisted checkpoint CRC32 at every open (the default).
90    Full,
91    /// Retain the CRC32 on write but skip its read-time verification.
92    WriteOnly,
93}
94
95impl CheckpointChecksumPolicy {
96    /// Select the process-wide checkpoint policy from `MTXDB_CHECKPOINT_CHECKSUM`.
97    /// Unknown values intentionally retain the safe default.
98    #[must_use]
99    pub fn from_env() -> Self {
100        match std::env::var("MTXDB_CHECKPOINT_CHECKSUM").as_deref() {
101            Ok("writeonly") => Self::WriteOnly,
102            Ok("full" | _) | Err(_) => Self::Full,
103        }
104    }
105
106    #[must_use]
107    fn verifies_reads(self) -> bool {
108        matches!(self, Self::Full)
109    }
110}

117/// A checkpoint read off disk, validated enough to trust for index rebuild.
118#[derive(Debug)]
119pub struct LoadedCheckpoint {
120    /// The pack-fingerprint the checkpoint was written against.
121    pub fingerprint: u64,
122    /// Journal LSN this checkpoint's index snapshot covers, captured atomically
123    /// with the snapshot. Zero when the checkpoint was written without a
124    /// journal.
125    pub covered_lsn: u64,
126    /// The highest redo `delta_seq` this checkpoint incorporates.
127    pub base_delta_seq: u64,
128    /// One entry per collection, in the checkpoint's directory order.
129    pub collections: Vec<LoadedCollection>,
130    /// Keeps the raw slot arrays alive for mmap-backed indexes built from
131    /// this checkpoint.
132    pub mmap: Arc<Mmap>,
133    /// `(writer's local shard slot, pack_id)` for every pack live at
134    /// checkpoint-write time. Every `slot` embedded in this checkpoint's
135    /// raw index slots (and in any delta frame that continues it) refers to
136    /// one of these slots — translate through this table to a reader's own
137    /// local slot rather than trusting the slot number directly. See the
138    /// `CHECKPOINT_VERSION` doc comment for why.
139    pub pack_table: Vec<(u16, u64)>,
140}

166/// Deterministic FNV-1a hash of the `(pack_id, file_len)` set a checkpoint
167/// (or an open store's current packs) describes. The set is sorted before
168/// hashing so the result is order-independent.
169#[must_use]
170pub fn pack_fingerprint(packs: &[(u64, u64)]) -> u64 {
171    let mut sorted: Vec<(u64, u64)> = packs.to_vec();
172    sorted.sort_unstable();
173    let mut hash: u64 = 0xcbf2_9ce4_8422_2325;
174    for (pack_id, file_len) in sorted {
175        for byte in pack_id.to_le_bytes() {
176            hash = (hash ^ u64::from(byte)).wrapping_mul(0x0000_0100_0000_01b3);
177        }
178        for byte in file_len.to_le_bytes() {
179            hash = (hash ^ u64::from(byte)).wrapping_mul(0x0000_0100_0000_01b3);
180        }
181    }
182    hash
183}

185/// Decode the 8-byte capacity prefix of a `LossyIndex::serialize` blob.
186fn blob_capacity(blob: &[u8]) -> Option<u64> {
187    let head: [u8; 8] = blob.get(..8)?.try_into().ok()?;
188    Some(u64::from_le_bytes(head))
189}
190
191/// Number of occupied (non-`0u64`) slots in a `LossyIndex::serialize` blob's
192/// slot section. The blob format is `[capacity_u64][slots × cap][homes × cap][tails × cap]`.
193fn blob_slot_count(blob: &[u8]) -> u32 {
194    let Some(capacity) = blob_capacity(blob) else {
195        return 0;
196    };
197    let cap = usize::try_from(capacity).unwrap_or(usize::MAX);
198    let slots_end = 8_usize.saturating_add(cap.saturating_mul(8));
199    let slots = blob.get(8..slots_end).unwrap_or(&[]);
200    u32::try_from(
201        slots
202            .chunks_exact(8)
203            .filter(|slot| *slot != [0u8; 8])
204            .count(),
205    )
206    .unwrap_or(u32::MAX)
207}

209/// Write a complete checkpoint atomically: temp file, fsync, rename over the
210/// final path. `collections` holds each collection's `serialize` blob plus the
211/// index generation it was written at (recorded in its directory entry so a
212/// delta log's frames can be gated against the same generation on replay),
213/// the first collection listed becoming the first in the directory order.
214///
215/// # Contract
216/// The `fingerprint` and every `collections` blob must be *captured
217/// atomically*: derived from pack-file lengths and index slots that were both
218/// observed while every collection's writer lock was held. This function only
219/// makes that already-consistent (pack set, index snapshot) pair durable; it
220/// cannot reconcile the two if a concurrent `put` changed either side between
221/// capture and this call. The sole caller
222/// (`PackfileStorage::persist_index_checkpoint`) satisfies this by pinning the
223/// pack lengths and all collection generations under every `put_mutex` (and
224/// rotating the delta epoch there too) before handing them to this function.
225///
226/// # Errors
227/// Returns `io::Error` on any failure; the previous checkpoint (if any) is
228/// left intact in that case.
229#[allow(
230    clippy::too_many_lines,
231    reason = "the three phases (directory, slots/homes/tails offsets, body write) are one atomic construction; splitting them would scatter overflow-checked arithmetic that must stay in lockstep"
232)]
233pub fn write_checkpoint(
234    path: &Path,
235    fingerprint: u64,
236    covered_lsn: u64,
237    base_delta_seq: u64,
238    collections: &[([u8; 16], u64, &[u8])],
239    pack_table: &[(u16, u64)],
240) -> std::io::Result<()> {
241    let pack_table_count = u32::try_from(pack_table.len())
242        .map_err(|_| std::io::Error::other("too many packs for checkpoint pack table u32"))?;
243    let pack_table_bytes = u64::try_from(pack_table.len())
244        .ok()
245        .and_then(|n| n.checked_mul(PACK_TABLE_ENTRY_LEN as u64))
246        .ok_or_else(|| std::io::Error::other("checkpoint pack table size overflow"))?;
247
248    let count = u32::try_from(collections.len())
249        .map_err(|_| std::io::Error::other("too many collections for checkpoint u32"))?;
250    let directory_bytes = u64::try_from(collections.len())
251        .ok()
252        .and_then(|n| n.checked_mul(COLLECTION_DIR_ENTRY_LEN as u64))
253        .ok_or_else(|| std::io::Error::other("checkpoint directory size overflow"))?;
254
255    // Compute per-section byte lengths from the serialize blobs.
256    // Blob format: [capacity_u64][slots × cap][homes × cap][tails × cap]
257    let mut slots_bytes: u64 = 0;
258    let mut homes_bytes: u64 = 0;
259    let mut tails_bytes: u64 = 0;
260    for (_, _, blob) in collections {
261        let capacity = blob_capacity(blob)
262            .ok_or_else(|| std::io::Error::other("collection slot blob missing capacity"))?;
263        let cap8 = capacity
264            .checked_mul(8)
265            .ok_or_else(|| std::io::Error::other("collection capacity × 8 overflow"))?;
266        slots_bytes = slots_bytes
267            .checked_add(cap8)
268            .ok_or_else(|| std::io::Error::other("checkpoint slots size overflow"))?;
269        // homes and tails are each capacity × 8 bytes when present in the blob
270        let blob_body = blob.len().saturating_sub(8);
271        let cap_usize = usize::try_from(capacity).unwrap_or(usize::MAX);
272        if blob_body >= cap_usize.saturating_mul(24) {
273            homes_bytes = homes_bytes
274                .checked_add(cap8)
275                .ok_or_else(|| std::io::Error::other("checkpoint homes size overflow"))?;
276            tails_bytes = tails_bytes
277                .checked_add(cap8)
278                .ok_or_else(|| std::io::Error::other("checkpoint tails size overflow"))?;
279        }
280    }
281
282    let body_bytes = slots_bytes
283        .checked_add(homes_bytes)
284        .and_then(|v| v.checked_add(tails_bytes))
285        .ok_or_else(|| std::io::Error::other("checkpoint body size overflow"))?;
286    let total_len = CHECKPOINT_HEADER_LEN
287        .saturating_add(usize::try_from(directory_bytes).unwrap_or(usize::MAX))
288        .saturating_add(usize::try_from(body_bytes).unwrap_or(usize::MAX))
289        .saturating_add(usize::try_from(pack_table_bytes).unwrap_or(usize::MAX));
290    // Reserve the header first, then build the body directly into the final
291    // buffer. This avoids retaining a second checkpoint-sized body merely to
292    // calculate the CRC carried by the header.
293    let mut buf = Vec::with_capacity(total_len);
294    buf.resize(CHECKPOINT_HEADER_LEN, 0);
295
296    // Phase 1: build directory entries, tracking per-collection offsets.
297    //
298    // Offsets must mirror Phase 3's actual write order exactly: each
299    // collection's slots/homes/tails are written back-to-back as one block
300    // (not as three separate all-slots/all-homes/all-tails sections), so a
301    // single running `body_offset` — not three independent per-kind
302    // counters — is what keeps a later collection's offsets from landing
303    // inside an earlier collection's bytes.
304    let mut body_offset: u64 = 0;
305    let mut dir_entries = Vec::with_capacity(collections.len());
306    for (collection_id, generation, blob) in collections {
307        let capacity = blob_capacity(blob)
308            .ok_or_else(|| std::io::Error::other("collection slot blob missing capacity"))?;
309        let capacity32 = u32::try_from(capacity)
310            .map_err(|_| std::io::Error::other("index capacity exceeds u32"))?;
311        let cap8 = capacity
312            .checked_mul(8)
313            .ok_or_else(|| std::io::Error::other("collection capacity × 8 overflow"))?;
314        let blob_body = blob.len().saturating_sub(8);
315        let cap_usize = usize::try_from(capacity).unwrap_or(usize::MAX);
316        let has_homes_tails = blob_body >= cap_usize.saturating_mul(24);
317
318        let slots_offset = body_offset;
319        let (homes_offset, tails_offset) = if has_homes_tails {
320            let homes_offset = body_offset
321                .checked_add(cap8)
322                .ok_or_else(|| std::io::Error::other("checkpoint homes offset overflow"))?;
323            let tails_offset = homes_offset
324                .checked_add(cap8)
325                .ok_or_else(|| std::io::Error::other("checkpoint tails offset overflow"))?;
326            (homes_offset, tails_offset)
327        } else {
328            (0, 0)
329        };
330        dir_entries.push(CollectionDirEntry {
331            collection_id: *collection_id,
332            generation: *generation,
333            slots_offset,
334            homes_offset,
335            tails_offset,
336            capacity: capacity32,
337            slot_count: blob_slot_count(blob),
338        });
339        let block_bytes = if has_homes_tails {
340            cap8.checked_mul(3)
341                .ok_or_else(|| std::io::Error::other("collection capacity × 24 overflow"))?
342        } else {
343            cap8
344        };
345        body_offset = body_offset
346            .checked_add(block_bytes)
347            .ok_or_else(|| std::io::Error::other("checkpoint body offset overflow"))?;
348    }
349
350    // Phase 2: write directory entries.
351    for entry in &dir_entries {
352        buf.extend_from_slice(&entry.encode());
353    }
354
355    // Phase 3: write slots, homes, tails as separate contiguous sections.
356    for (_, _, blob) in collections {
357        let capacity = blob_capacity(blob)
358            .ok_or_else(|| std::io::Error::other("collection slot blob missing capacity"))?;
359        let cap = usize::try_from(capacity).unwrap_or(usize::MAX);
360        let slots_end = 8_usize.saturating_add(cap.saturating_mul(8));
361        let homes_end = slots_end.saturating_add(cap.saturating_mul(8));
362        let tails_end = homes_end.saturating_add(cap.saturating_mul(8));
363        // Slots
364        if let Some(slots) = blob.get(8..slots_end) {
365            buf.extend_from_slice(slots);
366        }
367        // Homes
368        if let Some(homes) = blob.get(slots_end..homes_end) {
369            buf.extend_from_slice(homes);
370        }
371        // Tails
372        if let Some(tails) = blob.get(homes_end..tails_end) {
373            buf.extend_from_slice(tails);
374        }
375    }
376
377    // Phase 4: write the pack table (slot -> pack_id bindings), one fixed-
378    // width entry per currently-live pack. Appended after the slots/homes/
379    // tails body so existing offset math above is untouched.
380    for &(slot, pack_id) in pack_table {
381        buf.extend_from_slice(&PackTableEntry { slot, pack_id }.encode());
382    }
383
384    let content_crc32 = crc32fast::hash(&buf[CHECKPOINT_HEADER_LEN..]);
385
386    let header = CheckpointHeader {
387        magic: CHECKPOINT_MAGIC,
388        version: CHECKPOINT_VERSION,
389        collection_count: count,
390        directory_bytes,
391        slots_bytes,
392        pack_fingerprint: fingerprint,
393        content_crc32,
394        homes_bytes,
395        tails_bytes,
396        covered_lsn,
397        pack_table_count,
398        pack_table_bytes,
399        base_delta_seq,
400    };
401    buf[..CHECKPOINT_HEADER_LEN].copy_from_slice(&header.encode());
402
403    let unique = CHECKPOINT_TMP_COUNTER.fetch_add(1, Ordering::Relaxed);
404    let tmp_path = PathBuf::from(format!(
405        "{}.tmp.{}.{unique}",
406        path.display(),
407        std::process::id()
408    ));
409    let write_result = (|| -> std::io::Result<()> {
410        let mut tmp = fs::File::create(&tmp_path)?;
411        tmp.write_all(&buf)?;
412        // `sync_data` (fdatasync), not `sync_all` (fsync): per POSIX,
413        // fdatasync durably flushes the file's data plus whatever metadata
414        // is needed to retrieve it (the extended length included) — that's
415        // the whole content/length guarantee this write needs. It skips
416        // metadata that's *not* needed to read the data back (mtime/ctime).
417        // That's a real distinction, not a claim that all inode metadata is
418        // irrelevant here generally (permissions/ownership can still matter
419        // operationally) — it just happens that none of the metadata this
420        // skips is read back through this temp name: it's renamed over
421        // `path` immediately below and never opened by this name again. The
422        // caller (`persist_index_checkpoint`) separately fsyncs the base
423        // directory after the rename, which is what makes the rename itself
424        // durable; if that directory fsync fails, reopen falls back to
425        // whichever checkpoint (old or new) is actually intact rather than
426        // trusting the rename — this change doesn't alter that fallback.
427        tmp.sync_data()
428    })();
429    if let Err(e) = write_result {
430        let _ = fs::remove_file(&tmp_path);
431        return Err(e);
432    }
433    fs::rename(&tmp_path, path)
434}

436/// Read and validate a checkpoint. Returns `None` for any problem — a
437/// missing, truncated, or malformed file is treated identically to "no
438/// checkpoint", and the caller falls back to rebuilding from the packs.
439#[must_use]
440pub fn read_checkpoint(path: &Path) -> Option<LoadedCheckpoint> {
441    read_checkpoint_with_policy(path, CheckpointChecksumPolicy::from_env())
442}

444/// As [`read_checkpoint`], with an explicit integrity policy for callers that
445/// need deterministic configuration rather than the process environment.
446#[must_use]
447#[allow(
448    clippy::too_many_lines,
449    reason = "split across branches, too lazy to consolidate right now"
450)]
451pub fn read_checkpoint_with_policy(
452    path: &Path,
453    checksum_policy: CheckpointChecksumPolicy,
454) -> Option<LoadedCheckpoint> {
455    let file = fs::File::open(path).ok()?;
456    let mmap = Arc::new(crate::packfile::map_pack(&file).ok()?);
457    let buf: &[u8] = &mmap;
458    let header_bytes: [u8; CHECKPOINT_HEADER_LEN] =
459        buf.get(..CHECKPOINT_HEADER_LEN)?.try_into().ok()?;
460    let header = CheckpointHeader::decode(&header_bytes)?;
461    if header.magic != CHECKPOINT_MAGIC || header.version != CHECKPOINT_VERSION {
462        return None;
463    }
464    let count = header.collection_count as usize;
465    if header.directory_bytes
466        != u64::try_from(count)
467            .ok()?
468            .checked_mul(COLLECTION_DIR_ENTRY_LEN as u64)?
469    {
470        return None;
471    }
472    let slot_base =
473        CHECKPOINT_HEADER_LEN.checked_add(usize::try_from(header.directory_bytes).ok()?)?;
474    let body_bytes = header
475        .slots_bytes
476        .checked_add(header.homes_bytes)
477        .and_then(|v| v.checked_add(header.tails_bytes))?;
478    // Slots are unique u16 values, so a valid pack table cannot exceed 65,536
479    // entries; reject before `pack_table` and its dedupe sets reserve capacity
480    // from `pack_table_count`.
481    if header.pack_table_count > u32::from(u16::MAX).saturating_add(1) {
482        return None;
483    }
484    if header.pack_table_bytes
485        != u64::from(header.pack_table_count).checked_mul(PACK_TABLE_ENTRY_LEN as u64)?
486    {
487        return None;
488    }
489    let pack_table_base = slot_base.checked_add(usize::try_from(body_bytes).ok()?)?;
490    if buf.len() != pack_table_base.checked_add(usize::try_from(header.pack_table_bytes).ok()?)? {
491        return None;
492    }
493
494    // Verify the whole directory+slots body against the header's CRC in one
495    // pass, in place of re-deriving each collection's occupancy by walking
496    // its slots. This is strictly stronger than the old per-collection walk
497    // (it also covers the directory entries — `capacity`/`slot_count`
498    // themselves — which the walk never checked), and costs the same O(bytes)
499    // as a single SIMD-accelerated hash instead of a manual scan-and-branch
500    // loop over every slot of every collection.
501    if checksum_policy.verifies_reads() {
502        let content_crc32 = crc32fast::hash(buf.get(CHECKPOINT_HEADER_LEN..)?);
503        if content_crc32 != header.content_crc32 {
504            return None;
505        }
506    }
507
508    let mut collections = Vec::with_capacity(count);
509    for i in 0..count {
510        let entry_offset =
511            CHECKPOINT_HEADER_LEN.checked_add(i.checked_mul(COLLECTION_DIR_ENTRY_LEN)?)?;
512        let entry_bytes: [u8; COLLECTION_DIR_ENTRY_LEN] = buf
513            .get(entry_offset..entry_offset.saturating_add(COLLECTION_DIR_ENTRY_LEN))?
514            .try_into()
515            .ok()?;
516        let entry = CollectionDirEntry::decode(&entry_bytes)?;
517        if entry.capacity < 16 || !entry.capacity.is_power_of_two() {
518            return None;
519        }
520        // A corrupt or malicious directory entry could otherwise claim more
521        // occupied slots than the table has room for; a downstream consumer
522        // sizing an allocation off `slot_count` (e.g. a post-restart grow)
523        // would then trust a value with no upper bound instead of the
524        // checkpoint being rejected here. (An *understated* count — the
525        // failure mode the old occupancy walk additionally caught — is now
526        // caught by the CRC check above instead: any bit that makes
527        // `slot_count` disagree with the actual slots, or with what was
528        // written, changes the body's hash.)
529        if entry.slot_count > entry.capacity {
530            return None;
531        }
532        let slots_len = (entry.capacity as usize).checked_mul(8)?;
533        let region = slot_base.checked_add(usize::try_from(entry.slots_offset).ok()?)?;
534        if region.checked_add(slots_len)? > buf.len() {
535            return None;
536        }
537        let has_homes_tails = header.homes_bytes > 0 && header.tails_bytes > 0;
538        let homes_region = if has_homes_tails {
539            let r = slot_base.checked_add(usize::try_from(entry.homes_offset).ok()?)?;
540            if r.checked_add(slots_len)? > buf.len() {
541                return None;
542            }
543            r
544        } else {
545            0
546        };
547        let tails_region = if has_homes_tails {
548            let r = slot_base.checked_add(usize::try_from(entry.tails_offset).ok()?)?;
549            if r.checked_add(slots_len)? > buf.len() {
550                return None;
551            }
552            r
553        } else {
554            0
555        };
556        collections.push(LoadedCollection {
557            collection_id: entry.collection_id,
558            generation: entry.generation,
559            slots_offset: region,
560            homes_offset: homes_region,
561            tails_offset: tails_region,
562            capacity: entry.capacity,
563            slot_count: entry.slot_count,
564            has_homes_tails,
565        });
566    }
567
568    let mut pack_table = Vec::with_capacity(header.pack_table_count as usize);
569    let mut seen_slots = HashSet::with_capacity(header.pack_table_count as usize);
570    let mut seen_pack_ids = HashSet::with_capacity(header.pack_table_count as usize);
571    for i in 0..header.pack_table_count as usize {
572        let entry_offset = pack_table_base.checked_add(i.checked_mul(PACK_TABLE_ENTRY_LEN)?)?;
573        let entry_bytes: [u8; PACK_TABLE_ENTRY_LEN] = buf
574            .get(entry_offset..entry_offset.saturating_add(PACK_TABLE_ENTRY_LEN))?
575            .try_into()
576            .ok()?;
577        let entry = PackTableEntry::decode(&entry_bytes)?;
578        if !seen_slots.insert(entry.slot) || !seen_pack_ids.insert(entry.pack_id) {
579            return None;
580        }
581        pack_table.push((entry.slot, entry.pack_id));
582    }
583
584    Some(LoadedCheckpoint {
585        fingerprint: header.pack_fingerprint,
586        covered_lsn: header.covered_lsn,
587        base_delta_seq: header.base_delta_seq,
588        collections,
589        mmap,
590        pack_table,
591    })
592}

594/// Lightweight read of the persisted checkpoint's `pack_fingerprint`.
595///
596/// Returns `Ok(Some(fp))` for a valid checkpoint, `Ok(None)` if the file
597/// is missing (fresh store), and `Err` if the file exists but is truncated,
598/// has invalid magic/version, or an I/O error occurs.
599///
600/// # Errors
601///
602/// Returns a [`crate::index::delta::DeltaFingerprintError`] when an existing
603/// checkpoint cannot be read or validated.
604pub fn read_pack_fingerprint(
605    path: &Path,
606) -> Result<Option<u64>, crate::index::delta::DeltaFingerprintError> {
607    Ok(read_checkpoint_summary(path)?.map(|summary| summary.fingerprint))
608}

610/// What a checkpoint's header says, without reading its body.
611#[derive(Debug, Clone, Copy, PartialEq, Eq)]
612pub struct CheckpointSummary {
613    /// The pack fingerprint the checkpoint was written against.
614    pub fingerprint: u64,
615    /// The journal LSN its index covers, or zero when written without a journal.
616    pub covered_lsn: u64,
617    /// The highest redo `delta_seq` it incorporates.
618    pub base_delta_seq: u64,
619}
620
621/// Lightweight read of a checkpoint's header: its `pack_fingerprint` and the
622/// journal `covered_lsn` it records.
623///
624/// Same contract as [`read_pack_fingerprint`]: `Ok(None)` for a missing file,
625/// `Err` for a truncated one or one with an invalid magic or version.
626///
627/// # Errors
628///
629/// Returns a [`crate::index::delta::DeltaFingerprintError`] when an existing
630/// checkpoint cannot be read or validated.
631pub fn read_checkpoint_summary(
632    path: &Path,
633) -> Result<Option<CheckpointSummary>, crate::index::delta::DeltaFingerprintError> {
634    let file = match fs::File::open(path) {
635        Ok(f) => f,
636        Err(e) if e.kind() == io::ErrorKind::NotFound => return Ok(None),
637        Err(e) => return Err(crate::index::delta::io_error(path, "open checkpoint", e)),
638    };
639    let mut buf = [0u8; CHECKPOINT_HEADER_LEN];
640    let mut handle = file.take(CHECKPOINT_HEADER_LEN as u64);
641    handle
642        .read_exact(&mut buf)
643        .map_err(|e| crate::index::delta::io_error(path, "read checkpoint header", e))?;
644    let header = CheckpointHeader::decode(&buf)
645        .ok_or_else(|| crate::index::delta::invalid(path, "checkpoint header decode failed"))?;
646    if header.magic != CHECKPOINT_MAGIC || header.version != CHECKPOINT_VERSION {
647        return Err(crate::index::delta::invalid(
648            path,
649            "invalid checkpoint magic/version",
650        ));
651    }
652    Ok(Some(CheckpointSummary {
653        fingerprint: header.pack_fingerprint,
654        covered_lsn: header.covered_lsn,
655        base_delta_seq: header.base_delta_seq,
656    }))
657}

659/// Result of reading the durable fingerprint.
660#[derive(Debug, Clone, Copy)]
661pub struct DurableFingerprint {
662    /// The durable pack fingerprint (checkpoint fp advanced by delta log tail).
663    pub fingerprint: u64,
664    /// Whether the delta log ended with a torn tail (expected if writer crashed mid-append).
665    /// If true, a refresh may still be needed to pick up the incomplete batch.
666    pub torn_tail: bool,
667}
668
669/// Read the durable pack fingerprint for a store directory.
670///
671/// The durable fingerprint is the checkpoint's `pack_fingerprint`, advanced
672/// by the validated delta log's `tail_fingerprint` when one exists and its
673/// `base_fingerprint` matches the checkpoint.  This detects synced appends
674/// that only wrote to the delta log without rewriting the checkpoint.
675///
676/// Returns `Ok(Some(fp))` for a validated durable fingerprint, `Ok(None)`
677/// when there is no checkpoint (fresh store), and `Err` when the checkpoint
678/// is corrupt or the delta log cannot be reliably read. On `Err` the caller
679/// must treat the state as unknown and force a refresh.
680///
681/// # Errors
682///
683/// Returns a [`crate::index::delta::DeltaFingerprintError`] when the delta
684/// log exists but cannot be validated or read reliably.
685pub fn read_durable_fingerprint(
686    base_dir: &Path,
687) -> Result<Option<DurableFingerprint>, crate::index::delta::DeltaFingerprintError> {
688    let ckpt_path = base_dir.join(INDEX_CHECKPOINT_FILE);
689    let Some(ckpt_fp) = read_pack_fingerprint(&ckpt_path)? else {
690        return Ok(None);
691    };
692    let delta_path = base_dir.join(format!(
693        "{}.{:016x}",
694        crate::index::delta::INDEX_DELTA_FILE,
695        ckpt_fp,
696    ));
697    // Use the lightweight forward parser to avoid decoding every DeltaFrame.
698    match crate::index::delta::read_delta_tail_fingerprint(&delta_path) {
699        Ok(Some(tail_fp)) if tail_fp.base_fingerprint == ckpt_fp => Ok(Some(DurableFingerprint {
700            fingerprint: tail_fp.tail_fingerprint,
701            torn_tail: tail_fp.torn_tail,
702        })),
703        Ok(Some(_) | None) => Ok(Some(DurableFingerprint {
704            fingerprint: ckpt_fp,
705            torn_tail: false,
706        })),
707        Err(e) => Err(e),
708    }
709}
