35/// Probe: start writeback for a just-flushed range before the next flush, to
36/// test whether keeping the dirty frontier ahead of the writer shrinks the
37/// `folio_wait_bit` stall. Enabled by `MTXDB_SYNC_FILE_RANGE=1`. A pure hint:
38/// failure is ignored, since the bytes are already in the page cache.
39#[cfg(target_os = "linux")]
40#[allow(unsafe_code)]
41fn kick_writeback(file: &File, offset: u64, len: u64) {
42    use std::os::unix::io::AsRawFd;
43    const SYNC_FILE_RANGE_WRITE: u32 = 2;
44    static ENABLED: std::sync::OnceLock<bool> = std::sync::OnceLock::new();
45    if !*ENABLED.get_or_init(|| std::env::var("MTXDB_SYNC_FILE_RANGE").is_ok_and(|v| v != "0")) {
46        return;
47    }
48    // SAFETY: `file` holds a valid fd for the duration of the call, and the
49    // syscall only advances writeback for the given range — it cannot affect
50    // correctness.
51    let _ = unsafe {
52        sync_file_range(
53            file.as_raw_fd(),
54            i64::try_from(offset).unwrap_or(i64::MAX),
55            i64::try_from(len).unwrap_or(i64::MAX),
56            SYNC_FILE_RANGE_WRITE,
57        )
58    };
59}

61/// Flush a directory's entries to stable storage.
62///
63/// Unix can fsync a directory descriptor directly. Windows has no std API for
64/// it, and only lets you open a directory handle at all with
65/// `FILE_FLAG_BACKUP_SEMANTICS`; `sync_all` then maps to `FlushFileBuffers`,
66/// which persists the directory's entries — the same approach `SQLite`'s Win32
67/// VFS uses. `custom_flags` is safe, so this needs no FFI or new dependency.
68pub(crate) fn sync_directory(dir: &Path) -> io::Result<()> {
69    #[cfg(unix)]
70    {
71        File::open(dir)?.sync_all()
72    }
73    #[cfg(windows)]
74    {
75        use std::os::windows::fs::OpenOptionsExt;
76        const FILE_FLAG_BACKUP_SEMANTICS: u32 = 0x0200_0000;
77        std::fs::OpenOptions::new()
78            .read(true)
79            .write(true)
80            .custom_flags(FILE_FLAG_BACKUP_SEMANTICS)
81            .open(dir)?
82            .sync_all()
83    }
84    #[cfg(not(any(unix, windows)))]
85    {
86        let _ = dir;
87        Err(io::Error::new(
88            io::ErrorKind::Unsupported,
89            "directory sync is unsupported on this platform",
90        ))
91    }
92}

94/// Maximum number of shards in the pool.
95///
96/// This is a runtime policy cap, not a format limit: the shard slot is a
97/// `u16` that can represent 65,536 values, and nothing about the on-disk
98/// encoding forbids more than this count. The cap deliberately bounds one
99/// process's live shard objects — their pack-file descriptors, their
100/// (lazily created) mappings, and their buffered append memory — under a
101/// no-eviction lifecycle: once live, a shard is retired only when a repack
102/// reclaims it, never by closing and reopening its file.
103///
104/// It is also not a complete resource limit on its own. A process's usable
105/// shard count is bounded by its file-descriptor and VMA budgets, which are
106/// separate and often tighter (see [`ShardPool`]); a single
107/// `MAX_SHARDS`-sized pool can still reach `EMFILE` or exhaust
108/// `vm.max_map_count` first.
109pub const MAX_SHARDS: usize = 4096;

117/// Maximum shard size before rotation: `2^32 - 2` bytes (~4 GiB), the
118/// largest value for which every offset a shard can ever produce still
119/// fits `IndexEntry`'s 32-bit offset field (which reserves its all-zero
120/// encoding as the empty-slot sentinel, capping the max representable
121/// offset at `2^32 - 2` — see `index::IndexEntry`). Do not round this up
122/// to a clean `4 * 1024 * 1024 * 1024`: that's one byte over the ceiling and
123/// lets a shard produce an offset `IndexEntry::new` panics on.
124pub const MAX_SHARD_BYTES: u64 = crate::index::IndexEntry::MAX_OFFSET;

126/// Default in-memory threshold before a shard's buffered frames are written
127/// to disk as one positioned write. Keeps bulk writes from paying a
128/// `pwrite` (plus the per-record `try_clone` and file-length store) for
129/// every single record — the whole remaining bulk-write gap vs the bench's
130/// mdbx (which writes a memory-mapped in-memory transaction and pays
131/// persistence once at commit). The default [`AppendPolicy`] is
132/// [`AppendPolicy::Eager`], so this threshold only applies to pools opened
133/// with [`AppendPolicy::Buffered`].
134///
135/// The buffer is per shard, so the pool's buffered-memory commitment is this
136/// threshold times the number of shards holding unflushed data. That product
137/// is a worst-case upper bound, not a standing footprint: only a shard whose
138/// buffer is currently near full contributes this much, and `MAX_SHARDS`
139/// bounds how many can do so at once.
140pub(crate) const PENDING_FLUSH_BYTES: usize = 1 << 20;
141
142/// When a shard's appended frames are written to its underlying pack file.
143///
144/// The default is [`AppendPolicy::Eager`]: every `put_record` commits its
145/// own frame with one positioned write, matching the engine's historical
146/// behavior. An unsynced put is therefore immediately visible to other
147/// processes reading the pack files, and survives this process exiting (a
148/// crash or power loss still requires an explicit `sync_*` to be durable).
149///
150/// [`AppendPolicy::Buffered`] defers those writes into one positioned write
151/// per shard once `max_pending_bytes` of frames have accumulated (or at the
152/// rotation/sync/read/repack/checkpoint boundaries that flush first). This
153/// is far faster for bulk writes — one `pwrite` per ~1 MiB instead of one
154/// per record — but it changes the durability boundary:
155///
156/// **Records since the last flush are lost on process crash.**
157///
158/// Unflushed bytes live only in this process's memory, are invisible to
159/// other processes, and are gone if this process exits — there is
160/// deliberately no recovery path for them. (Recovery machinery here only
161/// covers bytes already committed to a pack file: checkpoint writes flush
162/// first, so the persisted fingerprint and index describe only committed
163/// lengths, never buffered ones.) A crash therefore leaves on-disk state
164/// self-consistent but drops everything still in the buffer. Buffered is a
165/// pure throughput option trading crash-loss of the unflushed tail; opt
166/// into it only where the caller already flushes or syncs on its own
167/// durability schedule (e.g. a batch import that ends in `sync_all`).
168#[derive(Clone, Copy, Debug, PartialEq, Eq)]
169pub enum AppendPolicy {
170    /// Commit each record's frame with its own positioned write, exactly as
171    /// mtxdb always did before buffering existed.
172    Eager,
173    /// Accumulate frames in memory and write them out once
174    /// `max_pending_bytes` have been buffered for a shard.
175    Buffered {
176        /// Per-shard in-memory flush threshold, in buffered frame bytes.
177        max_pending_bytes: usize,
178    },
179}
180
181impl AppendPolicy {
182    /// [`AppendPolicy::Buffered`] at the default flush threshold (~1 MiB).
183    #[must_use]
184    pub fn buffered() -> Self {
185        Self::Buffered {
186            max_pending_bytes: PENDING_FLUSH_BYTES,
187        }
188    }
189}

372/// Write the one-time store-creation marker, recording the `mtxdb`
373/// version (`CARGO_PKG_VERSION`) that created this store. Best-effort: a
374/// failure here doesn't fail store creation, since this is diagnostic
375/// metadata, not data the engine depends on to operate correctly.
376fn persist_store_meta(base_dir: &Path) {
377    let version = env!("CARGO_PKG_VERSION").as_bytes();
378    let Ok(version_len) = u8::try_from(version.len()) else {
379        return; // never true for a real semver string; just don't write garbage
380    };
381    let mut buf = Vec::with_capacity(6usize.saturating_add(version.len()));
382    buf.extend_from_slice(b"MTXS");
383    buf.push(STORE_META_VERSION);
384    buf.push(version_len);
385    buf.extend_from_slice(version);
386
387    let final_path = base_dir.join(STORE_META_FILENAME);
388    let tmp_path = final_path.with_extension(format!("meta.tmp.{}", std::process::id()));
389    let result = (|| -> io::Result<()> {
390        let mut tmp = File::create(&tmp_path)?;
391        tmp.write_all(&buf)?;
392        drop(tmp);
393        fs::rename(&tmp_path, &final_path)?;
394        Ok(())
395    })();
396    if result.is_err() {
397        let _ = fs::remove_file(&tmp_path);
398    }
399}

401/// Read back the `mtxdb` version that created this store, if the
402/// store was created by a build new enough to record it (`store.meta`
403/// predates this feature, so an older store — or one with an unreadable
404/// or corrupt marker — returns `None` rather than erroring: this is
405/// diagnostic-only information).
406#[must_use]
407pub fn store_created_by_version(base_dir: &Path) -> Option<String> {
408    let data = fs::read(base_dir.join(STORE_META_FILENAME)).ok()?;
409    if data.len() < 6 || &data[0..4] != b"MTXS" || data[4] != STORE_META_VERSION {
410        return None;
411    }
412    let version_len = usize::from(data[5]);
413    let version_end = 6usize.checked_add(version_len)?;
414    let version_bytes = data.get(6..version_end)?;
415    String::from_utf8(version_bytes.to_vec()).ok()
416}

494    }
495
496    /// Bytes of this shard's file that an fsync has made durable; see the field
497    /// doc for what this may and may not be used for.
498    #[must_use]

500        self.synced_len.load(Ordering::Acquire)
501    }
502
503    /// Bytes written to this shard's file (and so flushed to the page cache)
504    /// that no fsync has covered yet.
505    #[must_use]
506    pub fn unsynced_bytes(&self) -> u64 {
507        self.file_len().saturating_sub(self.synced_len())
508    }
509
510    /// fsync the file, then advance the durable watermark to the length read
511    /// *before* the fsync started: every byte at or below it was written before
512    /// the fsync began, so the fsync covers it, while bytes appended during the
513    /// fsync stay owed to the next one.
514    fn fsync_advancing_watermark(&self, data_only: bool) -> io::Result<()> {
515        let covered = self.file_len();
516        if data_only {
517            self.file.sync_data()?;
518        } else {
519            self.file.sync_all()?;
520        }
521        self.synced_len.fetch_max(covered, Ordering::AcqRel);
522        Ok(())
523    }
524
525    /// Restore counters from a persisted snapshot. Only called at startup,
526    /// before the shard is shared, so `Relaxed` stores are fine.
527    fn restore_stats(&self, stats: ShardStats) {
528        self.write_count.store(stats.write_count, Ordering::Relaxed);
529        self.bytes_written
530            .store(stats.bytes_written, Ordering::Relaxed);
531        self.sync_count.store(stats.sync_count, Ordering::Relaxed);
532    }
533}

535impl Shard {
536    /// Get the memory-mapped view, creating it if absent.
537    ///
538    /// Remaps when the file grows past the current mapping.
539    ///
540    /// # Errors
541    /// Returns `io::Error` if the packfile cannot be mapped.
542    pub fn mmap(&self) -> io::Result<parking_lot::RwLockReadGuard<'_, Option<Arc<Mmap>>>> {
543        let guard = self.mmap.read();
544        if guard.is_some() {
545            return Ok(guard);
546        }
547        drop(guard);
548        let mut guard = self.mmap.write();
549        if guard.is_none() {
550            *guard = Some(Arc::new(packfile::map_pack(&self.file)?));
551        }
552        drop(guard);
553        Ok(self.mmap.read())
554    }
555}

572/// Pool of global shard files shared across all collections.
573///
574/// At most `MAX_SHARDS` shard files are live at any time, which bounds the
575/// pool's own resource use regardless of collection count. That bound is a
576/// policy cap over live shard *objects*, not a promise about how many the
577/// process can actually hold; the real ceilings are the surrounding system
578/// budgets:
579///
580/// - File descriptors are per-process (`RLIMIT_NOFILE`): each live shard
581///   holds a persistent file descriptor, plus transient ones during rotation
582///   or recovery, so a process reaches `EMFILE` once
583///   `(limit - baseline - headroom) / fds_per_live_shard` shards are live.
584///   Under multiple readers on one host these descriptors also aggregate
585///   against the system-wide `fs.file-max` pool (`ENFILE`).
586/// - Mappings consume virtual address space and VMAs against the *per-process*
587///   `vm.max_map_count`. A mapping is created lazily, and because mappings are
588///   reference-counted a remap can leave more than one live while previously
589///   returned ranges still reference an older mapping — so an active shard
590///   accounts for one or more VMAs, not exactly one. VMA counts do not
591///   aggregate across processes.
592///
593/// Neither budget is fixed here: `fds_per_live_shard` and the per-shard VMA
594/// count both vary with workload and implementation, so the formulas stay
595/// symbolic rather than being reduced to constants.
596///
597/// The active write shard rotates when it exceeds `MAX_SHARD_BYTES`.
598pub struct ShardPool {
599    /// Fixed-size array of shard slots. `None` means unused.
600    shards: RwLock<Vec<Option<Arc<Shard>>>>,
601    /// Index of the shard currently accepting writes.
602    active_write: parking_lot::Mutex<u16>,
603    /// Serializes shard rotation (finding/creating the next shard).
604    rotation_lock: parking_lot::Mutex<()>,
605    base_dir: PathBuf,
606    /// Per-pool rotation threshold, in bytes. Defaults to `MAX_SHARD_BYTES`
607    /// but may be set lower (e.g. by a benchmark that wants many small
608    /// packs to exercise repack/locality behavior) — never higher, since
609    /// `MAX_SHARD_BYTES` is a hard ceiling imposed by `IndexEntry`'s 32-bit
610    /// offset field.
611    max_shard_bytes: u64,
612    /// Per-pool seed mixed into index bucket/tag derivation. Generated once
613    /// at first pool creation, persisted in `pool.meta`, immutable after
614    /// construction.
615    bucket_seed: u64,
616    /// Shard IDs written to since the last sync, for scoped fsync.
617    dirty: parking_lot::Mutex<HashSet<u16>>,
618    /// Monotonically increasing `pack_id` counter for pack filenames.
619    /// Each newly created shard file gets a unique `pack_id`, so a
620    /// reused slot never collides on-disk with a still-referenced old
621    /// shard at the same slot.
622    next_pack_id: AtomicU64,
623    /// Total number of shards retired (garbage-collected after a repack)
624    /// over the pool's lifetime.
625    retired_count: AtomicU64,
626    /// Number of `shard_stats.bin` snapshots renamed into place over the
627    /// pool's lifetime (counted only after the rename succeeds; the snapshot
628    /// is best-effort observability and is not fsynced). The snapshot is
629    /// observability data whose rewrites should be gated to dirty syncs;
630    /// exposing the count lets callers and tests confirm a steady-state
631    /// clean sync pays no stats metadata write.
632    stats_snapshots: AtomicU64,
633    /// Each collection's current "home" shard: writes for a collection are routed here
634    /// instead of always following the pool-wide active-write cursor, so a
635    /// collection's records stay contiguous within a shard rather than
636    /// interleaving with whichever other collections happen to write around the
637    /// same time. Shards remain shared — multiple collections' homes can and do
638    /// coincide, especially low-traffic collections — a collection just sticks to its
639    /// home until that shard fills, rather than following the global
640    /// cursor wherever unrelated collections have since moved it.
641    collection_home: RwLock<HashMap<[u8; 16], u16>>,
642    /// Unix-seconds timestamp of the currently-persisted stats snapshot,
643    /// if one has ever been restored or written by this pool — restored
644    /// at open time from an existing `shard_stats.bin`, and updated on
645    /// every successful `persist_stats`. Lets a caller (e.g. `mtxdb
646    /// shards`) show how stale the counters it's displaying are, since a
647    /// read-only pool only ever sees whatever the real writer last
648    /// flushed, not live in-process counters.
649    stats_persisted_at: RwLock<Option<u64>>,
650    /// Wall-clock instant of the last `maybe_persist_stats` flush (or
651    /// `None` if it's never been called), used to rate-limit that
652    /// timer-driven path — separate from `stats_persisted_at`, which is
653    /// the on-disk snapshot's own unix-seconds timestamp.
654    last_stats_flush: RwLock<Option<Instant>>,
655    /// (flush, fsync) wall-clock split of the last full `sync_all`, used to
656    /// attribute sync cost between buffered frame write-out and the fsync
657    /// calls themselves. `None` until the first sync.
658    last_sync_split: Mutex<Option<(Duration, Duration)>>,
659    /// Cumulative wall-clock time (microseconds) spent waiting to acquire
660    /// `dirty` inside `sync_dirty` and `sync_all`. Pure observability —
661    /// never resets, monotonically increasing. Lets operators confirm
662    /// whether the coarse dirty-set lock is actually a contention point
663    /// under concurrent load, before replacing it with a finer-grained
664    /// mechanism.
665    dirty_lock_wait: AtomicU64,
666    last_open_timings: Mutex<Option<ShardOpenTimings>>,
667    /// Whether this pool holds the writer lock on `base_dir` (see `open`
668    /// vs `open_read_only`). Gates `persist_stats`: a read-only pool
669    /// never writes anything, including its own (always-zero) stats
670    /// snapshot, so it can't race the real writer's.
671    writable: bool,
672    /// Whether records written through this pool are zstd-attempted (see
673    /// [`crate::packfile::write_record_with_options`]). `false` for a pool
674    /// whose payloads never shrink under compression (e.g. HAMT nodes),
675    /// to skip paying the compressor's cost on every write for no benefit.
676    compress: bool,
677    /// How much of the per-frame CRC32 this pool writes and verifies (see
678    /// [`packfile::ChecksumPolicy`]). Controls both what
679    /// [`Self::put_record`] hashes when appending and what
680    /// [`Self::read_at`] verifies on point lookups.
681    checksum_policy: packfile::ChecksumPolicy,
682    /// When a shard's appended frames are written to its pack file (see
683    /// [`AppendPolicy`]). Set at open; defaults to
684    /// [`AppendPolicy::Eager`], the historical per-put behavior. [`Self`]
685    /// owns it so `put_record`'s flush decision needs no extra indirection.
686    append_policy: AppendPolicy,
687    /// Present only for a writable pool — `open_read_only` takes no lock
688    /// at all, since it never writes or truncates anything (that risk
689    /// lives one layer up, in `PackfileStorage`'s collection-index rebuild, not
690    /// here), so there's nothing for a reader to need exclusivity from.
691    ///
692    /// Never read: this is an RAII guard held purely for its `Drop` side
693    /// effect (removing the `.mtxdb.lock` marker file when the pool
694    /// itself drops). Its value is deliberately never inspected — only
695    /// its lifetime matters — so `#[allow(dead_code)]` here is correct,
696    /// not a lint dodge: the field genuinely has no read access by
697    /// design, the same way a `MutexGuard` binding is never "used" either.
698    #[allow(dead_code)]
699    writer_lock: Option<WriterLock>,
700}

883    /// Discovers packfiles in the pool directory.
884    ///
885    /// Only v4 filenames are accepted: `pack_XXXXXXXXXXXXXXXX.pack`
886    /// where the 16 lowercase hex digits encode the `pack_id`. Uppercase
887    /// hex is rejected to prevent case-insensitive collisions on
888    /// case-insensitive filesystems and to enforce a single canonical
889    /// spelling per ID.
890    ///
891    /// A `pack_*.pack` file that doesn't match the v4 format, or a legacy
892    /// `shard_*.pack` file, is rejected with `Unsupported` — this is a hard
893    /// cutover, not a silent skip. Other applications' `.pack` files are
894    /// unrelated and ignored. Duplicate `pack_id`s are also rejected.
895    fn discover_pack_files(base_dir: &Path) -> io::Result<Vec<(u64, PathBuf)>> {
896        let mut pack_files = Vec::new();
897        let mut seen = std::collections::HashSet::new();
898        for entry in fs::read_dir(base_dir)? {
899            let entry = entry?;
900            let path = entry.path();
901            if !path.extension().is_some_and(|e| e == "pack") {
902                continue;
903            }
904            let Some(stem) = path.file_stem().and_then(|s| s.to_str()) else {
905                continue;
906            };
907            let id_hex = match stem.strip_prefix("pack_") {
908                Some(id_hex) => id_hex,
909                None if stem.starts_with("shard_") => {
910                    return Err(io::Error::new(
911                        io::ErrorKind::Unsupported,
912                        format!(
913                            "found pre-v4 shard file {}; \
914                             use a fresh pool or explicitly migrate/reset it",
915                            path.display()
916                        ),
917                    ));
918                }
919                None => continue,
920            };
921
922            // v4 format: exactly 16 lowercase hex digits for pack_id.
923            if id_hex.len() != 16 {
924                return Err(io::Error::new(
925                    io::ErrorKind::Unsupported,
926                    format!(
927                        "found pre-v4 pack file {}; \
928                         use a fresh pool or explicitly migrate/reset it",
929                        path.display()
930                    ),
931                ));
932            }
933            if !id_hex.chars().all(|c| c.is_ascii_hexdigit()) {
934                return Err(io::Error::new(
935                    io::ErrorKind::InvalidData,
936                    format!(
937                        "pack filename {} contains non-hex characters",
938                        path.display()
939                    ),
940                ));
941            }
942            if id_hex != id_hex.to_ascii_lowercase() {
943                return Err(io::Error::new(
944                    io::ErrorKind::InvalidData,
945                    format!("pack filename {} must use lowercase hex", path.display()),
946                ));
947            }
948            let pack_id = u64::from_str_radix(id_hex, 16).map_err(|e| {
949                io::Error::new(io::ErrorKind::InvalidData, format!("{id_hex}: {e}"))
950            })?;
951            if !seen.insert(pack_id) {
952                return Err(io::Error::new(
953                    io::ErrorKind::InvalidData,
954                    format!(
955                        "duplicate pack_id {pack_id:#018x} found in {}",
956                        path.display()
957                    ),
958                ));
959            }
960            pack_files.push((pack_id, path));
961        }
962        Ok(pack_files)
963    }

965    /// Discover and sort pack files in `base_dir`, enforcing the
966    /// `MAX_SHARDS` capacity bound.
967    fn discover_pack_files_sorted(base_dir: &Path) -> io::Result<Vec<(u64, PathBuf)>> {
968        let mut pack_files = Self::discover_pack_files(base_dir)?;
969        if pack_files.len() > MAX_SHARDS {
970            return Err(io::Error::new(
971                io::ErrorKind::InvalidData,
972                format!(
973                    "too many pack files: {} found, but the pool can hold at most {}",
974                    pack_files.len(),
975                    MAX_SHARDS
976                ),
977            ));
978        }
979        pack_files.sort_unstable_by_key(|(pack_id, _)| *pack_id);
980        Ok(pack_files)
981    }

983    /// Recover (if writable) and open every pack file, slotting them into
984    /// the shard vector. Consumes `pack_files` so each `PathBuf` moves into
985    /// its `Shard` rather than being cloned. Returns `(recovery_time,
986    /// recovery_calls, packfile_open_time, packfile_open_calls)`.
987    fn recover_and_open_packs(
988        writable: bool,
989        pack_files: Vec<(u64, PathBuf)>,
990        shards: &mut [Option<Arc<Shard>>],
991        next_slot: &mut u16,
992        max_pack_id: &mut u64,
993    ) -> io::Result<(Duration, u64, Duration, u64)> {
994        let mut recovery_time = Duration::ZERO;
995        let mut recovery_calls: u64 = 0;
996        let mut packfile_open_time = Duration::ZERO;
997        let mut packfile_open_calls: u64 = 0;
998        for (pack_id, path) in pack_files {
999            if writable {
1000                let recovery_started = Instant::now();
1001                // Only the safety of appending matters here (every frame and CRC
1002                // validated, a torn tail cut off); the records are indexed from
1003                // the checkpoint, or rescanned on the fallback path.
1004                packfile::recover_packfile(&path).map_err(|error| {
1005                    io::Error::new(
1006                        io::ErrorKind::InvalidData,
1007                        format!(
1008                            "corrupt pack {}; failed to scan and recover: {error}",
1009                            path.display()
1010                        ),
1011                    )
1012                })?;
1013                recovery_time = recovery_time.saturating_add(recovery_started.elapsed());
1014                recovery_calls = recovery_calls.saturating_add(1);
1015            }
1016
1017            let packfile_open_started = Instant::now();
1018            let file = packfile::open_packfile(&path, writable, pack_id).map_err(|error| {
1019                io::Error::new(
1020                    io::ErrorKind::InvalidData,
1021                    format!("corrupt pack {}; failed to open: {error}", path.display()),
1022                )
1023            })?;
1024            packfile_open_time = packfile_open_time.saturating_add(packfile_open_started.elapsed());
1025            packfile_open_calls = packfile_open_calls.saturating_add(1);
1026
1027            let file_len = file.metadata()?.len();
1028            let slot = *next_slot;
1029            *next_slot = next_slot.saturating_add(1);
1030            let shard = Arc::new(Shard::new(slot, pack_id, file, path, file_len));
1031            shards[slot as usize] = Some(shard);
1032            if pack_id >= *max_pack_id {
1033                *max_pack_id = pack_id.saturating_add(1);
1034            }
1035        }
1036        Ok((
1037            recovery_time,
1038            recovery_calls,
1039            packfile_open_time,
1040            packfile_open_calls,
1041        ))
1042    }

1044    /// Bootstrap a brand-new pool: create the first pack file, persist
1045    /// `pool.meta` (with a fresh seed if needed), and write `store.meta`.
1046    /// Returns `(next_pack_id, bucket_seed, store_meta_write_time,
1047    /// pool_meta_persist_time, initial_pack_create_time)`.
1048    ///
1049    /// # Errors
1050    /// Returns `io::Error` if the pool is empty and not writable.
1051    #[allow(clippy::too_many_arguments)]
1052    fn initialize_empty_pool(
1053        base_dir: &Path,
1054        shards: &mut [Option<Arc<Shard>>],
1055        mut next_pack_id: u64,
1056        mut bucket_seed: u64,
1057        writable: bool,
1058    ) -> io::Result<(u64, u64, Duration, Duration, Duration)> {
1059        if !writable {
1060            return Err(io::Error::new(
1061                io::ErrorKind::NotFound,
1062                format!(
1063                    "no shards found in {} (nothing to read)",
1064                    base_dir.display()
1065                ),
1066            ));
1067        }
1068        let pack_id = next_pack_id;
1069
1070        let t_store_meta = Instant::now();
1071        if !base_dir.join(STORE_META_FILENAME).exists() {
1072            persist_store_meta(base_dir);
1073        }
1074        let store_meta_write_time = t_store_meta.elapsed();
1075
1076        if bucket_seed == 0 {
1077            use std::collections::hash_map::RandomState;
1078            use std::hash::{BuildHasher, Hasher};
1079            bucket_seed = RandomState::new().build_hasher().finish();
1080        }
1081
1082        let t_pool_meta_persist = Instant::now();
1083        Self::persist_pool_meta_at_sync_dir(
1084            base_dir,
1085            pack_id.checked_add(1).expect("pack_id overflow"),
1086            bucket_seed,
1087            false,
1088        )?;
1089        let pool_meta_persist_time = t_pool_meta_persist.elapsed();
1090

1136    /// `open_internal` to keep the latter under the clippy line limit; the
1137    /// raw timing values travel as one [`ShardOpenTimings`] rather than a
1138    /// dozen positional `Duration`s, and the two derived fields
1139    /// (`metadata_unattributed`, `unattributed`) are computed here.
1140    #[allow(clippy::too_many_arguments)]
1141    fn build_pool(
1142        shards: Vec<Option<Arc<Shard>>>,
1143        base_dir: PathBuf,
1144        max_shard_bytes: u64,
1145        bucket_seed: u64,
1146        next_pack_id: u64,
1147        highest_active: u16,
1148        writable: bool,
1149        compress: bool,
1150        checksum_policy: packfile::ChecksumPolicy,
1151        stats_persisted_at: Option<u64>,
1152        mut timings: ShardOpenTimings,
1153        writer_lock: Option<WriterLock>,
1154    ) -> Self {
1155        let metadata_subphases_sum = timings
1156            .pool_meta_restore
1157            .saturating_add(timings.store_meta_write)
1158            .saturating_add(timings.pool_meta_persist)
1159            .saturating_add(timings.initial_pack_create)
1160            .saturating_add(timings.persisted_stats_restore);
1161        timings.metadata_unattributed = timings
1162            .metadata_restore
1163            .saturating_sub(metadata_subphases_sum);
1164
1165        let named_phases = timings
1166            .discovery
1167            .saturating_add(timings.writer_lock)
1168            .saturating_add(timings.packfile_recovery)
1169            .saturating_add(timings.packfile_open)
1170            .saturating_add(timings.metadata_restore);
1171        timings.unattributed = timings.total.saturating_sub(named_phases);
1172
1173        Self {
1174            shards: RwLock::new(shards),
1175            active_write: parking_lot::Mutex::new(highest_active),
1176            rotation_lock: parking_lot::Mutex::new(()),
1177            base_dir,
1178            max_shard_bytes,
1179            bucket_seed,
1180            dirty: parking_lot::Mutex::new(HashSet::new()),
1181            next_pack_id: AtomicU64::new(next_pack_id),
1182            retired_count: AtomicU64::new(0),
1183            stats_snapshots: AtomicU64::new(0),
1184            collection_home: RwLock::new(HashMap::new()),
1185            stats_persisted_at: RwLock::new(stats_persisted_at),
1186            last_stats_flush: RwLock::new(None),
1187            last_sync_split: Mutex::new(None),
1188            dirty_lock_wait: AtomicU64::new(0),
1189            last_open_timings: Mutex::new(Some(timings)),
1190            writable,
1191            compress,
1192            checksum_policy,
1193            append_policy: AppendPolicy::Eager,
1194            writer_lock,
1195        }
1196    }

1198    fn open_internal(
1199        base_dir: PathBuf,
1200        writable: bool,
1201        max_shard_bytes: u64,
1202        compress: bool,
1203        checksum_policy: packfile::ChecksumPolicy,
1204    ) -> io::Result<Self> {
1205        let open_started = Instant::now();
1206        if writable {
1207            fs::create_dir_all(&base_dir)?;
1208        } else if !base_dir.is_dir() {
1209            return Err(io::Error::new(
1210                io::ErrorKind::NotFound,
1211                format!(
1212                    "shard pool directory does not exist: {}",
1213                    base_dir.display()
1214                ),
1215            ));
1216        }
1217
1218        let writer_lock_started = Instant::now();
1219        let writer_lock = writable
1220            .then(|| Self::acquire_writer_lock(&base_dir))
1221            .transpose()?;
1222        let writer_lock_time = writer_lock_started.elapsed();
1223
1224        let mut shards: Vec<Option<Arc<Shard>>> = (0..MAX_SHARDS).map(|_| None).collect();
1225        let mut next_slot: u16 = 0;
1226        let mut max_pack_id: u64 = 0;
1227
1228        let discovery_started = Instant::now();
1229        let pack_files = Self::discover_pack_files_sorted(&base_dir)?;
1230        let discovery_time = discovery_started.elapsed();
1231
1232        let (recovery_time, recovery_calls, packfile_open_time, packfile_open_calls) =
1233            Self::recover_and_open_packs(
1234                writable,
1235                pack_files,
1236                &mut shards,
1237                &mut next_slot,
1238                &mut max_pack_id,
1239            )?;
1240
1241        let highest_active = next_slot.saturating_sub(1);
1242
1243        // Restore next_pack_id and bucket_seed from pool.meta if available,
1244        // falling back to max_pack_id computed from discovered files. The meta
1245        // file survives retired-and-deleted shards, so it's strictly more
1246        // conservative than the file scan. A corrupt pool.meta is a hard
1247        // error — it means pack_ids could be reused, which is data corruption.
1248        let metadata_started = Instant::now();
1249        let (next_pack_id, bucket_seed, pool_meta_restore_time) =
1250            Self::restore_pool_meta_state(&base_dir, max_pack_id)?;
1251
1252        let (
1253            store_meta_write_time,
1254            pool_meta_persist_time,
1255            initial_pack_create_time,
1256            shards,
1257            next_pack_id,
1258            bucket_seed,
1259        ) = if shards.iter().all(std::option::Option::is_none) {
1260            let (npi, bs, smw, pmp, ipc) = Self::initialize_empty_pool(
1261                &base_dir,
1262                &mut shards,
1263                next_pack_id,
1264                bucket_seed,
1265                writable,
1266            )?;
1267            (smw, pmp, ipc, shards, npi, bs)
1268        } else {
1269            (
1270                Duration::ZERO,
1271                Duration::ZERO,
1272                Duration::ZERO,
1273                shards,
1274                next_pack_id,
1275                bucket_seed,
1276            )
1277        };
1278
1279        // Restoring is a pure read of shard_stats.bin applied to our own
1280        // in-memory Shard objects — unconditional regardless of writable.
1281        let (stats_persisted_at, persisted_stats_restore_time) =
1282            Self::restore_stats_phase(&base_dir, &shards);
1283        let metadata_restore_time = metadata_started.elapsed();
1284
1285        let total = open_started.elapsed();
1286
1287        Ok(Self::build_pool(
1288            shards,
1289            base_dir,
1290            max_shard_bytes,
1291            bucket_seed,
1292            next_pack_id,
1293            highest_active,
1294            writable,
1295            compress,
1296            checksum_policy,
1297            stats_persisted_at,
1298            ShardOpenTimings {
1299                discovery: discovery_time,
1300                writer_lock: writer_lock_time,
1301                packfile_recovery: recovery_time,
1302                packfile_recovery_calls: recovery_calls,
1303                packfile_open: packfile_open_time,
1304                packfile_open_calls,
1305                metadata_restore: metadata_restore_time,
1306                pool_meta_restore: pool_meta_restore_time,
1307                persisted_stats_restore: persisted_stats_restore_time,
1308                store_meta_write: store_meta_write_time,
1309                pool_meta_persist: pool_meta_persist_time,
1310                initial_pack_create: initial_pack_create_time,
1311                total,
1312                ..Default::default()
1313            },
1314            writer_lock,
1315        ))
1316    }

1336    /// Claim the writer lock on `base_dir`: atomically create a
1337    /// `.mtxdb.lock` marker file, which fails with `AlreadyExists` if
1338    /// another writer already holds it. Pure `std::fs` — no OS-level
1339    /// advisory-lock API and no dependency, so it needs one thing real
1340    /// `flock` gives for free: recovery if the previous holder crashed
1341    /// without cleaning up (`WriterLock`'s `Drop` handles the clean-exit
1342    /// case). We store our `{pid, starttime}` in the file and, if creation
1343    /// fails because it already exists, check whether that exact process
1344    /// (not just that PID number) is still alive before concluding the
1345    /// lock is genuinely held — a stale file from a killed process is
1346    /// removed and retried once rather than wrongly blocking forever. The
1347    /// starttime is what makes this safe across PID reuse (see
1348    /// `lock_holder_is_dead`) — a bare PID is not enough on its own.
1349    fn acquire_writer_lock(base_dir: &Path) -> io::Result<WriterLock> {
1350        Self::acquire_lock_path(&base_dir.join(".mtxdb.lock"))
1351    }

1353    /// Acquire the same `{pid, starttime}`-guarded exclusive lock at an
1354    /// arbitrary path. Shared by the per-pool writer lock (`.mtxdb.lock`) and
1355    /// the database-root shared-WAL lock (`.mtxdb.wal.lock`); see
1356    /// `acquire_writer_lock`'s doc for the staleness contract.
1357    pub(crate) fn acquire_lock_path(lock_path: &Path) -> io::Result<WriterLock> {
1358        let mut file = File::options()
1359            .read(true)
1360            .write(true)
1361            .create(true)
1362            .truncate(false)
1363            .open(lock_path)?;
1364
1365        file.try_lock_exclusive().map_err(|error| {
1366            // fs2 exposes Windows ERROR_LOCK_VIOLATION directly, while std
1367            // does not map that Windows error to WouldBlock. Preserve the
1368            // platform-independent lock contract.
1369            let lock_contended = error.kind() == io::ErrorKind::WouldBlock
1370                || (cfg!(target_os = "windows")
1371                    && error.raw_os_error() == Some(WINDOWS_ERROR_LOCK_VIOLATION));
1372            if lock_contended {
1373                io::Error::new(
1374                    io::ErrorKind::WouldBlock,
1375                    format!(
1376                        "{} is already locked by another writer",
1377                        lock_path.display()
1378                    ),
1379                )
1380            } else {
1381                error
1382            }
1383        })?;
1384
1385        file.set_len(0)?;
1386        #[cfg(target_os = "linux")]
1387        let marker = Self::proc_start_time("self").map_or_else(
1388            || format!("{}\n", std::process::id()),
1389            |start| format!("{} {start}\n", std::process::id()),
1390        );
1391        #[cfg(not(target_os = "linux"))]
1392        let marker = format!("{}\n", std::process::id());
1393        file.write_all(marker.as_bytes())?;
1394        file.sync_all()?;
1395
1396        Ok(WriterLock { _file: file })
1397    }

1399    /// Atomically create the lock file and write our `{pid, starttime}`
1400    /// into it (Linux) or just our PID (elsewhere, where starttime can't be
1401    /// read and `lock_holder_is_dead` never trusts a bare PID anyway). No
1402    /// `sync_all` here: this is advisory only, and `lock_holder_is_dead`
1403    /// already fails closed (treats an unparsable file as "might be
1404    /// alive") on a torn write from a crash mid-write — there's no
1405    /// correctness reason to pay an fsync on every lock acquisition to
1406    /// protect against that.
1407    /// Reads field 22 (`starttime`, clock ticks since boot) out of
1408    /// `/proc/<pid_or_self>/stat`. Parses from the *last* `)` rather than
1409    /// splitting on whitespace from the start: the second field (`comm`,
1410    /// the executable name in parens) can itself contain spaces and
1411    /// parens, which would otherwise misalign every field after it — a
1412    /// classic `/proc/stat` parsing bug. Field 22 is the 20th
1413    /// whitespace-separated token after that closing paren (field 3 is the
1414    /// first token after it).
1415    #[cfg(target_os = "linux")]
1416    fn proc_start_time(pid_or_self: &str) -> Option<u64> {
1417        let contents = fs::read_to_string(format!("/proc/{pid_or_self}/stat")).ok()?;
1418        let after_comm = contents.rsplit_once(')')?.1;
1419        after_comm.split_whitespace().nth(19)?.parse::<u64>().ok()
1420    }

1422    /// Liveness check for whoever wrote `lock_path`. Only verifies
1423    /// anything on Linux (`/proc/<pid>`); everywhere else this
1424    /// conservatively assumes the holder might still be alive.
1425    ///
1426    /// A time-based fallback ("reclaim if the lock file is older than N")
1427    /// was considered and rejected: a long-lived, mostly-idle writer (a
1428    /// Synapse process sitting quiet for hours is normal, not an edge
1429    /// case) looks indistinguishable from an abandoned lock once it
1430    /// crosses any fixed age threshold, since nothing refreshes the
1431    /// timestamp while the lock is held — that trades a rare stuck-lock
1432    /// annoyance (fixable by deleting the file) for an occasional silent
1433    /// second writer and packfile corruption, the wrong side of that
1434    /// trade.
1435    ///
1436    /// A bare `/proc/<pid>` existence check is *not* enough on its own,
1437    /// either: PIDs get recycled by the OS, and a container commonly has
1438    /// long-lived sibling processes (postgres, redis, nginx, ...) that
1439    /// outlive many respawns of a crash-looping writer. Once the OS
1440    /// happens to reuse a dead writer's old PID number for one of those
1441    /// unrelated long-lived processes, a plain PID check reports "alive"
1442    /// forever — not because the original holder is still running, but
1443    /// because something else now coincidentally has that PID — poisoning
1444    /// recovery permanently. Comparing the recorded `starttime` (written
1445    /// at lock-acquisition time) against the current holder of that PID
1446    /// closes this: a reused PID almost certainly has a different
1447    /// starttime, so a mismatch means the original writer is gone even
1448    /// though its old PID number is (again) in use. A lock file written by
1449    /// an older binary (bare PID, no starttime) has nothing to compare
1450    /// against and fails closed exactly as before, same as any other
1451    /// unparsable content.
1452    #[must_use]
1453    pub fn lock_holder_is_dead(lock_path: &Path) -> bool {
1454        #[cfg(target_os = "linux")]
1455        {
1456            let Ok(contents) = fs::read_to_string(lock_path) else {
1457                return false;
1458            };
1459            let mut fields = contents.split_whitespace();
1460            let Some(pid) = fields.next().and_then(|s| s.parse::<u32>().ok()) else {
1461                return false;
1462            };
1463            if !Path::new(&format!("/proc/{pid}")).exists() {
1464                return true;
1465            }
1466            // The PID exists as a live process, but that alone doesn't mean
1467            // it's still our original writer (see doc comment above) —
1468            // disambiguate via starttime when we have one to compare.
1469            match fields.next().and_then(|s| s.parse::<u64>().ok()) {
1470                Some(recorded_start) => match Self::proc_start_time(&pid.to_string()) {
1471                    Some(current_start) => current_start != recorded_start,
1472                    // Couldn't read the current holder's stat (raced with
1473                    // its own exit, permissions, ...) — fail closed.
1474                    None => false,
1475                },
1476                // Old-format lock file, or starttime collection failed at
1477                // creation time — nothing to disambiguate a reuse with.
1478                None => false,
1479            }
1480        }
1481        #[cfg(not(target_os = "linux"))]
1482        {
1483            let _ = lock_path;
1484            false
1485        }
1486    }

1514    /// Discovers new pack files on disk and adds them to the pool.
1515    ///
1516    /// # Errors
1517    /// Returns `io::Error` if reading the directory, opening a discovered
1518    /// pack, reading its metadata, or allocating a shard slot fails.
1519    pub fn discover_shards(&self) -> io::Result<()> {
1520        let pack_files = Self::discover_pack_files(&self.base_dir)?;
1521        let mut shards = self.shards.write();
1522        let existing: std::collections::HashSet<u64> =
1523            shards.iter().flatten().map(|s| s.pack_id).collect();
1524        let mut sorted_files = pack_files;
1525        sorted_files.sort_unstable_by_key(|(id, _)| *id);
1526        for (pack_id, path) in sorted_files {
1527            if existing.contains(&pack_id) {
1528                continue;
1529            }
1530            let slot = shards.iter().position(Option::is_none).ok_or_else(|| {
1531                io::Error::new(
1532                    io::ErrorKind::InvalidData,
1533                    "shard pool full while discovering pack files",
1534                )
1535            })?;
1536            let file = crate::packfile::open_packfile(&path, false, pack_id)?;
1537            let file_len = file.metadata()?.len();
1538            let slot = u16::try_from(slot)
1539                .map_err(|_| io::Error::new(io::ErrorKind::InvalidData, "invalid shard slot"))?;
1540            shards[usize::from(slot)] = Some(std::sync::Arc::new(Shard::new(
1541                slot, pack_id, file, path, file_len,
1542            )));
1543        }
1544        Ok(())
1545    }

1600    /// Persist `next_pack_id` and `bucket_seed` to `pool.meta` using an atomic

1603    /// point never leaves a `pack_id` that could be reused.
1604    fn persist_pool_meta_at(base_dir: &Path, next: u64, bucket_seed: u64) -> io::Result<()> {

1608    /// Internal form of [`Self::persist_pool_meta_at`] allowing callers that
1609    /// create multiple files (such as fresh pool open) to defer directory
1610    /// synchronization until all renames have completed.
1611    fn persist_pool_meta_at_sync_dir(
1612        base_dir: &Path,
1613        next: u64,
1614        bucket_seed: u64,
1615        sync_dir: bool,
1616    ) -> io::Result<()> {
1617        let mut buf = Vec::with_capacity(21);
1618        buf.extend_from_slice(b"MTXP");
1619        buf.push(POOL_META_VERSION);
1620        buf.extend_from_slice(&next.to_le_bytes());
1621        buf.extend_from_slice(&bucket_seed.to_le_bytes());
1622
1623        let final_path = Self::pool_meta_path(base_dir);
1624        let tmp_path = final_path.with_extension(format!("meta.tmp.{}", std::process::id()));
1625        let write_result = (|| -> io::Result<()> {
1626            let mut tmp = File::create(&tmp_path)?;
1627            tmp.write_all(&buf)?;
1628            tmp.sync_all()?;
1629            drop(tmp);
1630            fs::rename(&tmp_path, &final_path)?;
1631            if sync_dir {

1691    /// shard's counters when its `pack_id` matches — a stale snapshot entry
1692    /// (from a shard since retired and deleted) is silently skipped
1693    /// rather than misapplied. Returns the snapshot's persisted-at
1694    /// unix timestamp, if the file was readable and current-format.
1695    ///
1696    /// Best-effort: a missing, truncated, or corrupt file just means no
1697    /// stats are restored — never a startup failure over stats alone.

1699        let path = Self::stats_path(base_dir);
1700        let buf = fs::read(&path).ok()?;
1701        if buf.len() < STATS_HEADER_LEN || &buf[0..4] != STATS_MAGIC || buf[4] != STATS_VERSION {
1702            return None;
1703        }
1704        let persisted_at = u64::from_le_bytes(buf[5..13].try_into().ok()?);
1705        let body = &buf[STATS_HEADER_LEN..];
1706        for chunk in body.chunks(STATS_RECORD_LEN) {
1707            let Ok(rec) = <&[u8; STATS_RECORD_LEN]>::try_from(chunk) else {
1708                break;
1709            };

1714                    break;
1715                }
1716            }
1717        }

1721    /// Persist every currently-open shard's IO/sync counters to disk,
1722    /// keyed by `pack_id` so a retired/reused slot's stale
1723    /// numbers are never mistakenly restored onto a new shard.
1724    ///
1725    /// Writes to a temp file and renames into place, so a crash mid-write
1726    /// leaves the previous snapshot (or none) rather than a torn file.
1727    ///
1728    /// # Errors
1729    /// Returns `io::Error` on write or rename failure.
1730    fn persist_stats(&self) -> io::Result<()> {
1731        let mut buf = Vec::new();
1732        buf.extend_from_slice(STATS_MAGIC);
1733        buf.push(STATS_VERSION);
1734        let persisted_at = SystemTime::now()
1735            .duration_since(UNIX_EPOCH)
1736            .map_or(0, |d| d.as_secs());
1737        buf.extend_from_slice(&persisted_at.to_le_bytes());
1738        for (_slot, shard) in self.all_shards() {
1739            shard.stats().encode(shard.pack_id, &mut buf);
1740        }
1741
1742        // Unique per (process, call) — the same base_dir can be opened by
1743        // more than one process at once (e.g. a long-running embedder plus
1744        // a short-lived `mtxdb shards` CLI invocation), and a fixed tmp
1745        // filename shared across them races: one process's rename() can
1746        // consume the shared tmp path out from under another's, which
1747        // then sees ENOENT on its own rename despite having written its
1748        // tmp file successfully moments earlier.
1749        let unique = STATS_TMP_COUNTER.fetch_add(1, Ordering::Relaxed);
1750        let tmp_path = Self::stats_path(&self.base_dir)
1751            .with_extension(format!("bin.tmp.{}.{unique}", std::process::id()));
1752        let final_path = Self::stats_path(&self.base_dir);
1753        // Deliberately no fsync here. The snapshot is observability data
1754        // (rebuilt from live counters; open tolerates a missing/old/short
1755        // file), and the rename below is never followed by a directory
1756        // fsync -- so the old `tmp.sync_all()` paid an fsync per persist for
1757        // a power-loss guarantee the rename didn't actually provide.
1758        let write_result = (|| -> io::Result<()> {
1759            let mut tmp = File::create(&tmp_path)?;
1760            tmp.write_all(&buf)?;
1761            Ok(())
1762        })();
1763        if let Err(e) = write_result {
1764            // Don't leave a half-written tmp file behind under its
1765            // now-unique name — best-effort, the write already failed.
1766            let _ = fs::remove_file(&tmp_path);
1767            return Err(e);
1768        }
1769        fs::rename(&tmp_path, &final_path)?;
1770        self.stats_snapshots.fetch_add(1, Ordering::Relaxed);
1771        *self.stats_persisted_at.write() = Some(persisted_at);
1772        Ok(())
1773    }

1919    /// Append a record to its collection's home shard (see `collection_home`).
1920    /// Returns `(slot, offset)`. Rotates the collection to a new home shard
1921    /// if its current one is full.
1922    ///
1923    /// Under the default [`AppendPolicy::Eager`] the frame is written to the
1924    /// page cache immediately — one positioned write per put, the historical
1925    /// behavior (unsynced bytes survive this process exiting; durability on
1926    /// crash/power loss still requires a `sync_*`). Under
1927    /// [`AppendPolicy::Buffered`] the frame is appended to the shard's
1928    /// in-memory pending region instead: offsets are handed out into the
1929    /// virtual `[file_len, file_len + pending)` range and committed by a
1930    /// later flush. Reads of a virtual offset first flush (see
1931    /// [`Self::read_at`]), so callers observing the returned offset always
1932    /// see the record either way.

1934    /// # Errors
1935    /// Returns `io::Error` on write, flush, or rotation failure.
1936    pub fn put_record(&self, record: &Record) -> io::Result<(u16, u64)> {

1938            .map(|(slot, offset, _)| (slot, offset))
1939    }

1941    /// Append a record and return its exact encoded frame size as well as its
1942    /// location. The extra size is used by incremental physical accounting.
1943    ///
1944    /// # Errors
1945    /// Returns an I/O or encoding error if the record cannot be appended.
1946    pub fn put_record_with_len(&self, record: &Record) -> io::Result<(u16, u64, u64)> {
1947        let mut shard = self.shard_for_collection(&record.collection_id);
1948        loop {
1949            if shard.is_poisoned() {
1950                return Err(io::Error::other(format!(
1951                    "shard {} is poisoned after an unrecoverable rollback failure",
1952                    shard.slot
1953                )));
1954            }
1955            // Uncompressed upper bound, used only for the pre-append
1956            // capacity check below — `write_record` may compress the
1957            // payload and write fewer bytes than this, but never more,
1958            // so checking against this bound never lets a shard overflow
1959            // MAX_SHARD_BYTES; it may just rotate a little earlier than
1960            // strictly necessary when compression would have made it fit.
1961            let max_record_len = record.serialized_len() as u64;
1962
1963            // `append_lock` serializes writers *and* flushers, and is held
1964            // from the capacity check through the append, the automatic
1965            // flush, and (on failure) the rollback. That is what makes the
1966            // rollback safe: a shard is shared by many collections (its
1967            // `collection_home` maps several collections onto the same
1968            // slot), so per-collection put mutexes DON'T serialize the
1969            // appenders — only this lock does. While it is held, no other
1970            // put, of any collection, can append to `pending`, so our frame
1971            // is provably the tail when the failed flush needs to retract
1972            // it.
1973            let append_guard = shard.append_lock.lock();
1974            // Re-check poison after acquiring the lock: the check above is
1975            // only a fast-path skip for the common case. A concurrent
1976            // put/flush can poison the shard between that check and here —
1977            // `append_lock` is what actually excludes further appends, so
1978            // it's the only place this check is load-bearing.
1979            if shard.is_poisoned() {
1980                return Err(io::Error::other(format!(
1981                    "shard {} is poisoned after an unrecoverable rollback failure",
1982                    shard.slot
1983                )));
1984            }
1985            // `file_len` is updated at flush time, making the virtual end
1986            // (committed length plus already-buffered frames) the
1987            // authoritative next offset. No syscalls happen per append.
1988            let committed = shard.file_len.load(Ordering::Acquire);
1989            let buffered = shard.pending.lock().len() as u64;
1990            let virtual_end = committed.checked_add(buffered).ok_or_else(|| {
1991                io::Error::new(io::ErrorKind::InvalidData, "file_len + pending overflow")
1992            })?;
1993
1994            // Check capacity while holding the append lock and against
1995            // the virtual end — avoids a TOCTOU race where two threads
1996            // both pass the check then one exceeds the limit.
1997            let fits = virtual_end
1998                .checked_add(max_record_len)
1999                .is_some_and(|sum| sum <= self.max_shard_bytes);
2000            if !fits && committed > packfile::HEADER_LEN as u64 {
2001                drop(append_guard);
2002                // First make room: push the buffered bytes to disk so
2003                // the capacity check is measured against real length.
2004                self.flush_shard(&shard)?;
2005                shard = self.rotate_collection_full_home(&record.collection_id, shard.slot)?;
2006                continue;
2007            }
2008
2009            let frame = packfile::encode_record_with_options(
2010                record,
2011                self.compress,
2012                self.checksum_policy.computes_checksum(),
2013            )?;
2014            let frame_len = frame.len();
2015            let pending_start = {
2016                let mut pending = shard.pending.lock();
2017                let start = pending.len();
2018                pending.extend_from_slice(&frame);
2019                shard.pending_records.fetch_add(1, Ordering::Relaxed);
2020                start
2021            };
2022
2023            let flush_result = match self.append_policy {
2024                AppendPolicy::Eager => self.flush_shard_with_guard(&shard, &append_guard),
2025                AppendPolicy::Buffered { max_pending_bytes } => {
2026                    if shard.pending.lock().len() >= max_pending_bytes {
2027                        self.flush_shard_with_guard(&shard, &append_guard)
2028                    } else {
2029                        Ok(())
2030                    }
2031                }
2032            };
2033            if let Err(e) = flush_result {
2034                // The caller sees `Err` and has not indexed this record, yet
2035                // its frame is still in `pending` — a later flush would
2036                // silently make a "failed" put durable, surfacing it as
2037                // stored data on restart's full scan. Retract it: a failed
2038                // flush leaves `pending` untouched, and `append_guard` is
2039                // still held from the append so nothing appended after our
2040                // frame, making the tail truncation exact. A torn write's
2041                // partially-written prefix is also reined back (`set_len` to
2042                // the still-current committed frontier) so leftover bytes
2043                // never read back as phantom records on a rebuild.
2044                let mut pending = shard.pending.lock();
2045                let end = pending_start.saturating_add(frame_len);
2046                if pending.len() >= end {
2047                    // Only discard the frame once the tail truncation has
2048                    // actually succeeded — otherwise the torn bytes stay on
2049                    // disk past `committed` with no frame left to account
2050                    // for them, and a restart's full scan (which reads real
2051                    // file length, not our tracked `file_len`) can surface
2052                    // them as phantom data. If truncation itself fails,
2053                    // leave `pending` untouched and poison the shard so
2054                    // nothing else appends past, or scans over, the corrupt
2055                    // tail.
2056                    match shard.file.set_len(committed) {
2057                        Ok(()) => {
2058                            pending.truncate(pending_start);
2059                            shard.pending_records.fetch_sub(1, Ordering::Relaxed);
2060                        }
2061                        Err(rollback_err) => {
2062                            drop(pending);
2063                            shard.poisoned.store(true, Ordering::Release);
2064                            return Err(io::Error::other(format!(
2065                                "flush failed ({e}) and rollback truncation also failed \
2066                                 ({rollback_err}); shard {} poisoned",
2067                                shard.slot
2068                            )));
2069                        }
2070                    }
2071                }
2072                return Err(e);
2073            }
2074            let disk_bytes = u64::try_from(frame_len).unwrap_or(u64::MAX);
2075            return Ok((shard.slot, virtual_end, disk_bytes));
2076        }
2077    }

2108    /// Commit one shard's buffered frames to disk as a single positioned
2109    /// write. No-op if the shard has nothing buffered. Updates `file_len`,
2110    /// IO/sync counters, and the dirty set only after a successful write,
2111    /// so a partial failure defers rather than loses the accounting.
2112    ///
2113    /// Safe to call concurrently with writers and other flushers: the
2114    /// additional buffered bytes are simply rolled into this (or the next)
2115    /// write, keeping `file_len` monotonic and offsets stable.
2116    ///
2117    /// Note: the shard→collection count bookkeeping is deliberately NOT
2118    /// here — it stays per-record in `PackfileStorage::put`/`put_many`
2119    /// (via `record_new_shard_collection`), entirely unchanged by
2120    /// buffering, so no flush-time aggregation or recompute can drift from
2121    /// its current semantics.
2122    ///
2123    /// # Errors
2124    /// Returns `io::Error` on flush failure.
2125    pub(crate) fn flush_shard(&self, shard: &Arc<Shard>) -> io::Result<()> {
2126        if shard.is_poisoned() {
2127            return Err(io::Error::other(format!(
2128                "shard {} is poisoned after an unrecoverable rollback failure",
2129                shard.slot
2130            )));
2131        }
2132        let guard = shard.append_lock.lock();
2133        self.flush_shard_with_guard(shard, &guard)
2134    }
2135
2136    /// Body of `flush_shard` under a caller-supplied `append_lock` guard, so
2137    /// a writer that already holds the lock (see `put_record`, which must
2138    /// keep it across append → flush → rollback) can flush without
2139    /// re-acquiring it.
2140    fn flush_shard_with_guard(
2141        &self,
2142        shard: &Arc<Shard>,
2143        _guard: &parking_lot::MutexGuard<'_, ()>,
2144    ) -> io::Result<()> {
2145        // The guard proves `append_lock` is held, so this is the single
2146        // point every flush path — direct, via `flush_shard`, or inline
2147        // from `put_record`'s own automatic flush — actually funnels
2148        // through under lock. Enforcing the poison check here, not just at
2149        // each call site, means correctness never depends on every current
2150        // and future caller remembering to check first.
2151        if shard.is_poisoned() {
2152            return Err(io::Error::other(format!(
2153                "shard {} is poisoned after an unrecoverable rollback failure",
2154                shard.slot
2155            )));
2156        }
2157        {
2158            let mut pending_guard = shard.pending.lock();
2159            if pending_guard.is_empty() {
2160                // Nothing buffered: nothing to commit or dirty.
2161                return Ok(());
2162            }
2163            let committed = shard.file_len.load(Ordering::Acquire);
2164            // Write directly from the still-buffered bytes rather than
2165            // draining them first: `append_lock` is held for this whole
2166            // block, so nothing else can append to `pending` meanwhile,
2167            // and keeping the bytes in place until the write actually
2168            // succeeds means a failed write leaves them buffered for the
2169            // next flush attempt instead of silently discarding already
2170            // "successful" puts whose offsets the index has already
2171            // handed out.
2172            // On Windows this is a seek followed by write_all rather than an
2173            // append write. It is safe because append_lock serializes every
2174            // flush and no other path moves the shared file pointer. The
2175            // committed file_len is initialized from file metadata on reopen
2176            // and advanced only after a successful flush, so this cannot
2177            // overwrite data or leave a gap at the on-disk EOF frontier.
2178            let file = shard.file.try_clone()?;

2184                file.seek(io::SeekFrom::Start(committed))?;
2185                file.write_all(&pending_guard)?;
2186            }

2188            kick_writeback(&file, committed, pending_guard.len() as u64);
2189            let new_len = pending_guard.len() as u64;

2191                io::Error::new(

2193                    "committed + buffered len overflow",
2194                )
2195            })?;

2199            shard.write_count.fetch_add(
2200                shard.pending_records.swap(0, Ordering::Relaxed),
2201                Ordering::Relaxed,
2202            );
2203            shard.bytes_written.fetch_add(new_len, Ordering::Relaxed);
2204            // Hold `append_lock` for the whole commit (snapshot → write →
2205            // length/accounting update) so no put can observe a file_len
2206            // that hasn't caught up with bytes already handed out.
2207        }
2208        self.dirty.lock().insert(shard.slot);

2210    }

2682    /// Rotate while `rotation_lock` is held by the caller.
2683    fn rotate_locked(&self) -> io::Result<()> {
2684        let current = *self.active_write.lock();
2685
2686        let mut shards = self.shards.write();
2687
2688        // Find the next available slot (skip current, prefer retired/empty)
2689        for offset in 1..=MAX_SHARDS_U16 {
2690            let candidate = current.wrapping_add(offset).wrapping_rem(MAX_SHARDS_U16);
2691            if shards[candidate as usize].is_none() {
2692                let pack_id = self.next_pack_id.load(Ordering::Relaxed);
2693                let next = pack_id.checked_add(1).expect("pack_id overflow");
2694
2695                // Persist the high-water mark BEFORE creating the pack file.
2696                // A crash after pack creation but before the next persist
2697                // would leave a pack_id in use with no pool.meta reservation.
2698                Self::persist_pool_meta_at(&self.base_dir, next, self.bucket_seed)?;
2699
2700                let (file, path) = Self::create_packfile_atomically(&self.base_dir, pack_id)?;
2701                let file_len = file.metadata()?.len();
2702                shards[candidate as usize] = Some(Arc::new(Shard::new(
2703                    candidate, pack_id, file, path, file_len,
2704                )));
2705                // Advance the in-memory counter only after successful creation.
2706                self.next_pack_id.store(next, Ordering::Relaxed);
2707                drop(shards);
2708                *self.active_write.lock() = candidate;
2709                return Ok(());
2710            }
2711        }
2712
2713        // All slots occupied and none retired — fail rather than silently
2714        // overwriting a shard that may still be referenced by collection indexes.
2715        // The caller must repack to reclaim retired shard slots before
2716        // rotating again.
2717        Err(io::Error::other(
2718            "shard pool full: all slots occupied by active shards; repack to reclaim",
2719        ))
2720    }

2786    /// tick (e.g. a ~1s flush loop), so a `mtxdb shards`-style reader in
2787    /// another process sees reasonably fresh counters between real
2788    /// `sync_all`/`sync_dirty` calls — those already persist stats as a
2789    /// side effect, but a write-heavy, rarely-syncing process could
2790    /// otherwise leave a live writer's snapshot stale indefinitely.
2791    ///
2792    /// No-op on a read-only pool, and a no-op if called again before
2793    /// `min_interval` has passed since the last flush from here — the
2794    /// caller can tick this on every write without it turning into
2795    /// `persist_stats`' full snapshot-write-and-rename on every call.
2796    /// `persist_stats` issues no fsync (the snapshot is best-effort
2797    /// observability), so this path never forces durability work onto a
2798    /// write.
2799    pub fn maybe_persist_stats(&self, min_interval: Duration) {
2800        if !self.writable {
2801            return;
2802        }
2803        let now = Instant::now();
2804        {
2805            let mut last = self.last_stats_flush.write();
2806            if last.is_some_and(|prev| now.duration_since(prev) < min_interval) {
2807                return;
2808            }
2809            *last = Some(now);
2810        }
2811        self.persist_stats_best_effort();
2812    }

2863    /// Sync all shards to disk.
2864    ///
2865    /// Commits any buffered frames first, so fsync covers everything a
2866    /// caller believes it has put.
2867    ///
2868    /// **Deliberately does not skip-based-coalesce.** Unlike [`Self::sync_dirty`],
2869    /// which may skip a shard once another caller has fsynced it, this fsyncs
2870    /// every shard unconditionally: its contract is that everything put before
2871    /// the call is durable, and "someone else fsynced this shard at some point"
2872    /// does not prove that. The per-shard `sync_lock` here therefore provides
2873    /// mutual exclusion only, not the coalescing win `sync_dirty` gets — two
2874    /// concurrent `sync_all` callers can each issue their own fsync, which is
2875    /// correct, just not deduplicated.
2876    ///
2877    /// # Errors
2878    /// Returns `io::Error` on sync failure.
2879    pub fn sync_all(&self) -> io::Result<()> {
2880        let flush_started = Instant::now();
2881        self.flush_all()?;
2882        let flush_elapsed = flush_started.elapsed();
2883        let fsync_started = Instant::now();
2884        let mut had_dirty = false;
2885        for (id, shard) in self.all_shards() {
2886            // Full-pool sync fsyncs every shard unconditionally (that is its
2887            // contract), so unlike `sync_dirty` it can't skip a clean shard —
2888            // but the per-shard lock still serializes it against a concurrent
2889            // `sync_dirty` fsyncing the same file, and the dirty bit is
2890            // claimed before the fsync rather than under the pool lock
2891            // across it.
2892            let _sync_guard = shard.sync_lock.lock();
2893            if self.take_dirty(id) {
2894                had_dirty = true;
2895            }
2896            if let Err(error) = shard.fsync_advancing_watermark(false) {
2897                self.restore_dirty(id);
2898                return Err(error);
2899            }
2900            shard.sync_count.fetch_add(1, Ordering::Relaxed);
2901        }
2902        let fsync_elapsed = fsync_started.elapsed();
2903        *self.last_sync_split.lock() = Some((flush_elapsed, fsync_elapsed));
2904        // The stats snapshot is observability, not state: only rewrite it when
2905        // a sync actually moved or committed data (parallel to `sync_dirty`),
2906        // so a steady-state writer that syncs between writes pays no metadata
2907        // write+rename for a file nothing changed. Periodic freshness is
2908        // `maybe_persist_stats`' job.
2909        if had_dirty {
2910            self.persist_stats_best_effort();
2911        }
2912        Ok(())
2913    }

2942    /// Sync only shards written to since the last sync.
2943    ///
2944    /// Each shard's dirty bit is claimed before its fsync and restored if the
2945    /// fsync fails, so a partial failure leaves the untried shards marked
2946    /// dirty for the next call.
2947    ///
2948    /// Concurrent callers coalesce per shard: the first to claim a shard's
2949    /// dirty bit fsyncs it; any other caller wanting the same shard waits on
2950    /// that shard's `sync_lock`, then finds the bit already cleared and skips
2951    /// its own fsync. The pool-wide `dirty` lock is never held across an
2952    /// fsync.
2953    ///
2954    /// Skipping is safe because a shard's dirty bit is only cleared when an
2955    /// fsync that *covers that shard's current bytes* has completed: every
2956    /// flush writes its bytes (`flush_shard_with_guard`) before marking the
2957    /// shard dirty, and each caller flushes before it claims — so any byte
2958    /// whose dirty bit this caller observes was written before the fsync that
2959    /// clears it, and an fsync covers all writes that completed before it.
2960    /// A write that lands after the claim re-marks the shard dirty and is
2961    /// caught by the next sync.
2962    ///
2963    /// Commits any buffered frames first, so the fsync covers everything a
2964    /// caller believes it has put.
2965    ///
2966    /// # Cross-process visibility
2967    ///
2968    /// This makes the dirty shard bytes durable, but it is not the
2969    /// transaction-level durability or publication barrier: it does not commit
2970    /// the journal/LSN boundary and does not advance the LSN a reader in
2971    /// another process observes, so nothing about another process's view
2972    /// follows from it returning `Ok`.
2973    ///
2974    /// * With a [journal](crate::journal) enabled, the transaction-level
2975    ///   durability barrier is
2976    ///   [`JournalCoordinator::sync_through`](crate::journal::JournalCoordinator::sync_through)
2977    ///   (reached via the packfile storage sync path), and shard fsyncs are
2978    ///   acceleration recovered from the journal on reopen. A separate process
2979    ///   observes committed data through the read-committed overlay
2980    ///   (`PackfileStorage::open_read_committed` / `get_read_committed`, the
2981    ///   `multi-reader` feature), whose boundary is the journal's committed
2982    ///   LSN. That boundary can advance *before* this call, and a
2983    ///   visible-but-uncommitted group is not crash-durable.
2984    /// * Without a journal, a peer sees shard bytes only after an explicit
2985    ///   [`StorageEngine::refresh_collection`](crate::storage::StorageEngine::refresh_collection)
2986    ///   re-scan; the engine has no ambient cross-process invalidation. A
2987    ///   successful `sync_dirty` then guarantees those bytes survive a crash,
2988    ///   not that a peer has picked them up.
2989    ///
2990    /// Same-process readers go through the live index; cross-process readers
2991    /// go through the journal overlay (or an explicit rescan).
2992    ///
2993    /// # Errors
2994    /// Returns `io::Error` on sync failure.
2995    pub fn sync_dirty(&self) -> io::Result<()> {
2996        let flush_started = Instant::now();
2997        self.flush_all()?;
2998        let flush_elapsed = flush_started.elapsed();
2999        let fsync_started = Instant::now();
3000        let mut synced_any = false;
3001        for id in self.dirty_snapshot() {
3002            let Some(shard) = self.get_shard(id) else {
3003                // Slot was retired between snapshot and lookup; leave any
3004                // remaining dirty bit for the next pass to resolve.
3005                continue;
3006            };
3007            let _sync_guard = shard.sync_lock.lock();
3008            if !self.take_dirty(id) {
3009                // A concurrent syncer we just waited for already fsynced
3010                // this shard — nothing left to do.
3011                continue;
3012            }
3013            // `sync_data` (fdatasync): the packfile is append-only and its
3014            // frame framing/size live in the data stream, so flushing data
3015            // plus the size metadata makes appended bytes readable after a
3016            // crash; the mode/timestamp metadata `sync_all` would also write
3017            // is not needed. The full-barrier `sync_all` above keeps
3018            // `sync_all`.
3019            if let Err(error) = shard.fsync_advancing_watermark(true) {
3020                self.restore_dirty(id);
3021                return Err(error);
3022            }
3023            shard.sync_count.fetch_add(1, Ordering::Relaxed);
3024            synced_any = true;
3025        }
3026        let fsync_elapsed = fsync_started.elapsed();
3027        *self.last_sync_split.lock() = Some((flush_elapsed, fsync_elapsed));
3028        if synced_any {
3029            // Rate-limit the snapshot on the hot dirty path: it is
3030            // observability data, and `sync_all` (on dirty shards) / `Drop`
3031            // persist it.
3032            self.maybe_persist_stats(STATS_FLUSH_MIN_INTERVAL);
3033        }
3034        Ok(())
3035    }

3037    /// Retire a shard: mark it for deletion and free its pool slot.
3038    ///
3039    /// The shard file is deleted when the last `Arc<Shard>` reference drops
3040    /// (via `Shard::drop` when `is_current == false`). Setting the slot to
3041    /// `None` allows `rotate()` to reuse it.
3042    ///
3043    /// Does nothing if the slot is already empty or is the active write shard.
3044    ///
3045    /// Also does nothing if the shard has buffered frames that fail to
3046    /// flush: retiring anyway would mark the shard for deletion (via
3047    /// `Shard::drop`) while data an index still references is only in
3048    /// memory, destroying it. Leaving the slot occupied means the next
3049    /// flush attempt (or another retire) gets another chance.
3050    pub fn retire_slot(&self, slot: u16) {
3051        // Commit any buffered frames first: records are only ever retired
3052        // after their bytes are on disk, so a retire can never strand data
3053        // that an index still references.
3054        if let Some(shard) = self.get_shard(slot) {
3055            if self.flush_shard(&shard).is_err() {
3056                return;
3057            }
3058        }
3059        let mut shards = self.shards.write();
3060        let mut dirty = self.dirty.lock();
3061
3062        if *self.active_write.lock() == slot {
3063            return; // leave dirty unchanged — sync_dirty must fsync later
3064        }
3065        if let Some(slot_entry) = shards.get_mut(slot as usize) {
3066            if let Some(shard) = slot_entry.take() {
3067                dirty.remove(&slot);
3068                shard.is_current.store(false, Ordering::Release);
3069                self.retired_count.fetch_add(1, Ordering::Relaxed);
3070                drop(shards);
3071                self.collection_home.write().retain(|_, home| *home != slot);
3072            }
3073        }
3074    }
