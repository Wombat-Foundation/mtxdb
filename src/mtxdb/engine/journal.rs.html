41/// Size of the unit a torn write can damage. 4096 matches common 4 KiB
42/// physical sectors (advanced-format disks), which is the usual atomic write unit, though no filesystem or device guarantees it: with 512-byte
43/// spacing both slots and the base header could share one such unit and a
44/// single torn write could reach all three. It costs 12 KiB per segment.
45const MARK_SECTOR_LEN: usize = 4096;
46/// The durability mark lives in two alternating slots, each alone in its own
47/// [`MARK_SECTOR_LEN`] unit after the one that holds the immutable base header. A torn write
48/// of one slot can therefore damage only that slot: never the other slot, and
49/// never the base header, which nothing rewrites after the file is created.
50/// Each slot is `generation(8) | durable_len(8) | crc(4)`; an
51/// all-zero slot has never been written and claims nothing.
52const MARK_SLOT_OFFSETS: [usize; 2] = [MARK_SECTOR_LEN, 2 * MARK_SECTOR_LEN];
53const MARK_SLOT_LEN: usize = 20;
54/// Where journal groups begin: after the base header's sector and both mark
55/// sectors. Everything that needs "the end of the header" uses this.
56const FILE_HEADER_LEN: usize = 3 * MARK_SECTOR_LEN;

94/// Segment size above which a sync forces a full index checkpoint, because only
95/// a full checkpoint records journal coverage and lets the segment be
96/// reclaimed. A quarter of [`MAX_SEGMENT_LEN`] leaves headroom for writes that
97/// land while the checkpoint runs. It is a margin, not a guarantee: a writer
98/// that appends more than the remaining space during one checkpoint still fills
99/// the segment.
100/// A coordinator starts with this and can be given another with
101/// [`JournalCoordinator::set_reclaim_trigger_len`].
102pub(crate) const RECLAIM_TRIGGER_LEN: u64 = MAX_SEGMENT_LEN / 4;

131/// Per-record write LSNs for one collection's live published suffix.
132///
133/// Authoritative for optimistic concurrency while the process runs, but
134/// reconstructible rather than the sole source of truth: replayed and newly
135/// published groups repopulate it, and a record absent here is seeded from its
136/// frame's `last_write_lsn` tag before a conditional write validates. Each
137/// record's version is the `last_lsn` of the journal group that last wrote it,
138/// so a conditional write can name the exact version it observed.
139///
140/// `deleted_at` records the collection's last delete LSN. A record removed by
141/// that delete has no entry, but still resolves to a version (the delete LSN)
142/// so an "absent at version N" observation cannot be mistaken for a match
143/// after a concurrent delete/recreate.
144#[derive(Debug, Default)]
145struct RecordVersions {
146    nodes: HashMap<[u8; 16], u64>,
147    deleted_at: Option<u64>,
148}

211pub enum TxnStageState {

295/// Transaction-local mutation buffer used by
296/// [`crate::database::DatabaseTransaction`].
297///
298/// Journal entries for a SQL transaction are buffered here and published from
299/// its post-commit callback. Call [`Self::discard`] from the transaction's
300/// error callback; that is required because Synapse retains after-callbacks
301/// across retry attempts.
302///

304///

306/// applies them to packs and indexes only after its caller declares commit,

381    /// Look several records up in the mutations staged for `pool`, in the order
382    /// of `node_ids`, taking the stage lock once and scanning the staged
383    /// mutations once however many records are asked for.
384    ///
385    /// This is what lets a transaction read its own uncommitted writes. The
386    /// staged mutations are in append order, so the newest one that names a
387    /// record wins. A staged `DeleteCollection` for the record's collection
388    /// hides everything older, both earlier staged puts and whatever the live
389    /// pool holds, while a put staged after it is newer and is seen. Only an
390    /// active stage answers: once the stage is discarded, or publication has
391    /// begun, every record is `Absent` and the caller reads the live store,
392    /// which by then serves the group through the transaction overlay.
393    ///
394    /// The single pass keeps the newest put position for each wanted record
395    /// and the position of the last delete for the collection, so the cost is
396    /// the staged mutations plus the records asked for, not their product.
397    /// Payloads are cloned only for records that resolve to a put.
398    #[must_use]
399    pub fn lookup_many(
400        &self,
401        pool: ShardType,
402        collection_id: &[u8; 16],
403        node_ids: &[[u8; 16]],
404    ) -> Vec<StagedLookup> {
405        #[cfg(test)]
406        self.lookup_many_calls
407            .fetch_add(1, std::sync::atomic::Ordering::Relaxed);
408        let data = self.data.lock();
409        if self.state.load(Ordering::Acquire) != Self::ACTIVE || node_ids.is_empty() {
410            return vec![StagedLookup::Absent; node_ids.len()];
411        }
412        let wanted: HashSet<[u8; 16]> = node_ids.iter().copied().collect();
413        let mut newest_put: HashMap<[u8; 16], (usize, Option<&Vec<u8>>)> = HashMap::new();
414        let mut last_delete: Option<usize> = None;
415        for (position, mutation) in data.pools[pool_index(pool)].iter().enumerate() {
416            match mutation {
417                Mutation::Put {
418                    collection_id: staged_collection,
419                    node_id,
420                    payload,
421                } if staged_collection == collection_id && wanted.contains(node_id) => {
422                    newest_put.insert(*node_id, (position, Some(payload)));
423                }
424                Mutation::RemoveRecord {
425                    collection_id: staged_collection,
426                    node_id,
427                } if staged_collection == collection_id && wanted.contains(node_id) => {
428                    newest_put.insert(*node_id, (position, None));
429                }
430                Mutation::DeleteCollection {
431                    collection_id: staged_collection,
432                } if staged_collection == collection_id => last_delete = Some(position),
433                _ => {}
434            }
435        }
436        node_ids
437            .iter()
438            .map(|node_id| match (newest_put.get(node_id), last_delete) {
439                // A put older than the delete is hidden by it.
440                (Some((position, _)), Some(delete)) if *position < delete => StagedLookup::Deleted,
441                (Some((_, Some(payload))), _) => StagedLookup::Put((*payload).clone()),
442                (Some((_, None)), _) | (None, Some(_)) => StagedLookup::Deleted,
443                (None, None) => StagedLookup::Absent,
444            })
445            .collect()
446    }

456    /// Discard an active attempt. Safe to call more than once.
457    ///
458    /// See the "Not a rollback" section on [`TxnStage`]: this drops only the
459    /// buffered journal mutations and never touches packs or the live index.
460    pub fn discard(&self) {
461        let mut data = self.data.lock();
462        if self.state.load(Ordering::Acquire) == Self::ACTIVE {
463            data.pools.iter_mut().for_each(Vec::clear);
464            data.applied.iter_mut().for_each(Vec::clear);
465            data.expectations.clear();
466            data.record_expectations.clear();
467            data.bytes = 0;
468            self.state.store(Self::DISCARDED, Ordering::Release);
469        }
470    }

719    /// Stage a normalized removal batch all-or-nothing: drop this stage's earlier
720    /// mutations for `pending`, stage removal markers for `live_ids`, and record
721    /// each pending key's version precondition. The budget is checked before
722    /// anything changes, so an `Err` leaves the stage exactly as it was.
723    ///
724    /// # Errors
725    /// Returns an error if the stage is not active or the result would exceed
726    /// the stage's byte limit.
727    pub fn stage_removals_normalized(
728        &self,
729        pool: ShardType,
730        collection_id: [u8; 16],
731        pending: &[([u8; 16], u64)],
732        live_ids: &[[u8; 16]],
733    ) -> io::Result<()> {
734        let mut data = self.data.lock();
735        if self.state.load(Ordering::Acquire) != Self::ACTIVE {
736            return Err(io::Error::other("transaction stage is not active"));
737        }
738        let forget: HashSet<[u8; 16]> = pending.iter().map(|(node_id, _)| *node_id).collect();
739        let index = pool_index(pool);
740        let mutation_charge = |mutation: &Mutation| match mutation {
741            Mutation::Put {
742                collection_id: c,
743                node_id,
744                payload,
745            } if *c == collection_id && forget.contains(node_id) => {
746                Some(payload.len().saturating_add(64))
747            }
748            Mutation::RemoveRecord {
749                collection_id: c,
750                node_id,
751            } if *c == collection_id && forget.contains(node_id) => Some(64),
752            _ => None,
753        };
754        let released = data.pools[index]
755            .iter()
756            .filter_map(mutation_charge)
757            .fold(0_usize, usize::saturating_add);
758        let overflow = || {
759            io::Error::new(
760                io::ErrorKind::InvalidInput,
761                "transaction stage size overflow",
762            )
763        };
764        let charge = live_ids.len().checked_mul(64).ok_or_else(overflow)?;
765        let total = data
766            .bytes
767            .saturating_sub(released)
768            .checked_add(charge)
769            .ok_or_else(overflow)?;
770        if total > self.limit_bytes {
771            return Err(stage_too_large_with_limit(total, self.limit_bytes));
772        }
773
774        data.pools[index].retain(|mutation| mutation_charge(mutation).is_none());
775        data.pools[index].extend(live_ids.iter().map(|node_id| Mutation::RemoveRecord {
776            collection_id,
777            node_id: *node_id,
778        }));
779        data.applied[index] = vec![false; data.pools[index].len()];
780        data.bytes = total;
781        for (node_id, expected) in pending {
782            data.record_expectations.push(RecordExpectation {
783                pool,
784                collection_id,
785                node_id: *node_id,
786                expected: *expected,
787            });
788        }
789        Ok(())
790    }

806    /// Drop every staged put or removal for the given keys in one pass over
807    /// the pool's staged mutations, releasing exactly the budget they held.
808    ///
809    /// # Errors
810    /// Returns an error if the stage is inactive.
811    pub fn forget_staged_records(
812        &self,
813        pool: ShardType,
814        collection_id: [u8; 16],
815        node_ids: &HashSet<[u8; 16]>,
816    ) -> io::Result<()> {
817        let mut data = self.data.lock();
818        if self.state.load(Ordering::Acquire) != Self::ACTIVE {
819            return Err(io::Error::other("transaction stage is not active"));
820        }
821        let index = pool_index(pool);
822        let mut released = 0usize;
823        data.pools[index].retain(|mutation| {
824            let (c, node, charge) = match mutation {
825                Mutation::Put {
826                    collection_id: c,
827                    node_id,
828                    payload,
829                } => (c, node_id, payload.len().saturating_add(64)),
830                Mutation::RemoveRecord {
831                    collection_id: c,
832                    node_id,
833                } => (c, node_id, 64),
834                Mutation::DeleteCollection { .. } => return true,
835            };
836            if *c == collection_id && node_ids.contains(node) {
837                released = released.saturating_add(charge);
838                false
839            } else {
840                true
841            }
842        });
843        // This helper only runs while the stage is active, before publication;
844        // no mutation can have been applied yet.
845        data.applied[index] = vec![false; data.pools[index].len()];
846        data.bytes = data.bytes.saturating_sub(released);
847        Ok(())
848    }

987    /// Append staged pool groups in dependency order: edges, event-DAG,
988    /// then state. Does not fsync; the ordinary coalesced sync remains the
989    /// durability boundary. Repeated calls are safe, including after a
990    /// partial error.
991    ///
992    /// # Errors
993    /// Returns an error if a staged pool has no coordinator or its journal
994    /// cannot append the group's complete framing and trailer.
995    pub fn publish(
996        &self,
997        coordinators: [Option<&JournalCoordinator>; ShardType::ALL.len()],
998    ) -> io::Result<()> {
999        let mut data = self.data.lock();
1000        match self.state.load(Ordering::Acquire) {
1001            Self::DISCARDED | Self::JOURNAL_PUBLISHED | Self::MATERIALIZING | Self::PUBLISHED => {
1002                return Ok(())
1003            }
1004            _ => {}
1005        }
1006        if coordinators.iter().all(Option::is_none) {
1007            // Journaling is disabled process-wide, so there is no journal to
1008            // publish into.
1009            self.state.store(Self::PUBLISHED, Ordering::Release);
1010            return Ok(());
1011        }
1012        // Preserve the historical per-pool append order for the existing
1013        // pools; ServerInfo is appended after them without changing the WAL
1014        // ordering of the other pools' frames.
1015        let pools = [
1016            ShardType::Edges,
1017            ShardType::EventDag,
1018            ShardType::State,
1019            ShardType::ServerInfo,
1020        ];
1021
1022        // A shared-WAL database gives every pool the same coordinator. Keep
1023        // the transaction as one journal group in that case, so readers never
1024        // observe only a prefix of a cross-pool commit. Standalone/per-pool
1025        // journals cannot provide that boundary and retain the ordered,
1026        // retry-safe fallback below.
1027        let active_pool_count = pools
1028            .iter()
1029            .filter(|pool| {
1030                let index = pool_index(**pool);
1031                !data.appended[index] && !data.pools[index].is_empty()
1032            })
1033            .count();
1034        if active_pool_count == 0 {
1035            self.state.store(Self::PUBLISHED, Ordering::Release);
1036            return Ok(());
1037        }
1038        let active_coordinators = pools
1039            .iter()
1040            .filter_map(|pool| {
1041                let index = pool_index(*pool);
1042                (!data.appended[index] && !data.pools[index].is_empty())
1043                    .then(|| coordinators[index])
1044            })
1045            .flatten()
1046            .collect::<Vec<_>>();
1047        if active_coordinators.len() == active_pool_count {
1048            if let Some(coordinator) = active_coordinators.first().copied() {
1049                if active_coordinators
1050                    .iter()
1051                    .all(|candidate| std::ptr::eq(*candidate, coordinator))
1052                {
1053                    let batches = pools
1054                        .iter()
1055                        .filter_map(|pool| {
1056                            let index = pool_index(*pool);
1057                            (!data.appended[index] && !data.pools[index].is_empty())
1058                                .then_some((*pool, data.pools[index].as_slice()))
1059                        })
1060                        .collect::<Vec<_>>();
1061                    let receipt = coordinator.publish_tagged_groups_checked(
1062                        &batches,
1063                        &data.expectations,
1064                        &data.record_expectations,
1065                    )?;
1066                    data.appended.fill(true);
1067                    data.receipt = Some(receipt);
1068                    self.mark_journal_published()?;
1069                    return Ok(());
1070                }
1071            }
1072        }
1073
1074        if active_pool_count > 1 {
1075            return Err(io::Error::new(
1076                io::ErrorKind::Unsupported,
1077                "cross-pool transactions require one shared journal coordinator",
1078            ));
1079        }
1080
1081        for pool in pools {
1082            let index = pool_index(pool);
1083            if data.appended[index] || data.pools[index].is_empty() {
1084                data.appended[index] = true;
1085                continue;
1086            }
1087            let coordinator = coordinators[pool_index(pool)].ok_or_else(|| {
1088                io::Error::new(io::ErrorKind::NotConnected, "staged pool has no journal")
1089            })?;
1090            let receipt = coordinator.publish_group_tagged_checked(
1091                pool,
1092                &data.pools[index],
1093                &data.expectations,
1094            )?;
1095            data.appended[index] = true;
1096            data.receipt = Some(receipt);
1097        }
1098        self.mark_journal_published()?;
1099        Ok(())
1100    }

1274#[derive(Default)]
1275struct DurableWaitTotals {
1276    calls: AtomicU64,
1277    total_ns: AtomicU64,
1278    max_ns: AtomicU64,
1279    buckets: [AtomicU64; 5],
1280}

1282impl DurableWaitTotals {
1283    fn observe(&self, duration: std::time::Duration) {
1284        let nanos = u64::try_from(duration.as_nanos()).unwrap_or(u64::MAX);
1285        self.calls.fetch_add(1, Ordering::Relaxed);
1286        self.total_ns.fetch_add(nanos, Ordering::Relaxed);
1287        self.max_ns.fetch_max(nanos, Ordering::Relaxed);
1288        let micros = duration.as_micros();
1289        let bucket = if micros < 1_000 {
1290            0
1291        } else if micros < 10_000 {
1292            1
1293        } else if micros < 100_000 {
1294            2
1295        } else if micros < 1_000_000 {
1296            3
1297        } else {
1298            4
1299        };
1300        self.buckets[bucket].fetch_add(1, Ordering::Relaxed);
1301    }
1302
1303    fn snapshot(&self) -> DurableWaitLatency {
1304        DurableWaitLatency {
1305            calls: self.calls.load(Ordering::Relaxed),
1306            total: std::time::Duration::from_nanos(self.total_ns.load(Ordering::Relaxed)),
1307            max: std::time::Duration::from_nanos(self.max_ns.load(Ordering::Relaxed)),
1308            buckets: std::array::from_fn(|i| self.buckets[i].load(Ordering::Relaxed)),
1309        }
1310    }

1467/// The last group of the longest prefix of `groups` that a shared segment may
1468/// drop: every pool with a frame in each group has reported coverage through the
1469/// group's `last_lsn`. `None` when even the first group is still needed.
1470///
1471/// The one place this rule lives: the directory and the scan fallback both call
1472/// it, so they cannot disagree.
1473fn boundary_through(
1474    groups: &[GroupMark],
1475    covered: &HashMap<ShardType, u64>,
1476) -> Option<(u64, Option<ShardType>)> {
1477    let mut boundary = None;
1478    for group in groups {
1479        for (index, pool) in ShardType::ALL.into_iter().enumerate() {
1480            if group.pools & (1_u8 << index) != 0
1481                && covered.get(&pool).is_none_or(|lsn| *lsn < group.last_lsn)
1482            {
1483                // This pool's coverage stops the cut here; report it as the
1484                // blocker so reclaim can say why the suffix was retained.
1485                return boundary.map(|lsn| (lsn, Some(pool)));
1486            }
1487        }
1488        boundary = Some(group.last_lsn);
1489    }
1490    boundary.map(|lsn| (lsn, None))
1491}

1739/// starting snapshot.

1746impl JournalReplayLease {

1748    /// group at or below it.
1749    ///
1750    /// Call this once the page through `through_lsn` has been durably applied:
1751    /// the history below it is then no longer needed, so a long rebuild does not
1752    /// pin the whole retained WAL. Advancing gives up the ability to resume from

1754    /// persisted. The floor is monotonic: a stale or out-of-order call can never
1755    /// lower it.

1757        let mut pins = self.registry.pins.lock();

1761    }

2530    /// Reclaim the longest prefix of the shared segment whose every group is
2531    /// durably covered by all of that group's own pools, returning the reclaim
2532    /// result or `None` when nothing is reclaimable yet.
2533    ///
2534    /// A group is reclaimed atomically, so it may only be dropped once every
2535    /// pool that contributed a frame to it has reported coverage through the
2536    /// group's `last_lsn`. The pools in one group are independent of the pools
2537    /// in another, so the reclaimable prefix is **not** a single minimum across
2538    /// every pool: an idle pool whose last frame sits in an early group must not
2539    /// pin later groups that its frames never reached. Walk the live groups in
2540    /// order and stop at the first one any contributing pool has not covered.
2541    ///
2542    /// # Errors
2543    /// Propagates a scan or segment-rewrite failure from [`Self::reclaim_through`].
2544    pub fn reclaim_shared(&self) -> io::Result<Option<Reclaim>> {
2545        // The group directory answers this without reading the segment. The
2546        // lock order matches `reclaim_through`: the sync lock, then the journal,
2547        // and coverage last.
2548        {
2549            let _sync = self.sync_lock.lock();
2550            let mut journal = self.journal.lock();
2551            let boundary = {
2552                let coverage = self.coverage.lock();
2553                journal.shared_reclaim_boundary(&coverage.covered)
2554            };
2555            match boundary {
2556                SharedBoundary::Through(covered_lsn, blocked_by) => {
2557                    // Shared reclaim computes its own per-group/per-pool
2558                    // boundary, so apply the replay floor after that decision
2559                    // and before the segment rewrite. The lease is global to
2560                    // this segment and therefore protects groups from every
2561                    // pool, including groups whose pool coverage is complete.
2562                    let replay_limit = self.replay_reclaim_limit(covered_lsn);
2563                    let lease_limited = replay_limit < covered_lsn;
2564                    let blocked_by = (!lease_limited).then_some(blocked_by).flatten();
2565                    let reclaimed = journal.reclaim_through_with_blocker(replay_limit, blocked_by);
2566                    if reclaimed.is_ok() {
2567                        self.bump_publish_signal();
2568                    }
2569                    let reclaimed = reclaimed.map(Some);
2570                    if lease_limited {
2571                        self.record_replay_lease_outcome(&journal);
2572                    } else {
2573                        // A cut held back only by an active replay lease is not a
2574                        // pool-coverage stall. Recording one would drive forced
2575                        // checkpoints and warning logs for as long as a rebuild
2576                        // pins its own window; the next reclaim after the lease
2577                        // is dropped records the real outcome.
2578                        let coverage = self.coverage.lock();
2579                        self.record_reclaim_outcome(&journal, &coverage.covered);
2580                        *self.replay_lease_stall.lock() = None;
2581                    }
2582                    return reclaimed;
2583                }
2584                SharedBoundary::NothingCovered => {
2585                    let coverage = self.coverage.lock();
2586                    self.record_reclaim_outcome(&journal, &coverage.covered);
2587                    return Ok(None);
2588                }
2589                SharedBoundary::Untrusted => {}
2590            }
2591        }
2592        // The directory did not match the file: decide from a scan of it.
2593        let scan = Journal::scan_read_only(&self.path)?;
2594        let marks = marks_from_scan(&scan);
2595        let boundary = {
2596            let coverage = self.coverage.lock();
2597            boundary_through(&marks, &coverage.covered)
2598        };
2599        match boundary {
2600            Some((covered_lsn, blocked_by)) => self
2601                .reclaim_through_with_blocker(covered_lsn, blocked_by)
2602                .map(Some),
2603            None => Ok(None),
2604        }
2605    }

2636    /// Note how a reclaim ended, with the journal and sync locks held. A
2637    /// segment still over the trigger means the reclaim is stalled: remember
2638    /// it, and say so once, naming the pools it is waiting on.
2639    fn record_reclaim_outcome(&self, journal: &Journal, covered: &HashMap<ShardType, u64>) {
2640        let len = journal.file_len;
2641        if len <= self.reclaim_trigger_len() {
2642            *self.reclaim_stall.lock() = None;
2643            *self.last_remediation.lock() = None;
2644            return;
2645        }
2646        let epoch = self.coverage_epoch.load(Ordering::Acquire);
2647        let mut stall = self.reclaim_stall.lock();
2648        let previous = *stall;
2649        let emergency = len >= journal.segment_cap.saturating_sub(journal.segment_cap / 4);
2650        // The first reclaim to leave the segment over the trigger is normal:
2651        // pools sync one after another, so the first to cross it cannot yet
2652        // reclaim what the others have not reported. It is only worth saying
2653        // once the segment has grown by an eighth of the trigger without a
2654        // reclaim getting it back under the trigger, or it reaches the
2655        // emergency zone. The
2656        // back-off and remediation below act on the stall from the start.
2657        let grace_from = previous.map_or(len, |old| old.grace_from);
2658        let already_reported = previous.is_some_and(|old| old.reported);
2659        let report = !already_reported
2660            && (emergency || len >= grace_from.saturating_add(self.reclaim_retry_growth()));
2661        *stall = Some(ReclaimStall {
2662            at_len: len,
2663            coverage_epoch: epoch,
2664            grace_from,
2665            reported: already_reported || report,
2666        });
2667        let entered_emergency = emergency
2668            && previous.is_some_and(|old| {
2669                old.at_len < journal.segment_cap.saturating_sub(journal.segment_cap / 4)
2670            });
2671        if report {
2672            self.reclaim_stalls.fetch_add(1, Ordering::Relaxed);
2673        }
2674        if report || (already_reported && entered_emergency) {
2675            eprintln!(
2676                "warning: shared WAL reclaim is stalled at {len} of {} bytes (trigger {}); \
2677                 waiting on pools {:?} to report coverage",
2678                journal.segment_cap,
2679                self.reclaim_trigger_len(),
2680                journal.blocking_pools(covered)
2681            );
2682        }
2683    }

2709    /// If a stalled reclaim has reached the emergency zone, make the pools
2710    /// holding it back checkpoint, except `caller`, which is mid-sync. Runs
2711    /// with no journal or persistence lock held, at most once per further
2712    /// eighth of the trigger of growth unless coverage moved, and never from
2713    /// inside another remediation.
2714    pub fn remediate_blockers(&self, caller: Option<ShardType>) {
2715        if REMEDIATING.with(std::cell::Cell::get) {
2716            return;
2717        }
2718        let len = self.segment_len();
2719        if len < self.emergency_len() || !self.is_reclaim_stalled() {
2720            return;
2721        }
2722        let epoch = self.coverage_epoch.load(Ordering::Acquire);
2723        {
2724            let mut last = self.last_remediation.lock();
2725            if let Some((last_len, last_epoch)) = *last {
2726                if last_epoch == epoch && len < last_len.saturating_add(self.reclaim_retry_growth())
2727                {
2728                    return;
2729                }
2730            }
2731            *last = Some((len, epoch));
2732        }
2733        let Some(remediate) = self.blocker_remediation.lock().clone() else {
2734            return;
2735        };
2736        REMEDIATING.with(|flag| flag.set(true));
2737        for pool in self.reclaim_blockers() {
2738            if Some(pool) != caller {
2739                remediate(pool);
2740            }
2741        }
2742        REMEDIATING.with(|flag| flag.set(false));
2743    }

2811    /// Pin the journal history required by `cursor` until the returned lease
2812    /// is dropped. The pin and reclaim share the `sync_lock` -> `journal` lock
2813    /// order, so reclaim either observes this lease or completes first and
2814    /// causes an explicit expired-cursor error here.
2815    ///
2816    /// # Errors
2817    /// Returns `InvalidInput` for a different segment, `Unsupported` without a
2818    /// publish signal, `InvalidData` for a previous writer incarnation or
2819    /// reclaimed history, and `Other` if lease IDs are exhausted.
2820    pub(crate) fn pin_replay_cursor(
2821        &self,
2822        cursor: &JournalCursor,
2823    ) -> io::Result<JournalReplayLease> {
2824        if cursor.path != self.path {
2825            return Err(io::Error::new(
2826                io::ErrorKind::InvalidInput,
2827                "journal cursor belongs to a different segment",
2828            ));
2829        }
2830        let signal = self.publish_signal().ok_or_else(|| {
2831            io::Error::new(
2832                io::ErrorKind::Unsupported,
2833                "replay leases require an installed publish signal",
2834            )
2835        })?;
2836
2837        let _sync = self.sync_lock.lock();
2838        let journal = self.journal.lock();
2839        if signal.snapshot().0 != cursor.incarnation {
2840            return Err(io::Error::new(
2841                io::ErrorKind::InvalidData,
2842                "journal cursor belongs to a previous writer incarnation",
2843            ));
2844        }
2845        if cursor.lsn.saturating_add(1) < journal.base_lsn {
2846            return Err(io::Error::new(
2847                io::ErrorKind::InvalidData,

2888    /// Read a bounded page of complete, durable groups after `cursor`.
2889    ///
2890    /// Pages preserve group boundaries and journal order. The durable horizon
2891    /// is sampled before the segment is read and returned as
2892    /// [`JournalChangesPage::horizon_lsn`]: `has_more` and the returned groups
2893    /// are complete with respect to that horizon, so a group that becomes
2894    /// durable after the sample is never silently skipped. Reclaim is allowed
2895    /// to remove history; if it has advanced past the cursor, this returns an
2896    /// `InvalidData` error so the caller can discard its partial rebuild and
2897    /// take a fresh snapshot. This method does not retain a lease across page
2898    /// calls. Hold a [`JournalReplayLease`] from the snapshot cursor while
2899    /// paging if reclaim must not expire it during a long replay.
2900    ///
2901    /// `limit` bounds the number of groups, not entries or bytes. Shared-journal
2902    /// pages include all pools and collections; callers filter the entries
2903    /// they need.
2904    ///
2905    /// # Errors
2906    /// Returns `InvalidInput` for a foreign cursor or zero page size,
2907    /// `Unsupported` without a publish signal, and `InvalidData` when the writer
2908    /// incarnation differs, required history has been reclaimed, or reclaim
2909    /// changes the segment's base LSN during the read. A concurrent reclaim
2910    /// may allow retrying the same cursor if its history remains available.
2911    /// Also propagates segment I/O and validation errors.
2912    pub fn changes_since(
2913        &self,
2914        cursor: &JournalCursor,
2915        limit: usize,
2916    ) -> io::Result<JournalChangesPage> {
2917        self.validate_changes_request(cursor, limit)?;
2918        let incarnation = self.replay_incarnation(cursor)?;
2919
2920        // Sample the durable high-water mark *before* reading the WAL. A group
2921        // at or below it was appended and fsynced before this read, so the tail
2922        // scan below cannot miss it. Sampling after the read could include a
2923        // group appended in between that the scan never saw, letting the page
2924        // report `has_more == false` with a durable group outstanding.
2925        let durable_lsn = self.committed_lsn();
2926
2927        // Capture the directory boundary and the byte offset needed for at most
2928        // `limit + 1` durable groups. The extra group determines `has_more`.
2929        // Holding the journal lock makes these offsets consistent with reclaim
2930        // and append; the actual I/O happens after releasing it.
2931        let window = self.changes_window(cursor, limit, durable_lsn)?;
2932
2933        // A test can force a commit after the horizon and byte range are fixed
2934        // but before the WAL is read. That group belongs to a later pass.
2935        #[cfg(test)]
2936        if let Some(hook) = take_changes_since_before_read_hook() {
2937            hook();
2938        }
2939
2940        let (scan, resumed) = self.scan_changes(cursor, &window)?;
2941        self.validate_changes_scan(cursor, incarnation, &window, &scan)?;
2942        if resumed {
2943            self.replay_page_stats
2944                .resumed_pages
2945                .fetch_add(1, Ordering::Relaxed);
2946        }
2947        Ok(self.changes_page(cursor, incarnation, durable_lsn, limit, scan))
2948    }

3032    /// Read the segment region a [`Self::changes_since`] page needs, resuming at
3033    /// the cursor's own byte offset so the retained prefix is never re-decoded.
3034    /// The directory only caps the read; when it is untrusted the read runs to
3035    /// the end of the segment.
3036    ///
3037    /// A resume offset is invalidated by a reclaim rewrite, a replacement, or a
3038    /// rotation. Any failure therefore falls back to a full scan, and
3039    /// [`Self::validate_changes_scan`] decides whether the cursor survived.
3040    fn scan_changes(
3041        &self,
3042        cursor: &JournalCursor,
3043        window: &ChangesWindow,
3044    ) -> io::Result<(Scan, bool)> {
3045        let end = if window.directory_trusted {
3046            window.end
3047        } else {
3048            u64::MAX
3049        };
3050        // A reclaim rewrites the retained prefix, moving every surviving group,
3051        // so the cursor's offset can point past the new end. That read succeeds
3052        // with no groups even though the window says groups follow; treat it as
3053        // invalid and rebuild. An empty read is otherwise trusted when the
3054        // window itself holds no group past the offset, or when the read stopped
3055        // exactly at the cursor's offset with the segment base unchanged — a
3056        // legitimately idle tail rather than a moved one.
3057        if let Ok(scan) = Journal::scan_read_only_range(
3058            &self.path,
3059            cursor.resume_offset,
3060            end,
3061            window.expected_lsn,
3062        ) {
3063            let idle_tail =
3064                scan.valid_len == cursor.resume_offset && scan.base_lsn == cursor.base_lsn;
3065            if !scan.groups.is_empty()
3066                || (!window.has_durable_group && end == cursor.resume_offset)
3067                || idle_tail
3068            {
3069                return Ok((scan, true));
3070            }
3071        }
3072        self.replay_page_stats
3073            .full_scan_fallbacks
3074            .fetch_add(1, Ordering::Relaxed);
3075        note_changes_since_full_scan();
3076        Journal::scan_read_only(&self.path).map(|scan| (scan, false))
3077    }

3119    /// Select the durable groups after `cursor` into a page capped at `limit`,
3120    /// and derive the next cursor and `has_more` from one extra group.
3121    fn changes_page(
3122        &self,
3123        cursor: &JournalCursor,
3124        incarnation: u64,
3125        durable_lsn: u64,
3126        limit: usize,
3127        scan: Scan,
3128    ) -> JournalChangesPage {
3129        let base_lsn = scan.base_lsn;
3130        let mut following = scan
3131            .groups
3132            .into_iter()

3134            .filter(|(group, _)| group.last_lsn > cursor.lsn && group.last_lsn <= durable_lsn);

3137        while selected.len() < limit {
3138            match following.next() {
3139                Some((group, end)) => {
3140                    resume_offset = end;

3152                incarnation,
3153                lsn: through_lsn,
3154                base_lsn,
3155                resume_offset,
3156            },
3157            through_lsn,
3158            has_more,

3210    /// Durably commit pending mutations and report where the time went.
3211    ///
3212    /// # Errors
3213    /// Returns an I/O error if the target is invalid, the journal is poisoned,
3214    /// or appending/fsyncing the group fails.
3215    pub fn sync_through_timed(
3216        &self,
3217        target_lsn: u64,
3218    ) -> io::Result<(Option<CommitReceipt>, JournalSyncTimings)> {
3219        self.sync_calls.fetch_add(1, Ordering::Relaxed);
3220        let mut timings = JournalSyncTimings::default();
3221        let in_flight = self
3222            .sync_in_flight
3223            .fetch_add(1, Ordering::AcqRel)
3224            .saturating_add(1);

3239        self.reject_if_poisoned()?;

3241        if self.covered_by_prior_commit(target_lsn, &mut timings) {
3242            return Ok((None, timings));
3243        }
3244        let previously_committed = self.committed_lsn.load(Ordering::Acquire);

3247        self.flush_captured_file(&capture, &mut timings)?;

3504    /// Validate `expectations` against the live collection versions. The caller
3505    /// must hold the publication lock, which also serializes the version bumps,
3506    /// so a passing check cannot race a competing publish.
3507    fn validate_expectations(
3508        &self,
3509        expectations: &[CollectionExpectation],
3510        record_expectations: &[RecordExpectation],
3511    ) -> io::Result<()> {
3512        if expectations.is_empty() && record_expectations.is_empty() {
3513            return Ok(());
3514        }
3515        let versions = self.collection_versions.lock();
3516        for expectation in expectations {
3517            let actual = versions
3518                .get(&(expectation.pool, expectation.collection_id))
3519                .copied()
3520                .unwrap_or(0);
3521            if actual != expectation.expected {
3522                return Err(io::Error::new(
3523                    io::ErrorKind::WouldBlock,
3524                    StaleVersion {
3525                        pool: expectation.pool,
3526                        collection_id: expectation.collection_id,
3527                        expected: expectation.expected,
3528                        actual,
3529                    },
3530                ));
3531            }
3532        }
3533        for expectation in record_expectations {
3534            // A missing entry is a record whose version predates this process
3535            // or was never seeded. Use the pool/collection baseline for that
3536            // case; the read path seeds cold records from frame metadata before
3537            // staging the expectation.
3538            let actual = self.absent_record_version(
3539                expectation.pool,
3540                &expectation.collection_id,
3541                &expectation.node_id,
3542            );
3543            if actual != expectation.expected {
3544                return Err(io::Error::new(
3545                    io::ErrorKind::WouldBlock,
3546                    StaleVersion {
3547                        pool: expectation.pool,
3548                        collection_id: expectation.collection_id,
3549                        expected: expectation.expected,
3550                        actual,
3551                    },
3552                ));
3553            }
3554        }
3555        Ok(())
3556    }

3573    /// Advance the collection and per-record version trackers for a published
3574    /// group, under the publication lock, so a later conditional commit
3575    /// validates against the version this group published.
3576    fn note_published_versions(&self, staged: &[(ShardType, &[Mutation])], last_lsn: u64) {
3577        let mut versions = self.collection_versions.lock();
3578        let mut records = self.record_versions.lock();
3579        for (pool, mutations) in staged {
3580            for mutation in *mutations {
3581                match mutation {
3582                    Mutation::Put {
3583                        collection_id,
3584                        node_id,
3585                        ..
3586                    }
3587                    | Mutation::RemoveRecord {
3588                        collection_id,
3589                        node_id,
3590                    } => {
3591                        let version = versions.entry((*pool, *collection_id)).or_insert(0);
3592                        *version = (*version).max(last_lsn);
3593                        let record = records.entry((*pool, *collection_id)).or_default();
3594                        let seen = record.nodes.entry(*node_id).or_insert(0);
3595                        *seen = (*seen).max(last_lsn);
3596                    }
3597                    Mutation::DeleteCollection { collection_id } => {
3598                        let version = versions.entry((*pool, *collection_id)).or_insert(0);
3599                        *version = (*version).max(last_lsn);
3600                        let record = records.entry((*pool, *collection_id)).or_default();
3601                        record.nodes.clear();
3602                        record.deleted_at = Some(
3603                            record
3604                                .deleted_at
3605                                .map_or(last_lsn, |deleted| deleted.max(last_lsn)),
3606                        );
3607                    }
3608                }
3609            }
3610        }

3613    /// Single publication path. Holds `publication` for the whole append so
3614    /// LSN assignment, the append and the visibility advance are one step.
3615    ///
3616    /// A transaction group (`transaction` set) is published before its writes
3617    /// are materialized into the packs. Until [`Self::transaction_materialized`]
3618    /// says otherwise, no pool may claim coverage at or past its first frame:
3619    /// the packs do not hold those frames yet, so a claim (and the reclaim it
3620    /// would license) could lose an acknowledged commit to a crash. It is
3621    /// registered here, under the journal lock, so no sync can promote a
3622    /// watermark past the group before the group is on the floor list.
3623    fn publish_groups(
3624        &self,
3625        staged: &[(ShardType, &[Mutation])],
3626        transaction: bool,
3627        expectations: &[CollectionExpectation],
3628        record_expectations: &[RecordExpectation],
3629    ) -> io::Result<Option<CommitReceipt>> {
3630        let _publication = self.publication.lock();
3631        self.validate_expectations(expectations, record_expectations)?;
3632        if self.poisoned.load(Ordering::Acquire) {
3633            return Err(io::Error::other(
3634                "journal is poisoned after a failed append",
3635            ));
3636        }
3637        let mut journal = self.journal.lock();
3638        if self.poisoned.load(Ordering::Acquire) {
3639            return Err(io::Error::other(
3640                "journal is poisoned after a failed append",
3641            ));
3642        }
3643        let group_mutations: Vec<(ShardType, Mutation)> = staged
3644            .iter()
3645            .flat_map(|(pool, mutations)| mutations.iter().cloned().map(|m| (*pool, m)))
3646            .collect();
3647        if group_mutations.is_empty() {
3648            return Ok(None);
3649        }
3650        let expected_first_lsn = journal.next_lsn;
3651        let expected_count = u64::try_from(group_mutations.len()).unwrap_or(u64::MAX);
3652        let sequence = self
3653            .sequence
3654            .as_ref()
3655            .map(|counter| counter.fetch_add(1, Ordering::Relaxed));
3656        let receipt = match journal.append_group_for_current_mode(&group_mutations, sequence) {
3657            Ok(receipt) => receipt,
3658            Err(error) => {
3659                if journal.poisoned {
3660                    self.poisoned.store(true, Ordering::Release);
3661                }
3662                return Err(self.explain_full_segment(error, &journal));
3663            }
3664        };
3665        if transaction {
3666            self.unmaterialized.lock().insert(receipt.first_lsn);
3667        }
3668        debug_assert_eq!(
3669            receipt.first_lsn, expected_first_lsn,
3670            "appended group must start at the journal tail"
3671        );
3672        debug_assert_eq!(
3673            receipt
3674                .last_lsn
3675                .saturating_sub(receipt.first_lsn)
3676                .saturating_add(1),
3677            expected_count,
3678            "appended group must cover every mutation"
3679        );
3680        // Queue this group's per-pool extents before it becomes visible, still
3681        // under the journal lock. A sync that captures the new visible LSN
3682        // then always finds the extents and promotes each pool's watermark;
3683        // queuing after the visibility advance would let a sync race past them.
3684        let extents = pool_extents(
3685            receipt.last_lsn,
3686            group_mutations.iter().map(|(pool, _)| *pool),
3687        );
3688        if !extents.is_empty() {
3689            self.pending_promotions
3690                .lock()
3691                .push((receipt.last_lsn, extents));
3692        }
3693        self.published_lsn
3694            .fetch_max(receipt.last_lsn, Ordering::Release);
3695        self.visible_lsn
3696            .fetch_max(receipt.last_lsn, Ordering::Release);
3697        // Advance every mutated collection's logical version to this group's
3698        // LSN, still under the publication lock, so a later conditional commit
3699        // validates against the version this group published.
3700        self.note_published_versions(staged, receipt.last_lsn);
3701        // Publish the read-committed change boundary for cross-process workers
3702        // at the exact point the group becomes visible, before any fsync: a
3703        // reader may observe a published-but-not-yet-durable group.
3704        self.bump_publish_signal();
3705        drop(journal);
3706        // Wake the committer only when a burst is at or above its early-flush
3707        // bound; this is the hot path, so ordinary publishes must not pay a
3708        // futex wake. `committer_wake_threshold` is zero when no committer is
3709        // running. Only the first publish at or above the bound wakes in a
3710        // commit epoch; any commit advances the epoch and re-arms the next
3711        // crossing.
3712        let threshold = self.committer_wake_threshold.load(Ordering::Relaxed);
3713        if threshold > 0 && self.pending_count() >= threshold && self.claim_threshold_wake() {
3714            #[cfg(test)]
3715            self.threshold_wakes.fetch_add(1, Ordering::Relaxed);
3716            self.wake_durable_waiters();
3717        }
3718        Ok(Some(receipt))
3719    }

3803                || len >= stall.at_len.saturating_add(self.reclaim_retry_growth());

3809            || len >= stall.at_len.saturating_add(self.reclaim_retry_growth())
3810            || pool.is_some_and(|pool| self.reclaim_blockers().contains(&pool))
3811    }

3934    /// [`Self::claim_threshold_wake`] with a hook run between a successful claim
3935    /// and the epoch recheck, so tests can force a commit into that window.
3936    fn claim_threshold_wake_with(&self, mut between_claim_and_recheck: impl FnMut()) -> bool {
3937        loop {
3938            let notified = self.threshold_notified_epoch.load(Ordering::Acquire);
3939            let epoch = self.commit_epoch.load(Ordering::Acquire);
3940            if notified == epoch {
3941                return false;
3942            }
3943            if self
3944                .threshold_notified_epoch
3945                .compare_exchange(notified, epoch, Ordering::AcqRel, Ordering::Acquire)
3946                .is_ok()
3947            {
3948                between_claim_and_recheck();
3949                if self.commit_epoch.load(Ordering::Acquire) == epoch {
3950                    return true;
3951                }
3952            }
3953        }

3971    /// Block until the group covering `token` is durable.
3972    ///
3973    /// With a background committer running this waits for it to flush; the
3974    /// bounded poll interval lets the waiter notice a committer that stopped or
3975    /// a poisoned journal. A committer that failed terminally is reported
3976    /// rather than waited on. Without a committer, this is exactly
3977    /// [`Self::sync_through`] on the token's LSN — the historical blocking
3978    /// behavior, so a caller that never opts into background commits is
3979    /// unaffected.
3980    ///
3981    /// # Errors
3982    /// Returns an error if the target was never published, the journal is
3983    /// poisoned, the committer failed, or the committing sync fails.
3984    pub fn wait_durable(&self, token: DurabilityToken) -> io::Result<Option<CommitReceipt>> {
3985        let target = token.lsn;
3986        if target == 0 || target <= self.committed_lsn.load(Ordering::Acquire) {
3987            self.durable_waits_already_durable
3988                .fetch_add(1, Ordering::Relaxed);
3989            return Ok(None);

3997    fn wait_durable_blocking(&self, target: u64) -> io::Result<Option<CommitReceipt>> {
3998        loop {
3999            if target <= self.committed_lsn.load(Ordering::Acquire) {
4000                return Ok(None);
4001            }
4002            // Recheck every predicate under `durable_lock`, the same lock the
4003            // notifiers take, so a state change cannot slip between check and
4004            // park. The poll interval remains a backstop, not a requirement.
4005            let mut guard = self.durable_lock.lock();
4006            if target <= self.committed_lsn.load(Ordering::Acquire) {
4007                return Ok(None);
4008            }
4009            if self.poisoned.load(Ordering::Acquire) {
4010                return Err(io::Error::other(
4011                    "journal is poisoned after a failed commit",
4012                ));
4013            }
4014            // A stopped committer falls through to a direct commit, preserving
4015            // the historical no-committer behavior and error text, but a
4016            // committer that failed must not be silently papered over.
4017            if let Some(failure) = self.background_failure_detail() {
4018                return Err(failure.into_io());
4019            }
4020            if !self.has_background_committer() {
4021                drop(guard);
4022                return self.sync_through(target);
4023            }
4024            // Only with a live committer is an unpublished target validated
4025            // here; the no-committer path above reaches `sync_through` first.
4026            if target > self.published_lsn.load(Ordering::Acquire) {
4027                return Err(io::Error::new(
4028                    io::ErrorKind::InvalidInput,
4029                    "sync target has not been published",
4030                ));
4031            }
4032            self.durable_cv.wait_for(&mut guard, DURABILITY_WAIT_POLL);
4033        }

4037    ///
4038    /// Used by the background committer's timer and by
4039    /// [`Self::stop_background_committer`]. A no-op returning `None` when
4040    /// nothing is pending (so an idle tick neither fsyncs nor inflates the
4041    /// coalescing counter). This is the journal-level durability boundary; a
4042    /// caller that also needs its packfile checkpoint rewritten should use the
4043    /// storage-level sync path instead.

4047    pub fn flush_durable(&self) -> io::Result<Option<CommitReceipt>> {
4048        let target = self.capture_sync_target();
4049        if target == 0 || target <= self.committed_lsn.load(Ordering::Acquire) {
4050            return Ok(None);
4051        }
4052        let receipt = self.sync_through(target)?;
4053        if receipt.is_some() {

4064    /// Start a background thread that fsyncs the pending WAL group on a bounded
4065    /// interval, coalescing every mutation published in the window into one
4066    /// group. Flushes early once `max_pending` records are outstanding.
4067    /// Idempotent: a call while one is running (or stopping) is a no-op.
4068    ///
4069    /// The thread holds only a [`Weak`] reference between iterations, so
4070    /// dropping the last `Arc<JournalCoordinator>` lets it exit within one
4071    /// interval without an `Arc` cycle. That is a bounded delay, not immediate
4072    /// teardown — call [`Self::stop_background_committer`] for deterministic
4073    /// cleanup and a final flush.
4074    ///
4075    /// # Errors
4076    /// Returns [`io::ErrorKind::InvalidInput`] if `interval` is zero or
4077    /// `max_pending` is zero (both would busy-loop or flush continuously), or
4078    /// another error if the committer thread cannot be spawned.
4079    pub fn start_background_committer(
4080        self: &Arc<Self>,
4081        config: GroupCommitConfig,
4082    ) -> io::Result<()> {
4083        if config.interval.is_zero() {
4084            return Err(io::Error::new(
4085                io::ErrorKind::InvalidInput,
4086                "background committer interval must be greater than zero",
4087            ));
4088        }
4089        if config.max_pending == 0 {
4090            return Err(io::Error::new(
4091                io::ErrorKind::InvalidInput,
4092                "background committer max_pending must be greater than zero",
4093            ));
4094        }
4095        let mut slot = self.background.lock();
4096        // Running or Stopping: never spawn a second committer.
4097        if !matches!(*slot, BackgroundState::Stopped) {
4098            return Ok(());
4099        }
4100        *self.background_failure.lock() = None;
4101        // A failed run can leave the notification epoch equal to the current
4102        // commit epoch, which would suppress the first threshold wake after a
4103        // restart. Reset it so the first crossing always wakes.
4104        self.threshold_notified_epoch
4105            .store(u64::MAX, Ordering::Release);
4106        let stop = Arc::new(AtomicBool::new(false));
4107        let weak = Arc::downgrade(self);
4108        let stop_clone = Arc::clone(&stop);

4110            .name("mtxdb-journal-committer".to_owned())
4111            .spawn(move || {

4113                // are released instead of polling forever on a dead worker.
4114                let outcome = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
4115                    Self::background_committer_loop(&weak, config, &stop_clone)
4116                }));

4119                        Ok(Ok(())) => None,
4120                        Ok(Err(failure)) => Some(failure),
4121                        Err(_) => Some(BackgroundFailure {
4122                            kind: io::ErrorKind::Other,
4123                            message: "background committer panicked".to_owned(),

4127                }
4128            })
4129            .map_err(|error| {
4130                io::Error::other(format!("failed to spawn journal committer: {error}"))
4131            })?;

4133            .store(config.max_pending, Ordering::Release);

4141    /// Signal the background committer to stop, join it, then flush any final
4142    /// pending group so a quiet stream's last write is durable before this
4143    /// returns. Safe to call when no committer is running (it still flushes).
4144    ///
4145    /// The committer slot is held in a `Stopping` state across the join, so a
4146    /// concurrent [`Self::start_background_committer`] cannot spawn a second
4147    /// worker mid-teardown.
4148    ///
4149    /// # Errors
4150    /// Propagates a failure from the final [`Self::flush_durable`], and surfaces
4151    /// a terminal committer failure recorded by the worker.
4152    pub fn stop_background_committer(&self) -> io::Result<()> {
4153        let handle = {
4154            let mut slot = self.background.lock();
4155            match std::mem::replace(&mut *slot, BackgroundState::Stopping) {
4156                BackgroundState::Running(mut committer) => {
4157                    committer.stop.store(true, Ordering::Release);
4158                    committer.handle.take()
4159                }
4160                other => {
4161                    *slot = other;
4162                    None
4163                }
4164            }
4165        };
4166        self.committer_wake_threshold.store(0, Ordering::Release);
4167        self.wake_durable_waiters();
4168        if let Some(handle) = handle {
4169            // A panic in the committer is caught and recorded by the worker's
4170            // wrapper; joining still succeeds and the failure is surfaced below.
4171            let _ = handle.join();
4172        }
4173        // Transition `Stopping -> Stopped` and take the recorded failure under
4174        // one `background` lock. Otherwise a `start_background_committer` that
4175        // sees `Stopped` can clear `background_failure` in the window between
4176        // the transition and the take, silently losing the worker's failure.
4177        let failure = {
4178            let mut slot = self.background.lock();
4179            if matches!(*slot, BackgroundState::Stopping) {
4180                *slot = BackgroundState::Stopped;
4181            }
4182            self.background_failure.lock().take()
4183        };
4184        self.flush_durable()?;
4185        if let Some(failure) = failure {
4186            return Err(failure.into_io());
4187        }
4188        Ok(())
4189    }

4191    /// Mark the committer slot stopped, publish any terminal failure, and wake
4192    /// waiters. Called by the worker itself as it exits.
4193    fn finish_background(&self, failure: Option<BackgroundFailure>) {
4194        // Transition, publish the failure, and clear the wake threshold under a
4195        // single `background` lock. A concurrent `start_background_committer`
4196        // (which holds that lock to clear `background_failure` and set its own
4197        // threshold) therefore either sees this worker still `Running` and does
4198        // nothing, or runs after the failure is already recorded; it cannot
4199        // erase the failure or have its threshold zeroed by this exiting worker.
4200        {
4201            let mut slot = self.background.lock();
4202            // Leave `Stopping` alone: `stop_background_committer` owns that
4203            // transition and takes the failure after joining this thread.
4204            if matches!(*slot, BackgroundState::Running(_)) {
4205                *slot = BackgroundState::Stopped;
4206                self.committer_wake_threshold.store(0, Ordering::Release);
4207            }
4208            if let Some(failure) = failure {
4209                *self.background_failure.lock() = Some(failure);
4210            }
4211        }
4212        self.wake_durable_waiters();
4213    }

4215    fn background_committer_loop(
4216        weak: &Weak<Self>,
4217        config: GroupCommitConfig,
4218        stop: &Arc<AtomicBool>,
4219    ) -> Result<(), BackgroundFailure> {
4220        let interval = config.interval;
4221        loop {
4222            let Some(coordinator) = weak.upgrade() else {
4223                return Ok(());
4224            };
4225            let now = std::time::Instant::now();
4226            let deadline = now.checked_add(interval).unwrap_or(now);
4227            {
4228                let mut guard = coordinator.durable_lock.lock();
4229                loop {
4230                    if stop.load(Ordering::Acquire) || coordinator.poisoned.load(Ordering::Acquire)
4231                    {
4232                        break;
4233                    }
4234                    let now = std::time::Instant::now();
4235                    if now >= deadline {
4236                        break;
4237                    }
4238                    if coordinator.pending_count() >= config.max_pending {
4239                        break;
4240                    }
4241                    let timeout = deadline.saturating_duration_since(now);
4242                    #[cfg(test)]
4243                    coordinator.committer_parks.fetch_add(1, Ordering::Release);
4244                    coordinator.durable_cv.wait_for(&mut guard, timeout);
4245                }
4246            }
4247            if stop.load(Ordering::Acquire) || coordinator.poisoned.load(Ordering::Acquire) {
4248                return Ok(());
4249            }
4250            // The final flush on stop is done by `stop_background_committer`
4251            // after join, so a stop that lands here does not double-fsync.
4252            if let Err(error) = coordinator.flush_durable() {
4253                if coordinator.poisoned.load(Ordering::Acquire) {
4254                    // Poison is already reported to waiters via the poison bit.
4255                    return Ok(());
4256                }
4257                return Err(BackgroundFailure::from_io(&error));
4258            }
4259        }
4260    }

4262    /// Compact the segment, dropping committed groups at or below
4263    /// `covered_lsn` while preserving any newer groups.
4264    ///
4265    /// Intended to run after a checkpoint durably records that LSN as applied,
4266    /// so it reclaims only data the packfiles already represent. Published
4267    /// groups are already appended; [`Self::sync_through`] makes them durable.
4268    /// Numbering is preserved across the rewrite. Active replay leases may
4269    /// reduce the reclaim boundary below
4270    /// `covered_lsn`.
4271    ///
4272    /// # Errors
4273    /// Returns `InvalidInput` if the effective reclaim boundary reaches an LSN
4274    /// that has not been appended. Also propagates poison, segment-read,
4275    /// validation, and rewrite errors.
4276    pub fn reclaim_through(&self, covered_lsn: u64) -> io::Result<Reclaim> {
4277        // Wait out any in-flight fsync: the rewrite replaces the segment file,
4278        // and a sync must not flush a handle to the replaced inode.
4279        let _sync = self.sync_lock.lock();
4280        let mut journal = self.journal.lock();
4281        let replay_limit = self.replay_reclaim_limit(covered_lsn);
4282        let reclaim = journal.reclaim_through(replay_limit)?;
4283        // The rewrite replaced the segment inode and moved its base LSN without
4284        // necessarily publishing a group. Advance the signal so a gated reader
4285        // rescans and detects the base jump instead of serving an index older
4286        // than the reclaimed prefix.
4287        self.bump_publish_signal();
4288        Ok(reclaim)
4289    }

4417    /// Read up to `max_len` bytes of the segment ending at `end_offset` from an
4418    /// already-open `file` into `window`, never reaching back into the file
4419    /// header. Returns `false` when the file is shorter than `end_offset` or
4420    /// has no group bytes before it, leaving `window` unspecified.
4421    ///
4422    /// A read-only overlay keeps this window from the end of the last group it
4423    /// consumed and compares it on later refreshes: file length alone cannot
4424    /// show that the bytes already consumed are still the same ones. The
4425    /// window is compared byte for byte instead of relying on the group
4426    /// trailer, because the trailer is not a content fingerprint: the group
4427    /// checksum covers `header || records` and every record already ends with
4428    /// its own CRC32, so for groups of the same shape it comes out identical
4429    /// whatever the payload. The window includes the last record's own CRC,
4430    /// which does depend on the record's full content.
4431    ///
4432    /// Reads through the caller's descriptor with a positioned read, so the
4433    /// refresh path pays one syscall and reuses `window`'s allocation instead
4434    /// of opening the file each time. The caller must notice a replaced file
4435    /// itself (its descriptor keeps the old inode). Never repairs or creates
4436    /// the file.

4447        if end_offset <= header_len {
4448            return Ok(false);
4449        }
4450        let start = end_offset.saturating_sub(max_len).max(header_len);

4452            .map_err(|_| invalid_data("tail window exceeds the address space"))?;
4453        window.clear();
4454        window.resize(window_len, 0);
4455        #[cfg(unix)]
4456        let read = std::os::unix::fs::FileExt::read_exact_at(file, window, start);
4457        #[cfg(not(unix))]
4458        let read = {
4459            let mut handle = file;
4460            handle
4461                .seek(SeekFrom::Start(start))
4462                .and_then(|_| handle.read_exact(window))

4535            file.set_len(0)?;
4536            file.seek(SeekFrom::Start(0))?;
4537            write_file_header(&mut file, 1, base_lsn)?;
4538            file.sync_all()?;
4539            sync_parent_dir(&path)?;
4540        }
4541
4542        let bytes = fs::read(&path)?;
4543        let (base_sequence, base_lsn) = validate_file_header(&bytes)?;
4544        if bytes.len() as u64 > MAX_SEGMENT_LEN {
4545            return Err(invalid_data("journal segment exceeds the 256 MiB limit"));
4546        }
4547        let file_len = u64::try_from(bytes.len()).unwrap_or(u64::MAX);
4548        let scan = scan_bytes(&bytes, base_sequence, base_lsn, || {
4549            read_durable_len_from_header(&bytes, file_len)
4550        })?;
4551        if scan.truncated_tail {
4552            file.set_len(scan.valid_len)?;
4553        }
4554        file.seek(SeekFrom::Start(scan.valid_len))?;
4555        if scan.truncated_tail || !scan.groups.is_empty() {
4556            // What recovery keeps may exist only in the page cache (a process
4557            // crash leaves it there), yet the coordinator will report it
4558            // committed. Make it durable now, and record that in the hint, so a
4559            // later sync never claims bytes that were not fsynced. The same
4560            // fsync makes a truncation durable, including one that dropped
4561            // every group, so a dropped tail cannot reappear.
4562            file.sync_all()?;
4563        }
4564        let next_sequence = scan
4565            .groups
4566            .last()
4567            .map_or(base_sequence, |group| group.sequence.saturating_add(1));
4568        let next_lsn = scan
4569            .groups
4570            .last()
4571            .map_or(base_lsn, |group| group.last_lsn.saturating_add(1));
4572
4573        let mark_generation = newest_mark(&bytes).map_or(0, |slot| slot.generation);
4574        let mut journal = Self {
4575            path,
4576            file,
4577            file_len: scan.valid_len,
4578            base_sequence,
4579            base_lsn,
4580            mark_generation,
4581            next_sequence,
4582            next_lsn,
4583            poisoned: false,
4584            segment_cap: MAX_SEGMENT_LEN,
4585            groups: marks_from_scan(&scan),
4586        };
4587        if !scan.groups.is_empty() {
4588            // Recovery fsynced what it kept above, so record that in the mark.
4589            journal.write_mark(scan.valid_len);
4590        }
4591        Ok((journal, scan))
4592    }

4661    /// Shared implementation for a group of mutations, each paired with its
4662    /// optional pool tag. The tag must match the segment's dialect (see
4663    /// the journal's single v1 format.
4664    fn append_group_inner(
4665        &mut self,
4666        mutations: &[(ShardType, Mutation)],
4667        sequence: Option<u64>,
4668    ) -> io::Result<CommitReceipt> {
4669        if self.poisoned {
4670            return Err(io::Error::other(
4671                "journal handle is poisoned after an earlier failed commit",
4672            ));
4673        }
4674        if mutations.is_empty() {
4675            return Err(io::Error::new(
4676                io::ErrorKind::InvalidInput,
4677                "cannot commit an empty journal group",
4678            ));
4679        }
4680
4681        let record_count = u32::try_from(mutations.len()).map_err(|_| {
4682            io::Error::new(io::ErrorKind::InvalidInput, "too many journal mutations")
4683        })?;
4684        let first_lsn = self.next_lsn;
4685        let last_lsn = first_lsn
4686            .checked_add(u64::from(record_count).saturating_sub(1))
4687            .ok_or_else(|| io::Error::other("journal LSN exhausted"))?;
4688        let sequence = sequence.unwrap_or(self.next_sequence);
4689        if sequence < self.next_sequence {
4690            return Err(io::Error::new(
4691                io::ErrorKind::InvalidInput,
4692                "journal group sequence would move backwards",
4693            ));
4694        }
4695
4696        let mut payload = Vec::new();
4697        for (index, (pool, mutation)) in mutations.iter().enumerate() {
4698            let lsn = first_lsn
4699                .checked_add(u64::try_from(index).map_err(|_| {
4700                    io::Error::new(io::ErrorKind::InvalidInput, "mutation index exceeds u64")
4701                })?)
4702                .ok_or_else(|| io::Error::other("journal LSN exhausted"))?;
4703            encode_mutation(lsn, *pool, mutation, &mut payload)?;
4704            if u64::try_from(payload.len()).unwrap_or(u64::MAX) > MAX_GROUP_LEN {
4705                return Err(io::Error::new(
4706                    io::ErrorKind::InvalidInput,
4707                    "journal group exceeds the 256 MiB format limit",
4708                ));
4709            }
4710        }
4711        let payload_len = u64::try_from(payload.len())
4712            .map_err(|_| io::Error::new(io::ErrorKind::InvalidInput, "group too large"))?;
4713        if payload_len > MAX_GROUP_LEN {
4714            return Err(io::Error::new(
4715                io::ErrorKind::InvalidInput,
4716                "journal group exceeds the 256 MiB segment limit",
4717            ));
4718        }
4719
4720        let header = encode_group_header(sequence, first_lsn, last_lsn, payload_len, record_count);
4721        let mut group_checksum = Hasher::new();
4722        group_checksum.update(&header);
4723        group_checksum.update(&payload);
4724        let trailer = encode_group_trailer(sequence, group_checksum.finalize());
4725
4726        let group_size = u64::try_from(
4727            header
4728                .len()
4729                .saturating_add(payload.len())
4730                .saturating_add(trailer.len()),
4731        )
4732        .map_err(|_| io::Error::new(io::ErrorKind::InvalidInput, "group too large"))?;
4733        if self.file_len.saturating_add(group_size) > self.segment_cap {
4734            return Err(io::Error::new(
4735                io::ErrorKind::WouldBlock,
4736                "journal segment is full; drain/rotate before accepting more writes",
4737            ));
4738        }
4739        let next_sequence = sequence
4740            .checked_add(1)
4741            .ok_or_else(|| io::Error::other("journal group sequence exhausted"))?;
4742        let next_lsn = last_lsn
4743            .checked_add(1)
4744            .ok_or_else(|| io::Error::other("journal LSN exhausted"))?;
4745
4746        // One write per group: every autocommit publish lands here, so the
4747        // group goes out as a single contiguous buffer instead of three
4748        // syscalls. A torn write still leaves an incomplete tail that recovery
4749        // truncates, exactly as before.
4750        let mut frame = Vec::with_capacity(
4751            header
4752                .len()
4753                .saturating_add(payload.len())
4754                .saturating_add(trailer.len()),
4755        );
4756        frame.extend_from_slice(&header);
4757        frame.extend_from_slice(&payload);
4758        frame.extend_from_slice(&trailer);
4759        let write_result = self.file.write_all(&frame);
4760        if let Err(error) = write_result {
4761            self.poisoned = true;
4762            return Err(error);
4763        }
4764        self.file_len = self.file_len.saturating_add(group_size);
4765        self.note_appended_group(sequence, first_lsn, last_lsn, mutations);
4766
4767        self.next_sequence = next_sequence;
4768        self.next_lsn = next_lsn;
4769
4770        Ok(CommitReceipt {
4771            sequence,
4772            first_lsn,
4773            last_lsn,
4774            bytes_written: group_size,
4775        })
4776    }

4778    /// Fsync the bytes [`Self::append_group`] wrote, making the group durable.
4779    ///
4780    /// Separate from [`Self::append_group`] so a caller can publish a
4781    /// visibility boundary after the group is complete but before the fsync.
4782    /// On failure this handle is poisoned: the appended group's on-disk state
4783    /// is uncertain, so the caller must reopen and rescan.
4784    ///
4785    /// # Errors
4786    /// Returns `io::Error` if the handle is poisoned or the sync fails.
4787    pub fn make_durable(&mut self) -> io::Result<()> {
4788        if self.poisoned {
4789            return Err(io::Error::other(
4790                "journal handle is poisoned after an earlier failed commit",
4791            ));
4792        }
4793        if let Err(error) = self.file.sync_all() {
4794            self.poisoned = true;
4795            return Err(error);
4796        }
4797        // The fsync returned, so everything appended so far is durable: record
4798        // that in the durability mark, after the fact.
4799        self.write_mark(self.file_len);
4800        Ok(())
4801    }

4803    /// Record, in the slot not holding the newest mark, that `durable_len`
4804    /// bytes are durable. Call it only after an fsync
4805    /// that covered them, so the mark can only under-claim.
4806    ///
4807    /// Only pool-tagged segments keep a mark. The slot is written with a
4808    /// positioned write, so it never moves the append cursor. A failed write
4809    /// is ignored: the previous mark stays in the other slot, or recovery sees
4810    /// no mark and truncates conservatively.
4811    fn write_mark(&mut self, durable_len: u64) {
4812        let generation = self.mark_generation.saturating_add(1);
4813        let bytes = encode_mark_slot(MarkSlot {
4814            generation,
4815            durable_len,
4816        });
4817        let slot = usize::try_from(generation % 2).unwrap_or(0);
4818        let offset = u64::try_from(MARK_SLOT_OFFSETS[slot]).unwrap_or(u64::MAX);
4819        #[cfg(unix)]
4820        let written = std::os::unix::fs::FileExt::write_all_at(&self.file, &bytes, offset);
4821        // Elsewhere a positioned write moves the shared cursor, so restore it
4822        // to the end of the segment, where appends continue. Appends take the
4823        // journal lock too, so nothing writes in between.
4824        #[cfg(not(unix))]
4825        let written = self
4826            .file
4827            .seek(SeekFrom::Start(offset))
4828            .and_then(|_| self.file.write_all(&bytes))
4829            .and_then(|()| self.file.seek(SeekFrom::Start(self.file_len)).map(drop));
4830        if written.is_ok() {
4831            self.mark_generation = generation;
4832        }
4833    }

5009    /// Reclaim using the group directory: pick the cut from it and read only
5010    /// the retained suffix from disk. The dropped prefix is never read.
5011    fn reclaim_through_directory(&mut self, covered_lsn: u64) -> io::Result<Reclaim> {
5012        let cut = self
5013            .groups
5014            .partition_point(|group| group.last_lsn <= covered_lsn);
5015        let Some(last_dropped) = cut.checked_sub(1).and_then(|index| self.groups.get(index)) else {
5016            return Ok(Reclaim {
5017                retained_groups: u64::try_from(self.groups.len()).unwrap_or(u64::MAX),
5018                reclaimed_bytes: 0,
5019                ..Default::default()
5020            });
5021        };
5022        let copy_started = std::time::Instant::now();
5023        let header_len = FILE_HEADER_LEN as u64;
5024        let cut_offset = last_dropped.end_offset;
5025        let retained = self.groups.get(cut..).unwrap_or_default();
5026        let (new_base_sequence, new_base_lsn) = retained
5027            .first()
5028            .map_or((self.next_sequence, self.next_lsn), |group| {
5029                (group.sequence, group.first_lsn)
5030            });
5031        let mut rebuilt = file_header_bytes(new_base_sequence, new_base_lsn);
5032        if let Some(first) = retained.first() {
5033            // Read the suffix straight into the buffer that becomes the new
5034            // segment, after its header, so it is allocated once.
5035            let header_len = rebuilt.len();
5036            read_range_into(
5037                &self.path,
5038                cut_offset,
5039                self.file_len.saturating_sub(cut_offset),
5040                &mut rebuilt,
5041            )?;
5042            // The suffix is copied byte for byte, so check it the way the full
5043            // scan would have: it must hold exactly the groups the directory
5044            // says, each valid. Anything else means the directory is stale, and
5045            // the scan path decides instead.
5046            let Ok(checked) = scan_groups_from(
5047                rebuilt.get(header_len..).unwrap_or_default(),
5048                cut_offset,
5049                first.sequence,
5050                first.first_lsn,
5051                new_base_lsn,
5052                || u64::MAX,
5053            ) else {
5054                return self.reclaim_through_scan(covered_lsn);
5055            };
5056            if checked.groups.len() != retained.len()
5057                || checked.valid_len != self.file_len
5058                || checked.truncated_tail
5059            {
5060                return self.reclaim_through_scan(covered_lsn);
5061            }
5062        }
5063        let reclaimed_bytes = self
5064            .file_len
5065            .saturating_sub(u64::try_from(rebuilt.len()).unwrap_or(u64::MAX));
5066        let retained_groups = u64::try_from(retained.len()).unwrap_or(u64::MAX);
5067        let shift = cut_offset.saturating_sub(header_len);
5068        let survivors = retained
5069            .iter()
5070            .map(|group| GroupMark {
5071                end_offset: group.end_offset.saturating_sub(shift),
5072                ..*group
5073            })
5074            .collect::<Vec<_>>();
5075        let copy = copy_started.elapsed();
5076        let rebuilt_len = rebuilt.len();
5077        let fsync_started = std::time::Instant::now();
5078        self.install_rebuilt(rebuilt, new_base_sequence, new_base_lsn)?;
5079        let fsync = fsync_started.elapsed();
5080        self.groups = survivors;
5081        Ok(Reclaim {
5082            retained_groups,
5083            reclaimed_bytes,
5084            retained_bytes: u64::try_from(rebuilt_len.saturating_sub(FILE_HEADER_LEN))
5085                .unwrap_or(u64::MAX),
5086            copy,
5087            fsync,
5088            ..Default::default()
5089        })
5090    }

5097        let bytes = fs::read(&self.path)?;
5098        let (base_sequence, base_lsn) = validate_file_header(&bytes)?;
5099        // Reclaim rewrites the segment from its own live file, so any invalid
5100        // group there is corruption, not a crash tail: treat every byte as
5101        // durable and let it fail.
5102        let scan = scan_bytes(&bytes, base_sequence, base_lsn, || u64::MAX)?;
5103        let boundary = scan_started.elapsed();
5104        let retained = scan
5105            .groups
5106            .iter()
5107            .filter(|group| group.last_lsn > covered_lsn)
5108            .collect::<Vec<_>>();
5109        if retained.len() == scan.groups.len() {
5110            self.groups = marks_from_scan(&scan);
5111            return Ok(Reclaim {
5112                retained_groups: u64::try_from(retained.len()).unwrap_or(u64::MAX),
5113                reclaimed_bytes: 0,
5114                boundary,
5115                ..Default::default()
5116            });
5117        }
5118
5119        let copy_started = std::time::Instant::now();
5120        let (new_base_sequence, new_base_lsn) = retained
5121            .first()
5122            .map_or((self.next_sequence, self.next_lsn), |group| {
5123                (group.sequence, group.first_lsn)
5124            });
5125        let mut rebuilt = file_header_bytes(new_base_sequence, new_base_lsn);
5126        let mut survivors = Vec::with_capacity(retained.len());
5127        for group in &retained {
5128            encode_group(group, &mut rebuilt)?;
5129            survivors.push(GroupMark {
5130                sequence: group.sequence,
5131                first_lsn: group.first_lsn,
5132                last_lsn: group.last_lsn,
5133                end_offset: u64::try_from(rebuilt.len()).unwrap_or(u64::MAX),
5134                pools: group
5135                    .entries
5136                    .iter()
5137                    .fold(0, |mask, entry| mask | pool_bit(entry.pool)),
5138            });
5139        }
5140        let copy = copy_started.elapsed();
5141        let retained_groups = u64::try_from(retained.len()).unwrap_or(u64::MAX);
5142        let rebuilt_len = rebuilt.len();
5143        let reclaimed_bytes =
5144            u64::try_from(bytes.len().saturating_sub(rebuilt.len())).unwrap_or(u64::MAX);
5145        let fsync_started = std::time::Instant::now();
5146        self.install_rebuilt(rebuilt, new_base_sequence, new_base_lsn)?;
5147        let fsync = fsync_started.elapsed();
5148        self.groups = survivors;
5149        Ok(Reclaim {
5150            retained_groups,
5151            reclaimed_bytes,
5152            retained_bytes: u64::try_from(rebuilt_len.saturating_sub(FILE_HEADER_LEN))
5153                .unwrap_or(u64::MAX),
5154            boundary,
5155            copy,
5156            fsync,
5157            ..Default::default()
5158        })
5159    }

5161    /// Durably replace the segment with `rebuilt` (a new header plus the
5162    /// retained groups) and point this handle at it.
5163    fn install_rebuilt(
5164        &mut self,
5165        mut rebuilt: Vec<u8>,
5166        new_base_sequence: u64,
5167        new_base_lsn: u64,
5168    ) -> io::Result<()> {
5169        // The rebuilt file is fsynced before it replaces the segment, so all of
5170        // it is durable. Put its mark inside it, first generation, so the mark
5171        // is durable with the data and the old segment's marks, which described
5172        // offsets that no longer exist, are simply gone.
5173        let rebuilt_mark = Some(MarkSlot {
5174            generation: 1,
5175            durable_len: u64::try_from(rebuilt.len()).unwrap_or(u64::MAX),
5176        });
5177        if let Some(mark) = rebuilt_mark {
5178            let offset = MARK_SLOT_OFFSETS[1];
5179            let end = offset.saturating_add(MARK_SLOT_LEN);
5180            rebuilt[offset..end].copy_from_slice(&encode_mark_slot(mark));
5181        }
5182
5183        let temp_path = self.path.with_extension("rotate");
5184        let write_result = (|| -> io::Result<()> {
5185            let mut temp = OpenOptions::new()
5186                .create(true)
5187                .write(true)
5188                .truncate(true)
5189                .open(&temp_path)?;
5190            temp.write_all(&rebuilt)?;
5191            temp.sync_all()?;
5192            drop(temp);
5193            fs::rename(&temp_path, &self.path)
5194        })();
5195        if let Err(error) = write_result {
5196            let _ = fs::remove_file(&temp_path);
5197            return Err(error);
5198        }
5199        // The rename has replaced the path, so from here onward any failure
5200        // must poison the handle: continuing to append through the old file
5201        // descriptor would write an unlinked inode and could acknowledge data
5202        // that is absent from the journal path.
5203        let replacement = match OpenOptions::new().read(true).write(true).open(&self.path) {
5204            Ok(file) => file,
5205            Err(error) => {
5206                self.poisoned = true;
5207                return Err(error);
5208            }
5209        };
5210        self.file = replacement;
5211        self.file_len = u64::try_from(rebuilt.len()).unwrap_or(u64::MAX);
5212        self.base_sequence = new_base_sequence;
5213        self.base_lsn = new_base_lsn;
5214        self.mark_generation = u64::from(rebuilt_mark.is_some());
5215        if let Err(error) = sync_parent_dir(&self.path) {
5216            self.poisoned = true;
5217            return Err(error);
5218        }
5219        if let Err(error) = self.file.seek(SeekFrom::End(0)) {
5220            self.poisoned = true;
5221            return Err(error);
5222        }
5223        Ok(())
5224    }
5225}

5298/// Decode one slot: `None` if it is unwritten (all zero) or fails its checksum.
5299fn decode_mark_slot(bytes: &[u8]) -> Option<MarkSlot> {
5300    let bytes = bytes.get(..MARK_SLOT_LEN)?;
5301    let mut crc = Hasher::new();
5302    crc.update(&bytes[..16]);
5303    if u32::from_le_bytes(bytes[16..20].try_into().ok()?) != crc.finalize() {
5304        return None;
5305    }
5306    let slot = MarkSlot {
5307        generation: u64::from_le_bytes(bytes[..8].try_into().ok()?),
5308        durable_len: u64::from_le_bytes(bytes[8..16].try_into().ok()?),
5309    };
5310    // Generation zero is never written, so a checksum-valid slot claiming it
5311    // is not a mark.
5312    (slot.generation != 0).then_some(slot)
5313}

5316/// slot holds one: unwritten, torn, or corrupt. `None` means "durability

5325/// Test-only: cut a **copy** of a journal segment back to its embedded
5326/// durability mark, keeping the header. That is the worst case of a power cut:
5327/// every frame written after the last fsync is lost.
5328///
5329/// # Errors
5330/// Returns an error if the copy cannot be read, has an invalid header, or cannot
5331/// be truncated.
5332#[cfg(test)]
5333pub(crate) fn cut_segment_image_to_durable_mark(path: &Path) -> io::Result<()> {
5334    let bytes = fs::read(path)?;
5335    if bytes.len() < FILE_HEADER_LEN {
5336        return Ok(());
5337    }
5338    let (_, _) = validate_file_header(&bytes)?;

5340    let mark = read_durable_len_from_header(&bytes, file_len);
5341    let header = u64::try_from(FILE_HEADER_LEN).unwrap_or(u64::MAX);

5344}

5385/// Re-encode a committed group with its original sequence and LSNs, for a
5386/// segment rewrite. Mirrors exactly the framing `commit_group` writes.
5387fn encode_group(group: &CommittedGroup, into: &mut Vec<u8>) -> io::Result<()> {
5388    let mut payload = Vec::new();
5389    for entry in &group.entries {
5390        encode_mutation(entry.lsn, entry.pool, &entry.mutation, &mut payload)?;
5391    }
5392    let record_count = u32::try_from(group.entries.len())
5393        .map_err(|_| io::Error::new(io::ErrorKind::InvalidData, "too many journal mutations"))?;
5394    let payload_len = u64::try_from(payload.len())
5395        .map_err(|_| io::Error::new(io::ErrorKind::InvalidData, "journal group too large"))?;
5396    let header = encode_group_header(
5397        group.sequence,
5398        group.first_lsn,
5399        group.last_lsn,
5400        payload_len,
5401        record_count,
5402    );
5403    let mut group_crc = Hasher::new();
5404    group_crc.update(&header);
5405    group_crc.update(&payload);
5406    let trailer = encode_group_trailer(group.sequence, group_crc.finalize());
5407    into.extend_from_slice(&header);
5408    into.extend_from_slice(&payload);
5409    into.extend_from_slice(&trailer);
5410    Ok(())
5411}

5413fn encode_mutation(
5414    lsn: u64,
5415    pool: ShardType,
5416    mutation: &Mutation,
5417    into: &mut Vec<u8>,
5418) -> io::Result<()> {
5419    // The flag byte is always the pool tag.
5420    let pool_byte = pool_tag(pool);
5421
5422    let start = into.len();
5423    // Zeroed fixed area; `MF_FLAGS` and the DeleteCollection-only fields stay
5424    // zero by construction. Index the fixed area by its named field ranges so
5425    // no raw offset arithmetic is needed.
5426    into.resize(start.saturating_add(FRAME_FIXED_LEN), 0);
5427    let payload = {
5428        let frame = &mut into[start..];
5429        frame[MF_KIND].copy_from_slice(&match mutation {
5430            Mutation::Put { .. } => [1],
5431            Mutation::DeleteCollection { .. } => [2],
5432            Mutation::RemoveRecord { .. } => [3],
5433        });
5434        frame[MF_POOL].copy_from_slice(&[pool_byte]);
5435        frame[MF_LSN].copy_from_slice(&lsn.to_le_bytes());
5436        match mutation {
5437            Mutation::Put {
5438                collection_id,
5439                node_id,
5440                payload,
5441            } => {
5442                frame[MF_COLLECTION].copy_from_slice(collection_id);
5443                frame[MF_NODE].copy_from_slice(node_id);
5444                let payload_len = u32::try_from(payload.len()).map_err(|_| {
5445                    io::Error::new(io::ErrorKind::InvalidInput, "mutation payload exceeds u32")
5446                })?;
5447                frame[MF_PAYLOAD_LEN].copy_from_slice(&payload_len.to_le_bytes());
5448                payload.as_slice()
5449            }
5450            Mutation::DeleteCollection { collection_id } => {
5451                frame[MF_COLLECTION].copy_from_slice(collection_id);
5452                &[]
5453            }
5454            Mutation::RemoveRecord {
5455                collection_id,
5456                node_id,
5457            } => {
5458                frame[MF_COLLECTION].copy_from_slice(collection_id);
5459                frame[MF_NODE].copy_from_slice(node_id);
5460                &[]
5461            }
5462        }
5463    };
5464    into.extend_from_slice(payload);
5465    let mut crc = Hasher::new();
5466    crc.update(&into[start..]);
5467    into.extend_from_slice(&crc.finalize().to_le_bytes());
5468    Ok(())
5469}

5503/// Verify a group's commit trailer and whole-group CRC, returning its payload.
5504fn verify_group_payload<'a>(
5505    bytes: &'a [u8],
5506    header: &[u8],
5507    group: &GroupHeader,
5508    payload_start: usize,
5509    payload_len: usize,
5510) -> io::Result<&'a [u8]> {
5511    let payload_end = payload_start
5512        .checked_add(payload_len)
5513        .ok_or_else(|| invalid_data("journal group length overflow"))?;
5514    let trailer_end = payload_end
5515        .checked_add(GROUP_TRAILER_LEN)
5516        .ok_or_else(|| invalid_data("journal group length overflow"))?;
5517    let trailer = bytes
5518        .get(payload_end..trailer_end)
5519        .ok_or_else(|| invalid_data("truncated journal commit trailer"))?;
5520    if &trailer[..4] != GROUP_COMMIT_MAGIC
5521        || u64::from_le_bytes(trailer[4..12].try_into().expect("fixed slice")) != group.sequence
5522    {
5523        return Err(invalid_data("invalid journal commit trailer"));
5524    }
5525    let payload = bytes
5526        .get(payload_start..payload_end)
5527        .ok_or_else(|| invalid_data("truncated journal group payload"))?;
5528    let mut group_crc = Hasher::new();
5529    group_crc.update(header);
5530    group_crc.update(payload);
5531    if u32::from_le_bytes(trailer[12..16].try_into().expect("fixed slice")) != group_crc.finalize()
5532    {
5533        return Err(invalid_data("journal committed group checksum mismatch"));
5534    }
5535    Ok(payload)
5536}

5582/// Validate the group starting at `cursor` in `bytes`.
5583fn scan_one_group(
5584    bytes: &[u8],
5585    cursor: usize,
5586    base_offset: u64,
5587    expected_sequence: u64,
5588    expected_lsn: u64,
5589) -> Result<GroupStep, GroupFault> {
5590    let remaining = bytes.len().saturating_sub(cursor);
5591    if remaining < GROUP_HEADER_LEN {
5592        return Ok(GroupStep::Incomplete);
5593    }
5594    let header = bytes
5595        .get(cursor..cursor.saturating_add(GROUP_HEADER_LEN))
5596        .ok_or_else(|| GroupFault::Crash(invalid_data("truncated journal group header")))?;
5597    let group = parse_group_header(header).map_err(GroupFault::Crash)?;
5598    if group.sequence < expected_sequence
5599        || group.first_lsn != expected_lsn
5600        || group.last_lsn < group.first_lsn
5601        || group
5602            .last_lsn
5603            .saturating_sub(group.first_lsn)
5604            .saturating_add(1)
5605            != u64::from(group.record_count)
5606        || group.record_count == 0
5607        || group.payload_len > MAX_GROUP_LEN
5608        || u64::from(group.record_count).saturating_mul(MIN_FRAME_LEN as u64) > group.payload_len
5609    {
5610        return Err(GroupFault::Fatal(invalid_data(
5611            "invalid journal sequence or group bounds",
5612        )));
5613    }
5614    let payload_len = usize::try_from(group.payload_len).map_err(|_| {
5615        GroupFault::Fatal(invalid_data("journal group length exceeds address space"))
5616    })?;
5617    let total_len = GROUP_HEADER_LEN
5618        .checked_add(payload_len)
5619        .and_then(|len| len.checked_add(GROUP_TRAILER_LEN))
5620        .ok_or_else(|| GroupFault::Fatal(invalid_data("journal group length overflow")))?;
5621    if remaining < total_len {
5622        return Ok(GroupStep::Incomplete);
5623    }
5624    let payload_index = cursor.saturating_add(GROUP_HEADER_LEN);
5625    let payload = verify_group_payload(bytes, header, &group, payload_index, payload_len)
5626        .map_err(GroupFault::Crash)?;
5627    let payload_offset = base_offset
5628        .saturating_add(u64::try_from(payload_index).unwrap_or(u64::MAX))
5629        .try_into()
5630        .unwrap_or(usize::MAX);
5631    let entries = decode_mutations(payload, group.first_lsn, group.record_count, payload_offset)
5632        .map_err(GroupFault::Fatal)?;
5633    Ok(GroupStep::Group {
5634        group,
5635        entries,
5636        total_len,
5637    })
5638}

5640/// Scan complete groups from `bytes`, which begins at absolute file offset
5641/// `base_offset`.
5642///
5643/// `expected_sequence` is a floor, not an exact match: a segment sharing a
5644/// global sequence with other pools may skip values. `expected_lsn` must match
5645/// the first group's `first_lsn` exactly, since LSNs stay contiguous within a
5646/// segment. `Scan::valid_len` is absolute in the file, not relative to `bytes`.
5647///
5648/// `durable_len` yields the embedded durability mark: how many
5649/// leading bytes of the file a completed fsync covered. It is called at most
5650/// once, and only when a group fails validation in a way a crash can explain. A group that fails validation
5651/// in a way a crash can explain ([`GroupFault::Crash`]) at or above it was never
5652/// acknowledged, so the scan stops there and reports a truncated tail, dropping
5653/// everything after it, even later groups that happen to be intact. The same
5654/// failure below it is corruption and is returned as an error. Return `u64::MAX`
5655/// to treat every byte as durable, so any invalid group fails; return `0` when
5656/// nothing is known, so the first invalid group ends the scan.
5657fn scan_groups_from(
5658    bytes: &[u8],
5659    base_offset: u64,
5660    mut expected_sequence: u64,
5661    mut expected_lsn: u64,
5662    segment_base_lsn: u64,
5663    durable_len: impl Fn() -> u64,
5664) -> io::Result<Scan> {
5665    let mut cursor = 0usize;
5666    let mut valid_len = base_offset;
5667    let mut groups = Vec::new();
5668    let mut group_ends = Vec::new();
5669    let mut truncated_tail = false;
5670
5671    while cursor < bytes.len() {
5672        let (group, entries, total_len) =
5673            match scan_one_group(bytes, cursor, base_offset, expected_sequence, expected_lsn) {
5674                Ok(GroupStep::Group {
5675                    group,
5676                    entries,
5677                    total_len,
5678                }) => (group, entries, total_len),
5679                Ok(GroupStep::Incomplete) => {
5680                    truncated_tail = true;
5681                    break;
5682                }
5683                Err(GroupFault::Fatal(error)) => return Err(error),
5684                Err(GroupFault::Crash(error)) => {
5685                    let at = base_offset.saturating_add(u64::try_from(cursor).unwrap_or(u64::MAX));
5686                    // The durability mark matters only here, on a failure a crash
5687                    // can explain, so it is read only now. Reading it on every scan
5688                    // would cost a file open and read on the hot incremental path
5689                    // for something a healthy segment never consults.
5690                    if at >= durable_len() {
5691                        truncated_tail = true;
5692                        break;
5693                    }
5694                    return Err(error);
5695                }
5696            };
5697        groups.push(CommittedGroup {
5698            sequence: group.sequence,
5699            first_lsn: group.first_lsn,
5700            last_lsn: group.last_lsn,
5701            entries,
5702        });
5703        cursor = cursor.saturating_add(total_len);
5704        valid_len = base_offset.saturating_add(u64::try_from(cursor).unwrap_or(u64::MAX));
5705        group_ends.push(valid_len);
5706        expected_sequence = group
5707            .sequence
5708            .checked_add(1)
5709            .ok_or_else(|| invalid_data("journal group sequence overflow"))?;
5710        expected_lsn = group
5711            .last_lsn
5712            .checked_add(1)
5713            .ok_or_else(|| invalid_data("journal LSN overflow"))?;
5714    }
5715
5716    let consumed_tail = bytes[cursor.saturating_sub(CONSUMED_TAIL_LEN)..cursor].to_vec();
5717    Ok(Scan {
5718        groups,
5719        valid_len,
5720        truncated_tail,
5721        base_lsn: segment_base_lsn,
5722        consumed_tail,
5723        group_ends,
5724    })
5725}

5727fn decode_mutations(
5728    payload: &[u8],
5729    first_lsn: u64,
5730    record_count: u32,
5731    payload_offset: usize,
5732) -> io::Result<Vec<JournalEntry>> {
5733    let mut cursor = 0_usize;
5734    let mut entries = Vec::with_capacity(usize::try_from(record_count).unwrap_or(0));
5735    for index in 0..record_count {
5736        let start = cursor;
5737        if payload.len().saturating_sub(cursor) < MIN_FRAME_LEN {
5738            return Err(invalid_data("truncated journal mutation frame"));
5739        }
5740        let fixed_end = cursor
5741            .checked_add(FRAME_FIXED_LEN)
5742            .ok_or_else(|| invalid_data("journal mutation length overflow"))?;
5743        let frame = payload
5744            .get(cursor..fixed_end)
5745            .ok_or_else(|| invalid_data("truncated journal mutation frame"))?;
5746        let kind = frame[MF_KIND.start];
5747        let tag = frame[MF_POOL.start];
5748        if frame[MF_FLAGS] != [0, 0] {
5749            return Err(invalid_data("unsupported journal mutation flags"));
5750        }
5751        let pool = pool_from_tag(tag)
5752            .ok_or_else(|| invalid_data("journal frame has no valid pool tag"))?;
5753        let lsn = u64::from_le_bytes(frame[MF_LSN].try_into().expect("fixed slice"));
5754        let expected = first_lsn
5755            .checked_add(u64::from(index))
5756            .ok_or_else(|| invalid_data("journal mutation LSN overflow"))?;
5757        if lsn != expected {
5758            return Err(invalid_data("journal mutation LSN out of order"));
5759        }
5760        let collection_id: [u8; 16] = frame[MF_COLLECTION].try_into().expect("fixed slice");
5761        let node_id: [u8; 16] = frame[MF_NODE].try_into().expect("fixed slice");
5762        let payload_len = usize::try_from(u32::from_le_bytes(
5763            frame[MF_PAYLOAD_LEN].try_into().expect("fixed slice"),
5764        ))
5765        .map_err(|_| invalid_data("journal mutation payload length exceeds address space"))?;
5766        let frame_end = fixed_end
5767            .checked_add(payload_len)
5768            .ok_or_else(|| invalid_data("journal mutation length overflow"))?;
5769        let crc_end = frame_end
5770            .checked_add(FRAME_TRAILER_LEN)
5771            .ok_or_else(|| invalid_data("journal mutation length overflow"))?;
5772        let data = payload
5773            .get(fixed_end..frame_end)
5774            .ok_or_else(|| invalid_data("journal mutation payload exceeds group"))?;
5775        let crc_bytes = payload
5776            .get(frame_end..crc_end)
5777            .ok_or_else(|| invalid_data("journal mutation checksum exceeds group"))?;
5778        let mut crc = Hasher::new();
5779        crc.update(
5780            payload
5781                .get(start..frame_end)
5782                .ok_or_else(|| invalid_data("journal mutation frame exceeds group"))?,
5783        );
5784        if u32::from_le_bytes(crc_bytes.try_into().expect("fixed slice")) != crc.finalize() {
5785            return Err(invalid_data("journal mutation checksum mismatch"));
5786        }
5787        let mutation = match kind {
5788            1 => Mutation::Put {
5789                collection_id,
5790                node_id,
5791                payload: data.to_vec(),
5792            },
5793            2 if payload_len == 0 && node_id == [0; 16] => {
5794                Mutation::DeleteCollection { collection_id }
5795            }
5796            3 if payload_len == 0 && node_id != [0; 16] => Mutation::RemoveRecord {
5797                collection_id,
5798                node_id,
5799            },
5800            _ => return Err(invalid_data("unknown or malformed journal mutation kind")),
5801        };
5802        entries.push(JournalEntry {
5803            lsn,
5804            offset: u64::try_from(payload_offset.saturating_add(start)).unwrap_or(u64::MAX),
5805            frame_len: u64::try_from(crc_end.saturating_sub(start)).unwrap_or(u64::MAX),
5806            pool,
5807            mutation,
5808        });
5809        cursor = crc_end;
5810    }
5811    if cursor != payload.len() {
5812        return Err(invalid_data("extra bytes after journal mutation frames"));
5813    }
5814    Ok(entries)
5815}

5829/// Exclusive writer claim on a database root's shared WAL.
5830///
5831/// One shared segment serves every pool, so a single writer process must own
5832/// it. This takes the same crash-safe `{pid, starttime}` lock the per-pool
5833/// writer lock uses, at `<db_root>/.mtxdb.wal.lock`, and releases it on drop
5834/// (see `ShardPool::acquire_writer_lock` for the staleness contract). The
5835/// holder opens the segment with [`Journal::open_shared`], builds one
5836/// [`JournalCoordinator`], and attaches each pool with
5837/// `PackfileStorage::enable_shared_journal`.
5838pub struct SharedWalLock {
5839    _lock: crate::shard::WriterLock,
5840}
5841
5842impl SharedWalLock {
5843    /// Acquire the root shared-WAL writer lock.
5844    ///
5845    /// # Errors
5846    /// Returns `WouldBlock` if another live writer holds it, or an I/O error
5847    /// if the lock file cannot be created.
5848    pub fn acquire(db_root: impl AsRef<Path>) -> io::Result<Self> {
5849        let db_root = db_root.as_ref();
5850        // Require a database root.
5851        if !crate::layout::is_database_root(db_root)? {
5852            return Err(io::Error::new(
5853                io::ErrorKind::InvalidData,
5854                format!(
5855                    "{} is not an mtxdb database root (missing {}); initialize it first",
5856                    db_root.display(),
5857                    crate::layout::DB_META_FILENAME
5858                ),
5859            ));
5860        }
5861        let path = db_root.join(crate::layout::WAL_LOCK_FILENAME);
5862        Ok(Self {
5863            _lock: crate::shard::ShardPool::acquire_lock_path(&path)?,
5864        })
5865    }
5866}
