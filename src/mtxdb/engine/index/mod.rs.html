15/// Per-database-instance tuning for [`LossyIndex`] construction.
16///
17/// `seed` is mixed into bucket *and* tag derivation to prevent adversarial
18/// precomputation against a fixed, unsalted content hash. Every real
19/// `PackfileStorage`-sourced config must carry the pool's actual persisted
20/// seed; the `Default` impl uses `seed: 0` only for tests and transient
21/// construction that never touches disk.
22#[derive(Debug, Clone, Copy)]
23pub struct IndexConfig {
24    /// Mixed into bucket and tag derivation to prevent adversarial
25    /// precomputation. See `index/mod.rs` seed docs.
26    pub seed: u64,
27    /// Minimum initial capacity for a newly created collection's index.
28    /// Composes with the hard 16-entry minimum: `effective_min = floor.max(16)`.
29    pub floor: u32,
30    /// Load factor (percent, 1..=90) at which the index grows.
31    pub load_factor_percent: u8,
32}
33
34impl Default for IndexConfig {
35    fn default() -> Self {
36        Self {
37            seed: 0,
38            floor: 64,
39            load_factor_percent: 75,
40        }
41    }
42}

44impl IndexConfig {
45    /// Validate the configuration.
46    ///
47    /// # Errors
48    /// Returns `Err(message)` if `load_factor_percent` is 0 or greater than
49    /// 90, or if `floor` is 0.
50    pub fn validate(&self) -> Result<(), &'static str> {
51        if self.load_factor_percent == 0 || self.load_factor_percent > 90 {
52            return Err("load_factor_percent must be in 1..=90");
53        }
54        if self.floor == 0 {
55            return Err("floor must be > 0");
56        }
57        Ok(())
58    }

61/// Splitmix64-style avalanche finalizer. Cheap, well-distributed, no
62/// dependencies. Used to mix the per-pool seed into bucket and tag
63/// derivation so an attacker cannot precompute hash → bucket/tag mappings
64/// without knowing the seed.
65#[inline]
66fn mix(x: u64) -> u64 {
67    let mut v = x;
68    v = v.wrapping_add(0x9E37_79B9_7F4A_7C15);
69    v = (v ^ (v >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
70    v = (v ^ (v >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
71    v ^ (v >> 31)
72}

74/// Per-entry entry in the lossy fanout index.
75///
76/// Layout: `[16-bit tag | 16-bit slot | 32-bit offset]` packed into a `u64`.
77///
78/// - **tag** (high 16 bits): truncated fingerprint for fast rejection.
79/// - **`slot`** (next 16 bits): low 12 bits identify the shard (the pool has
80///   4096 slots); bit 15 marks an explicit removed version. Bits 12..14 are
81///   reserved and must remain zero.
82/// - **offset** (low 32 bits): byte offset within the shard, stored as
83///   `offset + 1` so that the all-zeros encoding is reserved as the empty
84///   sentinel. Actual offset 0 is stored as 1, and `offset()` subtracts 1
85///   to recover the real value.
86///
87/// Empty slots are all-zeros. This is safe regardless of tag value: the
88/// `offset + 1` encoding guarantees a live entry's low 32 bits are never zero,
89/// so the all-zeros pattern can never be produced by a real insert — the tag
90/// bits play no role in reserving the sentinel.
91#[derive(Debug, Clone, Copy, PartialEq, Eq)]
92pub struct IndexEntry(u64);
93
94impl IndexEntry {
95    const EMPTY: Self = Self(0);
96
97    /// Largest byte offset representable by the 32-bit offset field while
98    /// reserving the all-zeros encoding for the empty-entry sentinel.
99    pub const MAX_OFFSET: u64 = (1u64 << 32) - 2;
100
101    const TAG_SHIFT: u64 = 48; // SHARD_BITS + OFFSET_BITS
102    const SHARD_SHIFT: u64 = 32; // OFFSET_BITS
103    const OFFSET_MASK: u64 = 0xFFFF_FFFF;
104    // Preserve the three currently-reserved bits in slot() so existing
105    // MAX_SHARDS validation rejects malformed encodings instead of masking
106    // them into a valid slot. Only the removal bit is excluded.
107    const SLOT_MASK: u16 = 0x7FFF;
108    const REMOVED_MASK: u16 = 0x8000;
109
110    /// Create a new entry from its components.
111    ///
112    /// The offset is stored as `offset + 1` so that the all-zeros `u64`
113    /// encoding is reserved as the empty sentinel. Actual offset 0 is
114    /// stored as 1 in the entry.
115    ///
116    /// # Panics
117    /// Panics if `tag` exceeds 16 bits, `slot` is not a valid live shard ID
118    /// (`>= crate::shard::MAX_SHARDS`), or `offset` exceeds
119    /// [`Self::MAX_OFFSET`].
120    #[must_use]
121    pub fn new(tag: u32, slot: u16, offset: u64) -> Self {
122        Self::new_with_removed(tag, slot, offset, false)
123    }
124
125    /// Create an entry with the explicit removed-version bit set when
126    /// `removed` is true.
127    ///
128    /// # Panics
129    /// Panics under the same invalid component conditions as [`Self::new`].
130    #[must_use]
131    pub fn new_with_removed(tag: u32, slot: u16, offset: u64, removed: bool) -> Self {
132        assert!(tag <= 0xFFFF, "tag must fit in 16 bits");
133        assert!(
134            usize::from(slot) < crate::shard::MAX_SHARDS,
135            "slot {slot} is not a valid live shard ID (must be < MAX_SHARDS)"
136        );
137        assert!(
138            offset <= Self::MAX_OFFSET,
139            "offset must fit in 32 bits minus 1 (reserved for empty sentinel)"
140        );
141        Self(
142            (u64::from(tag) << Self::TAG_SHIFT)
143                | (u64::from(slot | if removed { Self::REMOVED_MASK } else { 0 })
144                    << Self::SHARD_SHIFT)
145                | ((offset.wrapping_add(1)) & Self::OFFSET_MASK),
146        )
147    }
148
149    /// The sentinel value representing an empty entry.
150    #[must_use]
151    pub fn empty() -> Self {
152        Self::EMPTY
153    }
154
155    /// Returns `true` if this entry is the empty sentinel.
156    #[must_use]
157    pub fn is_empty(self) -> bool {
158        self.0 == 0
159    }
160
161    /// The 16-bit tag stored in this entry.
162    #[must_use]
163    pub fn tag(self) -> u32 {
164        ((self.0 >> Self::TAG_SHIFT) & 0xFFFF) as u32
165    }
166
167    /// The shard ID stored in this entry, excluding the removal flag.
168    ///
169    /// # Panics
170    /// Never panics: the packed field is masked to 16 bits before conversion.
171    #[must_use]
172    pub fn slot(self) -> u16 {
173        (u16::try_from((self.0 >> Self::SHARD_SHIFT) & 0xFFFF)
174            .expect("masked slot field fits in u16"))
175            & Self::SLOT_MASK
176    }
177
178    /// Whether this entry points to an explicit removed record version.
179    ///
180    /// # Panics
181    /// Never panics: the packed field is masked to 16 bits before conversion.
182    #[must_use]
183    pub fn is_removed(self) -> bool {
184        (u16::try_from((self.0 >> Self::SHARD_SHIFT) & 0xFFFF)
185            .expect("masked slot field fits in u16")
186            & Self::REMOVED_MASK)
187            != 0
188    }
189
190    /// Whether the packed slot contains reserved bits that are not defined.
191    ///
192    /// # Panics
193    /// Never panics: the packed field is masked to 16 bits before conversion.
194    #[must_use]
195    pub fn has_invalid_slot_bits(self) -> bool {
196        (u16::try_from((self.0 >> Self::SHARD_SHIFT) & 0xFFFF)
197            .expect("masked slot field fits in u16")
198            & 0x7000)
199            != 0
200    }
201
202    /// Validate the persisted slot portion without constructing an index.
203    #[must_use]
204    pub fn valid_raw_slot(raw: u64) -> bool {
205        let entry = Self(raw);
206        entry.is_empty()
207            || (!entry.has_invalid_slot_bits()
208                && usize::from(entry.slot()) < crate::shard::MAX_SHARDS)
209    }
210
211    /// The byte offset within the shard stored in this entry.
212    #[must_use]
213    pub fn offset(self) -> u64 {
214        (self.0 & Self::OFFSET_MASK).wrapping_sub(1)
215    }
216}

218/// A per-collection lossy fanout index for content-addressed records.
219///
220/// Uses open addressing with linear probing on a power-of-two sized table.
221/// The index is mmap-able (flat `u64` array) and small enough to stay
222/// resident in RAM for active collections.
223///
224/// Design decisions (from docs):
225/// - Partition by collection: ~8KB per 1000-node collection, 100 active collections < 1MB.
226/// - Empty entry terminates probe. Removed versions remain occupied entries,
227///   so removal never breaks a probe chain.
228/// - Tag collisions coexist in the probe chain; a tag is only a candidate
229///   filter and never an overwrite/equality proof.
230/// - Owned indexes keep `homes` and `tails` identity side tables: two
231///   anonymous `u64`s (16 bytes) per entry beyond the packed entry array.
232///   v1 checkpoints persist these tables as well.
233/// - Power-of-two capacity: shift-and-mask bucket selection, cache-aligned probes.
234#[derive(Debug)]
235pub struct LossyIndex {
236    /// Power-of-two capacity.
237    capacity: u32,
238    /// Bitmask for bucket selection: capacity - 1.
239    mask: u32,
240    /// Shift to extract top bits from hash for bucket index.
241    shift: u32,
242    /// Per-instance config: seed, floor, load factor.
243    config: IndexConfig,
244    /// The flat entry array. Checkpoint-loaded indexes borrow the immutable
245    /// raw slots from their mmap until their first write, which clones them
246    /// into the owned atomic representation.
247    slots: EntryStorage,
248    /// The first 64 bits of each entry's hash. Index slots intentionally pack
249    /// only a tag, shard, and offset; retaining the home hash separately lets
250    /// a live writer grow the table without rescanning the packfiles.
251    homes: Mutex<Vec<u64>>,
252    /// The remaining 40 bits of each live entry's hash, plus a high-bit
253    /// marker that says the identity is known. Together with `homes` and the
254    /// packed tag this makes overwrite equality exact without putting
255    /// full hashes in the checkpoint.
256    tails: Mutex<Vec<u64>>,
257    /// An index with incomplete identity tables remains readable, but must be
258    /// rebuilt from packfiles if it ever needs to grow. Hydrated checkpoint
259    /// indexes are growable without that pack scan.
260    can_grow: bool,
261    /// Number of occupied slots.
262    len: AtomicU32,
263    /// Longest linear-probe chain walked by any insert or lookup against
264    /// this index instance so far (never reset, monotonically
265    /// non-decreasing). Pure observability — insert has no probe-length
266    /// cap (see the doc on `InsertError::TableFull`: growth is what keeps
267    /// chains short, not a cap), so this exists to let an operator notice
268    /// a probe chain growing unexpectedly (e.g. an unseeded/misconfigured
269    /// deployment under adversarial content) without changing behavior.
270    max_probe_len: AtomicU32,
271}

273#[derive(Debug)]
274enum EntryStorage {
275    Owned(Vec<AtomicU64>),
276    Mmap { mmap: Arc<Mmap>, offset: usize },
277}
278
279impl EntryStorage {
280    #[inline]
281    fn get(&self, index: usize) -> u64 {
282        match self {
283            Self::Owned(slots) => slots[index].load(Ordering::Acquire),
284            Self::Mmap { mmap, offset } => {
285                let start = offset.saturating_add(index.saturating_mul(8));
286                let bytes: [u8; 8] = mmap[start..start.saturating_add(8)]
287                    .try_into()
288                    .expect("validated checkpoint entry range");
289                u64::from_le_bytes(bytes)
290            }
291        }
292    }
293
294    fn materialize(&self, capacity: usize) -> Vec<AtomicU64> {
295        (0..capacity)
296            .map(|index| AtomicU64::new(self.get(index)))
297            .collect()
298    }
299}

301impl LossyIndex {
302    /// Whether the persisted identity tables are sufficient to rehash every
303    /// occupied slot without consulting the packs. The probe-chain check is
304    /// intentional: a corrupted home value could otherwise make `grow()` lose
305    /// an entry while still looking superficially hydrated.
306    fn identity_tables_complete(
307        capacity: u32,
308        shift: u32,
309        slots: &EntryStorage,
310        homes: &[u64],
311        tails: &[u64],
312    ) -> bool {
313        let cap = capacity as usize;
314        if homes.len() != cap || tails.len() != cap {
315            return false;
316        }
317        let mask = usize::try_from(capacity.wrapping_sub(1)).unwrap_or(usize::MAX);
318        (0..cap).all(|bucket| {
319            if IndexEntry(slots.get(bucket)).is_empty() {
320                return true;
321            }
322            if tails[bucket] >> 63 == 0 {
323                return false;
324            }
325            let mut probe =
326                usize::try_from((homes[bucket] >> shift) & u64::from(capacity.wrapping_sub(1)))
327                    .unwrap_or(usize::MAX);
328            while probe != bucket {
329                if IndexEntry(slots.get(probe)).is_empty() {
330                    return false;
331                }
332                probe = probe.wrapping_add(1) & mask;
333            }
334            true
335        })
336    }
337
338    /// [`Self::identity_tables_complete`] over this index's current tables.
339    fn identities_now_complete(&self) -> bool {
340        let homes = self.homes.lock();
341        let tails = self.tails.lock();
342        Self::identity_tables_complete(self.capacity, self.shift, &self.slots, &homes, &tails)
343    }
344
345    /// A copy of an identity side table, or an all-zero (unhydrated) one of
346    /// `capacity` entries if the source has none.
347    fn identity_table(table: &Mutex<Vec<u64>>, capacity: u32) -> Vec<u64> {
348        let table = table.lock();
349        if table.len() == capacity as usize {
350            table.clone()
351        } else {
352            vec![0; capacity as usize]
353        }
354    }
355}

357impl Clone for LossyIndex {
358    fn clone(&self) -> Self {
359        Self {
360            capacity: self.capacity,
361            mask: self.mask,
362            shift: self.shift,
363            config: self.config,
364            slots: EntryStorage::Owned(self.slots.materialize(self.capacity as usize)),
365            // Keep whatever identity tables the source holds. A mapping loaded
366            // with its persisted `homes`/`tails` carries hydrated identities, and
367            // dropping them here made every entry lose its identity on the first
368            // write after a reopen (the next checkpoint then persisted zeros). A
369            // mapping loaded without tables has none, so it gets zeroed ones.
370            // Preserve the source's completeness gate: a hydrated checkpoint
371            // remains growable after materialization, while an incomplete one
372            // still falls back to a pack rebuild at a capacity boundary.
373            homes: Mutex::new(Self::identity_table(&self.homes, self.capacity)),
374            tails: Mutex::new(Self::identity_table(&self.tails, self.capacity)),
375            can_grow: self.can_grow,
376            len: AtomicU32::new(self.len.load(Ordering::Acquire)),
377            max_probe_len: AtomicU32::new(self.max_probe_len.load(Ordering::Relaxed)),
378        }
379    }
380}

404    /// Create a new index with at least `min_capacity` slots using the given config.
405    /// Capacity is rounded up to the next power of two. The configured `floor`
406    /// composes with the hard 16-entry minimum: `effective_min = floor.max(16)`.
407    ///
408    /// # Panics
409    /// Panics if `min_capacity` exceeds `u32::MAX` when rounded to a power of two,
410    /// or if `config` fails validation.
411    #[must_use]
412    pub fn with_config(min_capacity: usize, config: IndexConfig) -> Self {
413        config.validate().expect("invalid IndexConfig");
414        let effective_min = (config.floor.max(16)) as usize;
415        let capacity_usize = min_capacity.max(effective_min).next_power_of_two();
416        let capacity = u32::try_from(capacity_usize).expect("index capacity exceeds u32::MAX");
417        let shift = 64_u32.wrapping_sub(capacity.trailing_zeros());
418        Self {
419            mask: capacity.wrapping_sub(1),
420            capacity,
421            shift,
422            config,
423            slots: EntryStorage::Owned((0..capacity_usize).map(|_| AtomicU64::new(0)).collect()),
424            homes: Mutex::new(vec![0; capacity_usize]),
425            tails: Mutex::new(vec![0; capacity_usize]),
426            can_grow: true,
427            len: AtomicU32::new(0),
428            max_probe_len: AtomicU32::new(0),
429        }
430    }

432    /// This index's entry-table capacity (always a power of two).
433    #[must_use]
434    pub(crate) fn capacity(&self) -> u32 {
435        self.capacity
436    }

438    /// Longest linear-probe chain observed so far by any insert or lookup
439    /// against this index instance. Pure observability, never reset, never
440    /// influences control flow — see the field doc on `max_probe_len`.
441    #[must_use]
442    pub fn max_probe_len(&self) -> u32 {
443        self.max_probe_len.load(Ordering::Relaxed)
444    }

446    /// Record a probe chain of `len` steps as the new high-water mark if it
447    /// exceeds the current one. `Relaxed` is enough: this is an
448    /// observability counter with no ordering dependency on any other
449    /// field, and a lost update under concurrent writers only means a
450    /// transient undercount, never a wrong value that persists.
451    #[inline]
452    fn bump_max_probe_len(&self, len: u32) {
453        self.max_probe_len.fetch_max(len, Ordering::Relaxed);
454    }

456    /// Extract the bucket index from a 16-byte hash.
457    ///
458    /// The seed is `XORed` into the raw home value before the avalanche mix,
459    /// so the same content hash lands in different buckets across pools
460    /// with different seeds. The mixed value is what gets stored in `homes`,
461    /// so `grow()` (which rehashes from stored homes) sees the already-mixed
462    /// value and doesn't need to know the seed at all.
463    #[inline]
464    fn bucket(&self, hash: &[u8; 16]) -> usize {
465        let raw = u64::from_be_bytes(hash[..8].try_into().unwrap());
466        let home = mix(raw ^ self.config.seed);
467        self.bucket_for_home(home)
468    }
469
470    /// Mix a raw home value with this index's seed. Used by insert to store
471    /// the post-mix value in `homes`, and by `hydrate_slot_identity` to
472    /// reproduce the same home from a recovered hash.
473    #[inline]
474    fn mix_home(&self, raw_home: u64) -> u64 {
475        mix(raw_home ^ self.config.seed)
476    }
477
478    #[inline]
479    fn bucket_for_home(&self, home: u64) -> usize {
480        let masked = (home >> self.shift) & u64::from(self.mask);
481        // Safety: masked is always <= mask < capacity which fits in usize on all platforms
482        usize::try_from(masked).unwrap_or(usize::MAX)
483    }
484
485    /// Extract the 16-bit tag from a 16-byte hash, seeded with this index's seed.
486    ///
487    /// Uses bytes 8..12, disjoint from the bytes `bucket()` reads (0..8).
488    /// Seeding the tag prevents an attacker from grinding offline for
489    /// tag collisions against a known target bucket — without the seed,
490    /// a narrower 16-bit tag would make I/O amplification attacks feasible.
491    #[inline]
492    fn tag(&self, hash: &[u8; 16]) -> u32 {
493        let raw = u32::from_be_bytes(hash[8..12].try_into().unwrap());
494        let mixed = mix(u64::from(raw) ^ self.config.seed);
495        (mixed >> 48) as u32
496    }

498    /// The packed-entry tag for `hash`, exposed to crate-local recovery code
499    /// that must validate a persisted entry against its authoritative frame.
500    ///
501    /// Must use this index's own seed: every stored entry's tag was computed
502    /// via `self.tag(hash)` at insert time, so a recovery comparison against
503    /// an unseeded (or differently-seeded) tag would fail closed on every
504    /// occupied entry, not just colliding ones.
505    #[inline]
506    pub(crate) fn tag_for_hash(&self, hash: &[u8; 16]) -> u32 {
507        self.tag(hash)
508    }

510    #[inline]
511    fn tail(hash: &[u8; 16]) -> u64 {
512        const KNOWN: u64 = 1_u64 << 63;
513        let mut bytes = [0; 8];
514        bytes[3..].copy_from_slice(&hash[11..]);
515        KNOWN | u64::from_be_bytes(bytes)
516    }

585    ///

594    pub fn insert_undoable_with_removed(
595        &self,
596        hash: &[u8; 16],
597        slot: u16,
598        offset: u64,
599        removed: bool,
600    ) -> Result<(u32, u64, EntryUndo), InsertError> {
601        let tag = self.tag(hash);
602        let raw_home = u64::from_be_bytes(hash[..8].try_into().unwrap());
603        let home = self.mix_home(raw_home);
604        let mut bucket = self.bucket_for_home(home);
605        let mut probe_len: u32 = 0;
606
607        loop {
608            // `bucket` is masked by `self.mask` (capacity - 1) and capacity is
609            // a u32, so the conversion is lossless; it is done once here so the
610            // return/undo structs below don't each need a checked cast.
611            let bucket_wire = u32::try_from(bucket)
612                .expect("bucket is masked by the u32 capacity, so it always fits");
613            let EntryStorage::Owned(slots) = &self.slots else {
614                self.bump_max_probe_len(probe_len);
615                return Err(InsertError::TableFull);
616            };
617            let entry = IndexEntry(slots[bucket].load(Ordering::Acquire));
618            if entry.is_empty() {
619                // Only reject when actually inserting into a new entry
620                let threshold = max_occupied(self.capacity, self.config.load_factor_percent);
621                if self.len.load(Ordering::Relaxed) >= threshold {
622                    self.bump_max_probe_len(probe_len);
623                    return Err(InsertError::TableFull);
624                }
625                let old_home = self.homes.lock()[bucket];
626                let old_tail = self.tails.lock()[bucket];
627                // Writers are serialized by the collection put lock. Publish
628                // the entry last so concurrent readers see either the prior
629                // empty entry or a complete record location.
630                self.homes.lock()[bucket] = home;
631                self.tails.lock()[bucket] = Self::tail(hash);
632                let value = IndexEntry::new_with_removed(tag, slot, offset, removed).0;
633                slots[bucket].store(value, Ordering::Release);
634                self.len.fetch_add(1, Ordering::Relaxed);
635                let undo = EntryUndo {
636                    bucket: bucket_wire,
637                    old_entry: 0,
638                    old_home,
639                    old_tail,
640                    was_empty: true,
641                };
642                self.bump_max_probe_len(probe_len);
643                return Ok((bucket_wire, value, undo));
644            }
645            // A tag only filters candidates; it never proves key equality.
646            // Checkpoint-derived slots initially lack their 40-bit tail and
647            // ask storage to hydrate it from the frame before deciding.
648            if entry.tag() == tag {
649                let tail = self.tails.lock()[bucket];
650                if tail == 0 {
651                    self.bump_max_probe_len(probe_len);
652                    return Err(InsertError::NeedsIdentity {
653                        bucket: bucket_wire,
654                        slot: entry.slot(),
655                        offset: entry.offset(),
656                    });
657                }
658                if self.homes.lock()[bucket] == home && tail == Self::tail(hash) {
659                    let old_entry = entry.0;
660                    let old_home = self.homes.lock()[bucket];
661                    let value = IndexEntry::new_with_removed(tag, slot, offset, removed).0;
662                    slots[bucket].store(value, Ordering::Release);
663                    let undo = EntryUndo {
664                        bucket: bucket_wire,
665                        old_entry,
666                        old_home,
667                        old_tail: tail,
668                        was_empty: false,
669                    };
670                    self.bump_max_probe_len(probe_len);
671                    return Ok((bucket_wire, value, undo));
672                }
673            }
674            probe_len = probe_len.saturating_add(1);
675            bucket = bucket.wrapping_add(1) & self.mask as usize;
676        }
677    }

679    /// Reverse a write reported by [`Self::insert_undoable`]. Callers must
680    /// replay a batch's undo entries in the reverse order they were
681    /// produced, and only against the same live index that produced them
682    /// (never a cloned/owned index materialized after the fact).
683    ///
684    /// # Panics
685    /// Never panics; a no-op if called against a checkpoint-mmap-backed
686    /// index, since [`Self::insert_undoable`] cannot have produced an undo
687    /// entry against one (it errors with `TableFull` first).
688    pub fn rollback_slot(&self, undo: &EntryUndo) {
689        let EntryStorage::Owned(slots) = &self.slots else {
690            return;
691        };
692        let bucket = undo.bucket as usize;
693        self.homes.lock()[bucket] = undo.old_home;
694        self.tails.lock()[bucket] = undo.old_tail;
695        slots[bucket].store(undo.old_entry, Ordering::Release);
696        if undo.was_empty {
697            self.len.fetch_sub(1, Ordering::Relaxed);
698        }
699    }

701    /// Mark a checkpoint-derived entry with the full identity recovered from
702    /// its authoritative frame. Returns `false` if the entry changed or the
703    /// supplied hash cannot describe that entry.
704    pub(crate) fn hydrate_slot_identity(&self, bucket: u32, hash: &[u8; 16]) -> bool {
705        let bucket = bucket as usize;
706        if bucket >= self.capacity as usize {
707            return false;
708        }
709        let entry = IndexEntry(self.entry_at(bucket));
710        if entry.is_empty() || entry.tag() != self.tag(hash) {
711            return false;
712        }
713        let raw_home = u64::from_be_bytes(hash[..8].try_into().expect("16-byte hash"));

716        self.tails.lock()[bucket] = Self::tail(hash);

720    /// Apply recorded delta frames on top of a checkpoint-backed index,
721    /// replicating the live insert/overwrite sequence that produced them.
722    ///
723    /// Only valid on an owned (materialized) index — callers must clone an
724    /// mmap-backed index first. Frames are applied exactly as the live
725    /// session wrote them: an overwrite of a non-empty entry leaves the length
726    /// unchanged, a write into an empty entry increments it. The index's `len`
727    /// after replay therefore equals the checkpoint's occupancy plus the
728    /// number of fresh-entry writes, which is what the sidecar integrity sum
729    /// checks against.
730    ///
731    /// # Errors
732    /// Returns `DeltaReplayError` if a frame targets a bucket outside this
733    /// index's capacity, or carries the empty-entry sentinel (either is a
734    /// structurally inconsistent log, rejected wholesale), or the index is
735    /// still mmap-backed.
736    pub fn replay_frames(&self, frames: &[DeltaFrame]) -> Result<(), DeltaReplayError> {
737        let EntryStorage::Owned(slots) = &self.slots else {
738            return Err(DeltaReplayError::RequiresOwnedIndex);
739        };
740        // Validate every frame before mutating any entry: the contract is
741        // wholesale rejection of a structurally inconsistent log, so a
742        // caller must never observe a partially replayed index.
743        for frame in frames {
744            let bucket = frame.bucket as usize;
745            if bucket >= self.capacity as usize {
746                return Err(DeltaReplayError::FrameOutOfBounds {
747                    bucket: frame.bucket,
748                    capacity: self.capacity,
749                });
750            }
751            if frame.slot == 0 {
752                // A legitimate insert never logs the empty sentinel (see
753                // `record_delta`'s only caller). Storing it here would erase
754                // whatever live entry currently occupies this bucket and cut
755                // the probe chain short for anything beyond it — reject the
756                // whole log instead of silently losing data.
757                return Err(DeltaReplayError::EmptySlot {
758                    bucket: frame.bucket,
759                });
760            }
761            // The packed entry's slot field must name a live shard. A writer
762            // only ever encodes a slot from its own `ShardPool` (bounded by
763            // `MAX_SHARDS`; see `IndexEntry::new`), so a value at or above the
764            // cap can only be log corruption or a structurally invalid frame.
765            // Accepting it would install an entry no shard scan can resolve —
766            // `referenced_slot_ids` and friends skip `id >= MAX_SHARDS` — so
767            // the record would silently vanish from recovery.
768            let decoded = IndexEntry(frame.slot);
769            if decoded.has_invalid_slot_bits()
770                || usize::from(decoded.slot()) >= crate::shard::MAX_SHARDS
771            {
772                return Err(DeltaReplayError::SlotOutOfRange {
773                    bucket: frame.bucket,
774                    slot: decoded.slot(),
775                });
776            }
777        }
778        for frame in frames {
779            let bucket = frame.bucket as usize;
780            let previous = slots[bucket].load(Ordering::Acquire);
781            if previous == 0 {
782                self.len.fetch_add(1, Ordering::Relaxed);
783            }
784            slots[bucket].store(frame.slot, Ordering::Release);
785        }
786        Ok(())
787    }

789    /// Double this live index's capacity without reading packfiles.
790    ///
791    /// Returns `false` when the identity side tables are incomplete or the
792    /// index cannot grow further. Complete checkpoint identities can rehash
793    /// without reading packfiles; older/incomplete checkpoints retain the
794    /// packfile rebuild fallback.
795    pub fn grow(&self) -> Option<Self> {
796        // The gate is checked when growth is used, not only when the index was
797        // loaded: delta replay stores slot values straight into the table without
798        // identities, so a table that was complete at load can be incomplete by
799        // the time it has to grow.
800        if !self.can_grow || self.capacity > u32::MAX / 2 || !self.identities_now_complete() {
801            return None;
802        }
803        let grown = Self::with_config((self.capacity as usize).saturating_mul(2), self.config);
804        let homes = self.homes.lock();
805        let tails = self.tails.lock();
806        for (index, home) in homes.iter().copied().enumerate() {
807            let entry = IndexEntry(self.slots.get(index));
808            if entry.is_empty() {
809                continue;
810            }
811            let mut bucket = grown.bucket_for_home(home);
812            while !IndexEntry(grown.entry_at(bucket)).is_empty() {
813                bucket = bucket.wrapping_add(1) & grown.mask as usize;
814            }
815            grown.homes.lock()[bucket] = home;
816            grown.tails.lock()[bucket] = tails[index];
817            let EntryStorage::Owned(slots) = &grown.slots else {
818                unreachable!("new index is owned")
819            };
820            slots[bucket].store(entry.0, Ordering::Relaxed);
821            grown.len.fetch_add(1, Ordering::Relaxed);
822        }
823        Some(grown)
824    }

826    /// Double the table by recovering each occupied entry's full hash from
827    /// its stored location.
828    ///
829    /// Checkpoint-backed indexes deliberately omit `homes`, so their first
830    /// post-restart resize cannot use [`Self::grow`].  The packed slots still
831    /// retain every `(slot, offset)`, however.  A caller can therefore
832    /// recover only the `len` source hashes from the authoritative records
833    /// and rehash without rescanning every pack in the store. Locations are
834    /// visited in `(shard, offset)` order rather than hash-table order, so a
835    /// cold mmap walk follows pack order and lets the kernel coalesce page
836    /// faults/read-ahead. The callback must return the hash for precisely the
837    /// supplied location.
838    ///
839    /// `Ok(None)` means the table cannot grow further; callers retain their
840    /// full-rebuild fallback for that terminal case.
841    pub(crate) fn grow_by_recovering_hashes<E>(
842        &self,
843        mut hash_at: impl FnMut(u16, u64, u32) -> Result<[u8; 16], E>,
844    ) -> Result<Option<Self>, E> {
845        if self.capacity > u32::MAX / 2 {
846            return Ok(None);
847        }
848
849        let mut locations = Vec::with_capacity(self.len());
850        for index in 0..self.capacity as usize {
851            let entry = IndexEntry(self.entry_at(index));
852            if entry.is_empty() {
853                continue;
854            }
855            locations.push((
856                entry.slot(),
857                entry.offset(),
858                entry.tag(),
859                entry.is_removed(),
860            ));
861        }
862        locations.sort_unstable_by_key(|(slot, offset, _, _)| (*slot, *offset));
863
864        let grown = Self::with_config((self.capacity as usize).saturating_mul(2), self.config);
865        for (slot, offset, slot_tag, removed) in locations {
866            let hash = hash_at(slot, offset, slot_tag)?;
867            // A doubled table is at most 37.5% full because insertion only
868            // requests growth at 75%, so TableFull should be unreachable —
869            // but the load threshold has overflowed before (see
870            // `max_occupied`), and a silently dropped insert here would
871            // publish a grown index missing entries. Fail the growth closed
872            // instead: `Ok(None)` sends the caller to its pack-scan rebuild.
873            if grown
874                .insert_tracked_with_removed(&hash, slot, offset, removed)
875                .is_err()
876            {
877                return Ok(None);
878            }
879        }
880        Ok(Some(grown))
881    }

894    /// Whether the indexed latest version of `hash` is explicitly removed.
895    /// Returns `None` when no matching identity is present.
896    ///
897    /// # Panics
898    /// Never panics in practice: `hash` is exactly 16 bytes, so the fixed-size
899    /// conversion used to derive its home cannot fail.
900    #[must_use]
901    pub fn lookup_removed(&self, hash: &[u8; 16]) -> Option<bool> {
902        let tag = self.tag(hash);
903        let bucket = self.bucket(hash);
904        let raw_home = u64::from_be_bytes(hash[..8].try_into().expect("16-byte hash"));
905        let mixed_home = self.mix_home(raw_home);
906        let expected_tail = Self::tail(hash);
907        for step in 0..self.capacity as usize {
908            let current = bucket.wrapping_add(step) & self.mask as usize;
909            let entry = IndexEntry(self.entry_at(current));
910            if entry.is_empty() {
911                return None;
912            }
913            if entry.tag() != tag {
914                continue;
915            }
916            let tails = self.tails.lock();
917            let tail = tails.get(current).copied().unwrap_or(0);
918            drop(tails);
919            if tail == 0 {
920                continue;
921            }
922            let homes = self.homes.lock();
923            let home = homes.get(current).copied().unwrap_or(0);
924            drop(homes);
925            if home == mixed_home && tail == expected_tail {
926                return Some(entry.is_removed());
927            }
928        }
929        None
930    }

1091    /// Rewrites every occupied entry's `slot` through `remap`, in place.
1092    ///
1093    /// A checkpoint-loaded index's `slot`s were encoded by the writer's
1094    /// own local `ShardPool` slots at checkpoint-write time — a process-local
1095    /// handle, not a stable identity. `remap` (built by the caller from the
1096    /// checkpoint's persisted pack table, translated through this reader's
1097    /// own currently-open packs) reconciles that against this reader's own
1098    /// entry numbering, which can differ once any shard has ever been retired
1099    /// (see the checkpoint pack-table documentation).
1100    ///
1101    /// Checks first without touching storage: the common case (no shard has
1102    /// ever been retired, so a fresh reader's entry numbering already agrees
1103    /// with the checkpoint's) needs no rewrite at all, and mmap-backed
1104    /// indexes must stay mmap-backed when nothing actually changes — callers
1105    /// and tests depend on an unmodified checkpoint-loaded index staying
1106    /// zero-copy until its first real write. Only materializes into an owned
1107    /// copy (writes are never applied to the mmap) when at least one
1108    /// occupied entry's `slot` actually needs to change.
1109    ///
1110    /// Returns `false`, leaving the index unmodified, if any occupied entry's
1111    /// `slot` has no entry in `remap` — the checkpoint's pack table
1112    /// should cover every `slot` any of its own index entries use, so a
1113    /// miss means the checkpoint is internally inconsistent and the caller
1114    /// should fall back to a full rescan rather than serve unresolvable
1115    /// slots.
1116    #[must_use]
1117    pub fn remap_slots(&mut self, remap: &std::collections::HashMap<u16, u16>) -> bool {
1118        let capacity = self.capacity as usize;
1119        let mut needs_rewrite = false;
1120        for index in 0..capacity {
1121            let entry = IndexEntry(self.slots.get(index));
1122            if entry.is_empty() {
1123                continue;
1124            }
1125            let Some(&new_slot) = remap.get(&entry.slot()) else {
1126                return false;
1127            };
1128            if new_slot != entry.slot() {
1129                needs_rewrite = true;
1130            }
1131        }
1132        if !needs_rewrite {
1133            return true;
1134        }
1135        if !matches!(self.slots, EntryStorage::Owned(_)) {
1136            self.slots = EntryStorage::Owned(self.slots.materialize(capacity));
1137        }
1138        let EntryStorage::Owned(slots) = &self.slots else {
1139            unreachable!("just materialized to Owned above");
1140        };
1141        for cell in slots {
1142            let raw = cell.load(Ordering::Acquire);
1143            let entry = IndexEntry(raw);
1144            if entry.is_empty() {
1145                continue;
1146            }
1147            let Some(&new_slot) = remap.get(&entry.slot()) else {
1148                return false;
1149            };
1150            if new_slot == entry.slot() {
1151                continue;
1152            }
1153            let remapped = IndexEntry::new_with_removed(
1154                entry.tag(),
1155                new_slot,
1156                entry.offset(),
1157                entry.is_removed(),
1158            )
1159            .0;
1160            cell.store(remapped, Ordering::Relaxed);
1161        }
1162        true
1163    }

1174    /// Serialize the index to bytes for persistence.
1175    ///
1176    /// Format (all little-endian):
1177    /// ```text
1178    ///   [capacity: u64]
1179    ///   [slot_0: u64] ... [slot_{cap-1}: u64]     ← capacity × 8 B
1180    ///   [home_0: u64] ... [home_{cap-1}: u64]     ← capacity × 8 B (0 if unhydrated)
1181    ///   [tail_0: u64] ... [tail_{cap-1}: u64]     ← capacity × 8 B (0 if unhydrated)
1182    /// ```
1183    /// Total: `8 + capacity * 24` bytes.
1184    #[must_use]
1185    pub fn serialize(&self) -> Vec<u8> {
1186        let cap = self.capacity as usize;
1187        let byte_len = 8_usize.wrapping_add(cap.wrapping_mul(24));
1188        let mut buf = Vec::with_capacity(byte_len);
1189        buf.extend_from_slice(&u64::from(self.capacity).to_le_bytes());
1190        for index in 0..cap {
1191            let entry = IndexEntry(self.entry_at(index));
1192            buf.extend_from_slice(&entry.0.to_le_bytes());
1193        }
1194        let homes = self.homes.lock();
1195        let tails = self.tails.lock();
1196        for index in 0..cap {
1197            buf.extend_from_slice(&homes[index].to_le_bytes());
1198        }
1199        for index in 0..cap {
1200            buf.extend_from_slice(&tails[index].to_le_bytes());
1201        }
1202        buf
1203    }

1218    /// Deserialize an index from bytes with a given config (providing seed).
1219    ///
1220    /// Accepts both the slots-only format (`8 + cap*8` bytes) and the
1221    /// slots+homes+tails format (`8 + cap*24` bytes). Homes and tails are
1222    /// populated from the blob when present; zeroed otherwise.
1223    ///
1224    /// # Errors
1225    /// Returns `DeserializationError::TooShort` if the data is too short,
1226    /// or `DeserializationError::InvalidCapacity` if the capacity is not
1227    /// a power of two or is less than 16.
1228    ///
1229    /// # Panics
1230    /// Panics if the 8-byte capacity header cannot be read (guaranteed by the length check).
1231    pub fn deserialize_with_config(
1232        data: &[u8],
1233        config: IndexConfig,
1234    ) -> Result<Self, DeserializationError> {
1235        if data.len() < 8 {
1236            return Err(DeserializationError::TooShort);
1237        }
1238        let capacity_wire = u64::from_le_bytes(data[..8].try_into().unwrap());
1239        let capacity = u32::try_from(capacity_wire).map_err(|_| DeserializationError::TooShort)?;
1240        let capacity_usize = capacity as usize;
1241
1242        // Reject unsupported capacities: must be power of two and >= 16
1243        if capacity < 16 || !capacity.is_power_of_two() {
1244            return Err(DeserializationError::InvalidCapacity);
1245        }
1246
1247        let slots_len = 8_usize.wrapping_add(capacity_usize.wrapping_mul(8));
1248        let full_len = 8_usize.wrapping_add(capacity_usize.wrapping_mul(24));
1249        if data.len() < slots_len {
1250            return Err(DeserializationError::TooShort);
1251        }
1252        let has_homes_tails = data.len() >= full_len;
1253
1254        let mut slots = Vec::with_capacity(capacity_usize);
1255        let mut len: u32 = 0;
1256        for i in 0..capacity_usize {
1257            let offset = 8_usize.wrapping_add(i.wrapping_mul(8));
1258            let val = u64::from_le_bytes(data[offset..offset.wrapping_add(8)].try_into().unwrap());
1259            let entry = IndexEntry(val);
1260            if !IndexEntry::valid_raw_slot(val) {
1261                return Err(DeserializationError::InvalidSlot);
1262            }
1263            if !entry.is_empty() {
1264                len = len.wrapping_add(1);
1265            }
1266            slots.push(AtomicU64::new(entry.0));
1267        }
1268
1269        let shift = 64_u32.wrapping_sub(capacity.trailing_zeros());
1270
1271        let homes = if has_homes_tails {
1272            let base = slots_len;
1273            let mut h = Vec::with_capacity(capacity_usize);
1274            for i in 0..capacity_usize {
1275                let offset = base.wrapping_add(i.wrapping_mul(8));
1276                h.push(u64::from_le_bytes(
1277                    data[offset..offset.wrapping_add(8)].try_into().unwrap(),
1278                ));
1279            }
1280            h
1281        } else {
1282            vec![0; capacity_usize]
1283        };
1284
1285        let tails = if has_homes_tails {
1286            let base = slots_len.wrapping_add(capacity_usize.wrapping_mul(8));
1287            let mut t = Vec::with_capacity(capacity_usize);
1288            for i in 0..capacity_usize {
1289                let offset = base.wrapping_add(i.wrapping_mul(8));
1290                t.push(u64::from_le_bytes(
1291                    data[offset..offset.wrapping_add(8)].try_into().unwrap(),
1292                ));
1293            }
1294            t
1295        } else {
1296            vec![0; capacity_usize]
1297        };
1298
1299        let slots = EntryStorage::Owned(slots);
1300        let can_grow = Self::identity_tables_complete(capacity, shift, &slots, &homes, &tails);
1301        Ok(Self {
1302            mask: capacity.wrapping_sub(1),
1303            capacity,
1304            shift,
1305            config,
1306            slots,
1307            homes: Mutex::new(homes),
1308            tails: Mutex::new(tails),
1309            can_grow,
1310            len: AtomicU32::new(len),
1311            max_probe_len: AtomicU32::new(0),
1312        })
1313    }

1348    /// Build a read-only index directly over a validated checkpoint's raw
1349    /// slots with a given config, loading pre-hydrated homes and tails from
1350    /// the checkpoint mmap. This eliminates cold-start packfile reads for
1351    /// tag-collision verification when the checkpoint carries identity data.
1352    ///
1353    /// The slots remain mmap-backed; homes and tails are copied into owned
1354    /// `Vec<u64>` so they can be updated by `hydrate_slot_identity` if needed.
1355    ///
1356    /// # Panics
1357    /// Panics if `[homes_offset, homes_offset + capacity*8)` or
1358    /// `[tails_offset, tails_offset + capacity*8)` fall outside `mmap` —
1359    /// callers must only pass offsets validated against a checkpoint's own
1360    /// declared body length (see `checkpoint::read_checkpoint`).
1361    #[must_use]
1362    pub fn from_mmap_slots_with_homes_tails(
1363        mmap: Arc<Mmap>,
1364        slots_offset: usize,
1365        homes_offset: usize,
1366        tails_offset: usize,
1367        capacity: u32,
1368        len: u32,
1369        config: IndexConfig,
1370    ) -> Self {
1371        let cap = usize::try_from(capacity).unwrap_or(usize::MAX);
1372        let mut homes = Vec::with_capacity(cap);
1373        let mut tails = Vec::with_capacity(cap);
1374        let buf: &[u8] = &mmap;
1375        for i in 0..cap {
1376            let h_off = homes_offset.wrapping_add(i.wrapping_mul(8));
1377            let t_off = tails_offset.wrapping_add(i.wrapping_mul(8));
1378            homes.push(u64::from_le_bytes(
1379                buf[h_off..h_off.wrapping_add(8)].try_into().unwrap(),
1380            ));
1381            tails.push(u64::from_le_bytes(
1382                buf[t_off..t_off.wrapping_add(8)].try_into().unwrap(),
1383            ));
1384        }
1385        let slots = EntryStorage::Mmap {
1386            mmap,
1387            offset: slots_offset,
1388        };
1389        let can_grow = Self::identity_tables_complete(
1390            capacity,
1391            64_u32.wrapping_sub(capacity.trailing_zeros()),
1392            &slots,
1393            &homes,
1394            &tails,
1395        );
1396        Self {
1397            mask: capacity.wrapping_sub(1),
1398            capacity,
1399            shift: 64_u32.wrapping_sub(capacity.trailing_zeros()),
1400            config,
1401            slots,
1402            homes: Mutex::new(homes),
1403            tails: Mutex::new(tails),
1404            can_grow,
1405            len: AtomicU32::new(len),
1406            max_probe_len: AtomicU32::new(0),
1407        }
1408    }

1507/// Iterator over candidate offsets for a hash lookup.
1508///
1509/// Yields `(slot, offset)` for each entry whose tag matches and whose
1510/// in-memory identity (home + tail) matches the query hash, terminating at
1511/// the first empty entry or after the full capacity is probed.
1512///
1513/// When the index has hydrated identity (live/warm index), candidates are
1514/// verified in-memory before yielding — this eliminates disk I/O for tag
1515/// collisions on the normal path. For checkpoint-backed indexes without
1516/// hydrated identity (tail == 0), all tag-matching candidates are yielded
1517/// and the caller must verify against the packfile.
1518pub struct LookupIter<'a> {
1519    index: &'a LossyIndex,
1520    tag: u32,
1521    /// The mixed home for the query hash, for in-memory verification.
1522    mixed_home: u64,
1523    /// The expected tail for the query hash.
1524    expected_tail: u64,
1525    bucket: usize,
1526    mask: usize,
1527    remaining: usize,
1528    /// Steps walked so far, for `LossyIndex::max_probe_len` observability.
1529    steps: u32,
1530}
1531
1532impl Iterator for LookupIter<'_> {
1533    type Item = (u16, u64);
1534
1535    #[inline]
1536    fn next(&mut self) -> Option<Self::Item> {
1537        while self.remaining > 0 {
1538            self.remaining = self.remaining.wrapping_sub(1);
1539            let entry = IndexEntry(self.index.entry_at(self.bucket));
1540            if entry.is_empty() {
1541                self.index.bump_max_probe_len(self.steps);
1542                return None;
1543            }
1544            self.steps = self.steps.saturating_add(1);
1545            let current = self.bucket;
1546            self.bucket = self.bucket.wrapping_add(1) & self.mask;
1547            if entry.tag() != self.tag {
1548                continue;
1549            }
1550            // Tag matches. For a live index with hydrated identity, verify
1551            // home + tail in-memory to avoid yielding false positives that
1552            // would force the caller into an expensive packfile read.
1553            let tails = self.index.tails.lock();
1554            let tail = if tails.len() > current {
1555                tails[current]
1556            } else {
1557                0
1558            };
1559            drop(tails);
1560            if tail != 0 {
1561                // Identity is hydrated — check home and tail.
1562                let homes = self.index.homes.lock();
1563                let home = if homes.len() > current {
1564                    homes[current]
1565                } else {
1566                    0
1567                };
1568                drop(homes);
1569                if home == self.mixed_home && tail == self.expected_tail {
1570                    self.index.bump_max_probe_len(self.steps);
1571                    return Some((entry.slot(), entry.offset()));
1572                }
1573                // Tag matched but home/tail didn't — false positive, skip.
1574                continue;
1575            }
1576            // Checkpoint-backed without hydrated identity — yield and let
1577            // the caller verify against the packfile.
1578            self.index.bump_max_probe_len(self.steps);
1579            return Some((entry.slot(), entry.offset()));
1580        }
1581        self.index.bump_max_probe_len(self.steps);
1582        None
1583    }
1584}
