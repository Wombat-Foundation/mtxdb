1//! Database-root layout and named packfile pools.
2//!
3//! A [`crate::packfile::storage::PackfileStorage`] owns exactly one pool;
4//! it must be opened on one of this module's pool directories, never on the
5//! database root. Keeping that boundary explicit gives state, event-DAG, and
6//! edges data independent shard, GC, durability, and writer-lock
7//! lifecycles.
8
9use std::fs::{self, OpenOptions};
10use std::io::{self, Write};
11use std::path::{Path, PathBuf};
12use std::sync::atomic::{AtomicU64, Ordering};

14static DESCRIPTOR_TEMP_COUNTER: AtomicU64 = AtomicU64::new(0);

30/// Immutable descriptor at a database root.
31///
32/// Layout: `magic(4) | version(1) | seed(8) | version_len(1) | created_by |
33/// pool list`. The seed is random and nonzero, written once when the root is
34/// created; every pool's index seed derives from it (see
35/// [`DatabaseLayout::pool_seed`]), so a reader knows a pool's seed before the
36/// pool exists on disk. `created_by` is the `mtxdb` version that created the
37/// root (diagnostic only). The descriptor is parsed, not compared
38/// byte-for-byte: a mismatched version is rejected rather than misparsed.
39/// Every root has one shared WAL at its root, so the descriptor records no WAL
40/// layout. The shipped CLI, C ABI, and WASI entry points validate this root and
41/// open their selected named pool through this type;
42/// [`PackfileStorage`](crate::PackfileStorage) remains available as a
43/// lower-level single-pool API.
44const DB_META_MAGIC: &[u8; 4] = b"MTXD";
45const DB_META_VERSION: u8 = 1;
46const DB_META_SEED_LEN: usize = 8;
47/// The pool list a descriptor must carry, built from the pool directory names
48/// so it can never drift from them. A root whose descriptor lists other names
49/// is rejected by [`parse_db_meta`] instead of being opened with its data
50/// orphaned in directories nothing reads.
51fn db_meta_pool_list() -> Vec<u8> {
52    let mut list = Vec::new();
53    for shard_type in ShardType::ALL {
54        list.extend_from_slice(shard_type.as_str().as_bytes());
55        list.push(b'\n');
56    }
57    list
58}

60/// File name of the database-root descriptor.
61pub const DB_META_FILENAME: &str = "db.meta";

63/// Database-root shared-WAL lock file, held by the WAL owner and by any pool

65pub const WAL_LOCK_FILENAME: &str = ".mtxdb.wal.lock";

67/// `magic + version + seed + version_len` header length.
68const DB_META_FIXED_LEN: usize = 4 + 1 + DB_META_SEED_LEN + 1;

70/// A parsed, valid database descriptor.
71#[derive(Debug, Clone, PartialEq, Eq)]
72struct DbMeta {
73    seed: u64,
74    created_by: String,
75}

77/// A fresh nonzero random seed.
78fn random_seed() -> u64 {
79    use std::collections::hash_map::RandomState;
80    use std::hash::{BuildHasher, Hasher};
81    loop {
82        let seed = RandomState::new().build_hasher().finish();
83        if seed != 0 {
84            return seed;
85        }
86    }
87}

89/// Build the on-disk descriptor bytes for a database root.
90fn db_meta_bytes(seed: u64) -> Vec<u8> {
91    let pool_list = db_meta_pool_list();
92    let created_by = env!("CARGO_PKG_VERSION").as_bytes();
93    let created_by = &created_by[..created_by.len().min(usize::from(u8::MAX))];
94    let mut buf = Vec::with_capacity(
95        DB_META_FIXED_LEN
96            .saturating_add(created_by.len())
97            .saturating_add(pool_list.len()),
98    );
99    buf.extend_from_slice(DB_META_MAGIC);
100    buf.push(DB_META_VERSION);
101    buf.extend_from_slice(&seed.to_le_bytes());
102    buf.push(u8::try_from(created_by.len()).unwrap_or(u8::MAX));
103    buf.extend_from_slice(created_by);
104    buf.extend_from_slice(&pool_list);
105    buf
106}

108/// Parse a descriptor read from disk: correct magic, a version this build
109/// understands, a nonzero seed, and the expected pool list.
110fn parse_db_meta(contents: &[u8]) -> Option<DbMeta> {
111    if contents.len() < DB_META_FIXED_LEN
112        || &contents[..4] != DB_META_MAGIC
113        || contents[4] != DB_META_VERSION
114    {
115        return None;
116    }
117    let seed = u64::from_le_bytes(contents[5..5 + DB_META_SEED_LEN].try_into().ok()?);
118    if seed == 0 {
119        return None;
120    }
121    let version_len = usize::from(contents[DB_META_FIXED_LEN - 1]);
122    let created_end = DB_META_FIXED_LEN.checked_add(version_len)?;
123    let created_by =
124        String::from_utf8(contents.get(DB_META_FIXED_LEN..created_end)?.to_vec()).ok()?;
125    (contents[created_end..] == db_meta_pool_list()[..]).then_some(DbMeta { seed, created_by })
126}
127
128fn read_db_meta(path: &Path) -> io::Result<DbMeta> {
129    parse_db_meta(&fs::read(path)?).ok_or_else(|| {
130        io::Error::new(
131            io::ErrorKind::InvalidData,
132            format!("unrecognized mtxdb database descriptor: {}", path.display()),
133        )
134    })
135}

137/// The index seed of the pool at `pool_dir` when it lives at
138/// `<root>/pools/<pool>` inside a database root; `None` for a standalone pool
139/// directory, which keeps its own `pool.meta`.
140///
141/// # Errors
142/// Returns `InvalidData` if the enclosing `db.meta` is unrecognized.
143pub fn enclosing_pool_seed(pool_dir: &Path) -> io::Result<Option<u64>> {
144    let Some(name) = pool_dir.file_name().and_then(|name| name.to_str()) else {
145        return Ok(None);
146    };
147    let Some(shard_type) = ShardType::ALL.into_iter().find(|t| t.as_str() == name) else {
148        return Ok(None);
149    };
150    let Some(pools) = pool_dir
151        .parent()
152        .filter(|p| p.file_name().is_some_and(|n| n == "pools"))
153    else {
154        return Ok(None);
155    };
156    let Some(root) = pools.parent() else {
157        return Ok(None);
158    };
159    let meta_path = root.join(DB_META_FILENAME);
160    match fs::metadata(&meta_path) {
161        Ok(metadata) if metadata.is_file() => {}
162        Ok(_) => {
163            return Err(io::Error::new(
164                io::ErrorKind::InvalidData,
165                format!(
166                    "unrecognized mtxdb database descriptor: {}",
167                    meta_path.display()
168                ),
169            ));
170        }
171        Err(error) if error.kind() == io::ErrorKind::NotFound => return Ok(None),
172        Err(error) => return Err(error),
173    }
174    Ok(Some(pool_seed(read_db_meta(&meta_path)?.seed, shard_type)))
175}

177/// The `mtxdb` version that created the database root enclosing `pool_dir`.
178#[must_use]
179pub fn enclosing_created_by(pool_dir: &Path) -> Option<String> {
180    let root = pool_dir.parent()?.parent()?;
181    read_db_meta(&root.join(DB_META_FILENAME))
182        .ok()
183        .map(|meta| meta.created_by)
184}

186/// Derive a pool's index seed from the root seed and the pool's tag. Never zero.
187fn pool_seed(root_seed: u64, shard_type: ShardType) -> u64 {
188    let mut label = Vec::with_capacity(12);
189    label.extend_from_slice(&root_seed.to_le_bytes());
190    label.extend_from_slice(&shard_type.physical_pool_tag());
191    let digest = crate::storage::DigestAlgorithm::Blake3.digest(&label);
192    let seed = u64::from_le_bytes(digest[..8].try_into().unwrap_or([1; 8]));
193    if seed == 0 {
194        1
195    } else {
196        seed
197    }
198}

200/// Whether `root` is a database root: it has a valid `db.meta`. `Ok(false)`
201/// when there is no descriptor at all.
202///
203/// # Errors
204/// Returns `InvalidData` if a descriptor exists but is unrecognized, so a
205/// caller can never mistake a corrupt or old-format root for a missing one.
206pub fn is_database_root(root: &Path) -> io::Result<bool> {
207    let meta_path = root.join(DB_META_FILENAME);
208    let contents = match fs::read(&meta_path) {
209        Ok(contents) => contents,
210        Err(error) if error.kind() == io::ErrorKind::NotFound => return Ok(false),
211        Err(error) => return Err(error),
212    };
213    if parse_db_meta(&contents).is_none() {
214        return Err(io::Error::new(
215            io::ErrorKind::InvalidData,
216            format!(
217                "unrecognized mtxdb database descriptor: {}",
218                meta_path.display()
219            ),
220        ));
221    }
222    Ok(true)
223}

225/// Nearest enclosing database root for `dir`, if any.
226///
227/// Walks `dir` and its ancestors looking for a `db.meta` descriptor. This is
228/// how a store opened on `root/pools/<pool>` discovers that it lives inside a
229/// database root (see the per-pool journal gate in
230/// [`PackfileStorage::enable_journal`](crate::PackfileStorage::enable_journal)).
231///
232/// # Errors
233/// Returns `InvalidData` if a descriptor is found but unrecognized, so a
234/// caller can never mistake a corrupt root for a standalone store.
235pub fn enclosing_root(dir: &Path) -> io::Result<Option<PathBuf>> {
236    for ancestor in dir.ancestors() {
237        let meta_path = ancestor.join(DB_META_FILENAME);
238        if !meta_path.is_file() {
239            continue;
240        }
241        let contents = fs::read(&meta_path)?;
242        if parse_db_meta(&contents).is_none() {
243            return Err(io::Error::new(
244                io::ErrorKind::InvalidData,
245                format!(
246                    "unrecognized mtxdb database descriptor: {}",
247                    meta_path.display()
248                ),
249            ));
250        }
251        return Ok(Some(ancestor.to_path_buf()));
252    }
253    Ok(None)
254}

256/// A named independent packfile pool in an mtxdb database.
257#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
258pub enum ShardType {
259    /// HAMT nodes, roots, and state-group sidecars.
260    State,
261    /// Event JSON plus collection-DAG-oriented event data.
262    EventDag,
263    /// Edges pool: houses previous-event edges (`PREV`) and auth-chain edges (`AUTH`).
264    Edges,
265    /// Server metadata, including federation signing keys and raw key responses.
266    ServerInfo,
267}
268
269impl ShardType {
270    /// Every shard type defined by the current database layout.
271    pub const ALL: [Self; 4] = [Self::State, Self::EventDag, Self::Edges, Self::ServerInfo];
272
273    /// This pool's position in [`Self::ALL`], the canonical pool order. Use it to
274    /// index per-pool arrays so they cannot drift from `ALL` when a pool is added.
275    #[must_use]
276    pub const fn index(self) -> usize {
277        let mut position = 0;
278        while position < Self::ALL.len() {
279            if Self::ALL[position] as usize == self as usize {
280                return position;
281            }
282            position = position.saturating_add(1);
283        }
284        // Unreachable: every variant is in `ALL`, and a test checks it.
285        Self::ALL.len()
286    }
287
288    /// Stable on-disk directory name for this pool.
289    #[must_use]
290    pub const fn as_str(self) -> &'static str {
291        match self {
292            Self::State => "mtpl-state",
293            Self::EventDag => "mtpl-event",
294            Self::Edges => "mtpl-edges",
295            Self::ServerInfo => "mtpl-server-info",
296        }
297    }
298
299    /// Stable 4-byte physical pool tag used in physical layout and diagnostic labeling.
300    ///
301    /// # Note
302    /// This is a physical storage pool tag (e.g. `EDGE`), **not** a logical
303    /// member namespace (`PREV` / `AUTH`). Collection derivation MUST use logical
304    /// member namespaces, never this physical pool tag.
305    #[must_use]
306    pub const fn physical_pool_tag(self) -> [u8; 4] {
307        match self {
308            Self::State => *b"STAT",
309            Self::EventDag => *b"EVNT",
310            Self::Edges => *b"EDGE",
311            Self::ServerInfo => *b"SINF",
312        }
313    }
314}

316/// Validated database root from which named pool paths can be derived.
317#[derive(Debug, Clone)]
318pub struct DatabaseLayout {
319    root: PathBuf,
320    seed: u64,
321}
322
323impl DatabaseLayout {
324    /// Open or initialize a database root.
325    ///
326    /// # Errors
327    /// Returns an error if the root cannot be created/read or its descriptor
328    /// is unknown.
329    pub fn open(root: PathBuf) -> io::Result<Self> {
330        fs::create_dir_all(&root)?;
331        Self::reject_flat_store(&root)?;
332        let meta_path = root.join(DB_META_FILENAME);
333        if !meta_path.exists() {
334            Self::write_descriptor(&meta_path)?;
335        }
336        let seed = read_db_meta(&meta_path)?.seed;
337        Ok(Self { root, seed })
338    }
339
340    /// The index seed of `shard_type`'s pool, derived from the root seed so it
341    /// is known before the pool exists on disk.
342    #[must_use]
343    pub fn pool_seed(&self, shard_type: ShardType) -> u64 {
344        pool_seed(self.seed, shard_type)
345    }
346
347    /// Return a named pool's directory, creating its parent directory.
348    ///
349    /// The pool itself is initialized by `PackfileStorage::open`.
350    ///
351    /// # Errors
352    /// Returns an error if the pool parent cannot be created.
353    pub fn pool_dir(&self, shard_type: ShardType) -> io::Result<PathBuf> {
354        let path = self.root.join("pools").join(shard_type.as_str());
355        fs::create_dir_all(&path)?;
356        Ok(path)
357    }
358
359    /// A named pool's directory path, without creating or checking it. An
360    /// absent directory is an empty pool: the pool creates it with its first
361    /// pack.
362    #[must_use]
363    pub fn pool_path(&self, shard_type: ShardType) -> PathBuf {
364        self.root.join("pools").join(shard_type.as_str())
365    }
366
367    /// Return a named pool's directory without creating it or any parent.
368    ///
369    /// # Errors
370    /// Returns an error if the pool directory does not exist.
371    pub fn pool_dir_read_only(&self, shard_type: ShardType) -> io::Result<PathBuf> {
372        let path = self.root.join("pools").join(shard_type.as_str());
373        if !path.is_dir() {
374            return Err(io::Error::new(
375                io::ErrorKind::NotFound,
376                format!("pool directory does not exist: {}", path.display()),
377            ));
378        }
379        Ok(path)
380    }
381
382    /// Open a database root in read-only mode.
383    ///
384    /// Unlike [`Self::open`], this never creates directories or writes
385    /// `db.meta`. It validates the root already exists and contains a valid
386    /// descriptor.
387    ///
388    /// # Errors
389    /// Returns an error if the root does not exist, has no descriptor, or
390    /// contains an unrecognized one.
391    pub fn open_read_only(root: PathBuf) -> io::Result<Self> {
392        if !root.is_dir() {
393            return Err(io::Error::new(
394                io::ErrorKind::NotFound,
395                format!("database root does not exist: {}", root.display()),
396            ));
397        }
398        let meta_path = root.join(DB_META_FILENAME);
399        if !meta_path.exists() {
400            return Err(io::Error::new(
401                io::ErrorKind::NotFound,
402                format!("missing database descriptor: {}", meta_path.display()),
403            ));
404        }
405        let seed = read_db_meta(&meta_path)?.seed;
406        Self::reject_flat_store(&root)?;
407        Ok(Self { root, seed })
408    }
409
410    /// The database root this layout was opened from.
411    #[must_use]
412    pub fn root(&self) -> &Path {
413        &self.root
414    }
415
416    /// Path of the root-level shared WAL, `<root>/wal.bin`.
417    ///
418    /// Every root is driven through this one root-level segment, and no pool
419    /// directory holds a WAL of its own.
420    #[must_use]
421    pub fn shared_wal_path(&self) -> PathBuf {
422        self.root.join("wal.bin")
423    }
424
425    fn reject_flat_store(root: &Path) -> io::Result<()> {
426        for entry in fs::read_dir(root)? {
427            let path = entry?.path();
428            if path
429                .extension()
430                .is_some_and(|extension| extension == "pack")
431            {
432                return Err(io::Error::new(
433                    io::ErrorKind::InvalidData,
434                    format!(
435                        "unpooled flat mtxdb store at {}; migrate it or use a fresh database root before enabling named pools",
436                        root.display()
437                    ),
438                ));
439            }
440        }
441        Ok(())
442    }
443
444    /// Install a complete database descriptor from a synced temporary file.
445    ///
446    /// Accepts a valid descriptor installed by a concurrent opener. File creation,
447    /// write, sync, installation, and validation errors propagate; temporary-file
448    /// cleanup and parent-directory sync are best-effort.
449    fn write_descriptor(path: &Path) -> io::Result<()> {
450        let (temporary, file) = loop {
451            let suffix = DESCRIPTOR_TEMP_COUNTER.fetch_add(1, Ordering::Relaxed);
452            let name = format!(
453                ".{}.create.{}.{}",
454                path.file_name()
455                    .and_then(|name| name.to_str())
456                    .unwrap_or(DB_META_FILENAME),
457                std::process::id(),
458                suffix,
459            );
460            let temporary = path.with_file_name(name);
461            match OpenOptions::new()
462                .write(true)
463                .create_new(true)
464                .open(&temporary)
465            {
466                Ok(file) => break (temporary, file),
467                Err(error) if error.kind() == io::ErrorKind::AlreadyExists => {}
468                Err(error) => return Err(error),
469            }
470        };
471        let _temporary_guard = TempFileGuard::new(temporary.clone());
472        // Rebind so the handle is declared after the guard: on the `?` error
473        // paths `file` then closes first, and on Windows the guard's
474        // `remove_file` is not blocked by the still-open handle.
475        let mut file = file;
476        file.write_all(&db_meta_bytes(random_seed()))?;
477        file.sync_all()?;
478        drop(file);
479
480        // hard_link installs the fully-written file atomically without
481        // replacing a descriptor another opener may have installed first.
482        Self::install_descriptor_temp(&temporary, path, |source, destination| {
483            fs::hard_link(source, destination)
484        })
485    }
486
487    /// Install the synced first-create descriptor at `temporary` into `path`.
488    ///
489    /// Uses `hard_link`; there is no portable no-replace rename fallback, so
490    /// `Unsupported` and `PermissionDenied` are returned. An existing destination or a concurrently removed
491    /// temporary file is accepted only if the destination validates. Other
492    /// installation and validation errors propagate. Parent sync and cleanup
493    /// are best-effort.
494    fn install_descriptor_temp(
495        temporary: &Path,
496        path: &Path,
497        hard_link: impl FnOnce(&Path, &Path) -> io::Result<()>,
498    ) -> io::Result<()> {
499        match hard_link(temporary, path) {
500            Ok(()) => {
501                Self::sync_descriptor_parent(path)?;
502                Self::sweep_descriptor_temps(path, "create");
503                Ok(())
504            }
505            Err(error) if error.kind() == io::ErrorKind::AlreadyExists => {
506                Self::validate_existing_descriptor(path)?;
507                Self::sync_descriptor_parent(path)?;
508                Self::sweep_descriptor_temps(path, "create");
509                Ok(())
510            }
511            // A concurrent opener can install the canonical file and sweep
512            // this temp after we created it. In that case the canonical file
513            // is the result to validate, not a reason to fail initialization.
514            Err(error) if error.kind() == io::ErrorKind::NotFound => {
515                Self::validate_existing_descriptor(path)
516            }
517            Err(error)
518                if error.kind() == io::ErrorKind::Unsupported
519                    || error.kind() == io::ErrorKind::PermissionDenied =>
520            {
521                if error.kind() == io::ErrorKind::PermissionDenied && path.exists() {
522                    Self::validate_existing_descriptor(path)?;
523                    Self::sync_descriptor_parent(path)?;
524                    Self::sweep_descriptor_temps(path, "create");
525                    Ok(())
526                } else {
527                    Err(error)
528                }
529            }
530            Err(error) => Err(error),
531        }
532    }
533
534    /// Sync the descriptor's parent directory so the installed entry survives a
535    /// power loss. A rooted pool has no `pool.meta` fallback for the seed, so a
536    /// lost `db.meta` would be recreated with a different seed; failures
537    /// therefore propagate. Only filesystems that report directory sync as
538    /// unsupported are tolerated.
539    fn sync_descriptor_parent(path: &Path) -> io::Result<()> {
540        let parent = path
541            .parent()
542            .filter(|parent| !parent.as_os_str().is_empty())
543            .unwrap_or_else(|| Path::new("."));
544        match crate::shard::sync_directory(parent) {
545            Err(error) if error.kind() == io::ErrorKind::Unsupported => Ok(()),
546            other => other,
547        }
548    }
549
550    /// Remove regular sibling files named `.<descriptor>.<kind>.*`.
551    ///
552    /// Directory-read, metadata, and removal failures are ignored.
553    fn sweep_descriptor_temps(path: &Path, kind: &str) {
554        let parent = path
555            .parent()
556            .filter(|parent| !parent.as_os_str().is_empty())
557            .unwrap_or_else(|| Path::new("."));
558        let prefix = format!(
559            ".{}.{}.",
560            path.file_name()
561                .and_then(|name| name.to_str())
562                .unwrap_or(DB_META_FILENAME),
563            kind,
564        );
565        if let Ok(entries) = fs::read_dir(parent) {
566            for entry in entries.flatten() {
567                if entry
568                    .file_name()
569                    .to_str()
570                    .is_some_and(|name| name.starts_with(&prefix))
571                    && entry.file_type().is_ok_and(|kind| kind.is_file())
572                {
573                    let _ = fs::remove_file(entry.path());
574                }
575            }
576        }
577    }
578
579    /// Validate the current descriptor format and pool list at `path`.
580    ///
581    /// Propagates read errors and returns `InvalidData` for an unrecognized
582    /// descriptor.
583    fn validate_existing_descriptor(path: &Path) -> io::Result<()> {
584        read_db_meta(path).map(|_| ())
585    }
586}
