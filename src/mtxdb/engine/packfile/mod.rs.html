19/// Packfile format version byte following `MAGIC` in the header (see
20/// [`write_header_with_creation_seq`]/[`read_header`]).
21///
22/// Current v1 packfile format. This is a hard-cutover format: unrecognized
23/// version bytes are rejected and callers must rebuild or migrate the store.
24///
25/// There is no persisted numeric id, runtime slot, or creation sequence in the
26/// header: identity is the address, ordering is the caller's concern (by
27/// timestamp/offset or an explicit conflict policy), and the runtime
28/// slot/index is entirely internal to the storage implementation.
29pub const VERSION: u8 = 0x01;

45/// A pack's globally unique, immutable identity: 16 cryptographically random
46/// bytes assigned at creation.
47///
48/// Unlike the old pool-local monotonic counter, this address is unique across
49/// pools, databases, and hosts, so an extracted or imported pack keeps the same
50/// identity wherever it lands. It is written once at pack creation and never
51/// changes while the pack grows.
52#[derive(Clone, Copy, PartialEq, Eq, Hash, PartialOrd, Ord)]
53pub struct PackId(pub [u8; PACK_ID_LEN]);

55impl PackId {
56    /// Generate a fresh nonzero random address from the operating system's CSPRNG.
57    ///
58    /// Fills the full [`PACK_ID_LEN`] bytes from the operating system's CSPRNG
59    /// (`/dev/urandom` on unix, `BCryptGenRandom` on Windows). The identity is
60    /// externally visible and portable across pools and hosts, so
61    /// it must carry real entropy rather than a hash of a weaker seed: this
62    /// yields the full 128-bit collision bound the identity is documented to
63    /// have (birthday bound ≈ 2^64 packs).
64    ///
65    /// # Panics
66    /// Panics if the OS entropy source is unavailable. This cannot be handled
67    /// meaningfully at the call site: every caller (pack creation, extract,
68    /// import) needs a globally unique identity, and silently falling back to
69    /// a weaker source would reintroduce the collision risk this method exists
70    /// to avoid. OS entropy is available on every supported platform.
71    #[must_use]
72    pub fn random() -> Self {
73        // The all-zero address is the reserved empty sentinel, so a draw that
74        // lands on it (probability 2^-128) is discarded rather than returned:
75        // `random` guarantees a nonzero identity by contract.
76        loop {
77            let mut bytes = [0u8; PACK_ID_LEN];
78            entropy::fill(&mut bytes).expect("OS entropy source is available");
79            let id = Self(bytes);
80            if !id.is_zero() {
81                return id;
82            }
83        }
84    }

87    #[must_use]
88    pub fn as_bytes(&self) -> &[u8; PACK_ID_LEN] {
89        &self.0

92    /// Whether this is the all-zero address, which is reserved as the
93    /// "no pack"/empty sentinel and is never a valid pack identity.
94    ///
95    /// [`PackId::random`] never produces it; the zero value only appears as a
96    /// placeholder in tombstone records and empty table slots. Trust
97    /// boundaries (header reads, hex parsing, pack creation/import) reject it
98    /// so a stray zero can never be adopted as a real pack's identity.

101        self.0.iter().all(|&byte| byte == 0)
102    }

124    /// The filename to use when creating this pack among `siblings`.
125    ///
126    /// Normally the compact `pack_<16 hex digits>.pack`. If that 16-hex prefix
127    /// collides with an existing sibling, fall back to the full 32-hex name so
128    /// the choice is a single deterministic widening at creation time. This is
129    /// only consulted when *creating* a pack: an existing file's name is fixed
130    /// on disk and never recomputed, so adding a later sibling can never rename
131    /// an already-written pack.
132    ///
133    /// A pack placed in the pool out of band (a manual file copy) bypasses this
134    /// check, so a hand-copied pack should use its full 32-hex name: a 16-hex
135    /// name cannot be widened later, and a collision then surfaces as a
136    /// filename/header disagreement at discovery.
137    #[must_use]
138    pub fn filename_for(&self, siblings: &[PackId]) -> String {
139        let hex = self.as_hex();
140        let prefix = &hex[..PACK_FILENAME_PREFIX_HEX];
141        let collides = siblings
142            .iter()
143            .any(|other| other != self && other.as_hex()[..PACK_FILENAME_PREFIX_HEX] == *prefix);
144        if collides {
145            format!("pack_{hex}.pack")
146        } else {
147            self.filename()
148        }
149    }

151    /// Parse an address from a filename stem (the part before `.pack`). Accepts
152    /// the compact `pack_<16hex>` form and the full `pack_<32hex>` form emitted
153    /// by [`PackId::filename_for`] on collision.
154    ///
155    /// Returns the parsed prefix bytes and how many hex digits were present, so
156    /// callers can match against a full address's prefix.
157    ///
158    /// # Errors
159    /// Returns `None` if the stem is not one of those forms, or if it is the full
160    /// all-zero address (reserved, never a valid identity). A *truncated*
161    /// all-zero prefix is still accepted: it can legitimately belong to a
162    /// nonzero 128-bit address whose leading bits happen to be zero.
163    #[must_use]
164    pub fn parse_filename_prefix(stem: &str) -> Option<(Vec<u8>, usize)> {
165        let digits = stem.strip_prefix("pack_")?;
166        // `filename_for` emits only the 16-hex prefix or the full 32-hex
167        // address (on collision).
168        if digits.len() != PACK_FILENAME_PREFIX_HEX && digits.len() != PACK_ID_LEN * 2 {
169            return None;
170        }
171        if !digits
172            .bytes()
173            .all(|b| b.is_ascii_digit() || (b'a'..=b'f').contains(&b))
174        {
175            return None;
176        }
177        let byte_len = digits.len() / 2;
178        let mut bytes = vec![0u8; byte_len];
179        for (index, chunk) in digits.as_bytes().as_chunks::<2>().0.iter().enumerate() {
180            let hi = hex_value(chunk[0])?;
181            let lo = hex_value(chunk[1])?;
182            bytes[index] = (hi << 4) | lo;
183        }
184        if bytes.len() == PACK_ID_LEN && bytes.iter().all(|&byte| byte == 0) {
185            return None;
186        }
187        Some((bytes, digits.len()))
188    }

190    /// Parse a full address from exactly 32 lowercase hex digits, without a
191    /// `0x` prefix.
192    ///
193    /// # Errors
194    /// Returns `None` for the wrong length, any non-lowercase-hex character,
195    /// or the all-zero address (reserved, never a valid identity).
196    #[must_use]
197    pub fn from_hex(hex: &str) -> Option<Self> {
198        if hex.len() != PACK_ID_LEN * 2 {
199            return None;
200        }
201        let mut bytes = [0u8; PACK_ID_LEN];
202        for (index, chunk) in hex.as_bytes().as_chunks::<2>().0.iter().enumerate() {
203            let hi = hex_value(chunk[0])?;
204            let lo = hex_value(chunk[1])?;
205            bytes[index] = (hi << 4) | lo;
206        }
207        let id = Self(bytes);
208        if id.is_zero() {
209            return None;
210        }
211        Some(id)
212    }

218        hex_lower(&self.0).starts_with(prefix)

234/// Lowercase hex encoding without pulling in a formatter dependency.
235///
236/// Public so sibling modules (e.g. `shard`) can render address prefixes when
237/// validating filename agreement.
238#[must_use]
239pub fn hex_lower(bytes: &[u8]) -> String {
240    const HEX: &[u8; 16] = b"0123456789abcdef";
241    let mut out = String::with_capacity(bytes.len().saturating_mul(2));
242    for &byte in bytes {
243        out.push(char::from(HEX[usize::from(byte >> 4)]));
244        out.push(char::from(HEX[usize::from(byte & 0x0f)]));
245    }
246    out
247}

301/// Per-frame flag: `data` is zstd-compressed on disk; decompress to
302/// `uncompressed_len` bytes before returning it to the caller.
303///
304/// Public because inspection/recovery tooling needs to interpret the flag
305/// bits of arbitrary frames and `shard.rs`'s `mmap`-based inline frame parser

307/// the same flag bit instead of hardcoding it.
308pub const FLAG_COMPRESSED: u8 = 0x01;

310/// Per-frame flag: this version removes its logical record key. Readers that
311/// do not understand removal reject the frame through the known-flags check
312/// rather than interpreting the marker payload as a live value.
313pub const FLAG_REMOVED: u8 = 0x08;

315/// Per-frame flag: a versioned, length-delimited metadata area follows the
316/// fixed header and precedes the node bytes (see [`FrameMetadata`]).
317///
318/// Metadata is stored **outside** the (possibly compressed) node bytes so a
319/// reader can inspect identity/role without decompressing, and so generic
320/// records with no metadata are byte-for-byte unchanged. A reader that does
321/// not understand this flag MUST reject the frame rather than misparse it;
322/// the byte layout differs from the flag's absence.
323pub const FLAG_METADATA: u8 = 0x04;

325/// Any flag bit this reader understands. A frame carrying a bit outside this
326/// mask is rejected rather than misparsed.
327const KNOWN_FLAGS: u8 = FLAG_COMPRESSED | FLAG_CRC_DISABLED | FLAG_METADATA | FLAG_REMOVED;

356/// Per-frame flag: the 4-byte checksum field holds zeros, not a real CRC32,
357/// and must not be verified by any reader. Written when the store was
358/// configured with [`ChecksumPolicy::Disabled`] — the flag, not the checksum
359/// value, is what signals "don't verify", because a real CRC32 can also
360/// legitimately be zero and a Full-mode reader must never treat a
361/// CRC-disabled frame as corrupt.
362pub const FLAG_CRC_DISABLED: u8 = 0x02;
363
364/// How much of the per-frame CRC32 this store writes and verifies.
365///
366/// Frames are always *formatted* with a 4-byte checksum field (the framing
367/// is fixed); the policy controls whether that field is a real checksum and
368/// whether readers check it:
369///
370/// * `Full` (default) — every written record gets a CRC32, every read
371///   verifies it. Catches torn writes mid-file and silent bit-rot per record.
372/// * `WriteOnly` — records keep their CRC32 (so scanning/recovery tooling
373///   can still verify them), but point-lookup reads skip the hashing pass. A
374///   corrupt-but-plausibly-framed record may be returned as data.
375/// * `Disabled` — records are written with a zero checksum and the
376///   [`FLAG_CRC_DISABLED`] flag; nothing is computed or verified anywhere.
377///   Every such frame is permanently unverifiable.
378#[derive(Clone, Copy, Debug, PartialEq, Eq)]
379pub enum ChecksumPolicy {
380    /// Every written record gets a CRC32 and every read verifies it (default).
381    Full,
382    /// Records keep their CRC32, but point-lookup reads skip hashing them.
383    WriteOnly,
384    /// Records are written with a zero checksum and [`FLAG_CRC_DISABLED`];
385    /// nothing is computed or verified anywhere.
386    Disabled,
387}
388
389impl ChecksumPolicy {
390    /// Whether frames written under this policy carry a real CRC32.
391    #[must_use]
392    pub fn computes_checksum(self) -> bool {
393        !matches!(self, Self::Disabled)
394    }
395
396    /// Whether point-lookup reads under this policy hash and compare the CRC.
397    #[must_use]
398    pub fn verifies_reads(self) -> bool {
399        matches!(self, Self::Full)
400    }
401}

429/// Open a streaming scanner for a packfile.
430///
431/// When `verify_payload` is true, payloads and CRCs are consumed and
432/// validated. When false, only frame metadata and the declared frame bounds
433/// are checked, which is appropriate for diagnostic listing.
434///
435/// # Errors
436/// Returns an I/O error if the packfile cannot be opened or its header cannot
437/// be read or validated, or its length cannot be obtained. A file that
438/// [`read_header`] identifies as a non-packfile yields an empty scanner.
439pub fn scan_packfile_iter(path: &Path, verify_payload: bool) -> io::Result<PackfileScanner> {
440    let file = File::open(path)?;
441    scan_packfile_iter_from_file(file, verify_payload)
442}

444/// Build a scanner from an already-open shard handle. Collection snapshots
445/// open each shard while holding the collection lock, then perform the
446/// potentially long scan after releasing that lock.
447///
448/// The handle must be positioned at the start of the header. Captures its
449/// current length and uses the same `verify_payload` behavior as
450/// [`scan_packfile_iter`]. A non-packfile yields an empty scanner; metadata,
451/// header-read, and header-validation errors propagate.
452pub(crate) fn scan_packfile_iter_from_file(
453    file: File,
454    verify_payload: bool,
455) -> io::Result<PackfileScanner> {
456    let file_end = file.metadata()?.len();
457    let mut reader = BufReader::new(file);
458    let done = read_header(&mut reader)?.is_none();
459    Ok(PackfileScanner {
460        reader,
461        file_end,
462        verify_payload,
463        done,
464    })
465}

467impl Iterator for PackfileScanner {
468    type Item = io::Result<ScanEntry>;
469
470    fn next(&mut self) -> Option<Self::Item> {
471        if self.done {
472            return None;
473        }
474        let offset = match self.reader.stream_position() {
475            Ok(offset) => offset,
476            Err(error) => {
477                self.done = true;
478                return Some(Err(error));
479            }
480        };
481        let result = if self.verify_payload {
482            read_record_metadata(&mut self.reader)
483        } else {
484            read_record_metadata_skip_payload_with_end(&mut self.reader, self.file_end)
485        };
486        match result {
487            Ok(Some(metadata)) => Some(Ok((metadata.collection_id, metadata.hash, offset))),
488            Ok(None) => {
489                self.done = true;
490                None
491            }
492            Err(error) if error.kind() == io::ErrorKind::UnexpectedEof => {
493                self.done = true;
494                None
495            }
496            Err(error) => {
497                self.done = true;
498                Some(Err(error))
499            }
500        }
501    }
502}

504/// A single record in the packfile.
505///
506/// Frame layout on disk:
507/// ```text
508/// [u32 len]              — byte length of everything below down to (not including) crc32, little-endian
509/// [u8 flags]             — bit 0: node bytes are zstd-compressed
510/// [u32 uncompressed_len] — plaintext length of `data`; equals the node bytes' own length when flags == 0
511/// [16-byte collection_id]      — collection this record belongs to (for shard scan recovery)
512/// [16-byte hash]         — structural hash (index-rebuild metadata only, NOT for verification)
513/// [node bytes]           — opaque node payload, zstd-compressed iff flags bit 0 is set
514/// [u32 crc32]            — CRC32 covering len + flags + uncompressed_len + collection_id + hash + node_bytes
515/// ```
516///
517/// A frame is only ever written compressed when doing so makes it
518/// smaller — otherwise `data` is stored raw with `flags == 0` — so the
519/// on-disk frame length never exceeds the plaintext one.
520///
521/// **Security invariant:** The framed hash is index-rebuild metadata only.
522/// Verification always compares against the *caller-requested* hash, never
523/// against the hash stored in the frame. The CRC covers the bytes as
524/// written (compressed, when compressed) — it's verified *before*
525/// decompression is attempted, so a corrupt frame is never fed to the
526/// decompressor.
527#[derive(Debug, Clone, PartialEq, Eq)]
528pub struct Record {
529    /// The collection this record belongs to.
530    pub collection_id: [u8; 16],
531    /// The structural hash framed alongside the record (index-rebuild metadata only).
532    pub hash: [u8; 16],
533    /// The opaque node payload.
534    pub data: Bytes,
535    /// Optional versioned per-record metadata (see [`FrameMetadata`]). `None`
536    /// for a generic frame, which is byte-for-byte identical to a record
537    /// with no metadata area.
538    pub metadata: Option<FrameMetadata>,
539}
540
541/// Optional, versioned metadata carried in a frame when [`FLAG_METADATA`] is
542/// set. Removal state is carried independently by [`FLAG_REMOVED`]. Metadata
543/// is stored between the fixed header and the node bytes, outside any
544/// compression, so it can be inspected without decompressing.
545///
546/// The layout on disk is:
547///
548/// ```text
549/// [u8 metadata_version]  — must equal METADATA_VERSION
550/// [u32 metadata_len]     — byte length of the TLV block below (little-endian)
551/// [TLV block]            — metadata_len bytes, each entry:
552///     [u8 tag][u32 value_len][value_len bytes]
553/// ```
554///
555/// The metadata bytes and the `metadata_len` field are inside the frame's
556/// CRC-covered region. Unknown tags are preserved by the codec so a reader
557/// built against a newer writer can round-trip fields it does not interpret.
558#[derive(Debug, Clone, PartialEq, Eq, Default)]
559pub struct FrameMetadata {
560    /// Full 256-bit logical identity digest, if the template supplies one.
561    pub logical_id: Option<[u8; 32]>,
562    /// 256-bit digest of the stored bytes for this version, if supplied.
563    pub content_digest: Option<[u8; 32]>,
564    /// The hash function that produced [`Self::content_digest`]. Defaults to
565    /// SHA-256; recorded per record so a template or configuration can choose
566    /// another 256-bit algorithm without a format break.
567    pub digest_algorithm: DigestAlgorithm,
568    /// Role/schema identifier, if supplied.
569    pub role: Option<Vec<u8>>,
570    /// LSN of the journal group that last wrote this record. Absent on a
571    /// frame with no tag, which a reader treats as version `0`.
572    pub last_write_lsn: Option<u64>,
573    /// Whether this record version removes its logical key. Encoded in
574    /// [`FLAG_REMOVED`], not as a TLV, so old readers fail closed.
575    pub removed: bool,
576    /// Tags this build does not interpret, preserved verbatim so a
577    /// read/rewrite cycle does not discard a newer writer's fields.
578    pub unknown: Vec<(u8, Vec<u8>)>,
579}

581impl FrameMetadata {
582    /// Whether this metadata carries no fields at all. A frame with empty
583    /// metadata is encoded without [`FLAG_METADATA`], so it stays
584    /// byte-identical to a generic record.
585    #[must_use]
586    pub fn is_empty(&self) -> bool {
587        self.logical_id.is_none()
588            && self.content_digest.is_none()
589            && self.role.is_none()
590            && self.last_write_lsn.is_none()
591            && self.unknown.is_empty()
592    }
593
594    /// Encode the `[metadata_version][metadata_len][TLV...]` block.
595    ///
596    /// # Errors
597    /// Returns `InvalidInput` if the TLV block exceeds `u32::MAX` bytes.
598    pub fn encode(&self) -> io::Result<Vec<u8>> {
599        let mut tlv = Vec::new();
600        if let Some(id) = &self.logical_id {
601            push_tlv(&mut tlv, META_TAG_LOGICAL_ID, id)?;
602        }
603        if let Some(digest) = &self.content_digest {
604            push_tlv(&mut tlv, META_TAG_CONTENT_DIGEST, digest)?;
605            // The algorithm is only meaningful alongside a digest. Emit it
606            // even for the default so the on-disk record is self-describing
607            // rather than relying on a reader-side default.
608            push_tlv(
609                &mut tlv,
610                META_TAG_DIGEST_ALGORITHM,
611                &[self.digest_algorithm.id()],
612            )?;
613        }
614        if let Some(role) = &self.role {
615            push_tlv(&mut tlv, META_TAG_ROLE, role)?;
616        }
617        if let Some(lsn) = self.last_write_lsn {
618            push_tlv(&mut tlv, META_TAG_LAST_WRITE_LSN, &lsn.to_le_bytes())?;
619        }
620        for (tag, value) in &self.unknown {
621            push_tlv(&mut tlv, *tag, value)?;
622        }
623        let len = u32::try_from(tlv.len())
624            .map_err(|_| io::Error::new(io::ErrorKind::InvalidInput, "frame metadata too large"))?;
625        let mut out = Vec::with_capacity(1usize.saturating_add(4).saturating_add(tlv.len()));
626        out.push(METADATA_VERSION);
627        out.extend_from_slice(&len.to_le_bytes());
628        out.extend_from_slice(&tlv);
629        Ok(out)
630    }
631
632    /// Decode a metadata block from `bytes`, returning the metadata and the
633    /// number of bytes consumed.
634    ///
635    /// # Errors
636    /// Returns `InvalidData` on an unsupported version, a truncated block, a
637    /// mismatched declared length, or a duplicate/known-tag length violation.
638    pub fn decode(bytes: &[u8]) -> io::Result<(Self, usize)> {
639        let version = *bytes
640            .first()
641            .ok_or_else(|| invalid_data("frame metadata: missing version byte"))?;
642        if version != METADATA_VERSION {
643            return Err(invalid_data(&format!(
644                "frame metadata: unsupported version {version}"
645            )));
646        }
647        let len_bytes: [u8; 4] = bytes
648            .get(1..5)
649            .and_then(|slice| slice.try_into().ok())
650            .ok_or_else(|| invalid_data("frame metadata: truncated length"))?;
651        let tlv_len = u32::from_le_bytes(len_bytes) as usize;
652        let tlv_start = 5usize;
653        let tlv_end = tlv_start
654            .checked_add(tlv_len)
655            .ok_or_else(|| invalid_data("frame metadata: length overflow"))?;
656        let tlv = bytes
657            .get(tlv_start..tlv_end)
658            .ok_or_else(|| invalid_data("frame metadata: truncated TLV block"))?;
659
660        let mut metadata = Self::default();
661        let mut cursor = 0usize;
662        while cursor < tlv.len() {
663            let tag = *tlv
664                .get(cursor)
665                .ok_or_else(|| invalid_data("frame metadata: truncated tag"))?;
666            let value_len_start = cursor
667                .checked_add(1)
668                .ok_or_else(|| invalid_data("frame metadata: cursor overflow"))?;
669            let value_len_end = cursor
670                .checked_add(5)
671                .ok_or_else(|| invalid_data("frame metadata: cursor overflow"))?;
672            let value_len_bytes: [u8; 4] = tlv
673                .get(value_len_start..value_len_end)
674                .and_then(|slice| slice.try_into().ok())
675                .ok_or_else(|| invalid_data("frame metadata: truncated value length"))?;
676            let value_len = u32::from_le_bytes(value_len_bytes) as usize;
677            let value_start = value_len_end;
678            let value_end = value_start
679                .checked_add(value_len)
680                .ok_or_else(|| invalid_data("frame metadata: value length overflow"))?;
681            let value = tlv
682                .get(value_start..value_end)
683                .ok_or_else(|| invalid_data("frame metadata: truncated value"))?;
684            match tag {
685                META_TAG_LOGICAL_ID => {
686                    metadata.logical_id = Some(expect_32(value, "logical_id")?);
687                }
688                META_TAG_CONTENT_DIGEST => {
689                    metadata.content_digest = Some(expect_32(value, "content_digest")?);
690                }
691                META_TAG_DIGEST_ALGORITHM => {
692                    let id = *value
693                        .first()
694                        .ok_or_else(|| invalid_data("frame metadata: empty digest_algorithm"))?;
695                    if value.len() != 1 {
696                        return Err(invalid_data(
697                            "frame metadata: digest_algorithm must be 1 byte",
698                        ));
699                    }
700                    metadata.digest_algorithm = DigestAlgorithm::from_id(id);
701                }
702                META_TAG_ROLE => metadata.role = Some(value.to_vec()),
703                META_TAG_LAST_WRITE_LSN => {
704                    metadata.last_write_lsn = Some(expect_u64(value, "last_write_lsn")?);
705                }
706                _ => metadata.unknown.push((tag, value.to_vec())),
707            }
708            cursor = value_end;
709        }
710
711        Ok((metadata, tlv_end))
712    }
713}

715fn push_tlv(out: &mut Vec<u8>, tag: u8, value: &[u8]) -> io::Result<()> {
716    let len = u32::try_from(value.len()).map_err(|_| {
717        io::Error::new(
718            io::ErrorKind::InvalidInput,
719            "frame metadata value too large",
720        )
721    })?;
722    out.push(tag);
723    out.extend_from_slice(&len.to_le_bytes());
724    out.extend_from_slice(value);
725    Ok(())
726}

743/// The frame length for a record, refusing any a reader would reject: a frame
744/// the writer emits must be one [`read_record`] accepts, so the bound is the
745/// reader's [`MAX_FRAME_LEN`], not the prefix's `u32::MAX`.
746pub(crate) fn checked_frame_len(metadata_len: u32, data_len: u32) -> io::Result<u32> {
747    FRAME_FIXED_LEN
748        .checked_add(metadata_len)
749        .and_then(|len| len.checked_add(data_len))
750        .filter(|len| *len <= MAX_FRAME_LEN)
751        .ok_or_else(|| {
752            io::Error::new(
753                io::ErrorKind::InvalidInput,
754                "record frame exceeds the maximum frame length",
755            )
756        })
757}

759/// Largest frame length a reader accepts: a frame lives wholly inside one
760/// shard, and shard offsets are bounded by `IndexEntry::MAX_OFFSET`, so a
761/// longer prefix can only be corruption. This is a sanity ceiling, not a
762/// record-size policy (that belongs to the collection template).
763pub(crate) const MAX_FRAME_LEN: u32 = {
764    // `IndexEntry::MAX_OFFSET` is `(1 << 32) - 2`; express the same
765    // invariant in the destination type instead of narrowing a u64.
766    u32::MAX - 1
767};
768
769/// Append exactly `n` bytes from `reader` to `buf`, allocating only as bytes
770/// actually arrive. A corrupt length prefix on a short stream therefore fails
771/// with `UnexpectedEof` instead of pre-allocating `n` bytes up front.
772fn read_exact_growing(reader: &mut impl Read, buf: &mut Vec<u8>, n: usize) -> io::Result<()> {
773    let start = buf.len();
774    reader.take(n as u64).read_to_end(buf)?;
775    if buf.len().saturating_sub(start) != n {
776        return Err(io::ErrorKind::UnexpectedEof.into());
777    }
778    Ok(())
779}

781/// Read a record length prefix, distinguishing clean EOF from a torn prefix.
782///
783/// A zero-byte read is the normal end of an append-only pack. Once even one
784/// byte of the four-byte prefix exists, however, the frame is incomplete and
785/// must surface as `UnexpectedEof` so recovery can truncate it before a later
786/// append would land after corrupt bytes.
787fn read_frame_len_prefix(reader: &mut impl Read) -> io::Result<Option<[u8; 4]>> {
788    let mut len_buf = [0u8; 4];
789    let mut bytes_read = 0;
790    while bytes_read < 4 {
791        let n = reader.read(&mut len_buf[bytes_read..])?;
792        if n == 0 {
793            if bytes_read == 0 {
794                return Ok(None);
795            }
796            return Err(io::Error::new(
797                io::ErrorKind::UnexpectedEof,
798                "torn length prefix",
799            ));
800        }
801        bytes_read = bytes_read.saturating_add(n);
802    }
803    Ok(Some(len_buf))
804}

822/// Scaffold `Mmap::map`, isolating the single unsafe call in this crate.
823///
824/// # Safety
825/// This is the only `unsafe` block in the codebase. The mapping it creates
826/// is sound for the following reasons:
827///
828/// 1. **Append-only file growth.** A shard file only ever grows by appends
829///    that serialize through `put()`. `read_at` remaps when a read lands
830///    beyond the current mapping and the file has grown, so the mapping is
831///    never stale relative to the data being read.
832/// 2. **No truncation or size change.** A mapped `File` is never
833///    `set_len`-shrunk or extended after the mapping is created.
834/// 3. **Stable file descriptor.** The `Shard` struct owns the only `File`
835///    handle and is kept alive by `Arc`s, so the descriptor remains valid
836///    for the mapping's lifetime.
837///
838/// # Errors
839/// Returns `io::Error` if the file cannot be memory-mapped.
840#[allow(unsafe_code)]
841pub fn map_pack(file: &File) -> io::Result<memmap2::Mmap> {
842    // SAFETY: See the `map_pack` doc comment — append-only file, remapped on
843    // growth, never truncated or shrunk, and a stable fd for the mapping's
844    // lifetime.
845    unsafe { memmap2::Mmap::map(file) }
846}

848/// zstd compression level used for record payloads. A low level: packfile
849/// frames are written on the hot append
850/// path, so this favors write throughput over squeezing out the last few
851/// percent of ratio.
852#[cfg(feature = "zstd")]
853const ZSTD_LEVEL: i32 = 3;
854
855/// Compress `data` at [`ZSTD_LEVEL`], or `None` if the compressor errors (the
856/// caller then stores the frame raw). Only built with the `zstd` feature.
857#[cfg(feature = "zstd")]
858fn zstd_maybe_compress(data: &[u8]) -> Option<Vec<u8>> {
859    zstd::bulk::compress(data, ZSTD_LEVEL).ok()
860}

862/// Decompress a frame's node bytes to exactly `expected_len` (the framed
863/// `uncompressed_len`, a disk-provided u32, so only trusted once the output matches it). Shared by the
864/// buffered [`read_record`] path and the shard zero-copy decode path.
865#[cfg(feature = "zstd")]
866pub(crate) fn zstd_decompress(node_bytes: &[u8], expected_len: usize) -> io::Result<Vec<u8>> {
867    // Stream into a growing buffer: `bulk::decompress` allocates its full
868    // `capacity` up front, and `expected_len` is a disk-provided u32, so a
869    // corrupt header could otherwise demand up to 4 GiB before any byte is
870    // decoded. `take` also stops a frame that expands past its declared size.
871    let mut decompressed = Vec::new();
872    zstd::stream::read::Decoder::new(node_bytes)
873        .and_then(|decoder| {
874            decoder
875                .take((expected_len as u64).saturating_add(1))
876                .read_to_end(&mut decompressed)
877        })
878        .map_err(|e| {
879            io::Error::new(
880                io::ErrorKind::InvalidData,
881                format!("zstd decompress failed: {e}"),
882            )
883        })?;
884    if decompressed.len() != expected_len {
885        return Err(io::Error::new(
886            io::ErrorKind::InvalidData,
887            format!(
888                "decompressed length {} != framed uncompressed_len {expected_len}",
889                decompressed.len()
890            ),
891        ));
892    }
893    Ok(decompressed)
894}

896/// Write a single record into the packfile, compressing its payload with
897/// zstd when doing so makes the frame smaller (falls back to storing it
898/// raw otherwise — e.g. already-compressed or very small payloads often
899/// don't shrink).
900///
901/// # Errors
902/// Returns `io::Error` on write failure, or if `record.data` would make
903/// even an uncompressed frame exceed [`u32::MAX`] (checked against
904/// the plaintext on-disk frame length — `FRAME_FIXED_LEN` plus
905/// `record.data.len()` — before compression, so this bound is never
906/// looser than what [`read_record`] will actually accept).
907///
908/// # Panics
909/// Panics if the record's fixed fields plus plaintext data exceed `u32::MAX`.
910pub fn write_record(writer: &mut impl Write, record: &Record) -> io::Result<u64> {
911    write_record_with_options(writer, record, true)
912}

914/// Like [`write_record`], but lets the caller skip the zstd attempt
915/// entirely via `compress: false`.
916///
917/// Some pools (e.g. HAMT nodes/roots, whose bytes are dense structural
918/// hashes rather than text) never shrink under zstd — the compressor
919/// still has to run its full match-finding pass before falling back to
920/// raw storage, so a pool that never benefits pays that cost on every
921/// single write for nothing. `compress: false` skips the attempt
922/// altogether and always stores the payload raw. This never changes the
923/// on-disk frame format: a `false` caller just always takes the "doesn't
924/// shrink" branch that `compress: true` already falls back to at read
925/// time, so raw and compressed frames continue to coexist exactly as
926/// documented on [`Record`].
927///
928/// # Errors
929/// Same as [`write_record`].
930///
931/// # Panics
932/// Same as [`write_record`].
933pub fn write_record_with_options(
934    writer: &mut impl Write,
935    record: &Record,
936    compress: bool,
937) -> io::Result<u64> {
938    let buf = encode_record_with_options(record, compress, true)?;
939    writer.write_all(&buf)?;
940    Ok(buf.len() as u64)
941}

943/// Encode one record into its complete on-disk frame.
944///
945/// Kept separate from [`write_record_with_options`] so appenders which use
946/// positioned writes can retain their file cursor and avoid an `lseek` per
947/// record. The result includes the length prefix and checksum. When
948/// `write_checksum` is false the frame is emitted with a zero checksum and the
949/// [`FLAG_CRC_DISABLED`] flag, per [`ChecksumPolicy::Disabled`].
950///
951/// # Errors
952/// Returns `io::Error` if the payload is too large to frame.
953pub(crate) fn encode_record_with_options(
954    record: &Record,
955    compress: bool,
956    write_checksum: bool,
957) -> io::Result<Vec<u8>> {
958    let uncompressed_len = u32::try_from(record.data.len()).map_err(|_| {
959        io::Error::new(
960            io::ErrorKind::InvalidInput,
961            "record payload exceeds u32::MAX",
962        )
963    })?;
964    let metadata_block = match &record.metadata {
965        Some(metadata) if !metadata.is_empty() => Some(metadata.encode()?),
966        _ => None,
967    };
968    let metadata_len = metadata_block.as_ref().map_or(Ok(0), |block| {
969        u32::try_from(block.len()).map_err(|_| {
970            io::Error::new(
971                io::ErrorKind::InvalidInput,
972                "frame metadata exceeds u32::MAX",
973            )
974        })
975    })?;
976    // The frame length prefix is a u32; return an input error rather than
977    // panicking when the payload and metadata cannot fit in it.
978    checked_frame_len(metadata_len, uncompressed_len)?;
979
980    #[cfg(feature = "zstd")]
981    let compressed = compress
982        .then(|| zstd_maybe_compress(&record.data))
983        .flatten();
984    #[cfg(not(feature = "zstd"))]
985    let compressed: Option<Vec<u8>> = {
986        let _ = compress;
987        None
988    };
989    let (base_flags, node_bytes): (u8, &[u8]) = match &compressed {
990        Some(c) if c.len() < record.data.len() => (FLAG_COMPRESSED, c.as_slice()),
991        _ => (0, &record.data),
992    };
993    let mut flags = if write_checksum {
994        base_flags
995    } else {
996        base_flags | FLAG_CRC_DISABLED
997    };
998    if metadata_block.is_some() {
999        flags |= FLAG_METADATA;
1000    }
1001    if record
1002        .metadata
1003        .as_ref()
1004        .is_some_and(|metadata| metadata.removed)
1005    {
1006        flags |= FLAG_REMOVED;
1007    }
1008
1009    let frame_len = FRAME_FIXED_LEN
1010        .checked_add(metadata_len)
1011        .expect("bounded by u32::MAX")
1012        .checked_add(u32::try_from(node_bytes.len()).expect("bounded by u32::MAX"))
1013        .expect("bounded by u32::MAX");
1014    let total_len = 4_u64.wrapping_add(u64::from(frame_len)).wrapping_add(4);
1015
1016    // Assemble the whole frame in one buffer and issue a single
1017    // `write_all`, rather than one syscall per field. `writer` is
1018    // usually a raw `File` clone (shard files are written directly,
1019    // not through a `BufWriter`), so 7 separate small writes meant 7
1020    // separate `write()` syscalls per record — real, avoidable
1021    // overhead on the hottest path in the engine, unrelated to actual
1022    // disk speed.
1023    let mut buf = Vec::with_capacity(usize::try_from(total_len).unwrap_or(0));
1024    buf.extend_from_slice(&frame_len.to_le_bytes());
1025    buf.push(flags);
1026    buf.extend_from_slice(&uncompressed_len.to_le_bytes());
1027    buf.extend_from_slice(&record.collection_id);
1028    buf.extend_from_slice(&record.hash);
1029    if let Some(block) = &metadata_block {
1030        buf.extend_from_slice(block);
1031    }
1032    buf.extend_from_slice(node_bytes);
1033
1034    // CRC covers len + flags + uncompressed_len + collection_id + hash +
1035    // metadata + node_bytes, i.e. the bytes as written to disk (metadata
1036    // included, node bytes compressed when compressed) — exactly `buf`'s
1037    // contents so far, hashed in one pass since CRC32 over one contiguous
1038    // buffer is identical to the same bytes hashed via several `update`
1039    // calls. Under a period of `write_checksum == false` the field is
1040    // written as zeros (with FLAG_CRC_DISABLED set above) and no hashing
1041    // pass happens at all.
1042    let checksum = if write_checksum {
1043        let mut crc = crc32fast::Hasher::new();
1044        crc.update(&buf);
1045        crc.finalize()
1046    } else {
1047        0
1048    };
1049    buf.extend_from_slice(&checksum.to_le_bytes());
1050
1051    debug_assert_eq!(
1052        buf.len() as u64,
1053        total_len,
1054        "encoded frame length must match its length prefix"
1055    );
1056    Ok(buf)
1057}

1059/// Read a single record from the packfile. Returns `None` on EOF.
1060///
1061/// # Errors
1062/// Returns `io::Error` on read failure or `io::ErrorKind::InvalidData`
1063/// if the record length is invalid, an unrecognized flag bit is set, the
1064/// CRC check fails, or (for a compressed frame) decompression fails or
1065/// yields a length other than the framed `uncompressed_len`.
1066///
1067/// # Panics
1068/// Panics if the payload length does not fit in `usize` (always true on
1069/// 64-bit targets where `usize >= 32 bits`).
1070pub fn read_record(reader: &mut impl Read) -> io::Result<Option<Record>> {
1071    let Some(len_buf) = read_frame_len_prefix(reader)? else {
1072        return Ok(None);
1073    };
1074
1075    let frame_len = u32::from_le_bytes(len_buf);
1076    if !(FRAME_FIXED_LEN..=MAX_FRAME_LEN).contains(&frame_len) {
1077        return Err(io::Error::new(
1078            io::ErrorKind::InvalidData,
1079            format!("invalid record length: {frame_len}"),
1080        ));
1081    }
1082
1083    let mut payload = Vec::new();
1084    read_exact_growing(reader, &mut payload, frame_len as usize)?;
1085
1086    let mut crc_buf = [0u8; 4];
1087    reader.read_exact(&mut crc_buf)?;
1088
1089    let flags = payload[0];
1090    if flags & !KNOWN_FLAGS != 0 {
1091        return Err(io::Error::new(
1092            io::ErrorKind::InvalidData,
1093            format!("unsupported record flags: {flags:#04x}"),
1094        ));
1095    }
1096
1097    // Verify CRC over the bytes as written (compressed, if compressed) —
1098    // before any decompression is attempted, so a corrupt frame is never
1099    // fed to the decompressor. Frames carrying FLAG_CRC_DISABLED hold a
1100    // zero checksum and are skipped, not compared.
1101    if flags & FLAG_CRC_DISABLED == 0 {
1102        let mut crc = crc32fast::Hasher::new();
1103        crc.update(&len_buf);
1104        crc.update(&payload);
1105        let expected = crc.finalize();
1106        let actual = u32::from_le_bytes(crc_buf);
1107        if expected != actual {
1108            return Err(io::Error::new(
1109                io::ErrorKind::InvalidData,
1110                format!("CRC mismatch: expected {expected:08x}, got {actual:08x}"),
1111            ));
1112        }
1113    }
1114
1115    let uncompressed_len = u32::from_le_bytes(payload[1..5].try_into().unwrap());
1116    let mut collection_id = [0u8; 16];
1117    collection_id.copy_from_slice(&payload[5..21]);
1118    let mut hash = [0u8; 16];
1119    hash.copy_from_slice(&payload[21..37]);
1120
1121    // Optional metadata sits between the fixed header and the node bytes.
1122    let rest = &payload[37..];
1123    let (mut metadata, node_bytes) = if flags & FLAG_METADATA != 0 {
1124        let (metadata, consumed) = FrameMetadata::decode(rest)?;
1125        let node_bytes = rest
1126            .get(consumed..)
1127            .ok_or_else(|| invalid_data("frame metadata consumes past payload"))?;
1128        (Some(metadata), node_bytes)
1129    } else {
1130        (None, rest)
1131    };
1132    if flags & FLAG_REMOVED != 0 {
1133        metadata.get_or_insert_with(FrameMetadata::default).removed = true;
1134    }
1135
1136    let data = if flags & FLAG_COMPRESSED != 0 {
1137        #[cfg(feature = "zstd")]
1138        {
1139            Bytes::from(zstd_decompress(
1140                node_bytes,
1141                usize::try_from(uncompressed_len).expect("u32 fits in usize"),
1142            )?)
1143        }
1144        #[cfg(not(feature = "zstd"))]
1145        {
1146            return Err(io::Error::new(
1147                io::ErrorKind::InvalidData,
1148                "frame is zstd-compressed but this build was compiled without the `zstd` feature",
1149            ));
1150        }
1151    } else {
1152        if usize::try_from(uncompressed_len).expect("u32 always fits in usize") != node_bytes.len()
1153        {
1154            return Err(io::Error::new(
1155                io::ErrorKind::InvalidData,
1156                format!(
1157                    "raw node length {} != framed uncompressed_len {uncompressed_len}",
1158                    node_bytes.len()
1159                ),
1160            ));
1161        }
1162        Bytes::copy_from_slice(node_bytes)
1163    };
1164
1165    Ok(Some(Record {
1166        collection_id,
1167        hash,
1168        data,
1169        metadata,
1170    }))
1171}

1173/// A frame's `collection_id`/`hash` metadata, without its node payload — what
1174/// [`scan_packfile`]/[`scan_packfile_from`]/[`scan_and_recover_packfile`]
1175/// actually need. See [`read_record_metadata`].
1176#[derive(Debug, Clone, PartialEq, Eq)]
1177pub struct RecordMetadata {
1178    /// The collection this record belongs to.
1179    pub collection_id: [u8; 16],
1180    /// The structural hash framed alongside the record.
1181    pub hash: [u8; 16],
1182    /// Optional per-record metadata. Removal alone may set this to `Some`
1183    /// without [`FLAG_METADATA`], because its state comes from [`FLAG_REMOVED`].
1184    pub metadata: Option<FrameMetadata>,
1185}
1186
1187/// Buffer size for streaming node bytes through [`read_record_metadata`]
1188/// without allocating a buffer proportional to the frame's payload size.
1189/// Large enough to keep syscall/CRC-update overhead low, small enough to
1190/// stay a stack buffer.
1191const SCAN_DISCARD_BUF_LEN: usize = 8192;
1192
1193/// Shared header fields parsed from the fixed portion of a frame.
1194///
1195/// Extracted once by [`read_frame_header`] and consumed by either
1196/// [`read_record_metadata`] (CRC-verified) or
1197/// [`read_record_metadata_skip_payload`] (seek-past).
1198struct FrameHeader {
1199    len_buf: [u8; 4],
1200    frame_len: u32,
1201    flags: u8,
1202    fixed: [u8; FRAME_FIXED_LEN as usize],
1203    collection_id: [u8; 16],
1204    hash: [u8; 16],
1205    /// Optional metadata parsed from the frame. Removal alone may set this to
1206    /// `Some` without [`FLAG_METADATA`], based on [`FLAG_REMOVED`].
1207    metadata: Option<FrameMetadata>,
1208    /// The raw on-disk metadata block (`[version][len][TLV...]`), empty when
1209    /// absent. Retained so callers can feed it into the frame CRC (the block
1210    /// is covered by the checksum) and subtract its length from `frame_len`
1211    /// to find the node-bytes region.
1212    metadata_block: Vec<u8>,
1213}
1214
1215/// Resolve the on-disk metadata block length (`5 + tlv_len`) and bound it
1216/// against the frame that must contain it, *before* the caller allocates.
1217///
1218/// `tlv_len` comes straight from disk. The metadata block (its 5-byte prefix
1219/// plus the TLV bytes) shares the frame's post-header body with the node bytes,
1220/// so it must satisfy `block_len <= frame_len - FRAME_FIXED_LEN`. This rejects
1221/// a metadata block that overruns its containing frame before allocation.
1222fn checked_metadata_block_len(tlv_len: u32, frame_len: u32) -> io::Result<u32> {
1223    let block_len = 5u32
1224        .checked_add(tlv_len)
1225        .ok_or_else(|| invalid_data("frame metadata: length overflow"))?;
1226    let remaining_frame_body = frame_len
1227        .checked_sub(FRAME_FIXED_LEN)
1228        .ok_or_else(|| invalid_data("frame metadata overruns the frame"))?;
1229    if block_len > remaining_frame_body {
1230        return Err(invalid_data("frame metadata exceeds frame length"));
1231    }
1232    Ok(block_len)
1233}

1235/// Read and validate the frame length prefix and fixed header, returning the
1236/// parsed fields. The caller is responsible for consuming the payload and CRC
1237/// (or seeking past them).
1238fn read_frame_header(reader: &mut impl Read) -> io::Result<Option<FrameHeader>> {
1239    let Some(len_buf) = read_frame_len_prefix(reader)? else {
1240        return Ok(None);
1241    };
1242
1243    let frame_len = u32::from_le_bytes(len_buf);
1244    if !(FRAME_FIXED_LEN..=MAX_FRAME_LEN).contains(&frame_len) {
1245        return Err(io::Error::new(
1246            io::ErrorKind::InvalidData,
1247            format!("invalid record length: {frame_len}"),
1248        ));
1249    }
1250
1251    let mut fixed = [0u8; FRAME_FIXED_LEN as usize];
1252    reader.read_exact(&mut fixed)?;
1253
1254    let flags = fixed[0];
1255    if flags & !KNOWN_FLAGS != 0 {
1256        return Err(io::Error::new(
1257            io::ErrorKind::InvalidData,
1258            format!("unsupported record flags: {flags:#04x}"),
1259        ));
1260    }
1261
1262    // Read the optional metadata block (version byte + u32 length + TLV).
1263    let (mut metadata, metadata_block) = if flags & FLAG_METADATA != 0 {
1264        let mut prefix = [0u8; 5];
1265        reader.read_exact(&mut prefix)?;
1266        let version = prefix[0];
1267        if version != METADATA_VERSION {
1268            return Err(invalid_data(&format!(
1269                "frame metadata: unsupported version {version}"
1270            )));
1271        }
1272        let tlv_len = u32::from_le_bytes(prefix[1..5].try_into().expect("fixed slice"));
1273        // Resolve and bound the metadata block *before* allocating: `tlv_len`
1274        // is disk-controlled, so the allocation below must never be sized by it
1275        // unchecked.
1276        let block_len = checked_metadata_block_len(tlv_len, frame_len)?;
1277        let mut block = vec![0u8; usize::try_from(block_len).expect("u32 fits in usize")];
1278        block[..5].copy_from_slice(&prefix);
1279        reader.read_exact(&mut block[5..])?;
1280        let (metadata, consumed) = FrameMetadata::decode(&block)?;
1281        if consumed != block.len() {
1282            return Err(invalid_data(
1283                "frame metadata: trailing bytes after TLV block",
1284            ));
1285        }
1286        (Some(metadata), block)
1287    } else {
1288        (None, Vec::new())
1289    };
1290    if flags & FLAG_REMOVED != 0 {
1291        metadata.get_or_insert_with(FrameMetadata::default).removed = true;
1292    }
1293    let metadata_len = u32::try_from(metadata_block.len()).expect("bounded by frame length");
1294
1295    let node_region_len = frame_len
1296        .checked_sub(FRAME_FIXED_LEN)
1297        .and_then(|len| len.checked_sub(metadata_len))
1298        .ok_or_else(|| invalid_data("frame metadata overruns the frame"))?;
1299    let uncompressed_len = u32::from_le_bytes(fixed[1..5].try_into().unwrap());
1300    if flags & FLAG_COMPRESSED != 0 {
1301    } else if uncompressed_len != node_region_len {
1302        return Err(io::Error::new(
1303            io::ErrorKind::InvalidData,
1304            format!(
1305                "raw node length {node_region_len} != framed uncompressed_len {uncompressed_len}"
1306            ),
1307        ));
1308    }
1309
1310    let mut collection_id = [0u8; 16];
1311    collection_id.copy_from_slice(&fixed[5..21]);
1312    let mut hash = [0u8; 16];
1313    hash.copy_from_slice(&fixed[21..37]);
1314
1315    Ok(Some(FrameHeader {
1316        len_buf,
1317        frame_len,
1318        flags,
1319        fixed,
1320        collection_id,
1321        hash,
1322        metadata,
1323        metadata_block,
1324    }))
1325}

1327/// Read one frame's metadata (`collection_id`, `hash`) without allocating,
1328/// decompressing, or otherwise materializing its node payload — the scan
1329/// path's counterpart to [`read_record`], which fully decodes a frame for
1330/// callers that actually need its data.
1331///
1332/// Node bytes are streamed through a small fixed buffer and fed into the
1333/// running CRC as they're read, then discarded — this still detects
1334/// corruption in the payload region (the CRC covers the same bytes
1335/// [`read_record`] verifies), but never buffers or decompresses them.
1336/// Deliberately a streaming read, rather than seeking past the payload,
1337/// because seeking would skip CRC verification of the region.
1338///
1339/// Node bytes are streamed through a small fixed buffer and fed into the
1340/// running CRC as they're read, then discarded.
1341///
1342/// # Errors
1343/// Returns an I/O error for invalid length, unsupported flags, truncated
1344/// payload/CRC data, or a CRC mismatch.
1345///
1346/// # Panics
1347/// Never in practice: `read_frame_header` validates that `frame_len` is at
1348/// least `FRAME_FIXED_LEN` before the checked subtraction below.
1349pub fn read_record_metadata(reader: &mut impl Read) -> io::Result<Option<RecordMetadata>> {
1350    read_record_metadata_scratch(reader, &mut [0u8; SCAN_DISCARD_BUF_LEN])
1351}
1352
1353/// [`read_record_metadata`] with the payload-discard buffer supplied by the
1354/// caller. A scan over many frames passes one buffer for all of them: zeroing a
1355/// fresh 8 KiB for every frame costs more than parsing a small one.
1356fn read_record_metadata_scratch(
1357    reader: &mut impl Read,
1358    discard: &mut [u8; SCAN_DISCARD_BUF_LEN],
1359) -> io::Result<Option<RecordMetadata>> {
1360    let Some(header) = read_frame_header(reader)? else {
1361        return Ok(None);
1362    };
1363
1364    let mut crc = if header.flags & FLAG_CRC_DISABLED == 0 {
1365        let mut crc = crc32fast::Hasher::new();
1366        crc.update(&header.len_buf);
1367        crc.update(&header.fixed);
1368        crc.update(&header.metadata_block);
1369        Some(crc)
1370    } else {
1371        None
1372    };
1373
1374    // frame_len >= FRAME_FIXED_LEN is guaranteed by the range check above;
1375    // subtract the metadata block already consumed by read_frame_header.
1376    let mut remaining: usize = header
1377        .frame_len
1378        .checked_sub(FRAME_FIXED_LEN)
1379        .and_then(|len| len.checked_sub(u32::try_from(header.metadata_block.len()).ok()?))
1380        .expect("frame_len >= FRAME_FIXED_LEN + metadata_len, checked in read_frame_header")
1381        as usize;
1382
1383    // For a CRC-disabled frame the checksum field is zero and unused; the
1384    // payload still must be consumed to advance the stream, just without the
1385    // hashing pass.
1386    while remaining > 0 {
1387        let chunk_len = remaining.min(discard.len());
1388        reader.read_exact(&mut discard[..chunk_len])?;
1389        if let Some(crc) = &mut crc {
1390            crc.update(&discard[..chunk_len]);
1391        }
1392        remaining = remaining
1393            .checked_sub(chunk_len)
1394            .expect("chunk_len <= remaining by construction (min above)");
1395    }
1396
1397    let mut crc_buf = [0u8; 4];
1398    reader.read_exact(&mut crc_buf)?;
1399    if let Some(crc) = crc {
1400        let expected = crc.finalize();
1401        let actual = u32::from_le_bytes(crc_buf);
1402        if expected != actual {
1403            return Err(io::Error::new(
1404                io::ErrorKind::InvalidData,
1405                format!("CRC mismatch: expected {expected:08x}, got {actual:08x}"),
1406            ));
1407        }
1408    }
1409
1410    Ok(Some(RecordMetadata {
1411        collection_id: header.collection_id,
1412        hash: header.hash,
1413        metadata: header.metadata,
1414    }))
1415}

1417/// Read one frame's metadata and seek over its payload and CRC without
1418/// reading them. This is intended for diagnostic scans where payload
1419/// integrity is not being checked. The frame length, flags, and raw payload
1420/// length invariants are still validated.
1421///
1422/// # Errors
1423/// Returns an I/O error if the frame prefix or metadata cannot be read, if
1424/// the frame metadata is invalid, or if seeking past the payload fails.
1425///
1426/// # Panics
1427/// Never in practice: the fixed-size metadata buffer makes all internal
1428/// conversions and the frame-length subtraction provably valid after the
1429/// preceding bounds checks.
1430pub fn read_record_metadata_skip_payload(
1431    reader: &mut (impl Read + Seek),
1432) -> io::Result<Option<RecordMetadata>> {
1433    let payload_start = reader.stream_position()?;
1434    let file_end = reader.seek(io::SeekFrom::End(0))?;
1435    reader.seek(io::SeekFrom::Start(payload_start))?;
1436    read_record_metadata_skip_payload_with_end(reader, file_end)
1437}
1438
1439fn read_record_metadata_skip_payload_with_end(
1440    reader: &mut (impl Read + Seek),
1441    file_end: u64,
1442) -> io::Result<Option<RecordMetadata>> {
1443    let Some(header) = read_frame_header(reader)? else {
1444        return Ok(None);
1445    };
1446
1447    // The frame length excludes the four-byte CRC, while the fixed portion
1448    // includes the metadata and flags but not the node bytes or the
1449    // already-consumed metadata block.
1450    let skip = u64::from(
1451        header
1452            .frame_len
1453            .checked_sub(FRAME_FIXED_LEN)
1454            .and_then(|len| len.checked_sub(u32::try_from(header.metadata_block.len()).ok()?))
1455            .expect("frame_len >= FRAME_FIXED_LEN + metadata_len, checked in read_frame_header"),
1456    )
1457    .checked_add(4)
1458    .expect("frame payload length fits u64");
1459    let payload_start = reader.stream_position()?;
1460    let frame_end = payload_start
1461        .checked_add(skip)
1462        .ok_or_else(|| io::Error::new(io::ErrorKind::InvalidData, "record end overflows u64"))?;
1463    if frame_end > file_end {
1464        return Err(io::Error::new(
1465            io::ErrorKind::UnexpectedEof,
1466            "record payload or CRC is truncated",
1467        ));
1468    }
1469    reader.seek(io::SeekFrom::Start(frame_end))?;
1470    Ok(Some(RecordMetadata {
1471        collection_id: header.collection_id,
1472        hash: header.hash,
1473        metadata: header.metadata,
1474    }))
1475}

1477/// Write a new pack file's [`HEADER_LEN`]-byte reserved header: magic,
1478/// version, header length, `pack_id`, creation time, pool-local monotonic
1479/// creation sequence, and a CRC over all of the above — zero-padded to fill
1480/// `HEADER_LEN`. Written once, at creation, and never mutated again (see
1481/// [`VERSION`]'s doc for why).
1482///
1483/// # Errors
1484/// Returns `io::Error` if the pack ID is reserved, if `creation_seq` is zero
1485/// (`InvalidInput`; zero is not a valid sequence), if the system clock is
1486/// before the Unix epoch, or if the header cannot be written.
1487pub fn write_header_with_creation_seq(
1488    writer: &mut impl Write,
1489    pack_id: &PackId,
1490    creation_seq: u32,
1491) -> io::Result<()> {
1492    if pack_id.is_zero() {
1493        return Err(io::Error::new(
1494            io::ErrorKind::InvalidInput,
1495            "refusing to write the reserved all-zero pack id to a header",
1496        ));
1497    }
1498    let created_at = std::time::SystemTime::now()
1499        .duration_since(std::time::UNIX_EPOCH)
1500        .map_err(io::Error::other)?
1501        .as_secs();
1502
1503    write_header_with_created_at(writer, pack_id, created_at, creation_seq)
1504}

1506/// Write a pack header using an explicit creation time and sequence.
1507///
1508/// This is kept separate from [`write_header_with_creation_seq`] so tests and
1509/// recovery tools can construct headers without reaching into the encoded
1510/// layout. Callers creating new packs should use the system-time wrapper.
1511///
1512/// # Errors
1513/// Returns `io::Error` if the pack ID is reserved or the header cannot be
1514/// written.
1515///
1516/// # Panics
1517/// Never in practice: the internal conversion of `HEADER_LEN` to `u32` is a
1518/// compile-time constant within range.
1519pub fn write_header_with_created_at(
1520    writer: &mut impl Write,
1521    pack_id: &PackId,
1522    created_at: u64,
1523    creation_seq: u32,
1524) -> io::Result<()> {
1525    if creation_seq == 0 {
1526        return Err(io::Error::new(
1527            io::ErrorKind::InvalidInput,
1528            "refusing to write a pack header with creation_seq 0",
1529        ));
1530    }
1531    if pack_id.is_zero() {
1532        return Err(io::Error::new(
1533            io::ErrorKind::InvalidInput,
1534            "refusing to write the reserved all-zero pack id to a header",
1535        ));
1536    }
1537
1538    let mut buf = [0u8; HEADER_LEN];
1539    buf[0..4].copy_from_slice(&MAGIC);
1540    buf[4] = VERSION;
1541    let header_len = u32::try_from(HEADER_LEN).expect("HEADER_LEN fits in u32");
1542    buf[5..9].copy_from_slice(&header_len.to_le_bytes());
1543    let mut cursor: usize = 9;
1544    buf[cursor..cursor.saturating_add(PACK_ID_LEN)].copy_from_slice(pack_id.as_bytes());
1545    cursor = cursor.saturating_add(PACK_ID_LEN);
1546    buf[cursor..cursor.saturating_add(8)].copy_from_slice(&created_at.to_le_bytes());
1547    cursor = cursor.saturating_add(8);
1548    buf[cursor..cursor.saturating_add(4)].copy_from_slice(&creation_seq.to_le_bytes());
1549
1550    let crc = crc32fast::hash(&buf[..CRC_COVERED_LEN]);
1551    buf[CRC_COVERED_LEN..CRC_COVERED_LEN.wrapping_add(4)].copy_from_slice(&crc.to_le_bytes());
1552    // The rest of buf is already zero-initialized padding out to HEADER_LEN.
1553
1554    writer.write_all(&buf)
1555}

1557/// Read just the version byte from a packfile header (bytes 4..5, right
1558/// after the 4-byte [`MAGIC`]). Returns `Ok(None)` if the file is too
1559/// short or doesn't start with [`MAGIC`].
1560///
1561/// # Errors
1562///
1563/// Returns `io::Error` for a non-EOF read failure.
1564pub fn read_version(reader: &mut impl Read) -> io::Result<Option<u8>> {
1565    let mut magic = [0u8; 4];
1566    match reader.read_exact(&mut magic) {
1567        Ok(()) => {}
1568        Err(e) if e.kind() == io::ErrorKind::UnexpectedEof => return Ok(None),
1569        Err(e) => return Err(e),
1570    }
1571    if magic != MAGIC {
1572        return Ok(None);
1573    }
1574    let mut version = [0u8; 1];
1575    match reader.read_exact(&mut version) {
1576        Ok(()) => {}
1577        // `read_version` is deliberately only a lightweight probe.  A file
1578        // ending immediately after MAGIC is still too short to identify a
1579        // version, just like one ending part-way through MAGIC.
1580        Err(e) if e.kind() == io::ErrorKind::UnexpectedEof => return Ok(None),
1581        Err(e) => return Err(e),
1582    }
1583    Ok(Some(version[0]))
1584}

1586/// Read and validate a shard file's reserved header.
1587///
1588/// Returns `Ok(None)` for a file shorter than the four-byte magic or whose
1589/// first four bytes don't match [`MAGIC`]. Everything else that's wrong is a
1590/// real `Err`, not a quiet `None`, specifically so a caller (or one of the `scan_*` functions,
1591/// which all propagate this via `?`) can't mistake "this store is a
1592/// format I don't understand" for "this store has no data":
1593///
1594/// - Right magic, wrong version (most notably an older store): a
1595///   distinct, identifiable error naming the version found, not folded
1596///   into "not a packfile".
1597/// - Right magic and version, but truncated before a full `HEADER_LEN`
1598///   header, or a CRC mismatch: corruption, reported as such.
1599///
1600/// # Errors
1601/// Returns `io::Error` (`Unsupported`) if the version doesn't match
1602/// [`VERSION`], or (`InvalidData`) if the header is truncated after the magic,
1603/// declares the wrong length, has a CRC mismatch, or contains an all-zero pack
1604/// ID. Other I/O errors propagate.
1605///
1606/// # Panics
1607/// Never in practice: every internal `try_into`/`from_le_bytes` slices a
1608/// fixed, in-range region of the just-read header buffer.
1609pub fn read_header(reader: &mut impl Read) -> io::Result<Option<ShardHeader>> {
1610    // Phase 1: just enough to identify the format (magic + version),
1611    // before committing to reading a full HEADER_LEN buffer — a small
1612    // pre-cutover v1 file (bare 5-byte header) can easily be shorter
1613    // than HEADER_LEN in total, and version mismatch has to be detected
1614    // regardless of how much data follows it.
1615    let mut magic = [0u8; 4];
1616    match reader.read_exact(&mut magic) {
1617        Ok(()) => {}
1618        Err(e) if e.kind() == io::ErrorKind::UnexpectedEof => return Ok(None),
1619        Err(e) => return Err(e),
1620    }
1621    if magic != MAGIC {
1622        return Ok(None);
1623    }
1624    let mut version = [0u8; 1];
1625    match reader.read_exact(&mut version) {
1626        Ok(()) => {}
1627        Err(e) if e.kind() == io::ErrorKind::UnexpectedEof => {
1628            return Err(io::Error::new(
1629                io::ErrorKind::InvalidData,
1630                "missing version byte",
1631            ));
1632        }
1633        Err(e) => return Err(e),
1634    }
1635    let version = version[0];
1636    if version != VERSION {
1637        return Err(io::Error::new(
1638            io::ErrorKind::Unsupported,
1639            format!(
1640                "packfile format version {version:#04x} is not supported by this build \
1641                 (only {VERSION:#04x}) — this looks like a pre-cutover store; \
1642                 reset or migrate it rather than opening it with this version"
1643            ),
1644        ));
1645    }
1646
1647    // Phase 2: magic and version confirmed — read the rest of the
1648    // reserved header. Running out of bytes here means a genuinely
1649    // truncated header, which is corruption, not "not a packfile".
1650    let mut buf = [0u8; HEADER_LEN];
1651    buf[..4].copy_from_slice(&magic);
1652    buf[4] = version;
1653    reader.read_exact(&mut buf[5..]).map_err(|e| {
1654        if e.kind() == io::ErrorKind::UnexpectedEof {
1655            io::Error::new(
1656                io::ErrorKind::InvalidData,
1657                format!("truncated shard header: shorter than {HEADER_LEN} bytes"),
1658            )
1659        } else {
1660            e
1661        }
1662    })?;
1663
1664    let header_len = u32::from_le_bytes(buf[5..9].try_into().unwrap());
1665    if header_len as usize != HEADER_LEN {
1666        // A version byte we recognize but a header length we don't is
1667        // corruption too, for the same reason as above — VERSION and
1668        // HEADER_LEN have only ever shipped together.
1669        return Err(io::Error::new(
1670            io::ErrorKind::InvalidData,
1671            format!("shard header declares length {header_len}, expected {HEADER_LEN}"),
1672        ));
1673    }
1674
1675    let expected_crc = u32::from_le_bytes(
1676        buf[CRC_COVERED_LEN..CRC_COVERED_LEN.wrapping_add(4)]
1677            .try_into()
1678            .unwrap(),
1679    );
1680    let actual_crc = crc32fast::hash(&buf[..CRC_COVERED_LEN]);
1681    if actual_crc != expected_crc {
1682        return Err(io::Error::new(
1683            io::ErrorKind::InvalidData,
1684            format!("shard header CRC mismatch: expected {expected_crc:08x}, got {actual_crc:08x}"),
1685        ));
1686    }
1687
1688    let mut cursor: usize = 9;
1689    let mut pack_id = [0u8; PACK_ID_LEN];
1690    pack_id.copy_from_slice(&buf[cursor..cursor.saturating_add(PACK_ID_LEN)]);
1691    let pack_id = PackId(pack_id);
1692    // The all-zero address is the empty sentinel and never a real identity.
1693    // Checked after the CRC so the rejection is based on authenticated bytes.
1694    if pack_id.is_zero() {
1695        return Err(io::Error::new(
1696            io::ErrorKind::InvalidData,
1697            "shard header carries the reserved all-zero pack id",
1698        ));
1699    }
1700    cursor = cursor.saturating_add(PACK_ID_LEN);
1701    let created_at = u64::from_le_bytes(buf[cursor..cursor.saturating_add(8)].try_into().unwrap());
1702
1703    cursor = cursor.saturating_add(8);
1704    let creation_seq =
1705        u32::from_le_bytes(buf[cursor..cursor.saturating_add(4)].try_into().unwrap());
1706    if creation_seq == 0 {
1707        return Err(io::Error::new(
1708            io::ErrorKind::InvalidData,
1709            "shard header carries creation_seq 0",
1710        ));
1711    }
1712
1713    Ok(Some(ShardHeader {
1714        pack_id,
1715        created_at,
1716        creation_seq,
1717    }))
1718}

1721/// creation sequence and goes through [`open_packfile_with_creation_seq`].
1722///
1723/// With `writable` set, opens for appending; otherwise read-only. Either way
1724/// the header is validated and its `pack_id` checked against `pack_id`, the
1725/// caller's expected full identity. This function does not validate the

1728/// # Errors

1730/// Returns `Unsupported` for an unsupported header version, or `InvalidData`

1740        #[cfg(not(unix))]
1741        options.write(true);
1742        options.open(path)?
1743    } else {

1748}

1750/// Check that `file`'s header parses and names `pack_id`.

1807/// Scan an existing packfile and return `(collection_id, hash, offset)` entries.
1808/// Used during startup to rebuild per-collection indexes from shard files.
1809///
1810/// Does **not** truncate the file — safe to call on an active shard while
1811/// concurrent appends are landing. A torn tail (a crashed write that leaves
1812/// an incomplete final frame) stops the scan gracefully and returns the
1813/// entries found before it. Anything else `read_record` rejects — an
1814/// invalid length, a CRC mismatch — is mid-file corruption, not a torn
1815/// tail, and is propagated as an error rather than silently discarded.
1816///
1817/// # Errors
1818/// Returns `io::Error` on I/O failure during open or header read, or on
1819/// any read-loop error other than `UnexpectedEof`.
1820pub fn scan_packfile(path: &Path) -> io::Result<Vec<ScanEntry>> {
1821    let file = File::open(path)?;
1822    let mut reader = BufReader::new(file);
1823    let mut entries = Vec::new();
1824
1825    if read_header(&mut reader)?.is_none() {
1826        return Ok(entries);
1827    }
1828
1829    loop {
1830        let offset = reader.stream_position()?;
1831        match read_record_metadata(&mut reader) {
1832            Ok(Some(meta)) => {
1833                entries.push((meta.collection_id, meta.hash, offset));
1834            }
1835            Ok(None) => break,
1836            Err(e) if e.kind() == io::ErrorKind::UnexpectedEof => break,
1837            Err(e) => {
1838                // Propagate real corruption (CRC mismatch, invalid length)
1839                // rather than silently discarding it — the caller decides
1840                // whether to recover or fail.
1841                return Err(e);
1842            }
1843        }
1844    }
1845
1846    Ok(entries)
1847}

1876/// Scan a packfile starting from `start_offset`, returning only records
1877/// whose byte position is >= `start_offset`. Used for incremental repack:
1878/// the caller tracks how far through each shard file it has already scanned
1879/// and only needs the newly-appended bytes.
1880///
1881/// The offset must point to a valid record boundary (the start of a `[u32
1882/// len]` frame) — typically the byte position immediately after the last
1883/// record returned by a previous scan. Passing the file's end position
1884/// returns an empty vec without error.
1885///
1886/// Unlike [`scan_and_recover_packfile`], this never truncates — a
1887/// concurrent writer may be actively appending, so mutating the file is
1888/// unsafe. Torn tails are treated as EOF (same as [`scan_packfile`]).
1889///
1890/// # Errors
1891/// Returns `io::Error` on I/O failure other than `UnexpectedEof`.
1892pub fn scan_packfile_from(path: &Path, start_offset: u64) -> io::Result<Vec<ScanEntry>> {
1893    let file = File::open(path)?;
1894    let mut reader = BufReader::new(file);
1895
1896    // Seek past the reserved header (HEADER_LEN bytes) to the first
1897    // record. If start_offset is already past the header, seek directly
1898    // there.
1899    if start_offset == 0 {
1900        if read_header(&mut reader)?.is_none() {
1901            return Ok(Vec::new());
1902        }
1903    } else {
1904        use std::io::Seek;
1905        let file_len = reader.get_ref().metadata()?.len();
1906        if start_offset > file_len {
1907            return Err(io::Error::new(
1908                io::ErrorKind::InvalidInput,
1909                format!("start_offset {start_offset} exceeds file length {file_len} (likely a stale cursor from a previous epoch)"),
1910            ));
1911        }
1912        reader.seek(io::SeekFrom::Start(start_offset))?;
1913    }
1914
1915    let mut entries = Vec::new();
1916    loop {
1917        let offset = reader.stream_position()?;
1918        match read_record_metadata(&mut reader) {
1919            Ok(Some(meta)) => {
1920                entries.push((meta.collection_id, meta.hash, offset));
1921            }
1922            Ok(None) => break,
1923            Err(e) if e.kind() == io::ErrorKind::UnexpectedEof => break,
1924            Err(e) => return Err(e),
1925        }
1926    }
1927
1928    Ok(entries)
1929}

2093/// Extract all frames belonging to `target_collection` from `source_path` and write
2094/// them verbatim into a new valid packfile at `dest_path` with header `dest_pack_id`.
2095///
2096/// Every frame's framing and CRC is verified. Frame bytes (including compression
2097/// and optional frame metadata) are copied without decompressing or re-serializing,
2098/// ensuring byte-level fidelity. If the source pack ends in a torn tail, extraction
2099/// stops cleanly at the last valid frame and records `torn_tail: true` in the returned
2100/// stats.
2101fn validate_extracted_frame(frame_buf: &[u8], frame_len: u32) -> io::Result<()> {
2102    let total_frame_len = frame_buf.len();
2103    let flags = *frame_buf
2104        .get(4)
2105        .ok_or_else(|| invalid_data("truncated frame"))?;
2106    if flags & !KNOWN_FLAGS != 0 {
2107        return Err(io::Error::new(
2108            io::ErrorKind::InvalidData,
2109            format!("unsupported record flags: {flags:#04x}"),
2110        ));
2111    }
2112
2113    if flags & FLAG_METADATA != 0 {
2114        let meta_offset = (4_usize).saturating_add(FRAME_FIXED_LEN as usize);
2115        let header_end = meta_offset.saturating_add(5).saturating_add(4);
2116        if total_frame_len < header_end {
2117            return Err(invalid_data("frame too short for metadata header"));
2118        }
2119        let version = *frame_buf
2120            .get(meta_offset)
2121            .ok_or_else(|| invalid_data("missing metadata version"))?;
2122        if version != METADATA_VERSION {
2123            return Err(invalid_data(&format!(
2124                "frame metadata: unsupported version {version}"
2125            )));
2126        }
2127        let tlv_bytes: [u8; 4] = frame_buf
2128            .get(meta_offset.saturating_add(1)..meta_offset.saturating_add(5))
2129            .and_then(|s| s.try_into().ok())
2130            .ok_or_else(|| invalid_data("truncated metadata length prefix"))?;
2131        let tlv_len = u32::from_le_bytes(tlv_bytes);
2132        let _ = checked_metadata_block_len(tlv_len, frame_len)?;
2133    }
2134
2135    if flags & FLAG_CRC_DISABLED == 0 {
2136        let crc_start = total_frame_len.saturating_sub(4);
2137        let expected_bytes: [u8; 4] = frame_buf
2138            .get(crc_start..total_frame_len)
2139            .and_then(|s| s.try_into().ok())
2140            .ok_or_else(|| invalid_data("truncated CRC"))?;
2141        let expected_crc = u32::from_le_bytes(expected_bytes);
2142        let actual_crc = crc32fast::hash(&frame_buf[..crc_start]);
2143        if actual_crc != expected_crc {
2144            return Err(io::Error::new(
2145                io::ErrorKind::InvalidData,
2146                format!(
2147                    "frame CRC mismatch: expected {expected_crc:#010x}, got {actual_crc:#010x}"
2148                ),
2149            ));
2150        }
2151    }
2152
2153    Ok(())
2154}

2170/// lengths, framing, or CRCs return `InvalidData`.

2172    source_path: &Path,
2173    dest_path: &Path,
2174    target_collection: &[u8; 16],
2175    dest_pack_id: &PackId,

2178    let mut reader = BufReader::with_capacity(RECOVERY_BUFFER_BYTES, src_file);
2179

2181        return Err(io::Error::new(
2182            io::ErrorKind::InvalidData,
2183            format!(
2184                "not an mtxdb packfile (missing magic header): {}",
2185                source_path.display()
2186            ),
2187        ));
2188    };
2189
2190    let dst_file = File::create(dest_path)?;
2191    let mut writer = BufWriter::new(dst_file);
2192    // The extract is a subset copy of the source, so it inherits the source's
2193    // ordering key (creation time and pool-local sequence) rather than
2194    // claiming a new position. Allocating `source_seq + 1` would collide with
2195    // the pack the pool created right after the source and let the random
2196    // address decide ties. Equal keys against the source itself are harmless:
2197    // the copied records are identical.

2199        &mut writer,
2200        dest_pack_id,
2201        source_header.created_at,
2202        source_header.creation_seq,

2206        frames_extracted: 0,
2207        frame_bytes_written: 0,
2208        frames_scanned: 0,
2209        torn_tail: false,
2210    };
2211

2213

2216            Ok(Some(buf)) => buf,
2217            Ok(None) => break,
2218            Err(e) if e.kind() == io::ErrorKind::UnexpectedEof => {
2219                stats.torn_tail = true;

2221            }
2222            Err(e) => return Err(e),
2223        };

2226        if !(FRAME_FIXED_LEN..=MAX_FRAME_LEN).contains(&frame_len) {

2228                io::ErrorKind::InvalidData,

2230            ));
2231        }

2238            &mut reader,
2239            &mut frame_buf,
2240            total_frame_len.saturating_sub(4),

2244                stats.torn_tail = true;
2245                break;
2246            }
2247            Err(e) => return Err(e),

2253
2254        let mut frame_collection = [0u8; 16];

2270}

2312/// safe to append to: the same validation of every frame and CRC, and the same
2313/// truncation of a torn tail, but no per-record result is built.
2314///
2315/// # Errors
2316/// As [`scan_and_recover_packfile`].
2317pub fn recover_packfile(path: &Path) -> io::Result<PackRecovery> {
2318    recover_packfile_with(path, |_| {})
2319}

2321/// Buffer for a recovery scan: a few large sequential reads, not one per 8 KiB.
2322const RECOVERY_BUFFER_BYTES: usize = 1 << 20;
2323
2324/// A reader that counts the bytes it hands out, so the scan knows each frame's
2325/// offset without asking the file for its position (an `lseek` system call per
2326/// frame on a buffered reader).
2327struct CountingReader<R> {
2328    inner: R,
2329    position: u64,
2330}
2331
2332impl<R: Read> Read for CountingReader<R> {
2333    fn read(&mut self, buf: &mut [u8]) -> io::Result<usize> {
2334        let read = self.inner.read(buf)?;
2335        self.position = self
2336            .position
2337            .saturating_add(u64::try_from(read).unwrap_or(u64::MAX));
2338        Ok(read)
2339    }
2340}

2342/// The shared recovery walk: validates every frame, calls `on_record` for each
2343/// valid one, and truncates a torn tail.
2344fn recover_packfile_with(
2345    path: &Path,
2346    mut on_record: impl FnMut(ScanEntryWithRemoval),
2347) -> io::Result<PackRecovery> {
2348    let file = File::open(path)?;
2349    let mut reader = CountingReader {
2350        inner: BufReader::with_capacity(RECOVERY_BUFFER_BYTES, file),
2351        position: 0,
2352    };
2353    let mut recovery = PackRecovery {
2354        records: 0,
2355        valid_len: 0,
2356        truncated: false,
2357    };
2358
2359    if read_header(&mut reader)?.is_none() {
2360        return Ok(recovery);
2361    }
2362
2363    recovery.valid_len = reader.position;
2364    let mut scratch = [0u8; SCAN_DISCARD_BUF_LEN];
2365    loop {
2366        let offset = reader.position;
2367        match read_record_metadata_scratch(&mut reader, &mut scratch) {
2368            Ok(Some(meta)) => {
2369                on_record(ScanEntryWithRemoval {
2370                    collection_id: meta.collection_id,
2371                    hash: meta.hash,
2372                    offset,
2373                    removed: meta.metadata.as_ref().is_some_and(|frame| frame.removed),
2374                });
2375                recovery.records = recovery.records.saturating_add(1);
2376                recovery.valid_len = reader.position;
2377            }
2378            Ok(None) => break,
2379            Err(e) if e.kind() == io::ErrorKind::UnexpectedEof => {
2380                // Torn tail: a write crashed mid-frame. Truncate to the
2381                // last valid record boundary so future appends don't land
2382                // after a corrupt frame.
2383                recovery.truncated = true;
2384                break;
2385            }
2386            Err(e) => {
2387                // Real corruption (CRC mismatch, invalid length) — propagate
2388                // without truncating. Later valid records may exist past the
2389                // corrupt frame; truncating would lose them.
2390                return Err(e);
2391            }
2392        }
2393    }
2394
2395    // Only truncate if we actually hit a torn tail — not on clean EOF
2396    // or mid-file corruption (which was propagated as an error above).
2397    if recovery.truncated {
2398        drop(reader);
2399        let file = OpenOptions::new().write(true).open(path)?;
2400        file.set_len(recovery.valid_len)?;
2401        file.sync_all()?;
2402    }
2403
2404    Ok(recovery)
2405}
