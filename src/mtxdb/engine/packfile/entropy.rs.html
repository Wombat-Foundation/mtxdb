1//! Operating-system entropy without a third-party CSPRNG crate.
2//!
3//! `getrandom` pulls `wasip2` -> `wit-bindgen`, whose manifest requires Cargo's
4//! unstabilized `edition2024` feature. Cargo parses every target's dependency
5//! graph when it resolves, so merely depending on `getrandom` makes the 1.81
6//! MSRV build fail before any mtxdb code compiles. The crate only ever needs
7//! 16 bytes to seed a [`PackId`](super::PackId), so the supported platforms are
8//! handled directly: `/dev/urandom` on unix and `BCryptGenRandom` on Windows.
9
10use std::io;
11
12/// Fill every byte of `dest` with operating-system cryptographic entropy.
13///
14/// # Errors
15/// Returns the underlying [`io::Error`] if `/dev/urandom` cannot be read.
16#[cfg(unix)]
17pub(super) fn fill(dest: &mut [u8]) -> io::Result<()> {
18    use std::io::Read as _;
19    // `/dev/urandom` is the portable unix entropy device: it does not block
20    // once the kernel pool is initialized, needs no unsafe FFI, and exists on
21    // every unix target mtxdb builds for (Linux, macOS, the BSDs).
22    let mut source = std::fs::File::open("/dev/urandom")?;
23    source.read_exact(dest)
24}

26/// Fill every byte of `dest` with operating-system cryptographic entropy.
27///
28/// # Errors
29/// Returns an [`io::Error`] carrying the `BCryptGenRandom` NTSTATUS if the
30/// system RNG reports failure.
31///
32/// # Panics
33/// Panics if the destination is longer than a `u32`; callers only ever pass a
34/// 16-byte [`PackId`](super::PackId), so this is unreachable in practice.
35#[cfg(windows)]
36#[allow(unsafe_code)]
37pub(super) fn fill(dest: &mut [u8]) -> io::Result<()> {
38    use std::ffi::c_void;
39
40    // `bcrypt.dll` has shipped since Windows Vista. Passing a null algorithm
41    // handle together with this flag selects the system-preferred RNG, so no
42    // algorithm provider has to be opened or freed.
43    const BCRYPT_USE_SYSTEM_PREFERRED_RNG: u32 = 0x0000_0002;
44
45    #[link(name = "bcrypt")]
46    extern "system" {
47        fn BCryptGenRandom(
48            algorithm: *mut c_void,
49            buffer: *mut u8,
50            buffer_len: u32,
51            flags: u32,
52        ) -> i32;
53    }
54
55    // A `PackId` is always 16 bytes, but reject an oversized request rather
56    // than silently truncating the ABI length field.
57    let buffer_len = u32::try_from(dest.len()).expect("entropy request fits in u32");
58
59    // SAFETY: `dest` is a live, exclusively borrowed slice, so `as_mut_ptr`
60    // and `buffer_len` describe a writable buffer of exactly that length. A
61    // null algorithm handle plus BCRYPT_USE_SYSTEM_PREFERRED_RNG is the
62    // documented "use the system RNG" call, and no pointer is retained after
63    // return. This is the only unsafe call on the Windows entropy path.
64    let status = unsafe {
65        BCryptGenRandom(
66            std::ptr::null_mut(),
67            dest.as_mut_ptr(),
68            buffer_len,
69            BCRYPT_USE_SYSTEM_PREFERRED_RNG,
70        )
71    };
72
73    // `BCryptGenRandom` returns an NTSTATUS: >= 0 (STATUS_SUCCESS) on success,
74    // negative on failure. Surface the status rather than `GetLastError`,
75    // which the NTSTATUS return does not set.
76    if status < 0 {
77        Err(io::Error::other(format!(
78            "BCryptGenRandom failed with NTSTATUS {status:#010x}"
79        )))
80    } else {
81        Ok(())
82    }
83}
84
85#[cfg(not(any(unix, windows)))]
86compile_error!("mtxdb's OS entropy source supports unix and Windows only");
