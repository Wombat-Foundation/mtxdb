13/// How a record's stored payload is selected.
14#[derive(Debug, Clone, PartialEq, Eq)]
15pub enum PayloadPolicy {
16    /// Retain the complete received source record.
17    Source,
18    /// Retain only an explicit, derived projection of the source record.
19    Projection {
20        /// RFC 6901 pointers to fields that should be included in the projection.
21        include: Vec<String>,
22    },
23}

25/// Which bytes a frame's 256-bit logical identity (`logical_id`) is derived
26/// from.
27///
28/// Protocol-neutral by construction: this selects *what* gets hashed, while
29/// [`DigestAlgorithm`] selects *how*. Protocol-specific rules (Matrix reference
30/// hashes, redaction, and the like) are inputs to [`FrameIdPolicy::Canonical`]
31/// rather than a parallel family of enums.
32#[derive(Debug, Clone, PartialEq, Eq)]
33pub enum FrameIdPolicy {
34    /// Hash the value at an RFC 6901 pointer into the retained source record.
35    Pointer {
36        /// RFC 6901 pointer to a stable, source-level identity.
37        pointer: String,
38    },
39    /// Hash the stored payload bytes.
40    Payload,
41    /// Hash an immutable header/descriptor — the same strategy a pack uses for
42    /// its own address. `fields` names the descriptor fields to include, and is
43    /// honored by whoever assembles [`FrameIdInput::descriptor`].
44    HeaderDescriptor {
45        /// Descriptor field names to include in the hashed bytes.
46        fields: Vec<String>,
47    },
48    /// Hash a canonicalized form of the source record.
49    ///
50    /// `include` and `exclude_prefixes` are honored by the caller's
51    /// canonicalizer (e.g. a protocol extension applying a redaction policy);
52    /// the resulting canonical bytes are supplied in [`FrameIdInput::canonical`].
53    Canonical {
54        /// RFC 6901 pointers to fields retained in the canonical form.
55        include: Vec<String>,
56        /// Field-name prefixes stripped before canonicalization.
57        exclude_prefixes: Vec<String>,
58    },
59    /// The full logical identity is supplied by the caller and only
60    /// cross-checked, never derived.
61    ExternalCanonicalIdToCrosscheck,
62    /// The record ID is a caller-supplied key digest (e.g. BLAKE3(key)[..16]).
63    /// Validation and mapping to the underlying key are owned by the application layer.
64    Key,
65}

67/// The byte sources a [`FrameIdPolicy`] may draw from.
68pub struct FrameIdInput<'a> {
69    /// The stored payload bytes.
70    pub payload: &'a [u8],
71    /// The immutable header/descriptor bytes, if the frame has one.
72    pub descriptor: &'a [u8],
73    /// The canonicalized source bytes, if the caller produced them.
74    pub canonical: Option<&'a [u8]>,
75    /// Resolves an RFC 6901 pointer against the retained source record.
76    pub resolve: &'a dyn Fn(&str) -> Option<Vec<u8>>,
77}

79/// Derive a frame's 256-bit logical identity under `policy`.
80///
81/// Returns `None` when the policy's input is unavailable — a missing pointer or
82/// canonical form, or no descriptor — or when the identity is
83/// [`FrameIdPolicy::ExternalCanonicalIdToCrosscheck`] or [`FrameIdPolicy::Key`]
84/// and must be supplied by the caller instead.
85#[must_use]
86pub fn frame_digest(
87    policy: &FrameIdPolicy,
88    algorithm: DigestAlgorithm,
89    input: &FrameIdInput<'_>,
90) -> Option<Digest32> {
91    let bytes: Cow<'_, [u8]> = match policy {
92        FrameIdPolicy::Payload => Cow::Borrowed(input.payload),
93        FrameIdPolicy::HeaderDescriptor { .. } => Cow::Borrowed(input.descriptor),
94        FrameIdPolicy::Pointer { pointer } => Cow::Owned((input.resolve)(pointer)?),
95        FrameIdPolicy::Canonical { .. } => Cow::Borrowed(input.canonical?),
96        FrameIdPolicy::ExternalCanonicalIdToCrosscheck | FrameIdPolicy::Key => return None,
97    };
98    Some(algorithm.digest(&bytes))
99}

101/// Derive the 128-bit record routing key from a full identity digest.
102///
103/// The full digest remains the authoritative identity when it is available;
104/// this fixed-width prefix is only the core index key. Keeping the truncation
105/// here prevents adapters from independently choosing different slices.
106///
107/// # Panics
108///
109/// Never in practice: [`Digest32`] is exactly 32 bytes, so its first 16 bytes
110/// always fit [`NodeId`].
111#[must_use]
112pub fn record_logical_id(digest: &Digest32) -> NodeId {
113    digest[..16]
114        .try_into()
115        .expect("the first 16 bytes of Digest32 always fit NodeId")
116}

118/// 256-bit little-endian wrapping addition: `(a + b) mod 2^256`.
119///
120/// Unrolled 4-limb `u64` carry chain lowering to `ADD` / `ADC` in optimized codegen.
121#[inline]
122#[must_use]
123pub const fn wrapping_add_le(a: [u8; 32], b: [u8; 32]) -> [u8; 32] {
124    let a0 = u64::from_le_bytes([a[0], a[1], a[2], a[3], a[4], a[5], a[6], a[7]]);
125    let a1 = u64::from_le_bytes([a[8], a[9], a[10], a[11], a[12], a[13], a[14], a[15]]);
126    let a2 = u64::from_le_bytes([a[16], a[17], a[18], a[19], a[20], a[21], a[22], a[23]]);
127    let a3 = u64::from_le_bytes([a[24], a[25], a[26], a[27], a[28], a[29], a[30], a[31]]);
128
129    let b0 = u64::from_le_bytes([b[0], b[1], b[2], b[3], b[4], b[5], b[6], b[7]]);
130    let b1 = u64::from_le_bytes([b[8], b[9], b[10], b[11], b[12], b[13], b[14], b[15]]);
131    let b2 = u64::from_le_bytes([b[16], b[17], b[18], b[19], b[20], b[21], b[22], b[23]]);
132    let b3 = u64::from_le_bytes([b[24], b[25], b[26], b[27], b[28], b[29], b[30], b[31]]);
133
134    let (r0, carry0) = a0.overflowing_add(b0);
135    let (r1, carry1) = add_with_carry(a1, b1, carry0);
136    let (r2, carry2) = add_with_carry(a2, b2, carry1);
137    let (r3, _carry3) = add_with_carry(a3, b3, carry2);
138
139    let o0 = r0.to_le_bytes();
140    let o1 = r1.to_le_bytes();
141    let o2 = r2.to_le_bytes();
142    let o3 = r3.to_le_bytes();
143
144    [
145        o0[0], o0[1], o0[2], o0[3], o0[4], o0[5], o0[6], o0[7], o1[0], o1[1], o1[2], o1[3], o1[4],
146        o1[5], o1[6], o1[7], o2[0], o2[1], o2[2], o2[3], o2[4], o2[5], o2[6], o2[7], o3[0], o3[1],
147        o3[2], o3[3], o3[4], o3[5], o3[6], o3[7],
148    ]
149}

151#[inline]
152const fn add_with_carry(left: u64, right: u64, carry: bool) -> (u64, bool) {
153    let (sum, left_right_carry) = left.overflowing_add(right);
154    let (sum, carry_carry) = sum.overflowing_add(carry as u64);
155    (sum, left_right_carry || carry_carry)
156}

212/// Map a 4-byte member namespace to its fixed 32-byte bias constant.
213///
214/// Returns `Some(bias)` for recognized member namespaces (`EVNT`, `PREV`, `AUTH`, `STAT`, `INTL`).
215/// Returns `None` for unrecognized namespaces (such as physical pool tags like `b"EDGE"`).
216#[must_use]
217pub const fn namespace_bias(namespace: [u8; 4]) -> Option<[u8; 32]> {
218    if namespace[0] == b'E' && namespace[1] == b'V' && namespace[2] == b'N' && namespace[3] == b'T'
219    {
220        Some(NAMESPACE_BIAS_EVNT)
221    } else if namespace[0] == b'P'
222        && namespace[1] == b'R'
223        && namespace[2] == b'E'
224        && namespace[3] == b'V'
225    {
226        Some(NAMESPACE_BIAS_PREV)
227    } else if namespace[0] == b'A'
228        && namespace[1] == b'U'
229        && namespace[2] == b'T'
230        && namespace[3] == b'H'
231    {
232        Some(NAMESPACE_BIAS_AUTH)
233    } else if namespace[0] == b'S'
234        && namespace[1] == b'T'
235        && namespace[2] == b'A'
236        && namespace[3] == b'T'
237    {
238        Some(NAMESPACE_BIAS_STAT)
239    } else if namespace[0] == b'I'
240        && namespace[1] == b'N'
241        && namespace[2] == b'T'
242        && namespace[3] == b'L'
243    {
244        Some(NAMESPACE_BIAS_INTL)
245    } else {
246        None
247    }
248}

250/// Derive the 256-bit full logical identity for a collection group.
251///
252/// Formula: `BLAKE3-256("mtxdb/group/v1" || group_canonical_id)`
253#[must_use]
254pub fn derive_group_full_id(group_canonical_id: &[u8]) -> [u8; 32] {
255    let mut hasher = DigestAlgorithm::Blake3.hasher();
256    hasher.update(GROUP_DOMAIN_PREFIX);
257    hasher.update(group_canonical_id);
258    hasher.finalize()
259}

261/// Derive the 256-bit full logical identity for a group member directly from an
262/// already-computed 256-bit group logical ID.
263///
264/// Formula: `wrapping_add_le(group_full_id, NAMESPACE_BIAS[namespace])`
265///
266/// This avoids re-hashing `group_canonical_id` when deriving multiple member IDs
267/// for the same collection group (e.g. `EVNT`, `PREV`, `AUTH`, `STAT`).
268#[must_use]
269pub fn derive_member_full_id_from_group(
270    member_namespace: [u8; 4],
271    group_full_id: [u8; 32],
272) -> Option<[u8; 32]> {
273    let bias = namespace_bias(member_namespace)?;
274    Some(wrapping_add_le(group_full_id, bias))
275}

277/// Derive the 128-bit truncated collection ID for a group member directly from an
278/// already-computed 256-bit group logical ID.
279///
280/// Formula: `wrapping_add_le(group_full_id, NAMESPACE_BIAS[namespace])[..16]`
281#[must_use]
282pub fn derive_member_collection_id_from_group(
283    member_namespace: [u8; 4],
284    group_full_id: [u8; 32],
285) -> Option<[u8; 16]> {
286    let full_id = derive_member_full_id_from_group(member_namespace, group_full_id)?;
287    let mut collection_id = [0u8; 16];
288    collection_id.copy_from_slice(&full_id[..16]);
289    Some(collection_id)
290}

292/// Derive the 256-bit full logical identity for a group member.
293///
294/// Formula:
295/// ```text
296/// group_full_id = BLAKE3-256("mtxdb/group/v1" || group_canonical_id)

304    group_canonical_id: &[u8],
305) -> Option<[u8; 32]> {
306    let group_full_id = derive_group_full_id(group_canonical_id);
307    derive_member_full_id_from_group(member_namespace, group_full_id)
308}

327/// Try to derive the 256-bit full logical identity for a collection from its optional
328/// member namespace and group canonical ID.
329///
330/// When `member_namespace` is `Some(ns)`, delegates to [`derive_group_member_full_id`].
331/// When `None`, computes [`derive_group_full_id`].
332///
333/// Returns `None` if an unrecognized member namespace was provided.
334#[must_use]
335pub fn try_derive_collection_full_id(
336    member_namespace: Option<[u8; 4]>,
337    collection_canonical_id: &[u8],
338) -> Option<[u8; 32]> {
339    if let Some(ns) = member_namespace {
340        derive_group_member_full_id(ns, collection_canonical_id)
341    } else {
342        Some(derive_group_full_id(collection_canonical_id))
343    }
344}

362/// Derive a collection's 128-bit logical ID from its optional member namespace
363/// and group canonical ID (e.g. `b"!room:server"`).
364///
365/// Convenience helper intended for static, compile-time registered namespaces
366/// (such as `Some(MEMBER_NAMESPACE_EVNT)` or `None`).

378    try_derive_collection_id(member_namespace, collection_canonical_id)

382/// The reserved node id under which a collection's metadata (genesis) record
383/// is stored.
384///
385/// Reserved by construction: it is the only frame ever written under this id,
386/// so it needs no derived-hash ceremony — a value we control is exactly as
387/// collision-safe as a hash of a reserved string.
388pub const COLLECTION_METADATA_RECORD_ID: [u8; 16] = *b"mtxdb:metadata\0\0";

390/// A collection's first-class, immutable definition, written once as its
391/// metadata (genesis) record under [`COLLECTION_METADATA_RECORD_ID`].
392#[derive(Debug, Clone, PartialEq, Eq)]
393pub struct CollectionMetadata {
394    // --- Identity pre-image (hashed into `collection_logical_id`) ---
395    /// Member namespace (4 bytes, e.g. `b"EVNT"`, `b"PREV"`, `b"AUTH"`, `b"STAT"`).
396    /// Used with wrapping bias addition to derive `collection_logical_id`.
397    ///
398    /// Named `member_namespace` per contract, distinct from physical storage pool.
399    pub member_namespace: Option<[u8; 4]>,
400    /// The collection's **canonical** id — the caller-defined external key
401    /// (e.g. `!room:server`), canonicalized.
402    ///
403    /// # Group Canonical ID
404    /// Under the group derivation scheme, this represents the **group canonical ID**
405    /// from which all member collections (events, state, prev-edges, auth-edges)
406    /// deterministically derive their logical collection IDs via deterministic bias addition.
407    pub collection_canonical_id: Vec<u8>,
408    // --- Descriptive (NOT hashed; safe to change/extend) ---
409    /// How frames in this collection calculate their `record_logical_id`.
410    pub record_id_rule: RecordIdentityRule,
411    /// Source payload retention rule.
412    pub payload: PayloadPolicy,
413    /// Optional, opaque application payload owned by the template/protocol
414    /// layer — for example a self-describing Matrix room blob
415    /// (`{"ext":"matrix.room","fmt":1,"room_version":"10",...}`). Core never
416    /// interprets it; it is written once with the genesis metadata and handed
417    /// back to the application on open, so a reader can learn protocol
418    /// configuration (room version, creator) from the header without seeking
419    /// to and parsing the establishment record.
420    pub extension: Option<Vec<u8>>,
421    /// Logical role of this collection (e.g. "`event_dag`", "`state_hamt`", "`system_auxiliary`").
422    pub role: Option<String>,
423    /// Optional wire schema / format version string (e.g. "matrix.event.v1").
424    pub schema: Option<String>,
425}

462/// Append one `[tag:1][len:4 LE][value]` record.
463fn push_tlv(out: &mut Vec<u8>, tag: u8, value: &[u8]) {
464    out.push(tag);
465    out.extend_from_slice(
466        &u32::try_from(value.len())
467            .expect("collection-metadata field fits u32")
468            .to_le_bytes(),
469    );
470    out.extend_from_slice(value);
471}
472
473/// Encode a list of strings as repeated `[len:4 LE][utf8]`.
474fn encode_string_list(list: &[String]) -> Vec<u8> {
475    let mut out = Vec::new();
476    for item in list {
477        out.extend_from_slice(
478            &u32::try_from(item.len())
479                .expect("string fits u32")
480                .to_le_bytes(),
481        );
482        out.extend_from_slice(item.as_bytes());
483    }
484    out
485}

487/// Cursor over a TLV block.
488struct TlvReader<'a> {
489    bytes: &'a [u8],
490    cursor: usize,
491}
492
493impl<'a> TlvReader<'a> {
494    fn new(bytes: &'a [u8]) -> Self {
495        Self { bytes, cursor: 0 }
496    }
497
498    fn next(&mut self) -> Option<(u8, &'a [u8])> {
499        let tag = *self.bytes.get(self.cursor)?;
500        let len_start = self.cursor.checked_add(1)?;
501        let len_end = self.cursor.checked_add(5)?;
502        let len: [u8; 4] = self.bytes.get(len_start..len_end)?.try_into().ok()?;
503        let value_end = len_end.checked_add(u32::from_le_bytes(len) as usize)?;
504        let value = self.bytes.get(len_end..value_end)?;
505        self.cursor = value_end;
506        Some((tag, value))
507    }
508
509    fn is_exhausted(&self) -> bool {
510        self.cursor == self.bytes.len()
511    }
512}
513
514/// Decode repeated `[len:4 LE][utf8]` into a string list.
515fn decode_string_list(mut bytes: &[u8]) -> Option<Vec<String>> {
516    let mut out = Vec::new();
517    while !bytes.is_empty() {
518        let len: [u8; 4] = bytes.get(..4)?.try_into().ok()?;
519        let end = 4usize.checked_add(u32::from_le_bytes(len) as usize)?;
520        out.push(std::str::from_utf8(bytes.get(4..end)?).ok()?.to_owned());
521        bytes = bytes.get(end..)?;
522    }
523    Some(out)
524}
525
526fn encode_frame_id_policy(policy: &FrameIdPolicy) -> Vec<u8> {
527    let mut out = Vec::new();
528    match policy {
529        FrameIdPolicy::Pointer { pointer } => {
530            push_tlv(&mut out, POLICY_TAG_POINTER, pointer.as_bytes());
531        }
532        FrameIdPolicy::Payload => push_tlv(&mut out, POLICY_TAG_PAYLOAD, &[]),
533        FrameIdPolicy::HeaderDescriptor { fields } => {
534            push_tlv(
535                &mut out,
536                POLICY_TAG_HEADER_DESCRIPTOR,
537                &encode_string_list(fields),
538            );
539        }
540        FrameIdPolicy::Canonical {
541            include,
542            exclude_prefixes,
543        } => {
544            let mut inner = Vec::new();
545            push_tlv(
546                &mut inner,
547                CANONICAL_TAG_INCLUDE,
548                &encode_string_list(include),
549            );
550            push_tlv(
551                &mut inner,
552                CANONICAL_TAG_EXCLUDE_PREFIXES,
553                &encode_string_list(exclude_prefixes),
554            );
555            push_tlv(&mut out, POLICY_TAG_CANONICAL, &inner);
556        }
557        FrameIdPolicy::ExternalCanonicalIdToCrosscheck => {
558            push_tlv(&mut out, POLICY_TAG_EXTERNAL, &[]);
559        }
560        FrameIdPolicy::Key => {
561            push_tlv(&mut out, POLICY_TAG_KEY, &[]);
562        }
563    }
564    out
565}
566
567fn decode_frame_id_policy(bytes: &[u8]) -> Option<FrameIdPolicy> {
568    let mut reader = TlvReader::new(bytes);
569    let (tag, value) = reader.next()?;
570    if !reader.is_exhausted() {
571        return None;
572    }
573    Some(match tag {
574        POLICY_TAG_POINTER => FrameIdPolicy::Pointer {
575            pointer: std::str::from_utf8(value).ok()?.to_owned(),
576        },
577        POLICY_TAG_PAYLOAD => FrameIdPolicy::Payload,
578        POLICY_TAG_HEADER_DESCRIPTOR => FrameIdPolicy::HeaderDescriptor {
579            fields: decode_string_list(value)?,
580        },
581        POLICY_TAG_CANONICAL => {
582            let mut include = Vec::new();
583            let mut exclude_prefixes = Vec::new();
584            let mut reader = TlvReader::new(value);
585            while let Some((inner_tag, inner_value)) = reader.next() {
586                match inner_tag {
587                    CANONICAL_TAG_INCLUDE => include = decode_string_list(inner_value)?,
588                    CANONICAL_TAG_EXCLUDE_PREFIXES => {
589                        exclude_prefixes = decode_string_list(inner_value)?;
590                    }
591                    _ => {}
592                }
593            }
594            if !reader.is_exhausted() {
595                return None;
596            }
597            FrameIdPolicy::Canonical {
598                include,
599                exclude_prefixes,
600            }
601        }
602        POLICY_TAG_EXTERNAL => FrameIdPolicy::ExternalCanonicalIdToCrosscheck,
603        POLICY_TAG_KEY => FrameIdPolicy::Key,
604        _ => return None,
605    })
606}
607
608fn encode_record_id_rule(rule: &RecordIdentityRule) -> Vec<u8> {
609    let mut out = Vec::new();
610    push_tlv(
611        &mut out,
612        IDENTITY_TAG_DIGEST_ALGORITHM,
613        &[rule.digest_algorithm.id()],
614    );
615    push_tlv(
616        &mut out,
617        IDENTITY_TAG_POLICY,
618        &encode_frame_id_policy(&rule.policy),
619    );
620    out
621}
622
623fn decode_record_id_rule(bytes: &[u8]) -> Option<RecordIdentityRule> {
624    let mut rule = RecordIdentityRule {
625        policy: FrameIdPolicy::ExternalCanonicalIdToCrosscheck,
626        digest_algorithm: DigestAlgorithm::Sha256,
627    };
628    let mut has_digest_algorithm = false;
629    let mut has_policy = false;
630    let mut reader = TlvReader::new(bytes);
631    while let Some((tag, value)) = reader.next() {
632        match tag {
633            IDENTITY_TAG_DIGEST_ALGORITHM => {
634                rule.digest_algorithm = DigestAlgorithm::from_id(*value.first()?);
635                has_digest_algorithm = true;
636            }
637            IDENTITY_TAG_POLICY => {
638                rule.policy = decode_frame_id_policy(value)?;
639                has_policy = true;
640            }
641            _ => {}
642        }
643    }
644    (reader.is_exhausted() && has_digest_algorithm && has_policy).then_some(rule)
645}
646
647fn encode_payload(payload: &PayloadPolicy) -> Vec<u8> {
648    match payload {
649        PayloadPolicy::Source => vec![0x00],
650        PayloadPolicy::Projection { include } => {
651            let mut out = vec![0x01];
652            out.extend_from_slice(&encode_string_list(include));
653            out
654        }
655    }
656}
657
658fn decode_payload(bytes: &[u8]) -> Option<PayloadPolicy> {
659    match bytes.split_first()? {
660        (0x00, []) => Some(PayloadPolicy::Source),
661        (0x01, rest) => Some(PayloadPolicy::Projection {
662            include: decode_string_list(rest)?,
663        }),
664        _ => None,
665    }
666}

668impl CollectionMetadata {
669    /// Verify whether this metadata's canonical ID and member namespace reproduce the
670    /// given collection ID under the canonical derivation contract.
671    ///
672    /// Recomputes the collection ID using [`try_derive_collection_id`]. If the
673    /// metadata carries an unrecognized member namespace, this returns `false`.
674    #[must_use]
675    pub fn verify_collection_id(&self, collection_id: &[u8; 16]) -> bool {
676        match try_derive_collection_id(self.member_namespace, &self.collection_canonical_id) {
677            Some(derived) => derived == *collection_id,
678            None => false,
679        }
680    }
681
682    /// Check whether an existing stored metadata record conflicts in logical identity
683    /// with `self` (i.e. a 128-bit collection truncation collision where the stored
684    /// group canonical ID or member namespace disagrees).
685    #[must_use]
686    pub fn identity_collides_with(&self, other: &Self) -> bool {
687        self.collection_canonical_id != other.collection_canonical_id
688            || self.member_namespace != other.member_namespace
689    }
690
691    /// Validate collision between `self` (stored metadata) and `requested` metadata for a given
692    /// `collection_id`.
693    ///
694    /// Compares:
695    /// 1. `self.collection_canonical_id` (stored group canonical ID) vs `requested.collection_canonical_id`
696    /// 2. `self.member_namespace` (stored member namespace) vs `requested.member_namespace`
697    ///
698    /// If either differs, this is a 128-bit collection truncation collision.
699    /// It recomputes the full 256-bit group/member identity for both stored and requested
700    /// identities and verifies that both reproduce the 128-bit `collection_id`, proving a true
701    /// truncation collision.
702    ///
703    /// # Errors
704    /// - Returns `StorageError::Internal` if requested or stored metadata does not reproduce `collection_id`.
705    /// - Returns `StorageError::Collision` on true 128-bit truncation collision.
706    pub fn validate_identity_collision(
707        &self,
708        requested: &Self,
709        collection_id: &[u8; 16],
710    ) -> Result<(), crate::storage::StorageError> {
711        self.validate_identity_collision_with(
712            requested,
713            collection_id,
714            try_derive_collection_full_id,
715        )
716    }
717
718    pub(crate) fn validate_identity_collision_with(
719        &self,
720        requested: &Self,
721        collection_id: &[u8; 16],
722        derive: impl Fn(Option<[u8; 4]>, &[u8]) -> Option<[u8; 32]>,
723    ) -> Result<(), crate::storage::StorageError> {
724        if self.collection_canonical_id == requested.collection_canonical_id
725            && self.member_namespace == requested.member_namespace
726        {
727            return Ok(());
728        }
729
730        let stored_full = derive(self.member_namespace, &self.collection_canonical_id);
731        if stored_full.as_ref().map(|id| &id[..16]) != Some(&collection_id[..]) {
732            return Err(crate::storage::StorageError::Internal(
733                "stored collection metadata does not reproduce collection id".to_owned(),
734            ));
735        }
736
737        let requested_full = derive(
738            requested.member_namespace,
739            &requested.collection_canonical_id,
740        );
741        if requested_full.as_ref().map(|id| &id[..16]) != Some(&collection_id[..]) {
742            return Err(crate::storage::StorageError::Internal(
743                "requested collection metadata does not reproduce collection id".to_owned(),
744            ));
745        }
746
747        Err(crate::storage::StorageError::Collision(format!(
748            "collection identity truncation collision: stored (group={:?}, ns={:?}) disagrees with requested (group={:?}, ns={:?})",
749            String::from_utf8_lossy(&self.collection_canonical_id),
750            self.member_namespace.map(|ns| String::from_utf8_lossy(&ns).into_owned()),
751            String::from_utf8_lossy(&requested.collection_canonical_id),
752            requested.member_namespace.map(|ns| String::from_utf8_lossy(&ns).into_owned()),
753        )))
754    }
755
756    /// Encode this record as a TLV block.
757    #[must_use]
758    pub fn encode(&self) -> Vec<u8> {
759        let mut out = Vec::new();
760        if let Some(ns) = &self.member_namespace {
761            push_tlv(&mut out, META_TAG_MEMBER_NAMESPACE, ns);
762        }
763        push_tlv(
764            &mut out,
765            META_TAG_COLLECTION_CANONICAL_ID,
766            &self.collection_canonical_id,
767        );
768        push_tlv(
769            &mut out,
770            META_TAG_RECORD_ID_RULE,
771            &encode_record_id_rule(&self.record_id_rule),
772        );
773        push_tlv(&mut out, META_TAG_PAYLOAD, &encode_payload(&self.payload));
774        if let Some(extension) = &self.extension {
775            push_tlv(&mut out, META_TAG_EXTENSION, extension);
776        }
777        if let Some(role) = &self.role {
778            push_tlv(&mut out, META_TAG_ROLE, role.as_bytes());
779        }
780        if let Some(schema) = &self.schema {
781            push_tlv(&mut out, META_TAG_SCHEMA, schema.as_bytes());
782        }
783        out
784    }
785
786    /// Decode a TLV block. Unrecognized tags are ignored: this record is
787    /// write-once and never re-encoded, so there is nothing to preserve them
788    /// for.
789    #[must_use]
790    pub fn decode(bytes: &[u8]) -> Option<Self> {
791        let mut meta = Self {
792            member_namespace: None,
793            collection_canonical_id: Vec::new(),
794            record_id_rule: RecordIdentityRule {
795                policy: FrameIdPolicy::ExternalCanonicalIdToCrosscheck,
796                digest_algorithm: DigestAlgorithm::Sha256,
797            },
798            payload: PayloadPolicy::Source,
799            extension: None,
800            role: None,
801            schema: None,
802        };
803        let mut has_canonical_id = false;
804        let mut has_record_id_rule = false;
805        let mut reader = TlvReader::new(bytes);
806        while let Some((tag, value)) = reader.next() {
807            match tag {
808                META_TAG_MEMBER_NAMESPACE => {
809                    let ns: [u8; 4] = value.try_into().ok()?;
810                    namespace_bias(ns)?;
811                    meta.member_namespace = Some(ns);
812                }
813                META_TAG_COLLECTION_CANONICAL_ID => {
814                    meta.collection_canonical_id = value.to_vec();
815                    has_canonical_id = true;
816                }
817                META_TAG_RECORD_ID_RULE => {
818                    meta.record_id_rule = decode_record_id_rule(value)?;
819                    has_record_id_rule = true;
820                }
821                META_TAG_PAYLOAD => meta.payload = decode_payload(value)?,
822                META_TAG_EXTENSION => meta.extension = Some(value.to_vec()),
823                META_TAG_ROLE => {
824                    meta.role = Some(std::str::from_utf8(value).ok()?.to_owned());
825                }
826                META_TAG_SCHEMA => {
827                    meta.schema = Some(std::str::from_utf8(value).ok()?.to_owned());
828                }
829                _ => {}
830            }
831        }
832        (reader.is_exhausted()
833            && has_canonical_id
834            && !meta.collection_canonical_id.is_empty()
835            && has_record_id_rule)
836            .then_some(meta)
837    }
838}
