1//! Cross-process read-committed overlay, gated behind the `multi-reader`
2//! feature (see `Cargo.toml` and `docs/TODO.txt`'s extraction plan).
3//!
4//! Lets a separate OS process observe a live writer's committed-but-not-yet-
5//! checkpointed data by scanning the writer's journal segment read-only. A
6//! single-process embedded deployment never calls any of this.

18/// How many bytes ending at the last consumed group the overlay remembers to
19/// notice that the consumed prefix was rewritten.
20const TAIL_WINDOW: u64 = crate::journal::CONSUMED_TAIL_LEN as u64;
21
22/// A file's identity and change stamp: device, inode, length and change time.
23/// Equal stamps on a quiet file mean nothing wrote to it in between, so the
24/// consumed bytes need no re-read. The change time (ctime) is used because,
25/// unlike the modification time, no caller can set it backwards.
26#[derive(Clone, Copy, PartialEq, Eq)]
27struct FileStamp {
28    dev: u64,
29    ino: u64,
30    len: u64,
31    ctime_secs: i64,
32    ctime_nanos: i64,
33}
34
35impl FileStamp {
36    /// Whether two stamps name the same file, ignoring length and time.
37    fn same_file(self, other: Self) -> bool {
38        self.dev == other.dev && self.ino == other.ino
39    }
40
41    /// Whether the last change is older than [`QUIET_AFTER`], so any later
42    /// write gets a different change time, provided the filesystem's timestamp
43    /// granularity is no coarser than [`QUIET_AFTER`]. A change inside that
44    /// window could share a tick with a later write, so the stamp is not
45    /// trusted yet.
46    ///
47    /// This reads the wall clock. A backwards step makes the age negative and
48    /// the stamp counts as not quiet, which is the safe direction. A forward
49    /// step could make a just-changed file look quiet; that would also need a
50    /// same-tick rewrite to go unnoticed, so it is accepted rather than guarded.
51    fn is_quiet(self) -> bool {
52        let Ok(now) = std::time::SystemTime::now().duration_since(std::time::UNIX_EPOCH) else {
53            return false;
54        };
55        let Ok(secs) = u64::try_from(self.ctime_secs) else {
56            return false;
57        };
58        let changed = std::time::Duration::new(secs, u32::try_from(self.ctime_nanos).unwrap_or(0));
59        now.checked_sub(changed)
60            .is_some_and(|age| age > QUIET_AFTER)
61    }
62}
63
64/// How long a file must have been unchanged before an equal [`FileStamp`] is
65/// trusted to mean it was not rewritten in between.
66///
67/// This assumes the filesystem's change-time granularity is at most this long.
68/// Two seconds covers the coarse cases (FAT-class mounts, NFS, FUSE), and the
69/// fine ones (ext4, xfs, btrfs, tmpfs, NTFS) are far below it. The cost of
70/// being generous is only that a read-mostly file takes this long after its
71/// last write to reach the no-read fast path.
72#[cfg(not(test))]
73const QUIET_AFTER: std::time::Duration = std::time::Duration::from_secs(2);
74// The shortened test window must not leak into a real build, and the window
75// must stay above a one-second timestamp granularity.
76#[cfg(not(test))]
77const _: () = assert!(
78    QUIET_AFTER.as_secs() >= 2,
79    "the quiet window must stay above a one-second timestamp granularity"
80);
81/// Tests sleep past this window to reach the quiet fast path, and their files
82/// live on filesystems with fine timestamps, so a short window keeps them fast
83/// without changing what they check: an age above the window is quiet, below
84/// it is not.
85#[cfg(test)]
86const QUIET_AFTER: std::time::Duration = std::time::Duration::from_millis(150);

88/// The stamp of `meta`, when the platform exposes a reliable file identity.
89/// Inode 0 is not a real identity (some filesystems report it for every file),
90/// so it yields no stamp.
91#[cfg(unix)]
92fn file_stamp(meta: &fs::Metadata) -> Option<FileStamp> {
93    use std::os::unix::fs::MetadataExt;
94    if meta.ino() == 0 {
95        return None;
96    }
97    Some(FileStamp {
98        dev: meta.dev(),
99        ino: meta.ino(),
100        len: meta.len(),
101        ctime_secs: meta.ctime(),
102        ctime_nanos: meta.ctime_nsec(),
103    })
104}

108#[cfg(not(unix))]

110    None

113/// One committed value in the read-journal overlay: payload and its LSN.
114type ReadJournalValue = (bytes::Bytes, u64);
115/// Values for one collection, keyed by node ID.
116type ReadJournalCollection = HashMap<[u8; 16], ReadJournalValue>;

120/// Outcome of refreshing the overlay against the writer's segment.
121#[derive(Clone, Copy, PartialEq, Eq)]
122enum ReadRefresh {
123    /// The segment was scanned and applied; the overlay is current.
124    Applied,
125    /// The segment was reclaimed past the coverage this reader's index
126    /// incorporated. The checkpoint-bound index must be reloaded before the
127    /// overlay can be trusted again.
128    NeedsReload,
129}

131/// In-memory overlay of committed journal groups, built by scanning a
132/// writer's journal segment read-only. It is the read-committed source of
133/// truth for [`PackfileStorage::get_read_committed`]: entries at or below
134/// `observed_lsn` have a complete commit trailer, so they are committed even
135/// though the writer may not have fsynced or advanced the durable index yet.
136pub(super) struct ReadJournal {
137    /// Pool whose tagged frames this overlay applies, or `None` for a per-pool
138    /// (untagged, version-2) segment. A shared segment interleaves all pools'
139    /// frames, so a pool's reader must apply only frames tagged for it.
140    pool: Option<crate::layout::ShardType>,
141    /// Segment file scanned for committed groups.
142    path: PathBuf,
143    /// The journal LSN covered by the durable index this reader actually
144    /// loaded, fixed when the overlay was enabled.
145    ///
146    /// This is the *only* coverage entries may be filtered or pruned against.
147    /// Reading a fresher `journal.lsn` from disk would prune a committed entry
148    /// the reader's stale in-memory index does not yet contain, dropping the
149    /// record from both places. It is deliberately not advanced by a live
150    /// packfile rescan: only loading the corresponding checkpoint may advance
151    /// the index/coverage pair.
152    pub(super) covered: u64,
153    /// Raw file length at the last scan. Skips a rescan when unchanged, so an
154    /// unchanged partial tail is not re-probed on every read.
155    observed_len: u64,
156    /// Absolute offset after the last complete group consumed. A partial tail
157    /// is re-probed from here once more bytes land.
158    observed_valid_len: u64,
159    /// Highest applied LSN. Groups at or below this are already in the overlay.
160    observed_lsn: u64,
161    /// The last bytes consumed (up to [`TAIL_WINDOW`], ending at
162    /// `observed_valid_len`), or `None` before any group was applied.
163    ///
164    /// File length cannot show that the bytes already consumed are still the
165    /// ones on disk: after a writer restart that discarded a visible-but-
166    /// undurable group and reissued its LSN, or after a reclaim followed by
167    /// enough appends, the file can be the same length (or longer) while its
168    /// content differs. Comparing this window against the bytes now ending at
169    /// `observed_valid_len` catches that. It covers the most recently consumed
170    /// records only: a rewrite that leaves the whole window byte-identical is
171    /// not detected. The group trailer is not used because it is not a content
172    /// fingerprint (see [`Journal::read_tail_ending_at`]).
173    observed_tail: Option<Vec<u8>>,
174    /// Descriptor the tail window is read through, opened when the window was
175    /// recorded, so a refresh pays one positioned read instead of an open.
176    tail_file: Option<File>,
177    /// Stamp of the file when the tail window was recorded. The descriptor
178    /// keeps pointing at an old inode after the writer renames a replacement
179    /// into place (reclaim), so a path whose device or inode differs means the
180    /// segment was replaced and the overlay must be rebuilt. An equal stamp on
181    /// a file that was already quiet when recorded (`tail_quiet`) proves the
182    /// consumed bytes were not rewritten, so the window is not re-read.
183    tail_stamp: Option<FileStamp>,
184    /// Whether the file was quiet, per [`FileStamp::is_quiet`], when
185    /// `tail_stamp` was taken.
186    tail_quiet: bool,
187    /// Test-only count of window reads, to show the fingerprint skips them.
188    #[cfg(test)]
189    pub(super) tail_reads: u64,
190    /// Test-only total of segment bytes scanned across refreshes, to show a
191    /// retained overlay scans only what was appended.
192    #[cfg(test)]
193    pub(super) scanned_bytes: u64,
194    /// Test-only switch that makes every stamp absent, as on a platform without
195    /// inode numbers, to exercise the no-identity path on unix.
196    #[cfg(test)]
197    pub(super) force_no_identity: bool,
198    /// Test-only count of descriptors opened to record a tail. It does not
199    /// count the fresh open a no-identity comparison makes (see
200    /// `observed_tail_changed`), so `0` means "no descriptor is kept", not
201    /// "the file is never opened".
202    #[cfg(test)]
203    pub(super) tail_opens: u64,
204    /// Test-only count of overlay rebuilds forced by a changed tail.
205    #[cfg(test)]
206    pub(super) tail_resets: u64,
207    /// Scratch buffer the window is read into, reused across refreshes.
208    tail_scratch: Vec<u8>,
209    /// Committed puts as `(payload, lsn)`, keyed `collection_id -> node_id`.
210    pub(super) puts: ReadJournalPuts,
211    /// Highest committed delete LSN per collection. This is a persistent
212    /// boundary: durable fallback stays suppressed for the collection until the
213    /// checkpoint covers the delete, because the durable index may still hold
214    /// pre-delete records. A later put does not clear it.
215    pub(super) delete_lsn: HashMap<[u8; 16], u64>,
216    /// Whether this is the writer's transaction overlay, which is parked when
217    /// no transaction uses it, and not a read-only worker's overlay.
218    #[cfg(feature = "multi-reader")]
219    pub(super) transaction: bool,
220}

277    /// Drop all applied state so the next refresh rebuilds from the segment.
278    pub(super) fn reset_overlay(&mut self) {
279        self.puts.clear();
280        self.delete_lsn.clear();
281        self.observed_lsn = 0;
282        self.observed_valid_len = 0;
283        self.observed_tail = None;
284        self.tail_file = None;
285        self.tail_stamp = None;
286        self.tail_quiet = false;
287        // Zero the length too, so an equal-length segment after a reload still
288        // forces a full rescan instead of short-circuiting on `len == observed_len`.
289        self.observed_len = 0;
290    }
291
292    /// Discard overlay state the durable index this reader loaded now covers.
293    ///
294    /// Reclaim is best-effort, so a journal group can outlive the checkpoint
295    /// that covers it. Without this, an old journal put or delete could shadow
296    /// newer checkpointed state. The bound is [`Self::covered`] — the coverage
297    /// of the loaded index, not a detached disk read — so an entry is only
298    /// dropped once the fallback index is known to contain it.
299    fn prune_covered(&mut self, covered: u64) {
300        if covered == 0 {
301            return;
302        }
303        self.puts.retain(|_, entries| {
304            entries.retain(|_, (_, lsn)| *lsn > covered);
305            !entries.is_empty()
306        });
307        self.delete_lsn.retain(|_, lsn| *lsn > covered);
308    }
309
310    /// Whether the segment's base has advanced past the coverage this reader's
311    /// index incorporated, so the prefix may hold frames this index lacks.
312    ///
313    /// The reclaimed prefix is already gone from disk, so its contents cannot
314    /// be inferred from the surviving groups. A base jump is therefore treated
315    /// as a possible gap unconditionally; whether it is a *real* gap for this
316    /// pool is decided by the caller, which reloads the checkpoint-bound index
317    /// and only accepts the jump once that index is known to be current (see
318    /// [`PackfileStorage::refresh_read_journal`]). A `base_lsn` at or below
319    /// `covered + 1` means nothing at or below `covered` was dropped.
320    pub(super) fn reset_has_coverage_gap(&self, scan: &crate::journal::Scan) -> bool {
321        scan.base_lsn > self.covered.saturating_add(1)
322    }
323
324    /// Whether the bytes that ended the consumed prefix are no longer the ones
325    /// ending at `observed_valid_len` on disk. `stamp` is the segment path's
326    /// stamp right now. A replaced file, or missing or unreadable bytes, count
327    /// as changed: the consumed prefix is gone.
328    fn observed_tail_changed(&mut self, stamp: Option<FileStamp>) -> bool {
329        let Some(expected) = &self.observed_tail else {
330            // Groups were applied but their tail could not be recorded (or the
331            // recording was discarded), so nothing protects the overlay from a
332            // rewritten prefix: rebuild it and try recording again.
333            return self.observed_lsn > 0;
334        };
335        let mut unchanged_stamp = None;
336        // With a real identity the descriptor held since recording still names
337        // the file the overlay was built from. Without one (a platform that
338        // reports no inode numbers) a replaced file cannot be told apart by
339        // inode, so the path is opened afresh and what is there now is compared.
340        let fresh;
341        let file = match (stamp, self.tail_stamp) {
342            (Some(now), Some(then)) => {
343                if !now.same_file(then) {
344                    return true;
345                }
346                // Nothing wrote to a file that was already quiet when the
347                // window was recorded, so the consumed bytes are unchanged.
348                if self.tail_quiet && now == then {
349                    return false;
350                }
351                if now == then {
352                    unchanged_stamp = Some(now);
353                }
354                let Some(held) = &self.tail_file else {
355                    return true;
356                };
357                held
358            }
359            (None, None) => {
360                let Ok(opened) = File::open(&self.path) else {
361                    return true;
362                };
363                fresh = opened;
364                &fresh
365            }
366            // An identity that appeared or vanished since recording means the
367            // file is not the one the overlay was built from.
368            _ => return true,
369        };
370        #[cfg(test)]
371        {
372            self.tail_reads = self.tail_reads.saturating_add(1);
373        }
374        match Journal::read_tail_ending_at(
375            file,
376            self.observed_valid_len,
377            TAIL_WINDOW,
378            &mut self.tail_scratch,
379        ) {
380            Ok(true) if &self.tail_scratch != expected => true,
381            Ok(true) => {
382                // The window matches and the stamp is unchanged. If the file
383                // has by now been quiet longer than the timestamp granularity,
384                // any later write is guaranteed a different change time, so
385                // later refreshes can trust the stamp and skip this read. This
386                // is what lets a file that was still warm when recorded reach
387                // the fast path.
388                if unchanged_stamp.is_some_and(FileStamp::is_quiet) {
389                    self.tail_quiet = true;
390                }
391                false
392            }
393            Ok(false) | Err(_) => true,
394        }
395    }
396
397    /// Remember the bytes that end the consumed prefix, and keep a descriptor
398    /// for later refreshes to compare them through.
399    ///
400    /// The window is built from the bytes the scan actually consumed
401    /// (`consumed`), appended to the window already remembered, never from a
402    /// separate read of the file: a rewrite that lands after the scan can then
403    /// never be folded into what is remembered, and a later comparison is
404    /// always against what the overlay was built from. `scanned` is the
405    /// segment path's stamp taken before the scan; a write after that stat
406    /// changes the stamp, so the next refresh compares the window instead of
407    /// trusting it.
408    ///
409    /// While the segment is still the same file the held descriptor is reused,
410    /// so a refresh that only appended opens nothing. A new descriptor is
411    /// opened only after a reset or a replaced file, and if it refers to a
412    /// different file than `scanned` the segment was replaced mid-refresh and
413    /// the recording is discarded. On any failure nothing is remembered, which
414    /// [`Self::observed_tail_changed`] treats as untrusted, so the next refresh
415    /// rescans from the start instead of trusting the prefix.
416    fn record_observed_tail(&mut self, scanned: Option<FileStamp>, consumed: &[u8]) {
417        let mut window = self.observed_tail.take().unwrap_or_default();
418        // A stamp exists only where the platform reports a reliable file
419        // identity, so a `Some` pair means the held descriptor can be trusted
420        // to name the same file. With no identity there is no descriptor to
421        // reuse and none is ever held.
422        let reusable = match (scanned, self.tail_stamp) {
423            (Some(now), Some(then)) => now.same_file(then),
424            _ => false,
425        };
426        let held = if reusable {
427            self.tail_file.take()
428        } else {
429            None
430        };
431        self.tail_file = None;
432        self.tail_stamp = None;
433        self.tail_quiet = false;
434        if self.observed_valid_len == 0 {
435            return;
436        }
437        window.extend_from_slice(consumed);
438        if window.len() > crate::journal::CONSUMED_TAIL_LEN {
439            let excess = window
440                .len()
441                .saturating_sub(crate::journal::CONSUMED_TAIL_LEN);
442            window.drain(..excess);
443        }
444        // The remembered bytes must be exactly the window a later comparison
445        // reads. If they are not (nothing consumed to build them from), leave
446        // the overlay untrusted so it rebuilds.
447        let expected_len = crate::journal::consumed_tail_len(self.observed_valid_len);
448        // A segment with no group bytes yet (header only) has no window to
449        // protect and none to compare, so nothing is recorded.
450        if expected_len == 0 || window.len() != expected_len {
451            return;
452        }
453        // Without an identity no descriptor is kept: later comparisons open the
454        // path afresh (see `observed_tail_changed`).
455        let file = if scanned.is_none() {
456            None
457        } else if let Some(file) = held {
458            Some(file)
459        } else {
460            #[cfg(test)]
461            {
462                self.tail_opens = self.tail_opens.saturating_add(1);
463            }
464            let Ok(file) = File::open(&self.path) else {
465                return;
466            };
467            let opened = file.metadata().ok().and_then(|meta| file_stamp(&meta));
468            if let (Some(scanned), Some(opened)) = (scanned, opened) {
469                if !scanned.same_file(opened) {
470                    return;
471                }
472            }
473            Some(file)
474        };
475        self.observed_tail = Some(window);
476        self.tail_file = file;
477        self.tail_quiet = scanned.is_some_and(FileStamp::is_quiet);
478        self.tail_stamp = scanned;
479    }
480
481    /// Apply every committed group above `observed_lsn` to the overlay.
482    ///
483    /// A missing segment is treated as empty only before this reader has seen
484    /// any journal bytes; if an observed segment disappears, the caller must
485    /// reload the checkpoint. A segment that shrank since the last scan was
486    /// reclaimed, so the overlay is rebuilt from scratch. Never repairs or
487    /// creates the file, matching a read-only worker's constraints.
488    ///
489    /// `accept_reclaimed_prefix` is set by the caller once it has reloaded the
490    /// checkpoint-bound index to the writer's latest durable coverage for this
491    /// pool. On a shared segment a base jump beyond that coverage can only be
492    /// other pools' reclaimed frames, which this pool does not need, so the
493    /// jump is accepted instead of forcing another reload.
494    ///
495    /// Returns [`ReadRefresh::NeedsReload`] when a reset reveals the writer
496    /// reclaimed past this reader's incorporated coverage; the caller must
497    /// reload the checkpoint-bound index and rebind `covered` before retrying.
498    fn refresh(&mut self, accept_reclaimed_prefix: bool) -> Result<ReadRefresh, StorageError> {
499        // Coverage is fixed to the index this reader loaded. Reading
500        // `journal.lsn` fresh would prune entries the reader's stale index has
501        // not incorporated yet, dropping records from both sources.
502        let covered = self.covered;
503        let (len, stamp) = match fs::metadata(&self.path) {
504            Ok(meta) => (meta.len(), file_stamp(&meta)),
505            Err(error) if error.kind() == std::io::ErrorKind::NotFound => {
506                if self.observed_len != 0 || self.observed_lsn > covered {
507                    return Ok(ReadRefresh::NeedsReload);
508                }
509                (0, None)
510            }
511            Err(error) => return Err(StorageError::Io(error)),
512        };
513        #[cfg(test)]
514        let stamp = if self.force_no_identity { None } else { stamp };
515        // Length alone cannot show that the consumed prefix is unchanged, so
516        // check the bytes that ended it first. On a change, drop everything and
517        // rebuild: a full rescan is gap-checked like any other reset below.
518        if self.observed_tail_changed(stamp) {
519            #[cfg(test)]
520            {
521                self.tail_resets = self.tail_resets.saturating_add(1);
522            }
523            self.reset_overlay();
524        }
525        if len != self.observed_len {
526            let mut reset = false;
527            if len < self.observed_len {
528                // The segment was reclaimed/rotated; re-derive from the start.
529                self.reset_overlay();
530                reset = true;
531            }
532            let full_scan = self.observed_valid_len == 0;
533            #[cfg(test)]
534            let scan_from = self.observed_valid_len;
535            let scan = if full_scan {
536                Journal::scan_read_only(&self.path).map_err(StorageError::Io)?
537            } else {
538                // Read only the tail past the last complete group. A failure
539                // here means the segment was replaced under us (or a rotation
540                // changed its base), so fall back to a full rescan.
541                if let Ok(scan) = Journal::scan_read_only_from(
542                    &self.path,
543                    self.observed_valid_len,
544                    self.observed_lsn.saturating_add(1),
545                ) {
546                    scan
547                } else {
548                    self.reset_overlay();
549                    reset = true;
550                    Journal::scan_read_only(&self.path).map_err(StorageError::Io)?
551                }
552            };
553            // A full rescan has no continuity with the previous overlay, so it
554            // must be gap-checked too — not only an explicit shrink. Without
555            // this, a reload that rebinds `covered` below the segment base
556            // would silently serve a hole.
557            if (reset || full_scan)
558                && !accept_reclaimed_prefix
559                && self.reset_has_coverage_gap(&scan)
560            {
561                return Ok(ReadRefresh::NeedsReload);
562            }
563            for group in &scan.groups {
564                for entry in &group.entries {
565                    if entry.lsn <= self.observed_lsn || entry.lsn <= covered {
566                        continue;
567                    }
568                    // A shared segment interleaves all pools' frames; apply
569                    // only this pool's. `None` pool is a per-pool segment,
570                    // whose frames are all this store's.
571                    if self.pool.is_some_and(|pool| entry.pool != Some(pool)) {
572                        continue;
573                    }
574                    match &entry.mutation {
575                        JournalMutation::Put {
576                            collection_id,
577                            node_id,
578                            payload,
579                        } => {
580                            self.puts.entry(*collection_id).or_default().insert(
581                                *node_id,
582                                (bytes::Bytes::copy_from_slice(payload), entry.lsn),
583                            );
584                        }
585                        JournalMutation::DeleteCollection { collection_id } => {
586                            self.puts.remove(collection_id);
587                            self.delete_lsn
588                                .entry(*collection_id)
589                                .and_modify(|lsn| *lsn = (*lsn).max(entry.lsn))
590                                .or_insert(entry.lsn);
591                        }
592                    }
593                }
594                self.observed_lsn = self.observed_lsn.max(group.last_lsn);
595            }
596            // Resume from the last complete group, not the raw file length, so
597            // a partial tail is re-probed once its trailer lands.
598            #[cfg(test)]
599            {
600                self.scanned_bytes = self
601                    .scanned_bytes
602                    .saturating_add(scan.valid_len.saturating_sub(scan_from));
603            }
604            self.observed_valid_len = scan.valid_len;
605            self.observed_len = len;
606            self.record_observed_tail(stamp, &scan.consumed_tail);
607        }
608        // The checkpoint can advance without the segment changing (reclaim is
609        // best-effort), so prune overlay state it now covers.
610        self.prune_covered(covered);
611        Ok(ReadRefresh::Applied)
612    }
613}

616    /// Reload the durable index from the current on-disk checkpoint and rebind
617    /// [`Self::read_covered_lsn`] to it.
618    ///
619    /// Returns `false` when no checkpoint matching the current packs is on
620    /// disk, so the caller can fail closed instead of serving a stale index.
621    /// Coverage is captured before the checkpoint is read, so the new bound can
622    /// never be newer than the index loaded below.
623    pub(super) fn reload_index_from_checkpoint(&self) -> bool {
624        // A writer may have created new packs since this handle opened, and the
625        // checkpoint it wrote names that new pack set. Rediscover before
626        // building the fingerprint, or the reload can never match.
627        if self.shards.discover_shards().is_err() {
628            self.read_reload_failures.fetch_add(1, Ordering::Relaxed);
629            return false;
630        }
631        // Rebuild the fingerprint from files that still exist. This worker's
632        // shard table can retain open handles to packs the writer retired and
633        // unlinked during repack; including their cached lengths makes every
634        // checkpoint fingerprint mismatch forever. New/grown live packs are
635        // included at their current lengths, and the checkpoint delta log
636        // validates the suffix from the checkpoint's original fingerprint.
637        let mut open_shards: Vec<(u16, u64, PathBuf, u64)> = Vec::new();
638        for (id, shard) in self.shards.all_shards() {
639            match fs::metadata(&shard.path) {
640                Ok(metadata) => {
641                    open_shards.push((id, shard.pack_id, shard.path.clone(), metadata.len()));
642                }
643                Err(error) if error.kind() == std::io::ErrorKind::NotFound => {}
644                Err(_) => {
645                    self.read_reload_failures.fetch_add(1, Ordering::Relaxed);
646                    return false;
647                }
648            }
649        }
650        let deleted_collections = self.deleted_collections.lock().clone();
651        let mut timings = OpenTimings::default();
652        let Some((scan_out, collection_order, _delta_state, checkpoint_covered)) =
653            Self::checkpoint_scan_out(
654                &self.base_dir,
655                self.cache_capacity,
656                &self.shards,
657                &open_shards,
658                &deleted_collections,
659                false,
660                ReloadMode::JournalBound,
661                self.index_config,
662                &mut timings,
663            )
664        else {
665            self.read_reload_failures.fetch_add(1, Ordering::Relaxed);
666            return false;
667        };
668        {
669            let mut tables = self.index_tables.write();
670            tables.collections = scan_out.collections;
671            tables.collection_order = collection_order;
672            tables.shard_collections = scan_out.shard_collections;
673            tables.collection_shards = scan_out.collection_shards;
674        }
675        // Bind coverage to the checkpoint actually loaded, not to a
676        // separately-read `journal.lsn` that a concurrent checkpoint may
677        // already have advanced past this index.
678        self.read_covered_lsn
679            .store(checkpoint_covered, Ordering::Release);
680        self.read_reloads.fetch_add(1, Ordering::Relaxed);
681        true
682    }
683
684    /// Refresh the read-committed overlay, reloading the checkpoint-bound index
685    /// when the writer reclaimed past this reader's incorporated coverage.
686    ///
687    /// A reclaim that outruns the reader's index is the one case the overlay
688    /// cannot resolve from the segment alone; reloading the checkpoint advances
689    /// the index/coverage pair together, after which the retry is consistent.
690    /// Without a usable checkpoint the read fails closed.
691    ///
692    /// On success the held overlay guard is returned so the caller can read the
693    /// applied overlay without releasing and reacquiring the mutex, which could
694    /// otherwise expose an empty or partially rebuilt overlay to another reader.
695    pub(super) fn refresh_read_journal(
696        &self,
697    ) -> Result<parking_lot::MutexGuard<'_, Option<ReadJournal>>, StorageError> {
698        const RELOAD_ATTEMPTS: usize = 8;
699        const MAX_BACKOFF_MS: u64 = 64;
700        // Capture coverage once, before the first attempt. Comparing the final
701        // value against a per-iteration snapshot would only detect coverage that
702        // advanced on the *last* attempt, misreporting a moving checkpoint as a
703        // genuine gap. Reloads only ever advance `read_covered_lsn`, so a strict
704        // increase across the whole loop is exactly the retryable signal.
705        let coverage_before_attempts = self.read_covered_lsn.load(Ordering::Acquire);
706        // A failed reload attempt makes the outcome retryable even if coverage
707        // never advanced: a concurrent writer may simply not have written the
708        // checkpoint yet, so the same gap can resolve on a later read.
709        // Deliberately sticky: any failed reload makes the whole refresh
710        // retryable, even if a later attempt's reload succeeds.
711        let mut reload_failed = false;
712        for attempt in 0..RELOAD_ATTEMPTS {
713            let mut guard = self.read_journal.lock();
714            let Some(overlay) = guard.as_mut() else {
715                return Ok(guard);
716            };
717            if overlay.refresh(false)? == ReadRefresh::Applied {
718                return Ok(guard);
719            }
720            if !self.reload_index_from_checkpoint() {
721                reload_failed = true;
722                drop(guard);
723                let delay_ms = 1_u64
724                    .checked_shl(u32::try_from(attempt).unwrap_or(u32::MAX))
725                    .unwrap_or(MAX_BACKOFF_MS)
726                    .min(MAX_BACKOFF_MS);
727                std::thread::sleep(std::time::Duration::from_millis(delay_ms));
728                continue;
729            }
730            overlay.covered = self.read_covered_lsn.load(Ordering::Acquire);
731            overlay.reset_overlay();
732            // The reload bound this overlay to the newest checkpoint it could
733            // load. On a shared segment a base jump beyond that checkpoint is
734            // made up of other pools' reclaimed frames, which this pool never
735            // needs, so accept it once the loaded checkpoint is at least this
736            // pool's latest durable coverage. A per-pool segment holds only
737            // this store's frames, so it stays fail-closed.
738            let accept_reclaimed_prefix =
739                overlay.pool.is_some() && overlay.covered >= Self::read_journal_lsn(&self.base_dir);
740            // Rebuild immediately while still holding the guard: returning the
741            // guard only once the overlay is applied keeps another reader from
742            // observing the just-cleared state.
743            if overlay.refresh(accept_reclaimed_prefix)? == ReadRefresh::Applied {
744                return Ok(guard);
745            }
746            drop(guard);
747            if attempt.saturating_add(1) < RELOAD_ATTEMPTS {
748                let delay_ms = 1_u64
749                    .checked_shl(u32::try_from(attempt).unwrap_or(u32::MAX))
750                    .unwrap_or(MAX_BACKOFF_MS)
751                    .min(MAX_BACKOFF_MS);
752                std::thread::sleep(std::time::Duration::from_millis(delay_ms));
753            }
754        }
755        let advanced = self.read_covered_lsn.load(Ordering::Acquire) > coverage_before_attempts;
756        if reload_failed || advanced {
757            Err(StorageError::Io(std::io::Error::new(
758                std::io::ErrorKind::WouldBlock,
759                "read-committed reload failed or checkpoint coverage advanced; retry the read",
760            )))
761        } else {
762            Err(StorageError::Corrupt(
763                "read-committed journal segment skips LSNs not covered by the loaded checkpoint"
764                    .to_owned(),
765            ))
766        }
767    }

771    /// `path` is a writer's journal segment. The overlay is built with
772    /// [`Journal::scan_read_only`], which never creates, repairs, or locks the
773    /// segment, so this is safe from a read-only worker process. The durable
774    /// read API ([`crate::storage::StorageEngine::get_many`], [`Self::get_many_with_refresh`])
775    /// is unchanged and continues to hide unflushed writes; only
776    /// [`Self::get_read_committed`] consults the overlay.

778    /// # Errors

784        path: impl AsRef<std::path::Path>,

804    /// Enable the in-process overlay a writer's own pool uses while a published
805    /// transaction is being materialized.
806    ///
807    /// This is not the reader's setup. A read-only worker has an index loaded
808    /// from a checkpoint, so a reclaimed journal prefix past that checkpoint is
809    /// a gap it must reload to fill. A writer's live index already holds every
810    /// mutation it has applied, so a reclaimed prefix can never hide data from
811    /// it, and there is nothing to reload: its `read_covered_lsn` is never set
812    /// and a reload from the checkpoint cannot succeed against its own open
813    /// packs. Running the reader's gap check here made every commit fail with a
814    /// retry-forever `WouldBlock` once a checkpoint had reclaimed the shared
815    /// WAL past LSN 1.
816    ///
817    /// The overlay is bound to the pool's checkpoint coverage (`journal.lsn`):
818    /// frames at or below it are in the durable index, and frames above it are
819    /// served from the journal until the transaction is fully materialized.
820    /// Normal commits scan only the suffix appended since the previous
821    /// activation, because the previous overlay's position is kept; a full
822    /// scan from the segment start happens only for the first activation, after
823    /// a reclaim or replacement, or on recovery. Only `SharedDatabase`, the
824    /// segment's single writer, reaches this path; read-only workers use
825    /// [`Self::enable_read_journal`].
826    ///
827    /// It is built directly and refreshed once with `accept_reclaimed_prefix`
828    /// set, rather than going through `enable_read_journal_inner`, whose
829    /// refresh-then-reload path is the reader's and cannot succeed here.
830    ///
831    /// # Errors
832    /// Returns [`StorageError`] if the segment is unreadable or a committed
833    /// group fails validation.
834    #[cfg(feature = "multi-reader")]
835    pub(super) fn enable_transaction_read_journal(
836        &self,
837        path: &std::path::Path,
838        pool: crate::layout::ShardType,
839    ) -> Result<(), StorageError> {
840        let covered = self.durable_coverage();
841        // Keep the previous overlay's scan position and tail fingerprint, and
842        // drop only its entries. Every earlier transaction is fully
843        // materialized (no user held the overlay), so those entries are in the
844        // writer's index and would only shadow newer direct writes; but the
845        // position lets this refresh scan just the groups appended since, not
846        // the whole segment from the checkpoint. A reclaimed or replaced
847        // segment fails the length and fingerprint checks and rebuilds.
848        let mut overlay = match self.parked_transaction_overlay.lock().take() {
849            Some(mut previous)
850                if previous.pool == Some(pool) && previous.path.as_path() == path =>
851            {
852                previous.puts.clear();
853                previous.delete_lsn.clear();
854                previous.covered = covered;
855                previous
856            }
857            _ => ReadJournal::empty(path.to_path_buf(), covered, Some(pool)),
858        };
859        // The writer's index already holds everything it applied, so accept a
860        // reclaimed prefix instead of treating it as a gap. A fresh overlay has
861        // observed nothing and accepts the prefix, so the only outcome besides
862        // an error is `Applied`; anything else is a broken assumption and is
863        // surfaced instead of installing an overlay that was never built.
864        overlay.transaction = true;
865        #[cfg(test)]
866        let scanned_before = overlay.scanned_bytes;
867        let refreshed = overlay.refresh(true)?;
868        #[cfg(test)]
869        self.transaction_overlay_scanned.fetch_add(
870            overlay.scanned_bytes.saturating_sub(scanned_before),
871            Ordering::Relaxed,
872        );
873        match refreshed {
874            ReadRefresh::Applied => {}
875            ReadRefresh::NeedsReload => {
876                return Err(StorageError::Internal(
877                    "the writer's transaction overlay unexpectedly needed a checkpoint reload"
878                        .to_owned(),
879                ));
880            }
881        }
882        *self.read_journal.lock() = Some(overlay);
883        Ok(())
884    }
885
886    /// Take the transaction overlay out of the read path once no transaction
887    /// uses it. Its entries are materialized in the writer's index, so leaving
888    /// it installed would make every read-committed read refresh and consult a
889    /// journal that can only repeat what the index holds. Its scan position is
890    /// kept for the next activation.
891    #[cfg(feature = "multi-reader")]
892    pub(super) fn park_transaction_overlay(&self) {
893        let mut installed = self.read_journal.lock();
894        if installed
895            .as_ref()
896            .is_some_and(|overlay| overlay.transaction)
897        {
898            if let Some(mut overlay) = installed.take() {
899                overlay.puts.clear();
900                overlay.delete_lsn.clear();
901                *self.parked_transaction_overlay.lock() = Some(overlay);
902            }
903        }
904    }

906    fn enable_read_journal_inner(
907        &self,
908        path: impl AsRef<std::path::Path>,
909        pool: Option<crate::layout::ShardType>,
910    ) -> Result<(), StorageError> {
911        // Bind the overlay to the coverage of the index this handle loaded, not
912        // whatever `journal.lsn` says now: a concurrent checkpoint may already
913        // have advanced past this handle's in-memory index.
914        let covered = self.read_covered_lsn.load(Ordering::Acquire);
915        *self.read_journal.lock() = Some(ReadJournal::empty(
916            path.as_ref().to_path_buf(),
917            covered,
918            pool,
919        ));
920        if let Err(error) = self.refresh_read_journal() {
921            *self.read_journal.lock() = None;
922            return Err(error);
923        }
924        Ok(())
925    }

927    /// Read records at read-committed visibility: the durable index plus any
928    /// committed-but-unflushed journal group.
929    ///
930    /// This is the cross-process freshness path for read-only workers, and it
931    /// is deliberately separate from the durable API. A key absent from the
932    /// durable index is filled from a complete journal group (commit trailer
933    /// present) even before the writer fsyncs or advances the index
934    /// checkpoint. A committed collection delete shadows durable records.
935    ///
936    /// Without [`Self::enable_read_journal`] this degrades to
937    /// [`Self::get_many_with_refresh`].
938    ///
939    /// # Errors
940    /// Propagates errors from the durable read or journal scan. An unresolved
941    /// gap after journal reclamation returns [`StorageError::Corrupt`], or
942    /// [`StorageError::Io`] with `WouldBlock` if checkpoint coverage advanced
943    /// during the reload attempts.
944    pub fn get_read_committed(
945        &self,
946        collection_id: &[u8; 16],
947        ids: &[NodeId],
948    ) -> Result<Vec<Option<NodeData>>, StorageError> {
949        // Refresh outside the read lock: this may reload the checkpoint-bound
950        // index if the writer reclaimed past this reader's coverage. The
951        // returned guard keeps the applied overlay locked for the lookup below,
952        // so another reader cannot reset it in between.
953        let guard = self.refresh_read_journal()?;
954
955        // No overlay is installed (a writer with no transaction in flight):
956        // the live index is the whole answer, so skip the per-id bookkeeping
957        // below.
958        if guard.is_none() {
959            drop(guard);
960            let _fallback = FallbackReadGuard::enter();
961            return self.get_many_with_refresh(collection_id, ids);
962        }
963
964        let mut results: Vec<Option<NodeData>> = vec![None; ids.len()];
965        let mut unresolved: Vec<usize> = Vec::new();
966        {
967            if let Some(overlay) = guard.as_ref() {
968                match overlay.puts.get(collection_id) {
969                    Some(committed) => {
970                        for (index, id) in ids.iter().enumerate() {
971                            match committed.get(id) {
972                                Some((payload, _)) => {
973                                    results[index] = Some(NodeData::new(payload.clone()));
974                                }
975                                None => unresolved.push(index),
976                            }
977                        }
978                    }
979                    None => unresolved.extend(0..ids.len()),
980                }
981                // A delete the checkpoint does not yet cover means the durable
982                // index may still hold pre-delete records, so falling back for
983                // missing keys would resurrect them. Return the overlay's view
984                // (post-delete puts only) until the delete is covered.
985                if overlay.delete_lsn.contains_key(collection_id) {
986                    return Ok(results);
987                }
988            } else {
989                unresolved.extend(0..ids.len());
990            }
991        }
992        drop(guard);
993
994        if !unresolved.is_empty() {
995            let durable_ids: Vec<NodeId> = unresolved.iter().map(|&index| ids[index]).collect();
996            // Read the live index directly: `get_many` would redirect back here
997            // while a transaction overlay is active.
998            let _fallback = FallbackReadGuard::enter();
999            let durable = self.get_many_with_refresh(collection_id, &durable_ids)?;
1000            for (index, value) in unresolved.into_iter().zip(durable) {
1001                results[index] = value;
1002            }
1003        }
1004        Ok(results)
1005    }

1392    /// The remembered window is composed from consumed bytes across refreshes
1393    /// (the previous window plus what each scan consumed). The previous window
1394    /// ends exactly at the scan's start offset, `observed_valid_len`, and the
1395    /// scan's bytes begin there, so the two are contiguous. The composed window
1396    /// must always equal the window a later comparison reads from disk,
1397    /// including once the file outgrows the window, which is where the join and
1398    /// the trim in `record_observed_tail` matter.
1399    #[test]
1400    fn the_composed_window_matches_the_disk_after_every_refresh() {
1401        let wal = temp_wal("composed_window");
1402        let (mut journal, _) = Journal::open(&wal).unwrap();
1403        let mut overlay = ReadJournal::empty(wal.clone(), 0, None);
1404        let file = File::open(&wal).unwrap();
1405        for round in 0..12u8 {
1406            journal
1407                .append_group(&[put([round; 16], &[round; 300])])
1408                .unwrap();
1409            overlay.refresh(true).unwrap();
1410            let mut on_disk = Vec::new();
1411            assert!(Journal::read_tail_ending_at(
1412                &file,
1413                overlay.observed_valid_len,
1414                TAIL_WINDOW,
1415                &mut on_disk
1416            )
1417            .unwrap());
1418            assert_eq!(
1419                overlay.observed_tail.as_deref(),
1420                Some(on_disk.as_slice()),
1421                "round {round}: the remembered window must equal the disk"
1422            );
1423        }
1424        assert_eq!(overlay.tail_resets, 0);
1425    }
