16/// Packfile format version byte following `MAGIC` in the header (see
17/// [`write_header`]/[`read_header`]).
18///
19/// Version 4: replaces the v3 `shard_id`/`epoch` header fields with a
20/// single `pack_id: u64` — the pool-local monotonic identity for the
21/// shard. Adds `features: u32`. Filename changes from
22/// `shard_{slot:04x}_{epoch:016x}.pack` to `shard_{pack_id:016x}.pack`.
23/// Hard cutover: v3 files are rejected at open.
24pub const VERSION: u8 = 0x04;

26/// Total reserved header size in bytes: every shard file's first record
27/// starts at exactly this offset. One 4KiB page — ample collection for the
28/// descriptor fields plus future growth, with no benefit to a larger
29/// reservation (see [`write_header`] for the field layout).
30pub const HEADER_LEN: usize = 4096;
31
32/// Byte layout within the reserved header, up to where the CRC starts.
33/// Everything from `CRC_COVERED_LEN` to `HEADER_LEN` is the CRC itself
34/// (4 bytes) followed by zero padding.
35const CRC_COVERED_LEN: usize = 4 // magic
36    + 1 // version
37    + 4 // header_len (u32)
38    + 8 // pack_id (u64)
39    + 8 // created_at (u64, unix seconds)
40    + 4; // feature_flags (u32, reserved)

56/// Maximum record size (64KB). Bounds the *uncompressed* on-disk frame
57/// length (`FRAME_FIXED_LEN + data.len()`, i.e. what the frame would
58/// take up were it stored raw) — checked at write time against the
59/// caller's plaintext `data`, and at read time against the actual on-disk
60/// frame length, which never exceeds the uncompressed bound (compression
61/// is only ever used when it shrinks a frame; see [`write_record`]).
62/// Reject anything larger during recovery scan.
63pub const MAX_RECORD_LEN: u32 = 64 * 1024;
64
65/// Per-frame flag: `data` is zstd-compressed on disk; decompress to
66/// `uncompressed_len` bytes before returning it to the caller.
67///
68/// Public because inspection/recovery tooling needs to interpret the flag
69/// bits of arbitrary frames and `shard.rs`'s `mmap`-based inline frame parser
70/// (which duplicates this module's frame layout for zero-copy reads) shares
71/// the same flag bit instead of hardcoding it.
72pub const FLAG_COMPRESSED: u8 = 0x01;
73
74/// Per-frame flag: a versioned, length-delimited metadata area follows the
75/// fixed header and precedes the node bytes (see [`FrameMetadata`]).
76///
77/// Metadata is stored **outside** the (possibly compressed) node bytes so a
78/// reader can inspect identity/role without decompressing, and so generic
79/// records with no metadata are byte-for-byte unchanged. A reader that does
80/// not understand this flag MUST reject the frame rather than misparse it;
81/// the byte layout differs from the flag's absence.
82pub const FLAG_METADATA: u8 = 0x04;
83
84/// Any flag bit a v4 reader understands. A frame carrying a bit outside this
85/// mask is rejected rather than misparsed.
86const KNOWN_FLAGS: u8 = FLAG_COMPRESSED | FLAG_CRC_DISABLED | FLAG_METADATA;

88/// Version byte for the optional per-frame metadata area.
89pub const METADATA_VERSION: u8 = 0x01;

91/// TLV type: the full 256-bit logical identity digest. The value is exactly
92/// 32 bytes.
93pub const META_TAG_LOGICAL_ID: u8 = 0x01;

95/// TLV type: the optional 256-bit content digest of the stored bytes. The
96/// value is exactly 32 bytes. Distinct from [`META_TAG_LOGICAL_ID`]: a
97/// content digest is an attribute of a stored version, not an identity.
98pub const META_TAG_CONTENT_DIGEST: u8 = 0x02;
99
100/// TLV type: a role/schema identifier (UTF-8 bytes, length-delimited).
101pub const META_TAG_ROLE: u8 = 0x03;
102
103/// TLV type: the algorithm that produced [`META_TAG_CONTENT_DIGEST`]. The
104/// value is exactly 1 byte, matching [`DigestAlgorithm::id`]. Absent means the
105/// reader should assume the default algorithm (SHA-256) for older writers.
106pub const META_TAG_DIGEST_ALGORITHM: u8 = 0x04;

108/// Per-frame flag: the 4-byte checksum field holds zeros, not a real CRC32,
109/// and must not be verified by any reader. Written when the store was
110/// configured with [`ChecksumPolicy::Disabled`] — the flag, not the checksum
111/// value, is what signals "don't verify", because a real CRC32 can also
112/// legitimately be zero and a Full-mode reader must never treat a
113/// CRC-disabled frame as corrupt.
114pub const FLAG_CRC_DISABLED: u8 = 0x02;
115
116/// How much of the per-frame CRC32 this store writes and verifies.
117///
118/// Frames are always *formatted* with a 4-byte checksum field (the framing
119/// is fixed); the policy controls whether that field is a real checksum and
120/// whether readers check it:
121///
122/// * `Full` (default) — every written record gets a CRC32, every read
123///   verifies it. Catches torn writes mid-file and silent bit-rot per record.
124/// * `WriteOnly` — records keep their CRC32 (so scanning/recovery tooling
125///   can still verify them), but point-lookup reads skip the hashing pass. A
126///   corrupt-but-plausibly-framed record may be returned as data.
127/// * `Disabled` — records are written with a zero checksum and the
128///   [`FLAG_CRC_DISABLED`] flag; nothing is computed or verified anywhere.
129///   Every such frame is permanently unverifiable.
130#[derive(Clone, Copy, Debug, PartialEq, Eq)]
131pub enum ChecksumPolicy {
132    /// Every written record gets a CRC32 and every read verifies it (default).
133    Full,
134    /// Records keep their CRC32, but point-lookup reads skip hashing them.
135    WriteOnly,
136    /// Records are written with a zero checksum and [`FLAG_CRC_DISABLED`];
137    /// nothing is computed or verified anywhere.
138    Disabled,
139}
140
141impl ChecksumPolicy {
142    /// Whether frames written under this policy carry a real CRC32.
143    #[must_use]
144    pub fn computes_checksum(self) -> bool {
145        !matches!(self, Self::Disabled)
146    }
147
148    /// Whether point-lookup reads under this policy hash and compare the CRC.
149    #[must_use]
150    pub fn verifies_reads(self) -> bool {
151        matches!(self, Self::Full)
152    }
153}

175/// Open a streaming scanner for a packfile.
176///
177/// When `verify_payload` is true, payloads and CRCs are consumed and
178/// validated. When false, only frame metadata and the declared frame bounds
179/// are checked, which is appropriate for diagnostic listing.
180///
181/// # Errors
182/// Returns an I/O error if the packfile cannot be opened or its header cannot
183/// be read.
184pub fn scan_packfile_iter(path: &Path, verify_payload: bool) -> io::Result<PackfileScanner> {
185    let file = File::open(path)?;
186    let file_end = file.metadata()?.len();
187    let mut reader = BufReader::new(file);
188    let done = read_header(&mut reader)?.is_none();
189    Ok(PackfileScanner {
190        reader,
191        file_end,
192        verify_payload,
193        done,
194    })
195}

197impl Iterator for PackfileScanner {
198    type Item = io::Result<ScanEntry>;
199
200    fn next(&mut self) -> Option<Self::Item> {
201        if self.done {
202            return None;
203        }
204        let offset = match self.reader.stream_position() {
205            Ok(offset) => offset,
206            Err(error) => {
207                self.done = true;
208                return Some(Err(error));
209            }
210        };
211        let result = if self.verify_payload {
212            read_record_metadata(&mut self.reader)
213        } else {
214            read_record_metadata_skip_payload_with_end(&mut self.reader, self.file_end)
215        };
216        match result {
217            Ok(Some(metadata)) => Some(Ok((metadata.collection_id, metadata.hash, offset))),
218            Ok(None) => {
219                self.done = true;
220                None
221            }
222            Err(error) if error.kind() == io::ErrorKind::UnexpectedEof => {
223                self.done = true;
224                None
225            }
226            Err(error) => {
227                self.done = true;
228                Some(Err(error))
229            }
230        }
231    }
232}

234/// A single record in the packfile.
235///
236/// Frame layout on disk (v3):
237/// ```text
238/// [u32 len]              — byte length of everything below down to (not including) crc32, little-endian
239/// [u8 flags]             — bit 0: node bytes are zstd-compressed
240/// [u32 uncompressed_len] — plaintext length of `data`; equals the node bytes' own length when flags == 0
241/// [16-byte collection_id]      — collection this record belongs to (for shard scan recovery)
242/// [16-byte hash]         — structural hash (index-rebuild metadata only, NOT for verification)
243/// [node bytes]           — opaque node payload, zstd-compressed iff flags bit 0 is set
244/// [u32 crc32]            — CRC32 covering len + flags + uncompressed_len + collection_id + hash + node_bytes
245/// ```
246///
247/// A frame is only ever written compressed when doing so makes it
248/// smaller — otherwise `data` is stored raw with `flags == 0` — so the
249/// on-disk frame length never exceeds the plaintext one.
250///
251/// **Security invariant:** The framed hash is index-rebuild metadata only.
252/// Verification always compares against the *caller-requested* hash, never
253/// against the hash stored in the frame. The CRC covers the bytes as
254/// written (compressed, when compressed) — it's verified *before*
255/// decompression is attempted, so a corrupt frame is never fed to the
256/// decompressor.
257#[derive(Debug, Clone, PartialEq, Eq)]
258pub struct Record {
259    /// The collection this record belongs to.
260    pub collection_id: [u8; 16],
261    /// The structural hash framed alongside the record (index-rebuild metadata only).
262    pub hash: [u8; 16],
263    /// The opaque node payload.
264    pub data: Bytes,
265    /// Optional versioned per-record metadata (see [`FrameMetadata`]). `None`
266    /// for a generic frame, which is byte-for-byte identical to a v4 record
267    /// with no metadata area.
268    pub metadata: Option<FrameMetadata>,
269}
270
271/// Optional, versioned metadata carried in a frame when [`FLAG_METADATA`] is
272/// set. Stored between the fixed header and the node bytes, outside any
273/// compression, so it can be inspected without decompressing.
274///
275/// The layout on disk is:
276///
277/// ```text
278/// [u8 metadata_version]  — must equal METADATA_VERSION
279/// [u32 metadata_len]     — byte length of the TLV block below (little-endian)
280/// [TLV block]            — metadata_len bytes, each entry:
281///     [u8 tag][u32 value_len][value_len bytes]
282/// ```
283///
284/// The metadata bytes and the `metadata_len` field are inside the frame's
285/// CRC-covered region. Unknown tags are preserved by the codec so a reader
286/// built against a newer writer can round-trip fields it does not interpret.
287#[derive(Debug, Clone, PartialEq, Eq, Default)]
288pub struct FrameMetadata {
289    /// Full 256-bit logical identity digest, if the template supplies one.
290    pub logical_id: Option<[u8; 32]>,
291    /// 256-bit digest of the stored bytes for this version, if supplied.
292    pub content_digest: Option<[u8; 32]>,
293    /// The hash function that produced [`Self::content_digest`]. Defaults to
294    /// SHA-256; recorded per record so a template or configuration can choose
295    /// another 256-bit algorithm without a format break.
296    pub digest_algorithm: DigestAlgorithm,
297    /// Role/schema identifier, if supplied.
298    pub role: Option<Vec<u8>>,
299    /// Tags this build does not interpret, preserved verbatim so a
300    /// read/rewrite cycle does not discard a newer writer's fields.
301    pub unknown: Vec<(u8, Vec<u8>)>,
302}
303
304impl FrameMetadata {
305    /// Whether this metadata carries no fields at all. A frame with empty
306    /// metadata is encoded without [`FLAG_METADATA`], so it stays
307    /// byte-identical to a generic record.
308    #[must_use]
309    pub fn is_empty(&self) -> bool {
310        self.logical_id.is_none()
311            && self.content_digest.is_none()
312            && self.role.is_none()
313            && self.unknown.is_empty()
314    }
315
316    /// Encode the `[metadata_version][metadata_len][TLV...]` block.
317    ///
318    /// # Errors
319    /// Returns `InvalidInput` if the TLV block exceeds `u32::MAX` bytes.
320    pub fn encode(&self) -> io::Result<Vec<u8>> {
321        let mut tlv = Vec::new();
322        if let Some(id) = &self.logical_id {
323            push_tlv(&mut tlv, META_TAG_LOGICAL_ID, id)?;
324        }
325        if let Some(digest) = &self.content_digest {
326            push_tlv(&mut tlv, META_TAG_CONTENT_DIGEST, digest)?;
327            // The algorithm is only meaningful alongside a digest. Emit it
328            // even for the default so the on-disk record is self-describing
329            // rather than relying on a reader-side default.
330            push_tlv(
331                &mut tlv,
332                META_TAG_DIGEST_ALGORITHM,
333                &[self.digest_algorithm.id()],
334            )?;
335        }
336        if let Some(role) = &self.role {
337            push_tlv(&mut tlv, META_TAG_ROLE, role)?;
338        }
339        for (tag, value) in &self.unknown {
340            push_tlv(&mut tlv, *tag, value)?;
341        }
342        let len = u32::try_from(tlv.len())
343            .map_err(|_| io::Error::new(io::ErrorKind::InvalidInput, "frame metadata too large"))?;
344        let mut out = Vec::with_capacity(1usize.saturating_add(4).saturating_add(tlv.len()));
345        out.push(METADATA_VERSION);
346        out.extend_from_slice(&len.to_le_bytes());
347        out.extend_from_slice(&tlv);
348        Ok(out)
349    }
350
351    /// Decode a metadata block from `bytes`, returning the metadata and the
352    /// number of bytes consumed.
353    ///
354    /// # Errors
355    /// Returns `InvalidData` on an unsupported version, a truncated block, a
356    /// mismatched declared length, or a duplicate/known-tag length violation.
357    pub fn decode(bytes: &[u8]) -> io::Result<(Self, usize)> {
358        let version = *bytes
359            .first()
360            .ok_or_else(|| invalid_data("frame metadata: missing version byte"))?;
361        if version != METADATA_VERSION {
362            return Err(invalid_data(&format!(
363                "frame metadata: unsupported version {version}"
364            )));
365        }
366        let len_bytes: [u8; 4] = bytes
367            .get(1..5)
368            .and_then(|slice| slice.try_into().ok())
369            .ok_or_else(|| invalid_data("frame metadata: truncated length"))?;
370        let tlv_len = u32::from_le_bytes(len_bytes) as usize;
371        let tlv_start = 5usize;
372        let tlv_end = tlv_start
373            .checked_add(tlv_len)
374            .ok_or_else(|| invalid_data("frame metadata: length overflow"))?;
375        let tlv = bytes
376            .get(tlv_start..tlv_end)
377            .ok_or_else(|| invalid_data("frame metadata: truncated TLV block"))?;
378
379        let mut metadata = Self::default();
380        let mut cursor = 0usize;
381        while cursor < tlv.len() {
382            let tag = *tlv
383                .get(cursor)
384                .ok_or_else(|| invalid_data("frame metadata: truncated tag"))?;
385            let value_len_start = cursor
386                .checked_add(1)
387                .ok_or_else(|| invalid_data("frame metadata: cursor overflow"))?;
388            let value_len_end = cursor
389                .checked_add(5)
390                .ok_or_else(|| invalid_data("frame metadata: cursor overflow"))?;
391            let value_len_bytes: [u8; 4] = tlv
392                .get(value_len_start..value_len_end)
393                .and_then(|slice| slice.try_into().ok())
394                .ok_or_else(|| invalid_data("frame metadata: truncated value length"))?;
395            let value_len = u32::from_le_bytes(value_len_bytes) as usize;
396            let value_start = value_len_end;
397            let value_end = value_start
398                .checked_add(value_len)
399                .ok_or_else(|| invalid_data("frame metadata: value length overflow"))?;
400            let value = tlv
401                .get(value_start..value_end)
402                .ok_or_else(|| invalid_data("frame metadata: truncated value"))?;
403            match tag {
404                META_TAG_LOGICAL_ID => {
405                    metadata.logical_id = Some(expect_32(value, "logical_id")?);
406                }
407                META_TAG_CONTENT_DIGEST => {
408                    metadata.content_digest = Some(expect_32(value, "content_digest")?);
409                }
410                META_TAG_DIGEST_ALGORITHM => {
411                    let id = *value
412                        .first()
413                        .ok_or_else(|| invalid_data("frame metadata: empty digest_algorithm"))?;
414                    if value.len() != 1 {
415                        return Err(invalid_data(
416                            "frame metadata: digest_algorithm must be 1 byte",
417                        ));
418                    }
419                    metadata.digest_algorithm = DigestAlgorithm::from_id(id);
420                }
421                META_TAG_ROLE => metadata.role = Some(value.to_vec()),
422                _ => metadata.unknown.push((tag, value.to_vec())),
423            }
424            cursor = value_end;
425        }
426
427        Ok((metadata, tlv_end))
428    }
429}

431fn push_tlv(out: &mut Vec<u8>, tag: u8, value: &[u8]) -> io::Result<()> {
432    let len = u32::try_from(value.len()).map_err(|_| {
433        io::Error::new(
434            io::ErrorKind::InvalidInput,
435            "frame metadata value too large",
436        )
437    })?;
438    out.push(tag);
439    out.extend_from_slice(&len.to_le_bytes());
440    out.extend_from_slice(value);
441    Ok(())
442}

453/// Read a record length prefix, distinguishing clean EOF from a torn prefix.
454///
455/// A zero-byte read is the normal end of an append-only pack. Once even one
456/// byte of the four-byte prefix exists, however, the frame is incomplete and
457/// must surface as `UnexpectedEof` so recovery can truncate it before a later
458/// append would land after corrupt bytes.
459fn read_frame_len_prefix(reader: &mut impl Read) -> io::Result<Option<[u8; 4]>> {
460    let mut len_buf = [0u8; 4];
461    let mut bytes_read = 0;
462    while bytes_read < 4 {
463        let n = reader.read(&mut len_buf[bytes_read..])?;
464        if n == 0 {
465            if bytes_read == 0 {
466                return Ok(None);
467            }
468            return Err(io::Error::new(
469                io::ErrorKind::UnexpectedEof,
470                "torn length prefix",
471            ));
472        }
473        bytes_read = bytes_read.saturating_add(n);
474    }
475    Ok(Some(len_buf))
476}

480    /// fixed fields, plaintext node bytes, and CRC) — i.e. the size were
481    /// it written uncompressed. The actual on-disk size after

494/// Scaffold `Mmap::map`, isolating the single unsafe call in this crate.
495///
496/// # Safety
497/// This is the only `unsafe` block in the codebase. The mapping it creates
498/// is sound for the following reasons:
499///
500/// 1. **Append-only file growth.** A shard file only ever grows by appends
501///    that serialize through `put()`. `read_at` remaps when a read lands
502///    beyond the current mapping and the file has grown, so the mapping is
503///    never stale relative to the data being read.
504/// 2. **No truncation or size change.** A mapped `File` is never
505///    `set_len`-shrunk or extended after the mapping is created.
506/// 3. **Stable file descriptor.** The `Shard` struct owns the only `File`
507///    handle and is kept alive by `Arc`s, so the descriptor remains valid
508///    for the mapping's lifetime.
509///
510/// # Errors
511/// Returns `io::Error` if the file cannot be memory-mapped.
512#[allow(unsafe_code)]
513pub fn map_pack(file: &File) -> io::Result<memmap2::Mmap> {
514    // SAFETY: See the `map_pack` doc comment — append-only file, remapped on
515    // growth, never truncated or shrunk, and a stable fd for the mapping's
516    // lifetime.
517    unsafe { memmap2::Mmap::map(file) }
518}

520/// zstd compression level used for record payloads. A low level: packfile
521/// frames are small (<= [`MAX_RECORD_LEN`]) and written on the hot append
522/// path, so this favors write throughput over squeezing out the last few
523/// percent of ratio.
524#[cfg(feature = "zstd")]
525const ZSTD_LEVEL: i32 = 3;
526
527/// Compress `data` at [`ZSTD_LEVEL`], or `None` if the compressor errors (the
528/// caller then stores the frame raw). Only built with the `zstd` feature.
529#[cfg(feature = "zstd")]
530fn zstd_maybe_compress(data: &[u8]) -> Option<Vec<u8>> {
531    zstd::bulk::compress(data, ZSTD_LEVEL).ok()
532}

534/// Decompress a frame's node bytes to exactly `expected_len` (the framed
535/// `uncompressed_len`, already range-checked by the caller). Shared by the
536/// buffered [`read_record`] path and the shard zero-copy decode path.
537#[cfg(feature = "zstd")]
538pub(crate) fn zstd_decompress(node_bytes: &[u8], expected_len: usize) -> io::Result<Vec<u8>> {
539    let decompressed = zstd::bulk::decompress(node_bytes, expected_len).map_err(|e| {
540        io::Error::new(
541            io::ErrorKind::InvalidData,
542            format!("zstd decompress failed: {e}"),
543        )
544    })?;
545    if decompressed.len() != expected_len {
546        return Err(io::Error::new(
547            io::ErrorKind::InvalidData,
548            format!(
549                "decompressed length {} != framed uncompressed_len {expected_len}",
550                decompressed.len()
551            ),
552        ));
553    }
554    Ok(decompressed)
555}

572    write_record_with_options(writer, record, true)
573}

575/// Like [`write_record`], but lets the caller skip the zstd attempt
576/// entirely via `compress: false`.
577///
578/// Some pools (e.g. HAMT nodes/roots, whose bytes are dense structural
579/// hashes rather than text) never shrink under zstd — the compressor
580/// still has to run its full match-finding pass before falling back to
581/// raw storage, so a pool that never benefits pays that cost on every
582/// single write for nothing. `compress: false` skips the attempt
583/// altogether and always stores the payload raw. This never changes the
584/// on-disk frame format: a `false` caller just always takes the "doesn't
585/// shrink" branch that `compress: true` already falls back to at read
586/// time, so raw and compressed frames continue to coexist exactly as
587/// documented on [`Record`].
588///
589/// # Errors
590/// Same as [`write_record`].
591///
592/// # Panics
593/// Same as [`write_record`].
594pub fn write_record_with_options(
595    writer: &mut impl Write,
596    record: &Record,
597    compress: bool,
598) -> io::Result<u64> {
599    let buf = encode_record_with_options(record, compress, true)?;
600    writer.write_all(&buf)?;
601    Ok(buf.len() as u64)
602}

604/// Encode one record into its complete on-disk frame.
605///
606/// Kept separate from [`write_record_with_options`] so appenders which use
607/// positioned writes can retain their file cursor and avoid an `lseek` per
608/// record. The result includes the length prefix and checksum. When
609/// `write_checksum` is false the frame is emitted with a zero checksum and the
610/// [`FLAG_CRC_DISABLED`] flag, per [`ChecksumPolicy::Disabled`].
611///
612/// # Errors
613/// Returns `io::Error` if the payload is too large to frame.
614pub(crate) fn encode_record_with_options(
615    record: &Record,
616    compress: bool,
617    write_checksum: bool,
618) -> io::Result<Vec<u8>> {
619    let uncompressed_len =
620        u32::try_from(record.data.len()).expect("record payload exceeds u32::MAX");
621    let metadata_block = match &record.metadata {
622        Some(metadata) if !metadata.is_empty() => Some(metadata.encode()?),
623        _ => None,
624    };
625    let metadata_len = metadata_block.as_ref().map_or(0, |block| {
626        u32::try_from(block.len()).expect("bounded below")
627    });
628    let plaintext_frame_len = FRAME_FIXED_LEN
629        .checked_add(metadata_len)
630        .expect("record frame length exceeds u32::MAX")
631        .checked_add(uncompressed_len)
632        .expect("record frame length exceeds u32::MAX");
633    if plaintext_frame_len > MAX_RECORD_LEN {
634        return Err(io::Error::new(
635            io::ErrorKind::InvalidInput,
636            format!("record payload too large: {plaintext_frame_len} > {MAX_RECORD_LEN}"),
637        ));
638    }
639
640    #[cfg(feature = "zstd")]
641    let compressed = compress
642        .then(|| zstd_maybe_compress(&record.data))
643        .flatten();
644    #[cfg(not(feature = "zstd"))]
645    let compressed: Option<Vec<u8>> = {
646        let _ = compress;
647        None
648    };
649    let (base_flags, node_bytes): (u8, &[u8]) = match &compressed {
650        Some(c) if c.len() < record.data.len() => (FLAG_COMPRESSED, c.as_slice()),
651        _ => (0, &record.data),
652    };
653    let mut flags = if write_checksum {
654        base_flags
655    } else {
656        base_flags | FLAG_CRC_DISABLED
657    };
658    if metadata_block.is_some() {
659        flags |= FLAG_METADATA;
660    }
661
662    let frame_len = FRAME_FIXED_LEN
663        .checked_add(metadata_len)
664        .expect("bounded by MAX_RECORD_LEN")
665        .checked_add(u32::try_from(node_bytes.len()).expect("bounded by MAX_RECORD_LEN"))
666        .expect("bounded by MAX_RECORD_LEN");
667    let total_len = 4_u64.wrapping_add(u64::from(frame_len)).wrapping_add(4);
668
669    // Assemble the whole frame in one buffer and issue a single
670    // `write_all`, rather than one syscall per field. `writer` is
671    // usually a raw `File` clone (shard files are written directly,
672    // not through a `BufWriter`), so 7 separate small writes meant 7
673    // separate `write()` syscalls per record — real, avoidable
674    // overhead on the hottest path in the engine, unrelated to actual
675    // disk speed.
676    let mut buf = Vec::with_capacity(usize::try_from(total_len).unwrap_or(0));
677    buf.extend_from_slice(&frame_len.to_le_bytes());
678    buf.push(flags);
679    buf.extend_from_slice(&uncompressed_len.to_le_bytes());
680    buf.extend_from_slice(&record.collection_id);
681    buf.extend_from_slice(&record.hash);
682    if let Some(block) = &metadata_block {
683        buf.extend_from_slice(block);
684    }
685    buf.extend_from_slice(node_bytes);
686
687    // CRC covers len + flags + uncompressed_len + collection_id + hash +
688    // metadata + node_bytes, i.e. the bytes as written to disk (metadata
689    // included, node bytes compressed when compressed) — exactly `buf`'s
690    // contents so far, hashed in one pass since CRC32 over one contiguous
691    // buffer is identical to the same bytes hashed via several `update`
692    // calls. Under a period of `write_checksum == false` the field is
693    // written as zeros (with FLAG_CRC_DISABLED set above) and no hashing
694    // pass happens at all.
695    let checksum = if write_checksum {
696        let mut crc = crc32fast::Hasher::new();
697        crc.update(&buf);
698        crc.finalize()
699    } else {
700        0
701    };
702    buf.extend_from_slice(&checksum.to_le_bytes());
703
704    debug_assert_eq!(
705        buf.len() as u64,
706        total_len,
707        "encoded frame length must match its length prefix"
708    );
709    Ok(buf)
710}

712/// Read a single record from the packfile. Returns `None` on EOF.
713///
714/// # Errors
715/// Returns `io::Error` on read failure or `io::ErrorKind::InvalidData`
716/// if the record length is invalid, an unrecognized flag bit is set, the
717/// CRC check fails, or (for a compressed frame) decompression fails or
718/// yields a length other than the framed `uncompressed_len`.
719///
720/// # Panics
721/// Panics if the payload length does not fit in `usize` (always true on
722/// 64-bit targets where `usize >= 32 bits`).
723pub fn read_record(reader: &mut impl Read) -> io::Result<Option<Record>> {
724    let Some(len_buf) = read_frame_len_prefix(reader)? else {
725        return Ok(None);
726    };
727
728    let frame_len = u32::from_le_bytes(len_buf);
729    if !(FRAME_FIXED_LEN..=MAX_RECORD_LEN).contains(&frame_len) {
730        return Err(io::Error::new(
731            io::ErrorKind::InvalidData,
732            format!("invalid record length: {frame_len}"),
733        ));
734    }
735
736    let mut payload = vec![0u8; usize::try_from(frame_len).expect("u32 always fits in usize")];
737    reader.read_exact(&mut payload)?;
738
739    let mut crc_buf = [0u8; 4];
740    reader.read_exact(&mut crc_buf)?;
741
742    let flags = payload[0];
743    if flags & !KNOWN_FLAGS != 0 {
744        return Err(io::Error::new(
745            io::ErrorKind::InvalidData,
746            format!("unsupported record flags: {flags:#04x}"),
747        ));
748    }
749
750    // Verify CRC over the bytes as written (compressed, if compressed) —
751    // before any decompression is attempted, so a corrupt frame is never
752    // fed to the decompressor. Frames carrying FLAG_CRC_DISABLED hold a
753    // zero checksum and are skipped, not compared.
754    if flags & FLAG_CRC_DISABLED == 0 {
755        let mut crc = crc32fast::Hasher::new();
756        crc.update(&len_buf);
757        crc.update(&payload);
758        let expected = crc.finalize();
759        let actual = u32::from_le_bytes(crc_buf);
760        if expected != actual {
761            return Err(io::Error::new(
762                io::ErrorKind::InvalidData,
763                format!("CRC mismatch: expected {expected:08x}, got {actual:08x}"),
764            ));
765        }
766    }
767
768    let uncompressed_len = u32::from_le_bytes(payload[1..5].try_into().unwrap());
769    let mut collection_id = [0u8; 16];
770    collection_id.copy_from_slice(&payload[5..21]);
771    let mut hash = [0u8; 16];
772    hash.copy_from_slice(&payload[21..37]);
773
774    // Optional metadata sits between the fixed header and the node bytes.
775    let rest = &payload[37..];
776    let (metadata, node_bytes) = if flags & FLAG_METADATA != 0 {
777        let (metadata, consumed) = FrameMetadata::decode(rest)?;
778        let node_bytes = rest
779            .get(consumed..)
780            .ok_or_else(|| invalid_data("frame metadata consumes past payload"))?;
781        (Some(metadata), node_bytes)
782    } else {
783        (None, rest)
784    };
785
786    let data = if flags & FLAG_COMPRESSED != 0 {
787        if uncompressed_len > MAX_DATA_LEN {
788            return Err(io::Error::new(
789                io::ErrorKind::InvalidData,
790                format!("framed uncompressed_len too large: {uncompressed_len} > {MAX_DATA_LEN}"),
791            ));
792        }
793        #[cfg(feature = "zstd")]
794        {
795            Bytes::from(zstd_decompress(
796                node_bytes,
797                usize::try_from(uncompressed_len).expect("checked above"),
798            )?)
799        }
800        #[cfg(not(feature = "zstd"))]
801        {
802            return Err(io::Error::new(
803                io::ErrorKind::InvalidData,
804                "frame is zstd-compressed but this build was compiled without the `zstd` feature",
805            ));
806        }
807    } else {
808        if usize::try_from(uncompressed_len).expect("u32 always fits in usize") != node_bytes.len()
809        {
810            return Err(io::Error::new(
811                io::ErrorKind::InvalidData,
812                format!(
813                    "raw node length {} != framed uncompressed_len {uncompressed_len}",
814                    node_bytes.len()
815                ),
816            ));
817        }
818        Bytes::copy_from_slice(node_bytes)
819    };
820
821    Ok(Some(Record {
822        collection_id,
823        hash,
824        data,
825        metadata,
826    }))
827}

829/// A frame's `collection_id`/`hash` metadata, without its node payload — what
830/// [`scan_packfile`]/[`scan_packfile_from`]/[`scan_and_recover_packfile`]
831/// actually need. See [`read_record_metadata`].
832#[derive(Debug, Clone, PartialEq, Eq)]
833pub struct RecordMetadata {
834    /// The collection this record belongs to.
835    pub collection_id: [u8; 16],
836    /// The structural hash framed alongside the record.
837    pub hash: [u8; 16],
838    /// Optional per-record metadata, if the frame carried [`FLAG_METADATA`].
839    pub metadata: Option<FrameMetadata>,
840}
841
842/// Buffer size for streaming node bytes through [`read_record_metadata`]
843/// without allocating a buffer proportional to the frame's payload size.
844/// Large enough to keep syscall/CRC-update overhead low, small enough to
845/// stay a stack buffer.
846const SCAN_DISCARD_BUF_LEN: usize = 8192;
847
848/// Shared header fields parsed from the fixed portion of a frame.
849///
850/// Extracted once by [`read_frame_header`] and consumed by either
851/// [`read_record_metadata`] (CRC-verified) or
852/// [`read_record_metadata_skip_payload`] (seek-past).
853struct FrameHeader {
854    len_buf: [u8; 4],
855    frame_len: u32,
856    flags: u8,
857    fixed: [u8; FRAME_FIXED_LEN as usize],
858    collection_id: [u8; 16],
859    hash: [u8; 16],
860    /// Optional metadata parsed from the frame, if [`FLAG_METADATA`] was set.
861    metadata: Option<FrameMetadata>,
862    /// The raw on-disk metadata block (`[version][len][TLV...]`), empty when
863    /// absent. Retained so callers can feed it into the frame CRC (the block
864    /// is covered by the checksum) and subtract its length from `frame_len`
865    /// to find the node-bytes region.
866    metadata_block: Vec<u8>,
867}
868
869/// Resolve the on-disk metadata block length (`5 + tlv_len`) and bound it
870/// against the frame that must contain it, *before* the caller allocates.
871///
872/// `tlv_len` comes straight from disk. The metadata block (its 5-byte prefix
873/// plus the TLV bytes) shares the frame's post-header body with the node bytes,
874/// so it must satisfy `block_len <= frame_len - FRAME_FIXED_LEN`. Because
875/// `frame_len` is itself capped at [`MAX_RECORD_LEN`], this both rejects a
876/// frame whose metadata overruns it and caps the metadata allocation at the
877/// record-size limit instead of the `u32` field's ~4 GiB range.
878fn checked_metadata_block_len(tlv_len: u32, frame_len: u32) -> io::Result<u32> {
879    let block_len = 5u32
880        .checked_add(tlv_len)
881        .ok_or_else(|| invalid_data("frame metadata: length overflow"))?;
882    let remaining_frame_body = frame_len
883        .checked_sub(FRAME_FIXED_LEN)
884        .ok_or_else(|| invalid_data("frame metadata overruns the frame"))?;
885    if block_len > remaining_frame_body {
886        return Err(invalid_data("frame metadata exceeds frame length"));
887    }
888    Ok(block_len)
889}

891/// Read and validate the frame length prefix and fixed header, returning the
892/// parsed fields. The caller is responsible for consuming the payload and CRC
893/// (or seeking past them).
894fn read_frame_header(reader: &mut impl Read) -> io::Result<Option<FrameHeader>> {
895    let Some(len_buf) = read_frame_len_prefix(reader)? else {
896        return Ok(None);
897    };
898
899    let frame_len = u32::from_le_bytes(len_buf);
900    if !(FRAME_FIXED_LEN..=MAX_RECORD_LEN).contains(&frame_len) {
901        return Err(io::Error::new(
902            io::ErrorKind::InvalidData,
903            format!("invalid record length: {frame_len}"),
904        ));
905    }
906
907    let mut fixed = [0u8; FRAME_FIXED_LEN as usize];
908    reader.read_exact(&mut fixed)?;
909
910    let flags = fixed[0];
911    if flags & !KNOWN_FLAGS != 0 {
912        return Err(io::Error::new(
913            io::ErrorKind::InvalidData,
914            format!("unsupported record flags: {flags:#04x}"),
915        ));
916    }
917
918    // Read the optional metadata block (version byte + u32 length + TLV).
919    let (metadata, metadata_block) = if flags & FLAG_METADATA != 0 {
920        let mut prefix = [0u8; 5];
921        reader.read_exact(&mut prefix)?;
922        let version = prefix[0];
923        if version != METADATA_VERSION {
924            return Err(invalid_data(&format!(
925                "frame metadata: unsupported version {version}"
926            )));
927        }
928        let tlv_len = u32::from_le_bytes(prefix[1..5].try_into().expect("fixed slice"));
929        // Resolve and bound the metadata block *before* allocating: `tlv_len`
930        // is disk-controlled, so the allocation below must never be sized by it
931        // unchecked.
932        let block_len = checked_metadata_block_len(tlv_len, frame_len)?;
933        let mut block = vec![0u8; usize::try_from(block_len).expect("u32 fits in usize")];
934        block[..5].copy_from_slice(&prefix);
935        reader.read_exact(&mut block[5..])?;
936        let (metadata, consumed) = FrameMetadata::decode(&block)?;
937        if consumed != block.len() {
938            return Err(invalid_data(
939                "frame metadata: trailing bytes after TLV block",
940            ));
941        }
942        (Some(metadata), block)
943    } else {
944        (None, Vec::new())
945    };
946    let metadata_len = u32::try_from(metadata_block.len()).expect("bounded by frame length");
947
948    let node_region_len = frame_len
949        .checked_sub(FRAME_FIXED_LEN)
950        .and_then(|len| len.checked_sub(metadata_len))
951        .ok_or_else(|| invalid_data("frame metadata overruns the frame"))?;
952    let uncompressed_len = u32::from_le_bytes(fixed[1..5].try_into().unwrap());
953    if flags & FLAG_COMPRESSED != 0 {
954        if uncompressed_len > MAX_DATA_LEN {
955            return Err(io::Error::new(
956                io::ErrorKind::InvalidData,
957                format!("framed uncompressed_len too large: {uncompressed_len} > {MAX_DATA_LEN}"),
958            ));
959        }
960    } else if uncompressed_len != node_region_len {
961        return Err(io::Error::new(
962            io::ErrorKind::InvalidData,
963            format!(
964                "raw node length {node_region_len} != framed uncompressed_len {uncompressed_len}"
965            ),
966        ));
967    }
968
969    let mut collection_id = [0u8; 16];
970    collection_id.copy_from_slice(&fixed[5..21]);
971    let mut hash = [0u8; 16];
972    hash.copy_from_slice(&fixed[21..37]);
973
974    Ok(Some(FrameHeader {
975        len_buf,
976        frame_len,
977        flags,
978        fixed,
979        collection_id,
980        hash,
981        metadata,
982        metadata_block,
983    }))
984}

986/// Read one frame's metadata (`collection_id`, `hash`) without allocating,
987/// decompressing, or otherwise materializing its node payload — the scan
988/// path's counterpart to [`read_record`], which fully decodes a frame for
989/// callers that actually need its data.
990///
991/// Node bytes are streamed through a small fixed buffer and fed into the
992/// running CRC as they're read, then discarded — this still detects
993/// corruption in the payload region (the CRC covers the same bytes
994/// [`read_record`] verifies), but never buffers or decompresses them.
995/// Deliberately a streaming read, rather than seeking past the payload,
996/// because seeking would skip CRC verification of the region.
997///
998/// Node bytes are streamed through a small fixed buffer and fed into the
999/// running CRC as they're read, then discarded.
1000///
1001/// # Errors
1002/// Returns an I/O error for invalid length, unsupported flags, truncated
1003/// payload/CRC data, or a CRC mismatch.
1004///
1005/// # Panics
1006/// Never in practice: `read_frame_header` validates that `frame_len` is at
1007/// least `FRAME_FIXED_LEN` before the checked subtraction below.
1008pub fn read_record_metadata(reader: &mut impl Read) -> io::Result<Option<RecordMetadata>> {
1009    read_record_metadata_scratch(reader, &mut [0u8; SCAN_DISCARD_BUF_LEN])
1010}
1011
1012/// [`read_record_metadata`] with the payload-discard buffer supplied by the
1013/// caller. A scan over many frames passes one buffer for all of them: zeroing a
1014/// fresh 8 KiB for every frame costs more than parsing a small one.
1015fn read_record_metadata_scratch(
1016    reader: &mut impl Read,
1017    discard: &mut [u8; SCAN_DISCARD_BUF_LEN],
1018) -> io::Result<Option<RecordMetadata>> {
1019    let Some(header) = read_frame_header(reader)? else {
1020        return Ok(None);
1021    };
1022
1023    let mut crc = if header.flags & FLAG_CRC_DISABLED == 0 {
1024        let mut crc = crc32fast::Hasher::new();
1025        crc.update(&header.len_buf);
1026        crc.update(&header.fixed);
1027        crc.update(&header.metadata_block);
1028        Some(crc)
1029    } else {
1030        None
1031    };
1032
1033    // frame_len >= FRAME_FIXED_LEN is guaranteed by the range check above;
1034    // subtract the metadata block already consumed by read_frame_header.
1035    let mut remaining: usize = header
1036        .frame_len
1037        .checked_sub(FRAME_FIXED_LEN)
1038        .and_then(|len| len.checked_sub(u32::try_from(header.metadata_block.len()).ok()?))
1039        .expect("frame_len >= FRAME_FIXED_LEN + metadata_len, checked in read_frame_header")
1040        as usize;
1041
1042    // For a CRC-disabled frame the checksum field is zero and unused; the
1043    // payload still must be consumed to advance the stream, just without the
1044    // hashing pass.
1045    while remaining > 0 {
1046        let chunk_len = remaining.min(discard.len());
1047        reader.read_exact(&mut discard[..chunk_len])?;
1048        if let Some(crc) = &mut crc {
1049            crc.update(&discard[..chunk_len]);
1050        }
1051        remaining = remaining
1052            .checked_sub(chunk_len)
1053            .expect("chunk_len <= remaining by construction (min above)");
1054    }
1055
1056    let mut crc_buf = [0u8; 4];
1057    reader.read_exact(&mut crc_buf)?;
1058    if let Some(crc) = crc {
1059        let expected = crc.finalize();
1060        let actual = u32::from_le_bytes(crc_buf);
1061        if expected != actual {
1062            return Err(io::Error::new(
1063                io::ErrorKind::InvalidData,
1064                format!("CRC mismatch: expected {expected:08x}, got {actual:08x}"),
1065            ));
1066        }
1067    }
1068
1069    Ok(Some(RecordMetadata {
1070        collection_id: header.collection_id,
1071        hash: header.hash,
1072        metadata: header.metadata,
1073    }))
1074}

1076/// Read one frame's metadata and seek over its payload and CRC without
1077/// reading them. This is intended for diagnostic scans where payload
1078/// integrity is not being checked. The frame length, flags, and raw payload
1079/// length invariants are still validated.
1080///
1081/// # Errors
1082/// Returns an I/O error if the frame prefix or metadata cannot be read, if
1083/// the frame metadata is invalid, or if seeking past the payload fails.
1084///
1085/// # Panics
1086/// Never in practice: the fixed-size metadata buffer makes all internal
1087/// conversions and the frame-length subtraction provably valid after the
1088/// preceding bounds checks.
1089pub fn read_record_metadata_skip_payload(
1090    reader: &mut (impl Read + Seek),
1091) -> io::Result<Option<RecordMetadata>> {
1092    let payload_start = reader.stream_position()?;
1093    let file_end = reader.seek(io::SeekFrom::End(0))?;
1094    reader.seek(io::SeekFrom::Start(payload_start))?;
1095    read_record_metadata_skip_payload_with_end(reader, file_end)
1096}
1097
1098fn read_record_metadata_skip_payload_with_end(
1099    reader: &mut (impl Read + Seek),
1100    file_end: u64,
1101) -> io::Result<Option<RecordMetadata>> {
1102    let Some(header) = read_frame_header(reader)? else {
1103        return Ok(None);
1104    };
1105
1106    // The frame length excludes the four-byte CRC, while the fixed portion
1107    // includes the metadata and flags but not the node bytes or the
1108    // already-consumed metadata block.
1109    let skip = u64::from(
1110        header
1111            .frame_len
1112            .checked_sub(FRAME_FIXED_LEN)
1113            .and_then(|len| len.checked_sub(u32::try_from(header.metadata_block.len()).ok()?))
1114            .expect("frame_len >= FRAME_FIXED_LEN + metadata_len, checked in read_frame_header"),
1115    )
1116    .checked_add(4)
1117    .expect("frame payload length fits u64");
1118    let payload_start = reader.stream_position()?;
1119    let frame_end = payload_start
1120        .checked_add(skip)
1121        .ok_or_else(|| io::Error::new(io::ErrorKind::InvalidData, "record end overflows u64"))?;
1122    if frame_end > file_end {
1123        return Err(io::Error::new(
1124            io::ErrorKind::UnexpectedEof,
1125            "record payload or CRC is truncated",
1126        ));
1127    }
1128    reader.seek(io::SeekFrom::Start(frame_end))?;
1129    Ok(Some(RecordMetadata {
1130        collection_id: header.collection_id,
1131        hash: header.hash,
1132        metadata: header.metadata,
1133    }))
1134}

1136/// Write a new pack file's [`HEADER_LEN`]-byte reserved header: magic,
1137/// version, header length, `pack_id`, creation time, and a CRC over all
1138/// of the above — zero-padded to fill `HEADER_LEN`. Written once, at
1139/// creation, and never mutated again (see [`VERSION`]'s doc for why).
1140///
1141/// # Errors
1142/// Returns `io::Error` on write failure, or if the system clock is
1143/// before the Unix epoch (treated as a hard error rather than silently
1144/// recording a wrong creation time).
1145///
1146/// # Panics
1147/// Never in practice: the only internal conversion (`HEADER_LEN` as
1148/// `u32`) is a compile-time constant well within range.
1149pub fn write_header(writer: &mut impl Write, pack_id: u64) -> io::Result<()> {
1150    let created_at = std::time::SystemTime::now()
1151        .duration_since(std::time::UNIX_EPOCH)
1152        .map_err(io::Error::other)?
1153        .as_secs();
1154
1155    let mut buf = [0u8; HEADER_LEN];
1156    buf[0..4].copy_from_slice(&MAGIC);
1157    buf[4] = VERSION;
1158    let header_len = u32::try_from(HEADER_LEN).expect("HEADER_LEN fits in u32");
1159    buf[5..9].copy_from_slice(&header_len.to_le_bytes());
1160    buf[9..17].copy_from_slice(&pack_id.to_le_bytes());
1161    buf[17..25].copy_from_slice(&created_at.to_le_bytes());
1162    // buf[25..29] (feature_flags) stays zero — reserved for future use.
1163
1164    let crc = crc32fast::hash(&buf[..CRC_COVERED_LEN]);
1165    buf[CRC_COVERED_LEN..CRC_COVERED_LEN.wrapping_add(4)].copy_from_slice(&crc.to_le_bytes());
1166    // The rest of buf is already zero-initialized padding out to HEADER_LEN.
1167
1168    writer.write_all(&buf)
1169}

1171/// Read just the version byte from a packfile header (bytes 4..5, right
1172/// after the 4-byte [`MAGIC`]). Returns `Ok(None)` if the file is too
1173/// short or doesn't start with [`MAGIC`].
1174///
1175/// # Errors
1176///
1177/// Returns `io::Error` for a non-EOF read failure.
1178pub fn read_version(reader: &mut impl Read) -> io::Result<Option<u8>> {
1179    let mut magic = [0u8; 4];
1180    match reader.read_exact(&mut magic) {
1181        Ok(()) => {}
1182        Err(e) if e.kind() == io::ErrorKind::UnexpectedEof => return Ok(None),
1183        Err(e) => return Err(e),
1184    }
1185    if magic != MAGIC {
1186        return Ok(None);
1187    }
1188    let mut version = [0u8; 1];
1189    match reader.read_exact(&mut version) {
1190        Ok(()) => {}
1191        // `read_version` is deliberately only a lightweight probe.  A file
1192        // ending immediately after MAGIC is still too short to identify a
1193        // version, just like one ending part-way through MAGIC.
1194        Err(e) if e.kind() == io::ErrorKind::UnexpectedEof => return Ok(None),
1195        Err(e) => return Err(e),
1196    }
1197    Ok(Some(version[0]))
1198}

1200/// Read and validate a shard file's reserved header.
1201///
1202/// Returns `Ok(None)` only for something that genuinely isn't an mdb
1203/// packfile at all — an empty file, or one whose first bytes don't match
1204/// [`MAGIC`]. Everything else that's wrong is a real `Err`, not a quiet
1205/// `None`, specifically so a caller (or one of the `scan_*` functions,
1206/// which all propagate this via `?`) can't mistake "this store is a
1207/// format I don't understand" for "this store has no data":
1208///
1209/// - Right magic, wrong version (most notably a pre-v2 store): a
1210///   distinct, identifiable error naming the version found, not folded
1211///   into "not a packfile".
1212/// - Right magic and version, but truncated before a full `HEADER_LEN`
1213///   header, or a CRC mismatch: corruption, reported as such.
1214///
1215/// # Errors
1216/// Returns `io::Error` (`Unsupported`) if the version doesn't match
1217/// [`VERSION`], or (`InvalidData`) if the header is truncated or its CRC
1218/// doesn't match, or on I/O failure.
1219///
1220/// # Panics
1221/// Never in practice: every internal `try_into`/`from_le_bytes` slices a
1222/// fixed, in-range region of the just-read header buffer.
1223pub fn read_header(reader: &mut impl Read) -> io::Result<Option<ShardHeader>> {
1224    // Phase 1: just enough to identify the format (magic + version),
1225    // before committing to reading a full HEADER_LEN buffer — a small
1226    // pre-cutover v1 file (bare 5-byte header) can easily be shorter
1227    // than HEADER_LEN in total, and version mismatch has to be detected
1228    // regardless of how much data follows it.
1229    let mut magic = [0u8; 4];
1230    match reader.read_exact(&mut magic) {
1231        Ok(()) => {}
1232        Err(e) if e.kind() == io::ErrorKind::UnexpectedEof => return Ok(None),
1233        Err(e) => return Err(e),
1234    }
1235    if magic != MAGIC {
1236        return Ok(None);
1237    }
1238    let mut version = [0u8; 1];
1239    match reader.read_exact(&mut version) {
1240        Ok(()) => {}
1241        Err(e) if e.kind() == io::ErrorKind::UnexpectedEof => {
1242            return Err(io::Error::new(
1243                io::ErrorKind::InvalidData,
1244                "missing version byte",
1245            ));
1246        }
1247        Err(e) => return Err(e),
1248    }
1249    let version = version[0];
1250    if version != VERSION {
1251        return Err(io::Error::new(
1252            io::ErrorKind::Unsupported,
1253            format!(
1254                "packfile format version {version:#04x} is not supported by this build \
1255                 (only {VERSION:#04x}) — this looks like a pre-cutover store; \
1256                 reset or migrate it rather than opening it with this version"
1257            ),
1258        ));
1259    }
1260
1261    // Phase 2: magic and version confirmed — read the rest of the
1262    // reserved header. Running out of bytes here means a genuinely
1263    // truncated v2 header, which is corruption, not "not a packfile".
1264    let mut buf = [0u8; HEADER_LEN];
1265    buf[..4].copy_from_slice(&magic);
1266    buf[4] = version;
1267    reader.read_exact(&mut buf[5..]).map_err(|e| {
1268        if e.kind() == io::ErrorKind::UnexpectedEof {
1269            io::Error::new(
1270                io::ErrorKind::InvalidData,
1271                format!("truncated v2 shard header: shorter than {HEADER_LEN} bytes"),
1272            )
1273        } else {
1274            e
1275        }
1276    })?;
1277
1278    let header_len = u32::from_le_bytes(buf[5..9].try_into().unwrap());
1279    if header_len as usize != HEADER_LEN {
1280        // A version byte we recognize but a header length we don't is
1281        // corruption too, for the same reason as above — VERSION and
1282        // HEADER_LEN have only ever shipped together.
1283        return Err(io::Error::new(
1284            io::ErrorKind::InvalidData,
1285            format!("shard header declares length {header_len}, expected {HEADER_LEN}"),
1286        ));
1287    }
1288
1289    let expected_crc = u32::from_le_bytes(
1290        buf[CRC_COVERED_LEN..CRC_COVERED_LEN.wrapping_add(4)]
1291            .try_into()
1292            .unwrap(),
1293    );
1294    let actual_crc = crc32fast::hash(&buf[..CRC_COVERED_LEN]);
1295    if actual_crc != expected_crc {
1296        return Err(io::Error::new(
1297            io::ErrorKind::InvalidData,
1298            format!("shard header CRC mismatch: expected {expected_crc:08x}, got {actual_crc:08x}"),
1299        ));
1300    }
1301
1302    let pack_id = u64::from_le_bytes(buf[9..17].try_into().unwrap());
1303    let created_at = u64::from_le_bytes(buf[17..25].try_into().unwrap());
1304
1305    Ok(Some(ShardHeader {
1306        pack_id,
1307        created_at,
1308    }))
1309}

1311/// Open or create a packfile, writing the header if it's new.
1312///
1313/// `pack_id` is the caller's expectation for this file — derived from
1314/// its filename. On creation it's written into the new header; on
1315/// opening an existing file it's cross-checked against what the header
1316/// actually says, so a shard file that's been copied or renamed
1317/// inconsistently with its own recorded identity is caught here rather
1318/// than silently trusted.
1319///
1320/// # Errors
1321/// Returns `io::Error` on open/write failure, `io::ErrorKind::InvalidData`
1322/// if the existing header is invalid or its CRC fails, or
1323/// `io::ErrorKind::InvalidData` if the header's recorded `pack_id`
1324/// doesn't match what the filename says it should be.
1325pub fn open_packfile(path: &Path, create: bool, pack_id: u64) -> io::Result<File> {
1326    if create {
1327        let mut options = OpenOptions::new();
1328        options.read(true).create(true);
1329        // Unix flushes use positioned writes through the append-capable
1330        // handle. Windows' append access masks FILE_WRITE_DATA, which is
1331        // required by set_len during torn-tail rollback; its write path
1332        // therefore opens an ordinary writable handle instead.
1333        #[cfg(unix)]
1334        options.append(true);
1335        #[cfg(not(unix))]
1336        options.write(true);
1337        let mut file = options.open(path)?;
1338
1339        if file.metadata()?.len() == 0 {
1340            write_header(&mut file, pack_id)?;
1341            file.sync_all()?;
1342        } else {
1343            let mut reader = BufReader::new(&file);
1344            let Some(header) = read_header(&mut reader)? else {
1345                return Err(io::Error::new(
1346                    io::ErrorKind::InvalidData,
1347                    "invalid packfile header",
1348                ));
1349            };
1350            if header.pack_id != pack_id {
1351                return Err(io::Error::new(
1352                    io::ErrorKind::InvalidData,
1353                    format!(
1354                        "shard file {} identifies itself as pack_id {:#018x} in its header, \
1355                         but its filename says pack_id {pack_id:#018x} — \
1356                         copied or renamed inconsistently with its own history",
1357                        path.display(),
1358                        header.pack_id,
1359                    ),
1360                ));
1361            }
1362        }
1363        Ok(file)
1364    } else {
1365        // Read-only path: open with read-only permissions, validate
1366        // the header, never write or create.
1367        let file = File::open(path)?;
1368        let mut reader = BufReader::new(&file);
1369        let Some(header) = read_header(&mut reader)? else {
1370            return Err(io::Error::new(
1371                io::ErrorKind::InvalidData,
1372                "invalid packfile header",
1373            ));
1374        };
1375        if header.pack_id != pack_id {
1376            return Err(io::Error::new(
1377                io::ErrorKind::InvalidData,
1378                format!(
1379                    "shard file {} identifies itself as pack_id {:#018x} in its header, \
1380                     but its filename says pack_id {pack_id:#018x} — \
1381                     copied or renamed inconsistently with its own history",
1382                    path.display(),
1383                    header.pack_id,
1384                ),
1385            ));
1386        }
1387        Ok(file)
1388    }
1389}

1391/// Scan an existing packfile and return `(collection_id, hash, offset)` entries.
1392/// Used during startup to rebuild per-collection indexes from shard files.
1393///
1394/// Does **not** truncate the file — safe to call on an active shard while
1395/// concurrent appends are landing. A torn tail (a crashed write that leaves
1396/// an incomplete final frame) stops the scan gracefully and returns the
1397/// entries found before it. Anything else `read_record` rejects — an
1398/// invalid length, a CRC mismatch — is mid-file corruption, not a torn
1399/// tail, and is propagated as an error rather than silently discarded.
1400///
1401/// # Errors
1402/// Returns `io::Error` on I/O failure during open or header read, or on
1403/// any read-loop error other than `UnexpectedEof`.
1404pub fn scan_packfile(path: &Path) -> io::Result<Vec<ScanEntry>> {
1405    let file = File::open(path)?;
1406    let mut reader = BufReader::new(file);
1407    let mut entries = Vec::new();
1408
1409    if read_header(&mut reader)?.is_none() {
1410        return Ok(entries);
1411    }
1412
1413    loop {
1414        let offset = reader.stream_position()?;
1415        match read_record_metadata(&mut reader) {
1416            Ok(Some(meta)) => {
1417                entries.push((meta.collection_id, meta.hash, offset));
1418            }
1419            Ok(None) => break,
1420            Err(e) if e.kind() == io::ErrorKind::UnexpectedEof => break,
1421            Err(e) => {
1422                // Propagate real corruption (CRC mismatch, invalid length)
1423                // rather than silently discarding it — the caller decides
1424                // whether to recover or fail.
1425                return Err(e);
1426            }
1427        }
1428    }
1429
1430    Ok(entries)
1431}

1460/// Scan a packfile starting from `start_offset`, returning only records
1461/// whose byte position is >= `start_offset`. Used for incremental repack:
1462/// the caller tracks how far through each shard file it has already scanned
1463/// and only needs the newly-appended bytes.
1464///
1465/// The offset must point to a valid record boundary (the start of a `[u32
1466/// len]` frame) — typically the byte position immediately after the last
1467/// record returned by a previous scan. Passing the file's end position
1468/// returns an empty vec without error.
1469///
1470/// Unlike [`scan_and_recover_packfile`], this never truncates — a
1471/// concurrent writer may be actively appending, so mutating the file is
1472/// unsafe. Torn tails are treated as EOF (same as [`scan_packfile`]).
1473///
1474/// # Errors
1475/// Returns `io::Error` on I/O failure other than `UnexpectedEof`.
1476pub fn scan_packfile_from(path: &Path, start_offset: u64) -> io::Result<Vec<ScanEntry>> {
1477    let file = File::open(path)?;
1478    let mut reader = BufReader::new(file);
1479
1480    // Seek past the reserved header (HEADER_LEN bytes) to the first
1481    // record. If start_offset is already past the header, seek directly
1482    // there.
1483    if start_offset == 0 {
1484        if read_header(&mut reader)?.is_none() {
1485            return Ok(Vec::new());
1486        }
1487    } else {
1488        use std::io::Seek;
1489        let file_len = reader.get_ref().metadata()?.len();
1490        if start_offset > file_len {
1491            return Err(io::Error::new(
1492                io::ErrorKind::InvalidInput,
1493                format!("start_offset {start_offset} exceeds file length {file_len} (likely a stale cursor from a previous epoch)"),
1494            ));
1495        }
1496        reader.seek(io::SeekFrom::Start(start_offset))?;
1497    }
1498
1499    let mut entries = Vec::new();
1500    loop {
1501        let offset = reader.stream_position()?;
1502        match read_record_metadata(&mut reader) {
1503            Ok(Some(meta)) => {
1504                entries.push((meta.collection_id, meta.hash, offset));
1505            }
1506            Ok(None) => break,
1507            Err(e) if e.kind() == io::ErrorKind::UnexpectedEof => break,
1508            Err(e) => return Err(e),
1509        }
1510    }
1511
1512    Ok(entries)
1513}

1612/// Extract all frames belonging to `target_collection` from `source_path` and write
1613/// them verbatim into a new valid packfile at `dest_path` with header `dest_pack_id`.
1614///
1615/// Every frame's framing and CRC is verified. Frame bytes (including compression
1616/// and optional frame metadata) are copied without decompressing or re-serializing,
1617/// ensuring byte-level fidelity. If the source pack ends in a torn tail, extraction
1618/// stops cleanly at the last valid frame and records `torn_tail: true` in the returned
1619/// stats.
1620fn validate_extracted_frame(frame_buf: &[u8], frame_len: u32) -> io::Result<()> {
1621    let total_frame_len = frame_buf.len();
1622    let flags = *frame_buf
1623        .get(4)
1624        .ok_or_else(|| invalid_data("truncated frame"))?;
1625    if flags & !KNOWN_FLAGS != 0 {
1626        return Err(io::Error::new(
1627            io::ErrorKind::InvalidData,
1628            format!("unsupported record flags: {flags:#04x}"),
1629        ));
1630    }
1631
1632    if flags & FLAG_METADATA != 0 {
1633        let meta_offset = (4_usize).saturating_add(FRAME_FIXED_LEN as usize);
1634        let header_end = meta_offset.saturating_add(5).saturating_add(4);
1635        if total_frame_len < header_end {
1636            return Err(invalid_data("frame too short for metadata header"));
1637        }
1638        let version = *frame_buf
1639            .get(meta_offset)
1640            .ok_or_else(|| invalid_data("missing metadata version"))?;
1641        if version != METADATA_VERSION {
1642            return Err(invalid_data(&format!(
1643                "frame metadata: unsupported version {version}"
1644            )));
1645        }
1646        let tlv_bytes: [u8; 4] = frame_buf
1647            .get(meta_offset.saturating_add(1)..meta_offset.saturating_add(5))
1648            .and_then(|s| s.try_into().ok())
1649            .ok_or_else(|| invalid_data("truncated metadata length prefix"))?;
1650        let tlv_len = u32::from_le_bytes(tlv_bytes);
1651        let _ = checked_metadata_block_len(tlv_len, frame_len)?;
1652    }
1653
1654    if flags & FLAG_CRC_DISABLED == 0 {
1655        let crc_start = total_frame_len.saturating_sub(4);
1656        let expected_bytes: [u8; 4] = frame_buf
1657            .get(crc_start..total_frame_len)
1658            .and_then(|s| s.try_into().ok())
1659            .ok_or_else(|| invalid_data("truncated CRC"))?;
1660        let expected_crc = u32::from_le_bytes(expected_bytes);
1661        let actual_crc = crc32fast::hash(&frame_buf[..crc_start]);
1662        if actual_crc != expected_crc {
1663            return Err(io::Error::new(
1664                io::ErrorKind::InvalidData,
1665                format!(
1666                    "frame CRC mismatch: expected {expected_crc:#010x}, got {actual_crc:#010x}"
1667                ),
1668            ));
1669        }
1670    }
1671
1672    Ok(())
1673}

1734                format!("invalid record length: {frame_len}"),
1735            ));
1736        }
1737
1738        let total_frame_len = (frame_len as usize).saturating_add(8);
1739        frame_buf.resize(total_frame_len, 0);
1740        frame_buf[0..4].copy_from_slice(&len_buf);
1741
1742        match reader.read_exact(&mut frame_buf[4..total_frame_len]) {
1743            Ok(()) => {}
1744            Err(e) if e.kind() == io::ErrorKind::UnexpectedEof => {
1745                stats.torn_tail = true;
1746                break;
1747            }
1748            Err(e) => return Err(e),
1749        }
1750
1751        validate_extracted_frame(&frame_buf, frame_len)?;
1752
1753        stats.frames_scanned = stats.frames_scanned.saturating_add(1);
1754
1755        let mut frame_collection = [0u8; 16];
1756        frame_collection.copy_from_slice(&frame_buf[9..25]);
1757
1758        if frame_collection == *target_collection {
1759            writer.write_all(&frame_buf)?;
1760            stats.frames_extracted = stats.frames_extracted.saturating_add(1);
1761            stats.frame_bytes_written = stats
1762                .frame_bytes_written
1763                .saturating_add(u64::try_from(total_frame_len).unwrap_or(u64::MAX));
1764        }
1765    }
1766
1767    writer.flush()?;
1768    writer.get_ref().sync_all()?;
1769
1770    Ok(stats)
1771}

1773/// Scan a packfile and truncate any torn tail at the last valid record
1774/// boundary. Used during explicit recovery to repair a packfile before
1775/// reopening for append.
1776///
1777/// Only truncates on `UnexpectedEof` (a torn tail from a crashed write).
1778/// Other errors (CRC mismatch, invalid length) are propagated without
1779/// truncating — mid-file corruption should not discard valid records
1780/// that follow the corrupt frame.
1781///
1782/// # Errors
1783/// Returns `io::Error` on read or truncate failure.
1784pub fn scan_and_recover_packfile(path: &Path) -> io::Result<Vec<ScanEntry>> {
1785    let mut entries = Vec::new();
1786    recover_packfile_with(path, |entry| entries.push(entry))?;
1787    Ok(entries)
1788}
1789
1790/// What [`recover_packfile`] found in one pack.
1791#[derive(Debug, Clone, Copy, PartialEq, Eq)]
1792pub struct PackRecovery {
1793    /// Complete, CRC-valid records in the pack.
1794    pub records: u64,
1795    /// Offset just past the last valid record (the header's end for an empty pack).
1796    pub valid_len: u64,
1797    /// Whether a torn tail was cut off at `valid_len`.
1798    pub truncated: bool,
1799}

1801/// [`scan_and_recover_packfile`] for a caller that only needs the pack made
1802/// safe to append to: the same validation of every frame and CRC, and the same
1803/// truncation of a torn tail, but no per-record result is built.
1804///
1805/// # Errors
1806/// As [`scan_and_recover_packfile`].
1807pub fn recover_packfile(path: &Path) -> io::Result<PackRecovery> {
1808    recover_packfile_with(path, |_| {})
1809}
1810
1811/// Buffer for a recovery scan: a few large sequential reads, not one per 8 KiB.
1812const RECOVERY_BUFFER_BYTES: usize = 1 << 20;
1813
1814/// A reader that counts the bytes it hands out, so the scan knows each frame's
1815/// offset without asking the file for its position (an `lseek` system call per
1816/// frame on a buffered reader).
1817struct CountingReader<R> {
1818    inner: R,
1819    position: u64,
1820}
1821
1822impl<R: Read> Read for CountingReader<R> {
1823    fn read(&mut self, buf: &mut [u8]) -> io::Result<usize> {
1824        let read = self.inner.read(buf)?;
1825        self.position = self
1826            .position
1827            .saturating_add(u64::try_from(read).unwrap_or(u64::MAX));
1828        Ok(read)
1829    }
1830}
1831
1832/// The shared recovery walk: validates every frame, calls `on_record` for each
1833/// valid one, and truncates a torn tail.
1834fn recover_packfile_with(
1835    path: &Path,
1836    mut on_record: impl FnMut(ScanEntry),
1837) -> io::Result<PackRecovery> {
1838    let file = File::open(path)?;
1839    let mut reader = CountingReader {
1840        inner: BufReader::with_capacity(RECOVERY_BUFFER_BYTES, file),
1841        position: 0,
1842    };
1843    let mut recovery = PackRecovery {
1844        records: 0,
1845        valid_len: 0,
1846        truncated: false,
1847    };
1848
1849    if read_header(&mut reader)?.is_none() {
1850        return Ok(recovery);
1851    }
1852
1853    recovery.valid_len = reader.position;
1854    let mut scratch = [0u8; SCAN_DISCARD_BUF_LEN];
1855    loop {
1856        let offset = reader.position;
1857        match read_record_metadata_scratch(&mut reader, &mut scratch) {
1858            Ok(Some(meta)) => {
1859                on_record((meta.collection_id, meta.hash, offset));
1860                recovery.records = recovery.records.saturating_add(1);
1861                recovery.valid_len = reader.position;
1862            }
1863            Ok(None) => break,
1864            Err(e) if e.kind() == io::ErrorKind::UnexpectedEof => {
1865                // Torn tail: a write crashed mid-frame. Truncate to the
1866                // last valid record boundary so future appends don't land
1867                // after a corrupt frame.
1868                recovery.truncated = true;
1869                break;
1870            }
1871            Err(e) => {
1872                // Real corruption (CRC mismatch, invalid length) — propagate
1873                // without truncating. Later valid records may exist past the
1874                // corrupt frame; truncating would lose them.
1875                return Err(e);
1876            }
1877        }
1878    }
1879
1880    // Only truncate if we actually hit a torn tail — not on clean EOF
1881    // or mid-file corruption (which was propagated as an error above).
1882    if recovery.truncated {
1883        drop(reader);
1884        let file = OpenOptions::new().write(true).open(path)?;
1885        file.set_len(recovery.valid_len)?;
1886        file.sync_all()?;
1887    }
1888
1889    Ok(recovery)
1890}
