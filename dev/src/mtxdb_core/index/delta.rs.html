1//! The incremental index delta log (`index.delta`).
2//!
3//! Packfiles stay authoritative and the checkpoint stays a rebuildable
4//! acceleration structure; this log is a further acceleration on top of the
5//! checkpoint. Between full checkpoint rewrites, a session's index changes are
6//! recorded as fixed-width [`DeltaFrame`] records so that `sync()` can append
7//! a few kilobyte batch instead of rewriting the whole checkpoint. A crash at
8//! any point degrades to the fingerprint gates below and an ordinary rescan.
9//!
10//! # Trusted gates (all must pass, else the log is ignored and the opener
11//! falls back to a full rescan / the next sync rewrites the checkpoint):
12//!
13//! - The log's `base_fingerprint` must equal the checkpoint it continues.
14//! - The log's `tail_fingerprint` must equal the fingerprint of the packs
15//!   currently on disk (the state the frames were written against).
16//! - Every frame's `generation` must match the checkpoint's recorded
17//!   generation for that collection (guards against a delta for a pre-resize
18//!   table being applied to a resized/repacked one).
19//!
20//! # Layout (all little-endian, fixed width — see [`crate::index::format`]):
21//!
22//! ```text
23//!   [DeltaLogHeader 16B]       magic "MDLG" | version | reserved | base_fingerprint
24//!   [DeltaBatchHeader 8B]      magic "MDLB" | frame_count
25//!   [DeltaFrame * frame_count] fixed 36B records
26//!   [DeltaLogTrailer 16B]      magic "DLTR" | reserved | tail_fingerprint
27//!   [DeltaBatchHeader ..]*     -- further batches, each count-framed and trailer-terminated
28//! ```
29//!
30//! Framing is arithmetic, not magic-sniffed: a reader advances by exactly
31//! `frame_count * DELTA_FRAME_LEN` bytes from each batch header, and only then
32//! expects the trailer. A batch header's `frame_count` (not any record's
33//! payload bytes) determines where the next record begins — a collection ID
34//! that happens to spell a magic value can never misdirect the parser. A torn
35//! suffix is exactly "not enough bytes remain for `frame_count` frames plus a
36//! trailer", and only complete batches are trusted.
37

45/// File name of the incremental index delta log inside a store's base dir.
46pub const INDEX_DELTA_FILE: &str = "index.delta";
47/// Bytes in one log header: magic(4) + version(1) + reserved(3) + base fingerprint(8).
48pub const DELTA_LOG_HEADER_LEN: usize = 16;
49/// Bytes in one batch header: magic(4) + `frame_count`(4). Determines the
50/// number of following 36-byte frames arithmetically.
51pub const DELTA_BATCH_HEADER_LEN: usize = 8;
52/// Bytes in one batch trailer: magic(4) + reserved(4) + tail fingerprint(8).
53pub const DELTA_LOG_TRAILER_LEN: usize = 16;
54
55/// Hard ceiling on the delta log file size this reader will allocate for.
56/// The writer (`storage.rs`) forces a full checkpoint rewrite well before
57/// this — its own cap is 8 MiB — so a well-behaved log never approaches this
58/// value; it exists purely so a corrupted or maliciously oversized
59/// `index.delta` (e.g. a truncated-looking length field, or the file
60/// replaced wholesale) can't make an opener allocate an unbounded buffer
61/// before any structural validation runs. Generous relative to the writer's
62/// own cap so it never rejects a real log.
63const MAX_DELTA_LOG_FILE_BYTES: u64 = 64 * 1024 * 1024;
64
65const DELTA_LOG_MAGIC: &[u8; 4] = b"MDLG";
66/// Current wire version (see the header's version byte).
67///
68/// v2 repurposes the trailer's 4 reserved bytes as a CRC32 covering the
69/// batch's frame bytes and its tail fingerprint, so a batch corrupted after
70/// being written — whether in the frames or just the fingerprint — but which
71/// still happens to pass the fingerprint/generation gates in `storage.rs` is
72/// caught and dropped at read time instead of being replayed as valid index
73/// state. A v1 log fails the version check below and falls back to a full
74/// rescan, same as any other structurally rejected log.
75const DELTA_LOG_VERSION: u8 = 2;
76const DELTA_BATCH_MAGIC: &[u8; 4] = b"MDLB";
77const DELTA_LOG_TRAILER_MAGIC: &[u8; 4] = b"DLTR";
78
79const BASE_FINGERPRINT_OFFSET: usize = 8;
80const TRAILER_FINGERPRINT_OFFSET: usize = 8;
81/// Offset of the batch's frame-bytes CRC32 within the trailer's reserved
82/// field (bytes 4..8, between the magic and the tail fingerprint).
83const TRAILER_CRC_OFFSET: usize = 4;
84
85/// A decoded delta log that passed its structural gates.
86#[derive(Debug)]
87pub struct DeltaLog {
88    /// The checkpoint fingerprint this log continues from.
89    pub base_fingerprint: u64,
90    /// Every committed frame in file order (torn trailing batches are absent).
91    pub frames: Vec<DeltaFrame>,
92    /// The pack fingerprint the final committed batch was written against.
93    pub tail_fingerprint: u64,
94    /// Byte length of the committed region in the log file (through the last
95    /// committed trailer). A session continuing this log appends at this
96    /// boundary — it is the answer to "has a base header already been
97    /// written", independent of any fresh-session default.
98    pub file_len: u64,
99    /// Whether a torn tail was encountered after the last committed batch.
100    pub torn_tail: bool,
101}

103/// Encode the fixed-width log header for `base_fingerprint`.
104#[must_use]
105pub fn encode_header(base_fingerprint: u64) -> [u8; DELTA_LOG_HEADER_LEN] {
106    let mut bytes = [0u8; DELTA_LOG_HEADER_LEN];
107    bytes[..4].copy_from_slice(DELTA_LOG_MAGIC);
108    bytes[4] = DELTA_LOG_VERSION;
109    bytes[BASE_FINGERPRINT_OFFSET..16].copy_from_slice(&base_fingerprint.to_le_bytes());
110    bytes
111}
112
113/// Encode the fixed-width batch header for `frame_count`.
114#[must_use]
115pub fn encode_batch_header(frame_count: u32) -> [u8; DELTA_BATCH_HEADER_LEN] {
116    let mut bytes = [0u8; DELTA_BATCH_HEADER_LEN];
117    bytes[..4].copy_from_slice(DELTA_BATCH_MAGIC);
118    bytes[4..8].copy_from_slice(&frame_count.to_le_bytes());
119    bytes
120}
121
122/// Encode the fixed-width batch trailer for `tail_fingerprint`, with `crc`
123/// the CRC32 of the batch's frame bytes and `tail_fingerprint` together
124/// (stored at byte offset 4 of the trailer's reserved field).
125#[must_use]
126pub fn encode_trailer(tail_fingerprint: u64, crc: u32) -> [u8; DELTA_LOG_TRAILER_LEN] {
127    let mut bytes = [0u8; DELTA_LOG_TRAILER_LEN];
128    bytes[..4].copy_from_slice(DELTA_LOG_TRAILER_MAGIC);
129    bytes[TRAILER_CRC_OFFSET..8].copy_from_slice(&crc.to_le_bytes());
130    bytes[TRAILER_FINGERPRINT_OFFSET..16].copy_from_slice(&tail_fingerprint.to_le_bytes());
131    bytes
132}
133
134/// CRC32 covering one batch's frame bytes *and* its `tail_fingerprint`, as
135/// stored in the trailer's reserved field. Folding the fingerprint into the
136/// CRC means fingerprint-only corruption (the frame bytes untouched) is also
137/// caught here, rather than relying solely on the fingerprint gate in
138/// `storage.rs` matching it against the wrong thing.
139#[must_use]
140fn batch_crc(frames_bytes: &[u8], tail_fingerprint: u64) -> u32 {
141    let mut hasher = crc32fast::Hasher::new();
142    hasher.update(frames_bytes);
143    hasher.update(&tail_fingerprint.to_le_bytes());
144    hasher.finalize()
145}

147/// Validated result of a lightweight delta-log scan.
148#[derive(Debug, Clone, Copy)]
149pub struct DeltaTailFingerprint {
150    /// The `base_fingerprint` from the log header (matches the checkpoint
151    /// that this delta continues).
152    pub base_fingerprint: u64,
153    /// The last committed `tail_fingerprint` from the validated batches.
154    pub tail_fingerprint: u64,
155    /// Whether a torn tail was encountered after the last committed batch.
156    /// `true` means the file ended mid-batch (expected if writer crashed);

159}

161/// Failure while inspecting a delta log's durable fingerprint.
162#[derive(Debug)]
163pub enum DeltaFingerprintError {
164    /// The delta log could not be accessed or read.
165    Io {
166        /// Delta-log path involved in the operation.
167        path: PathBuf,
168        /// Operation being performed when the I/O error occurred.
169        operation: &'static str,
170        /// Underlying operating-system error.
171        source: io::Error,
172    },
173    /// The delta log has an invalid or unsupported on-disk structure.
174    Invalid {
175        /// Delta-log path involved in the validation failure.
176        path: PathBuf,
177        /// Human-readable validation failure.
178        reason: String,
179    },
180}
181
182impl fmt::Display for DeltaFingerprintError {
183    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
184        match self {
185            Self::Io {
186                path,
187                operation,
188                source,
189            } => write!(formatter, "{} {}: {source}", operation, path.display()),
190            Self::Invalid { path, reason } => {
191                write!(
192                    formatter,
193                    "invalid delta fingerprint {}: {reason}",
194                    path.display()
195                )
196            }
197        }
198    }
199}

222/// Lightweight forward scan of the delta log's last committed
223/// `tail_fingerprint`.
224///
225/// Walks each batch from the log header forward — reading batch headers,
226/// computing frame/trailer boundaries with checked arithmetic, and
227/// verifying CRC — retaining the last valid `tail_fingerprint`.  Stops at
228/// the first torn or corrupt suffix.  This avoids decoding individual
229/// [`DeltaFrame`] records while still validating every batch boundary.
230///
231/// When a torn trailing batch is encountered the function returns the
232/// last _valid_ `tail_fingerprint` (from a prior committed batch), which
233/// is the correct durable generation for the committed prefix.  Returns
234/// `None` only when the file is missing, too short, has an unrecognized
235/// magic/version, or has no committed batches at all. A missing file is
236/// returned as `Ok(None)` so callers can fall back to the checkpoint.
237///
238/// # Errors
239///
240/// Returns [`DeltaFingerprintError`] when the file cannot be read or its
241/// committed structure is invalid.
242#[allow(
243    clippy::too_many_lines,
244    reason = "parses the fixed-width log format in one pass"
245)]
246pub fn read_delta_tail_fingerprint(
247    path: &Path,
248) -> Result<Option<DeltaTailFingerprint>, DeltaFingerprintError> {
249    let mut file = match fs::File::open(path) {
250        Ok(file) => file,
251        Err(error) if error.kind() == io::ErrorKind::NotFound => return Ok(None),
252        Err(error) => return Err(io_error(path, "open delta log", error)),
253    };
254    let len = file
255        .metadata()
256        .map_err(|error| io_error(path, "stat delta log", error))?
257        .len();
258    let log_header_len = u64::try_from(DELTA_LOG_HEADER_LEN)
259        .map_err(|_| invalid(path, "log header length overflows u64"))?;
260    let batch_header_len = u64::try_from(DELTA_BATCH_HEADER_LEN)
261        .map_err(|_| invalid(path, "batch header length overflows u64"))?;
262    let trailer_len = u64::try_from(DELTA_LOG_TRAILER_LEN)
263        .map_err(|_| invalid(path, "trailer length overflows u64"))?;
264    let delta_frame_len =
265        u64::try_from(DELTA_FRAME_LEN).map_err(|_| invalid(path, "frame length overflows u64"))?;
266
267    // Reject files that exceed the writer's own cap.
268    if len > MAX_DELTA_LOG_FILE_BYTES {
269        return Err(invalid(path, "file exceeds maximum delta-log size"));
270    }
271
272    // --- log header ---
273    if len < log_header_len {
274        return Err(invalid(path, "truncated log header"));
275    }
276    let mut log_hdr = [0u8; DELTA_LOG_HEADER_LEN];
277    file.seek(std::io::SeekFrom::Start(0))
278        .map_err(|error| io_error(path, "seek delta log header", error))?;
279    file.read_exact(&mut log_hdr)
280        .map_err(|error| io_error(path, "read delta log header", error))?;
281    if &log_hdr[..4] != DELTA_LOG_MAGIC || log_hdr[4] != DELTA_LOG_VERSION {
282        return Err(invalid(path, "unrecognized magic or version"));
283    }
284    let base_fp = u64::from_le_bytes(
285        log_hdr[8..16]
286            .try_into()
287            .map_err(|_| invalid(path, "invalid base fingerprint"))?,
288    );
289
290    let mut pos = log_header_len;
291    let mut tail_fingerprint: Option<u64> = None;
292
293    loop {
294        // Enough room for a batch header + trailer?
295        let batch_header_end = pos
296            .checked_add(batch_header_len)
297            .ok_or_else(|| invalid(path, "batch header offset overflow"))?;
298        if batch_header_end
299            .checked_add(trailer_len)
300            .ok_or_else(|| invalid(path, "batch trailer boundary overflow"))?
301            > len
302        {
303            break;
304        }
305        // --- batch header ---
306        let mut batch_hdr = [0u8; DELTA_BATCH_HEADER_LEN];
307        file.seek(std::io::SeekFrom::Start(pos))
308            .map_err(|error| io_error(path, "seek batch header", error))?;
309        file.read_exact(&mut batch_hdr)
310            .map_err(|error| io_error(path, "read batch header", error))?;
311        if &batch_hdr[..4] != DELTA_BATCH_MAGIC {
312            return match tail_fingerprint {
313                Some(tail_fingerprint) => Ok(Some(DeltaTailFingerprint {
314                    base_fingerprint: base_fp,
315                    tail_fingerprint,
316                    torn_tail: pos < len,
317                })),
318                None => Err(invalid(
319                    path,
320                    format!("invalid batch header at offset {pos}"),
321                )),
322            };
323        }
324        let frame_count = u64::from(u32::from_le_bytes(
325            batch_hdr[4..8]
326                .try_into()
327                .map_err(|_| invalid(path, "invalid frame count"))?,
328        ));
329        let frame_bytes_len = frame_count
330            .checked_mul(delta_frame_len)
331            .ok_or_else(|| invalid(path, "frame byte length overflow"))?;
332        let trailer_start = batch_header_end
333            .checked_add(frame_bytes_len)
334            .ok_or_else(|| invalid(path, "trailer offset overflow"))?;
335        if trailer_start
336            .checked_add(trailer_len)
337            .ok_or_else(|| invalid(path, "trailer boundary overflow"))?
338            > len
339        {
340            // Torn tail: not enough bytes for complete batch + trailer
341            break;
342        }
343        // --- trailer ---
344        let mut trailer = [0u8; DELTA_LOG_TRAILER_LEN];
345        file.seek(std::io::SeekFrom::Start(trailer_start))
346            .map_err(|error| io_error(path, "seek delta trailer", error))?;
347        file.read_exact(&mut trailer)
348            .map_err(|error| io_error(path, "read delta trailer", error))?;
349        if &trailer[..4] != DELTA_LOG_TRAILER_MAGIC {
350            break;
351        }
352        let tfp = u64::from_le_bytes(
353            trailer[TRAILER_FINGERPRINT_OFFSET..16]
354                .try_into()
355                .map_err(|_| invalid(path, "invalid trailer fingerprint"))?,
356        );
357        let stored_crc = u32::from_le_bytes(
358            trailer[TRAILER_CRC_OFFSET..8]
359                .try_into()
360                .map_err(|_| invalid(path, "invalid trailer CRC"))?,
361        );
362        // --- frame bytes for CRC ---
363        let frames_start = batch_header_end;
364        let frames_len_usize = usize::try_from(frame_bytes_len)
365            .map_err(|_| invalid(path, "frame bytes do not fit in memory"))?;
366        let mut frames = vec![0u8; frames_len_usize];
367        file.seek(std::io::SeekFrom::Start(frames_start))
368            .map_err(|error| io_error(path, "seek delta frames", error))?;
369        file.read_exact(&mut frames)
370            .map_err(|error| io_error(path, "read delta frames", error))?;
371        if batch_crc(&frames, tfp) != stored_crc {
372            if tail_fingerprint.is_some() {
373                break;
374            }
375            return Err(invalid(path, format!("invalid batch CRC at offset {pos}")));
376        }
377        tail_fingerprint = Some(tfp);
378        pos = trailer_start
379            .checked_add(trailer_len)
380            .ok_or_else(|| invalid(path, "next batch offset overflow"))?;
381    }
382    // Determine if we stopped due to a torn tail (insufficient bytes for next batch)
383    // or because we reached EOF cleanly after a complete batch.
384    let torn_tail = pos < len;
385    match tail_fingerprint {
386        Some(tail_fingerprint) => Ok(Some(DeltaTailFingerprint {
387            base_fingerprint: base_fp,
388            tail_fingerprint,
389            torn_tail,
390        })),
391        None => Err(invalid(path, "no committed delta batch")),
392    }
393}

395/// Length in bytes of one fully framed batch carrying `frame_count` frames.
396#[must_use]
397pub fn batch_len(frame_count: usize) -> Option<usize> {
398    DELTA_BATCH_HEADER_LEN
399        .checked_add(frame_count.checked_mul(DELTA_FRAME_LEN)?)?
400        .checked_add(DELTA_LOG_TRAILER_LEN)
401}

403/// Read and structurally validate the delta log. Returns `None` for a missing
404/// file, a bad header, a log with no complete committed batch (entirely torn),
405/// or any other structural inconsistency — the caller treats that as "no
406/// delta".
407///
408/// Batch boundaries come from each batch header's `frame_count`, never from
409/// scanning for magic, so no frame payload can redirect the parse. A torn
410/// trailing batch (fewer bytes than `frame_count` frames + a trailer) is
411/// dropped; everything up to the last complete batch is returned.
412#[must_use]
413pub fn read_delta_log(path: &Path) -> Option<DeltaLog> {
414    // Check the size before allocating and then read from the *same* handle:
415    // `metadata` + a separate `fs::read` would size the buffer off the
416    // metadata call, letting the file be swapped or grown between the two so
417    // the allocation exceeds the limit. Opening once pins the inode and make
418    // the measured length and the bytes read describe the same file; a
419    // bounded read from that handle then cannot allocate past
420    // `MAX_DELTA_LOG_FILE_BYTES`.
421    use std::io::Read;
422    let mut file = fs::File::open(path).ok()?;
423    let len = file.metadata().ok()?.len();
424    if len > MAX_DELTA_LOG_FILE_BYTES {
425        return None;
426    }
427    let mut buf = vec![0u8; usize::try_from(len).ok()?];
428    file.read_exact(&mut buf).ok()?;
429    if buf.len() < DELTA_LOG_HEADER_LEN {
430        return None;
431    }
432    if &buf[..4] != DELTA_LOG_MAGIC || buf[4] != DELTA_LOG_VERSION {
433        return None;
434    }
435    let base_fingerprint = u64::from_le_bytes(buf[BASE_FINGERPRINT_OFFSET..16].try_into().ok()?);
436
437    let mut frames: Vec<DeltaFrame> = Vec::new();
438    let mut tail_fingerprint: Option<u64> = None;
439    let mut offset = DELTA_LOG_HEADER_LEN;
440    while offset < buf.len() {
441        if buf.len().saturating_sub(offset) < DELTA_BATCH_HEADER_LEN {
442            break;
443        }
444        if &buf[offset..offset.saturating_add(4)] != DELTA_BATCH_MAGIC {
445            break;
446        }
447        let Some(header_count) = buf.get(offset.saturating_add(4)..offset.saturating_add(8)) else {
448            break;
449        };
450        let frame_count = u32::from_le_bytes(header_count.try_into().ok()?) as usize;
451        let Some(batch_len) = batch_len(frame_count) else {
452            break;
453        };
454        let Some(batch_end) = offset.checked_add(batch_len) else {
455            break;
456        };
457        if batch_end > buf.len() {
458            // Torn final batch: fewer bytes than frame_count frames + trailer.
459            break;
460        }
461        let frames_start = offset.saturating_add(DELTA_BATCH_HEADER_LEN);
462        let trailer_start = batch_end.saturating_sub(DELTA_LOG_TRAILER_LEN);
463        if &buf[trailer_start..trailer_start.saturating_add(4)] != DELTA_LOG_TRAILER_MAGIC {
464            break;
465        }
466        let frames_bytes = &buf[frames_start..trailer_start];
467        let stored_crc = u32::from_le_bytes(
468            buf[trailer_start.saturating_add(TRAILER_CRC_OFFSET)..trailer_start.saturating_add(8)]
469                .try_into()
470                .ok()?,
471        );
472        let batch_tail_fingerprint = u64::from_le_bytes(
473            buf[trailer_start.saturating_add(TRAILER_FINGERPRINT_OFFSET)
474                ..trailer_start.saturating_add(16)]
475                .try_into()
476                .ok()?,
477        );
478        if batch_crc(frames_bytes, batch_tail_fingerprint) != stored_crc {
479            // The frame bytes or the tail fingerprint were corrupted after
480            // being written (bit rot, torn/partial write not otherwise
481            // caught by the length framing, etc). Trust nothing from this
482            // batch onward — same as a torn trailer, everything already
483            // accumulated from earlier committed batches is kept, but this
484            // and any later batch are dropped rather than replayed as valid
485            // index state.
486            break;
487        }
488        for frame_bytes in frames_bytes.chunks_exact(DELTA_FRAME_LEN) {
489            frames.push(DeltaFrame::decode(frame_bytes)?);
490        }
491        tail_fingerprint = Some(batch_tail_fingerprint);
492        offset = batch_end;
493    }
494    let torn_tail = offset < buf.len();
495    Some(DeltaLog {
496        base_fingerprint,
497        frames,
498        tail_fingerprint: tail_fingerprint?,
499        // `offset` is the frontier of the last committed batch: the loop only
500        // breaks past a committed trailer or before a torn/unparseable tail.
501        file_len: u64::try_from(offset).unwrap_or(u64::MAX),
502        torn_tail,
503    })
504}

506/// Append one batch (optional header, then a count-framed batch) to the delta
507/// log and fsync it. Returns the number of bytes appended. `write_header` must
508/// be `true` when writing to a freshly created (or truncated) file that
509/// doesn't yet carry the base header.
510///
511/// # Errors
512/// Returns `io::Error` on any write or fsync failure.
513pub fn append_batch(
514    path: &Path,
515    write_header: bool,
516    base_fingerprint: u64,
517    frames: &[DeltaFrame],
518    tail_fingerprint: u64,
519) -> std::io::Result<usize> {
520    let frame_count = u32::try_from(frames.len())
521        .map_err(|_| std::io::Error::other("delta batch exceeds u32 frame count"))?;
522    let mut appended = 0usize;
523    let mut file = fs::OpenOptions::new()
524        .create(true)
525        .append(true)
526        .open(path)?;
527    if write_header {
528        file.write_all(&encode_header(base_fingerprint))?;
529        appended = appended.saturating_add(DELTA_LOG_HEADER_LEN);
530    }
531    file.write_all(&encode_batch_header(frame_count))?;
532    appended = appended.saturating_add(DELTA_BATCH_HEADER_LEN);
533    let mut frames_bytes = Vec::with_capacity(frames.len().saturating_mul(DELTA_FRAME_LEN));
534    for frame in frames {
535        frames_bytes.extend_from_slice(&frame.encode());
536    }
537    file.write_all(&frames_bytes)?;
538    appended = appended.saturating_add(frames_bytes.len());
539    file.write_all(&encode_trailer(
540        tail_fingerprint,
541        batch_crc(&frames_bytes, tail_fingerprint),
542    ))?;
543    appended = appended.saturating_add(DELTA_LOG_TRAILER_LEN);
544    file.sync_all()?;
545    Ok(appended)
546}

548/// Errors that can arise while applying a replayed delta log to a
549/// checkpoint-backed index. Any of these means the log is structurally
550/// inconsistent with the checkpoint it claims to continue, and the whole log
551/// must be rejected (opener falls back to a rescan).
552#[derive(Debug, PartialEq, Eq)]
553pub enum DeltaReplayError {
554    /// A frame targets a bucket outside the checkpoint index's capacity.
555    FrameOutOfBounds {
556        /// The frame's target bucket.
557        bucket: u32,
558        /// The checkpoint index's capacity.
559        capacity: u32,
560    },
561    /// Replay requires an owned (materialized) slot array.
562    RequiresOwnedIndex,
563    /// A frame carried the empty-slot sentinel (`0`). A legitimate insert
564    /// never produces one — `record_delta` only ever logs a freshly built,
565    /// non-empty `IndexSlot` — so this can only be log corruption or a
566    /// structurally invalid frame. Storing it as-is would silently erase
567    /// whatever live entry currently occupies that bucket and truncate the
568    /// probe chain past it, stranding any entries beyond it.
569    EmptySlot {
570        /// The frame's target bucket.
571        bucket: u32,
572    },
573}
574
575impl std::fmt::Display for DeltaReplayError {
576    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
577        match self {
578            Self::FrameOutOfBounds { bucket, capacity } => {
579                write!(
580                    f,
581                    "delta frame bucket {bucket} exceeds index capacity {capacity}"
582                )
583            }
584            Self::RequiresOwnedIndex => write!(f, "delta replay requires an owned index"),
585            Self::EmptySlot { bucket } => {
586                write!(
587                    f,
588                    "delta frame at bucket {bucket} carries the empty-slot sentinel"
589                )
590            }
591        }
592    }
593}

738        let path = dir.join(INDEX_DELTA_FILE);

751        let log = read_delta_log(&path).expect("first committed batch must still be trusted");
752        assert_eq!(log.tail_fingerprint, 11, "corrupted batch must be dropped");
753        assert_eq!(log.frames.len(), 1);
