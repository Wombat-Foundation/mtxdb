1//! The incremental index delta log (`index.delta`).
2//!
3//! Packfiles stay authoritative and the checkpoint stays a rebuildable
4//! acceleration structure; this log is a further acceleration on top of the
5//! checkpoint. Between full checkpoint rewrites, a session's index changes are
6//! recorded as fixed-width [`DeltaFrame`] records so that `sync()` can append
7//! a few kilobyte batch instead of rewriting the whole checkpoint. A crash at
8//! any point degrades to the fingerprint gates below and an ordinary rescan.
9//!
10//! # Trusted gates (all must pass, else the log is ignored and the opener
11//! falls back to a full rescan / the next sync rewrites the checkpoint):
12//!
13//! - The log's `base_fingerprint` must equal the checkpoint it continues.
14//! - The log's `tail_fingerprint` must equal the fingerprint of the packs
15//!   currently on disk (the state the frames were written against).
16//! - Every frame's `generation` must match the checkpoint's recorded
17//!   generation for that collection (guards against a delta for a pre-resize
18//!   table being applied to a resized/repacked one).
19//!
20//! # Layout (all little-endian, fixed width — see [`crate::index::format`]):
21//!
22//! ```text
23//!   [DeltaLogHeader 16B]       magic "MDLG" | version | reserved | base_fingerprint
24//!   [DeltaBatchHeader 8B]      magic "MDLB" | frame_count
25//!   [DeltaFrame * frame_count] fixed 36B records
26//!   [DeltaLogTrailer 16B]      magic "DLTR" | reserved | tail_fingerprint
27//!   [DeltaBatchHeader ..]*     -- further batches, each count-framed and trailer-terminated
28//! ```
29//!
30//! Framing is arithmetic, not magic-sniffed: a reader advances by exactly
31//! `frame_count * DELTA_FRAME_LEN` bytes from each batch header, and only then
32//! expects the trailer. A batch header's `frame_count` (not any record's
33//! payload bytes) determines where the next record begins — a collection ID
34//! that happens to spell a magic value can never misdirect the parser. A torn
35//! suffix is exactly "not enough bytes remain for `frame_count` frames plus a
36//! trailer", and only complete batches are trusted.
37

44/// File name of the incremental index delta log inside a store's base dir.
45pub const INDEX_DELTA_FILE: &str = "index.delta";
46/// Bytes in one log header: magic(4) + version(1) + reserved(3) + base fingerprint(8).
47pub const DELTA_LOG_HEADER_LEN: usize = 16;
48/// Bytes in one batch header: magic(4) + `frame_count`(4). Determines the
49/// number of following 36-byte frames arithmetically.
50pub const DELTA_BATCH_HEADER_LEN: usize = 8;
51/// Bytes in one batch trailer: magic(4) + reserved(4) + tail fingerprint(8).
52pub const DELTA_LOG_TRAILER_LEN: usize = 16;
53
54/// Hard ceiling on the delta log file size this reader will allocate for.
55/// The writer (`storage.rs`) forces a full checkpoint rewrite well before
56/// this — its own cap is 8 MiB — so a well-behaved log never approaches this
57/// value; it exists purely so a corrupted or maliciously oversized
58/// `index.delta` (e.g. a truncated-looking length field, or the file
59/// replaced wholesale) can't make an opener allocate an unbounded buffer
60/// before any structural validation runs. Generous relative to the writer's
61/// own cap so it never rejects a real log.
62const MAX_DELTA_LOG_FILE_BYTES: u64 = 64 * 1024 * 1024;
63
64const DELTA_LOG_MAGIC: &[u8; 4] = b"MDLG";
65/// Current wire version (see the header's version byte).
66///
67/// v2 repurposes the trailer's 4 reserved bytes as a CRC32 covering the
68/// batch's frame bytes and its tail fingerprint, so a batch corrupted after
69/// being written — whether in the frames or just the fingerprint — but which
70/// still happens to pass the fingerprint/generation gates in `storage.rs` is
71/// caught and dropped at read time instead of being replayed as valid index
72/// state. A v1 log fails the version check below and falls back to a full
73/// rescan, same as any other structurally rejected log.
74const DELTA_LOG_VERSION: u8 = 2;
75const DELTA_BATCH_MAGIC: &[u8; 4] = b"MDLB";
76const DELTA_LOG_TRAILER_MAGIC: &[u8; 4] = b"DLTR";
77
78const BASE_FINGERPRINT_OFFSET: usize = 8;
79const TRAILER_FINGERPRINT_OFFSET: usize = 8;
80/// Offset of the batch's frame-bytes CRC32 within the trailer's reserved
81/// field (bytes 4..8, between the magic and the tail fingerprint).
82const TRAILER_CRC_OFFSET: usize = 4;
83
84/// A decoded delta log that passed its structural gates.
85#[derive(Debug)]
86pub struct DeltaLog {
87    /// The checkpoint fingerprint this log continues from.
88    pub base_fingerprint: u64,
89    /// Every committed frame in file order (torn trailing batches are absent).
90    pub frames: Vec<DeltaFrame>,
91    /// The pack fingerprint the final committed batch was written against.
92    pub tail_fingerprint: u64,
93    /// Byte length of the committed region in the log file (through the last
94    /// committed trailer). A session continuing this log appends at this
95    /// boundary — it is the answer to "has a base header already been
96    /// written", independent of any fresh-session default.
97    pub file_len: u64,
98}

100/// Encode the fixed-width log header for `base_fingerprint`.
101#[must_use]
102pub fn encode_header(base_fingerprint: u64) -> [u8; DELTA_LOG_HEADER_LEN] {
103    let mut bytes = [0u8; DELTA_LOG_HEADER_LEN];
104    bytes[..4].copy_from_slice(DELTA_LOG_MAGIC);
105    bytes[4] = DELTA_LOG_VERSION;
106    bytes[BASE_FINGERPRINT_OFFSET..16].copy_from_slice(&base_fingerprint.to_le_bytes());
107    bytes
108}
109
110/// Encode the fixed-width batch header for `frame_count`.
111#[must_use]
112pub fn encode_batch_header(frame_count: u32) -> [u8; DELTA_BATCH_HEADER_LEN] {
113    let mut bytes = [0u8; DELTA_BATCH_HEADER_LEN];
114    bytes[..4].copy_from_slice(DELTA_BATCH_MAGIC);
115    bytes[4..8].copy_from_slice(&frame_count.to_le_bytes());
116    bytes
117}
118
119/// Encode the fixed-width batch trailer for `tail_fingerprint`, with `crc`
120/// the CRC32 of the batch's frame bytes and `tail_fingerprint` together
121/// (stored at byte offset 4 of the trailer's reserved field).
122#[must_use]
123pub fn encode_trailer(tail_fingerprint: u64, crc: u32) -> [u8; DELTA_LOG_TRAILER_LEN] {
124    let mut bytes = [0u8; DELTA_LOG_TRAILER_LEN];
125    bytes[..4].copy_from_slice(DELTA_LOG_TRAILER_MAGIC);
126    bytes[TRAILER_CRC_OFFSET..8].copy_from_slice(&crc.to_le_bytes());
127    bytes[TRAILER_FINGERPRINT_OFFSET..16].copy_from_slice(&tail_fingerprint.to_le_bytes());
128    bytes
129}
130
131/// CRC32 covering one batch's frame bytes *and* its `tail_fingerprint`, as
132/// stored in the trailer's reserved field. Folding the fingerprint into the
133/// CRC means fingerprint-only corruption (the frame bytes untouched) is also
134/// caught here, rather than relying solely on the fingerprint gate in
135/// `storage.rs` matching it against the wrong thing.
136#[must_use]
137fn batch_crc(frames_bytes: &[u8], tail_fingerprint: u64) -> u32 {
138    let mut hasher = crc32fast::Hasher::new();
139    hasher.update(frames_bytes);
140    hasher.update(&tail_fingerprint.to_le_bytes());
141    hasher.finalize()
142}
143
144/// Length in bytes of one fully framed batch carrying `frame_count` frames.
145#[must_use]
146pub fn batch_len(frame_count: usize) -> Option<usize> {
147    DELTA_BATCH_HEADER_LEN
148        .checked_add(frame_count.checked_mul(DELTA_FRAME_LEN)?)?
149        .checked_add(DELTA_LOG_TRAILER_LEN)
150}

152/// Read and structurally validate the delta log. Returns `None` for a missing
153/// file, a bad header, a log with no complete committed batch (entirely torn),
154/// or any other structural inconsistency — the caller treats that as "no
155/// delta".
156///
157/// Batch boundaries come from each batch header's `frame_count`, never from
158/// scanning for magic, so no frame payload can redirect the parse. A torn
159/// trailing batch (fewer bytes than `frame_count` frames + a trailer) is
160/// dropped; everything up to the last complete batch is returned.
161#[must_use]
162pub fn read_delta_log(path: &Path) -> Option<DeltaLog> {
163    // Check the size before allocating: `fs::read` would otherwise size its
164    // buffer off an unvalidated on-disk length, letting a corrupted or
165    // oversized file exhaust memory before any structural check runs.
166    if fs::metadata(path).ok()?.len() > MAX_DELTA_LOG_FILE_BYTES {
167        return None;
168    }
169    let buf = fs::read(path).ok()?;
170    if buf.len() < DELTA_LOG_HEADER_LEN {
171        return None;
172    }
173    if &buf[..4] != DELTA_LOG_MAGIC || buf[4] != DELTA_LOG_VERSION {
174        return None;
175    }
176    let base_fingerprint = u64::from_le_bytes(buf[BASE_FINGERPRINT_OFFSET..16].try_into().ok()?);
177
178    let mut frames: Vec<DeltaFrame> = Vec::new();
179    let mut tail_fingerprint: Option<u64> = None;
180    let mut offset = DELTA_LOG_HEADER_LEN;
181    while offset < buf.len() {
182        if buf.len().saturating_sub(offset) < DELTA_BATCH_HEADER_LEN {
183            break;
184        }
185        if &buf[offset..offset.saturating_add(4)] != DELTA_BATCH_MAGIC {
186            break;
187        }
188        let Some(header_count) = buf.get(offset.saturating_add(4)..offset.saturating_add(8)) else {
189            break;
190        };
191        let frame_count = u32::from_le_bytes(header_count.try_into().ok()?) as usize;
192        let Some(batch_len) = batch_len(frame_count) else {
193            break;
194        };
195        let Some(batch_end) = offset.checked_add(batch_len) else {
196            break;
197        };
198        if batch_end > buf.len() {
199            // Torn final batch: fewer bytes than frame_count frames + trailer.
200            break;
201        }
202        let frames_start = offset.saturating_add(DELTA_BATCH_HEADER_LEN);
203        let trailer_start = batch_end.saturating_sub(DELTA_LOG_TRAILER_LEN);
204        if &buf[trailer_start..trailer_start.saturating_add(4)] != DELTA_LOG_TRAILER_MAGIC {
205            break;
206        }
207        let frames_bytes = &buf[frames_start..trailer_start];
208        let stored_crc = u32::from_le_bytes(
209            buf[trailer_start.saturating_add(TRAILER_CRC_OFFSET)..trailer_start.saturating_add(8)]
210                .try_into()
211                .ok()?,
212        );
213        let batch_tail_fingerprint = u64::from_le_bytes(
214            buf[trailer_start.saturating_add(TRAILER_FINGERPRINT_OFFSET)
215                ..trailer_start.saturating_add(16)]
216                .try_into()
217                .ok()?,
218        );
219        if batch_crc(frames_bytes, batch_tail_fingerprint) != stored_crc {
220            // The frame bytes or the tail fingerprint were corrupted after
221            // being written (bit rot, torn/partial write not otherwise
222            // caught by the length framing, etc). Trust nothing from this
223            // batch onward — same as a torn trailer, everything already
224            // accumulated from earlier committed batches is kept, but this
225            // and any later batch are dropped rather than replayed as valid
226            // index state.
227            break;
228        }
229        for frame_bytes in frames_bytes.chunks_exact(DELTA_FRAME_LEN) {
230            frames.push(DeltaFrame::decode(frame_bytes)?);
231        }
232        tail_fingerprint = Some(batch_tail_fingerprint);
233        offset = batch_end;
234    }
235    Some(DeltaLog {
236        base_fingerprint,
237        frames,
238        tail_fingerprint: tail_fingerprint?,
239        // `offset` is the frontier of the last committed batch: the loop only
240        // breaks past a committed trailer or before a torn/unparseable tail.
241        file_len: u64::try_from(offset).unwrap_or(u64::MAX),
242    })
243}

245/// Append one batch (optional header, then a count-framed batch) to the delta
246/// log and fsync it. Returns the number of bytes appended. `write_header` must
247/// be `true` when writing to a freshly created (or truncated) file that
248/// doesn't yet carry the base header.
249///
250/// # Errors
251/// Returns `io::Error` on any write or fsync failure.
252pub fn append_batch(
253    path: &Path,
254    write_header: bool,
255    base_fingerprint: u64,
256    frames: &[DeltaFrame],
257    tail_fingerprint: u64,
258) -> std::io::Result<usize> {
259    let frame_count = u32::try_from(frames.len())
260        .map_err(|_| std::io::Error::other("delta batch exceeds u32 frame count"))?;
261    let mut appended = 0usize;
262    let mut file = fs::OpenOptions::new()
263        .create(true)
264        .append(true)
265        .open(path)?;
266    if write_header {
267        file.write_all(&encode_header(base_fingerprint))?;
268        appended = appended.saturating_add(DELTA_LOG_HEADER_LEN);
269    }
270    file.write_all(&encode_batch_header(frame_count))?;
271    appended = appended.saturating_add(DELTA_BATCH_HEADER_LEN);
272    let mut frames_bytes = Vec::with_capacity(frames.len().saturating_mul(DELTA_FRAME_LEN));
273    for frame in frames {
274        frames_bytes.extend_from_slice(&frame.encode());
275    }
276    file.write_all(&frames_bytes)?;
277    appended = appended.saturating_add(frames_bytes.len());
278    file.write_all(&encode_trailer(
279        tail_fingerprint,
280        batch_crc(&frames_bytes, tail_fingerprint),
281    ))?;
282    appended = appended.saturating_add(DELTA_LOG_TRAILER_LEN);
283    file.sync_all()?;
284    Ok(appended)
285}

287/// Errors that can arise while applying a replayed delta log to a
288/// checkpoint-backed index. Any of these means the log is structurally
289/// inconsistent with the checkpoint it claims to continue, and the whole log
290/// must be rejected (opener falls back to a rescan).
291#[derive(Debug, PartialEq, Eq)]
292pub enum DeltaReplayError {
293    /// A frame targets a bucket outside the checkpoint index's capacity.
294    FrameOutOfBounds {
295        /// The frame's target bucket.
296        bucket: u32,
297        /// The checkpoint index's capacity.
298        capacity: u32,
299    },
300    /// Replay requires an owned (materialized) slot array.
301    RequiresOwnedIndex,
302}
303
304impl std::fmt::Display for DeltaReplayError {
305    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
306        match self {
307            Self::FrameOutOfBounds { bucket, capacity } => {
308                write!(
309                    f,
310                    "delta frame bucket {bucket} exceeds index capacity {capacity}"
311                )
312            }
313            Self::RequiresOwnedIndex => write!(f, "delta replay requires an owned index"),
314        }
315    }
316}
