1//! On-disk serialization of the per-collection indexes (`index.checkpoint`).
2//!
3//! Packfiles stay authoritative: this file is a rebuildable acceleration
4//! structure. A full `rewrite` (see `PackfileStorage::persist_index_checkpoint`)
5//! is done atomically (temp file + rename) whenever the delta log can't
6//! continue — a structural change, a broken continuation, or the log's size
7//! cap. Between rewrites, incremental index mutations are persisted as
8//! batches in the `index.delta` log ([`crate::index::delta`]); the checkpoint
9//! pins the exact pack set those frames continue, and a crash at any point
10//! leaves a fingerprint mismatch the next open resolves with a rescan — never
11//! a wrong replay.
12//!
13//! Layout (all little-endian, fixed width — see [`crate::index::format`]):
14//!
15//! ```text
16//!   [CheckpointHeader 64B]
17//!   [CollectionDirEntry * count, 40B each]
18//!   [collection 0 raw slots: capacity * 8B]
19//!   [collection 1 raw slots: capacity * 8B]
20//!   ...
21//! ```
22//!
23//! The header's `pack_fingerprint` is a deterministic hash of the
24//! `(pack_id, length)` set the checkpoint describes. An opener only trusts
25//! the file when that fingerprint matches the packs currently on disk (optionally
26//! with a gated delta replay continuing it); any append, rotation, or repack
27//! changes it and forces a rescan.
28

41/// Magic identifying the persisted-index checkpoint format.
42pub const CHECKPOINT_MAGIC: [u8; 8] = *b"MTXIDX01";
43/// Current wire version (see [`CheckpointHeader::version`]).
44///
45/// Bumped to 2 when `content_crc32` was added to the header (replacing the
46/// read-time occupancy walk with a single checksum verification — see
47/// `read_checkpoint`). Not deployed anywhere yet, so no migration: a v1
48/// checkpoint simply fails the version check below and falls through to the
49/// existing full-rescan fallback, same as any other unreadable checkpoint.
50pub const CHECKPOINT_VERSION: u32 = 2;
51/// File name of the persisted index checkpoint inside a store's base dir.
52pub const INDEX_CHECKPOINT_FILE: &str = "index.checkpoint";
53
54/// Whether an opener verifies the checkpoint body's CRC32.
55///
56/// This policy is intentionally independent from [`crate::packfile::ChecksumPolicy`]:
57/// frame checksums, delta-log checksums, and checkpoint integrity protect
58/// different on-disk structures and must not be disabled together by accident.
59/// `WriteOnly` retains the CRC when writing a checkpoint for offline/recovery
60/// tooling, but trusts it during normal opens.
61#[derive(Debug, Clone, Copy, PartialEq, Eq)]
62pub enum CheckpointChecksumPolicy {
63    /// Verify the persisted checkpoint CRC32 at every open (the default).
64    Full,
65    /// Retain the CRC32 on write but skip its read-time verification.
66    WriteOnly,
67}
68
69impl CheckpointChecksumPolicy {
70    /// Select the process-wide checkpoint policy from `MTXDB_CHECKPOINT_CHECKSUM`.
71    /// Unknown values intentionally retain the safe default.
72    #[must_use]
73    pub fn from_env() -> Self {
74        match std::env::var("MTXDB_CHECKPOINT_CHECKSUM").as_deref() {
75            Ok("writeonly") => Self::WriteOnly,
76            Ok("full" | _) | Err(_) => Self::Full,
77        }
78    }
79
80    #[must_use]
81    fn verifies_reads(self) -> bool {
82        matches!(self, Self::Full)
83    }
84}

91/// A checkpoint read off disk, validated enough to trust for index rebuild.
92#[derive(Debug)]
93pub struct LoadedCheckpoint {
94    /// The pack-fingerprint the checkpoint was written against.
95    pub fingerprint: u64,
96    /// One entry per collection, in the checkpoint's directory order.
97    pub collections: Vec<LoadedCollection>,
98    /// Keeps the raw slot arrays alive for mmap-backed indexes built from
99    /// this checkpoint.
100    pub mmap: Arc<Mmap>,
101}

103/// One collection's raw slots in the checkpoint mapping.
104#[derive(Debug)]
105pub struct LoadedCollection {
106    /// The collection whose slots these are.
107    pub collection_id: [u8; 16],
108    /// Checkpoint generation for this collection (reserved for a future
109    /// delta-log; always 0 today).
110    pub generation: u64,
111    /// Absolute byte offset of the raw `capacity * 8` slot array.
112    pub slots_offset: usize,
113    /// Allocated index capacity.
114    pub capacity: u32,
115    /// Number of non-empty slots, recorded at checkpoint write time.
116    pub slot_count: u32,
117}

119/// Deterministic FNV-1a hash of the `(pack_id, file_len)` set a checkpoint
120/// (or an open store's current packs) describes. The set is sorted before
121/// hashing so the result is order-independent.
122#[must_use]
123pub fn pack_fingerprint(packs: &[(u64, u64)]) -> u64 {
124    let mut sorted: Vec<(u64, u64)> = packs.to_vec();
125    sorted.sort_unstable();
126    let mut hash: u64 = 0xcbf2_9ce4_8422_2325;
127    for (pack_id, file_len) in sorted {
128        for byte in pack_id.to_le_bytes() {
129            hash = (hash ^ u64::from(byte)).wrapping_mul(0x0000_0100_0000_01b3);
130        }
131        for byte in file_len.to_le_bytes() {
132            hash = (hash ^ u64::from(byte)).wrapping_mul(0x0000_0100_0000_01b3);
133        }
134    }
135    hash
136}

138/// Decode the 8-byte capacity prefix of a `LossyIndex::serialize` blob.
139fn blob_capacity(blob: &[u8]) -> Option<u64> {
140    let head: [u8; 8] = blob.get(..8)?.try_into().ok()?;
141    Some(u64::from_le_bytes(head))
142}
143
144/// Number of occupied (non-`0u64`) slots in a `LossyIndex::serialize` blob's
145/// slot section.
146fn blob_slot_count(blob: &[u8]) -> u32 {
147    u32::try_from(
148        blob[8..]
149            .chunks_exact(8)
150            .filter(|slot| *slot != [0u8; 8])
151            .count(),
152    )
153    .unwrap_or(u32::MAX)
154}

156/// Write a complete checkpoint atomically: temp file, fsync, rename over the
157/// final path. `collections` holds each collection's `serialize` blob plus the
158/// index generation it was written at (recorded in its directory entry so a
159/// delta log's frames can be gated against the same generation on replay),
160/// the first collection listed becoming the first in the directory order.
161///
162/// # Contract
163/// The `fingerprint` and every `collections` blob must be *captured
164/// atomically*: derived from pack-file lengths and index slots that were both
165/// observed while every collection's writer lock was held. This function only
166/// makes that already-consistent (pack set, index snapshot) pair durable; it
167/// cannot reconcile the two if a concurrent `put` changed either side between
168/// capture and this call. The sole caller
169/// (`PackfileStorage::persist_index_checkpoint`) satisfies this by pinning the
170/// pack lengths and all collection generations under every `put_mutex` (and
171/// rotating the delta epoch there too) before handing them to this function.
172///
173/// # Errors
174/// Returns `io::Error` on any failure; the previous checkpoint (if any) is
175/// left intact in that case.
176pub fn write_checkpoint(
177    path: &Path,
178    fingerprint: u64,
179    collections: &[([u8; 16], u64, &[u8])],
180) -> std::io::Result<()> {
181    let count = u32::try_from(collections.len())
182        .map_err(|_| std::io::Error::other("too many collections for checkpoint u32"))?;
183    let directory_bytes = u64::try_from(collections.len())
184        .ok()
185        .and_then(|n| n.checked_mul(COLLECTION_DIR_ENTRY_LEN as u64))
186        .ok_or_else(|| std::io::Error::other("checkpoint directory size overflow"))?;
187    let mut slots_bytes: u64 = 0;
188    for (_, _, blob) in collections {
189        let blob_len = u64::try_from(blob.len().saturating_sub(8))
190            .map_err(|_| std::io::Error::other("collection slot array too large"))?;
191        slots_bytes = slots_bytes
192            .checked_add(blob_len)
193            .ok_or_else(|| std::io::Error::other("checkpoint slots size overflow"))?;
194    }
195
196    let total_len = CHECKPOINT_HEADER_LEN
197        .saturating_add(usize::try_from(directory_bytes).unwrap_or(usize::MAX))
198        .saturating_add(usize::try_from(slots_bytes).unwrap_or(usize::MAX));
199    // Reserve the header first, then build the body directly into the final
200    // buffer. This avoids retaining a second checkpoint-sized body merely to
201    // calculate the CRC carried by the header.
202    let mut buf = Vec::with_capacity(total_len);
203    buf.resize(CHECKPOINT_HEADER_LEN, 0);
204
205    let mut slots_offset: u64 = 0;
206    let mut dir_entries = Vec::with_capacity(collections.len());
207    for (collection_id, generation, blob) in collections {
208        let capacity = blob_capacity(blob)
209            .ok_or_else(|| std::io::Error::other("collection slot blob missing capacity"))?;
210        let capacity32 = u32::try_from(capacity)
211            .map_err(|_| std::io::Error::other("index capacity exceeds u32"))?;
212        let slot_array_len: u64 = blob[8..]
213            .len()
214            .try_into()
215            .map_err(|_| std::io::Error::other("collection slot array too large"))?;
216        dir_entries.push(CollectionDirEntry {
217            collection_id: *collection_id,
218            generation: *generation,
219            slots_offset,
220            capacity: capacity32,
221            slot_count: blob_slot_count(blob),
222        });
223        slots_offset = slots_offset
224            .checked_add(slot_array_len)
225            .ok_or_else(|| std::io::Error::other("checkpoint slots offset overflow"))?;
226    }
227    for entry in &dir_entries {
228        buf.extend_from_slice(&entry.encode());
229    }
230    for (_, _, blob) in collections {
231        buf.extend_from_slice(&blob[8..]);
232    }
233    let content_crc32 = crc32fast::hash(&buf[CHECKPOINT_HEADER_LEN..]);
234
235    let header = CheckpointHeader {
236        magic: CHECKPOINT_MAGIC,
237        version: CHECKPOINT_VERSION,
238        collection_count: count,
239        directory_bytes,
240        slots_bytes,
241        pack_fingerprint: fingerprint,
242        content_crc32,
243    };
244    buf[..CHECKPOINT_HEADER_LEN].copy_from_slice(&header.encode());
245
246    let unique = CHECKPOINT_TMP_COUNTER.fetch_add(1, Ordering::Relaxed);
247    let tmp_path = PathBuf::from(format!(
248        "{}.tmp.{}.{unique}",
249        path.display(),
250        std::process::id()
251    ));
252    let write_result = (|| -> std::io::Result<()> {
253        let mut tmp = fs::File::create(&tmp_path)?;
254        tmp.write_all(&buf)?;
255        // `sync_data` (fdatasync), not `sync_all` (fsync): per POSIX,
256        // fdatasync durably flushes the file's data plus whatever metadata
257        // is needed to retrieve it (the extended length included) — that's
258        // the whole content/length guarantee this write needs. It skips
259        // metadata that's *not* needed to read the data back (mtime/ctime).
260        // That's a real distinction, not a claim that all inode metadata is
261        // irrelevant here generally (permissions/ownership can still matter
262        // operationally) — it just happens that none of the metadata this
263        // skips is read back through this temp name: it's renamed over
264        // `path` immediately below and never opened by this name again. The
265        // caller (`persist_index_checkpoint`) separately fsyncs the base
266        // directory after the rename, which is what makes the rename itself
267        // durable; if that directory fsync fails, reopen falls back to
268        // whichever checkpoint (old or new) is actually intact rather than
269        // trusting the rename — this change doesn't alter that fallback.
270        tmp.sync_data()
271    })();
272    if let Err(e) = write_result {
273        let _ = fs::remove_file(&tmp_path);
274        return Err(e);
275    }
276    fs::rename(&tmp_path, path)
277}

279/// Read and validate a checkpoint. Returns `None` for any problem — a
280/// missing, truncated, or malformed file is treated identically to "no
281/// checkpoint", and the caller falls back to rebuilding from the packs.
282#[must_use]
283pub fn read_checkpoint(path: &Path) -> Option<LoadedCheckpoint> {
284    read_checkpoint_with_policy(path, CheckpointChecksumPolicy::from_env())
285}

287/// As [`read_checkpoint`], with an explicit integrity policy for callers that
288/// need deterministic configuration rather than the process environment.
289#[must_use]
290pub fn read_checkpoint_with_policy(
291    path: &Path,
292    checksum_policy: CheckpointChecksumPolicy,
293) -> Option<LoadedCheckpoint> {
294    let file = fs::File::open(path).ok()?;
295    let mmap = Arc::new(crate::packfile::map_pack(&file).ok()?);
296    let buf: &[u8] = &mmap;
297    let header_bytes: [u8; CHECKPOINT_HEADER_LEN] =
298        buf.get(..CHECKPOINT_HEADER_LEN)?.try_into().ok()?;
299    let header = CheckpointHeader::decode(&header_bytes)?;
300    if header.magic != CHECKPOINT_MAGIC || header.version != CHECKPOINT_VERSION {
301        return None;
302    }
303    let count = header.collection_count as usize;
304    if header.directory_bytes
305        != u64::try_from(count)
306            .ok()?
307            .checked_mul(COLLECTION_DIR_ENTRY_LEN as u64)?
308    {
309        return None;
310    }
311    let slot_base =
312        CHECKPOINT_HEADER_LEN.checked_add(usize::try_from(header.directory_bytes).ok()?)?;
313    if buf.len() != slot_base.checked_add(usize::try_from(header.slots_bytes).ok()?)? {
314        return None;
315    }
316
317    // Verify the whole directory+slots body against the header's CRC in one
318    // pass, in place of re-deriving each collection's occupancy by walking
319    // its slots. This is strictly stronger than the old per-collection walk
320    // (it also covers the directory entries — `capacity`/`slot_count`
321    // themselves — which the walk never checked), and costs the same O(bytes)
322    // as a single SIMD-accelerated hash instead of a manual scan-and-branch
323    // loop over every slot of every collection.
324    if checksum_policy.verifies_reads() {
325        let content_crc32 = crc32fast::hash(buf.get(CHECKPOINT_HEADER_LEN..)?);
326        if content_crc32 != header.content_crc32 {
327            return None;
328        }
329    }
330
331    let mut collections = Vec::with_capacity(count);
332    for i in 0..count {
333        let entry_offset =
334            CHECKPOINT_HEADER_LEN.checked_add(i.checked_mul(COLLECTION_DIR_ENTRY_LEN)?)?;
335        let entry_bytes: [u8; COLLECTION_DIR_ENTRY_LEN] = buf
336            .get(entry_offset..entry_offset.saturating_add(COLLECTION_DIR_ENTRY_LEN))?
337            .try_into()
338            .ok()?;
339        let entry = CollectionDirEntry::decode(&entry_bytes)?;
340        if entry.capacity < 16 || !entry.capacity.is_power_of_two() {
341            return None;
342        }
343        // A corrupt or malicious directory entry could otherwise claim more
344        // occupied slots than the table has room for; a downstream consumer
345        // sizing an allocation off `slot_count` (e.g. a post-restart grow)
346        // would then trust a value with no upper bound instead of the
347        // checkpoint being rejected here. (An *understated* count — the
348        // failure mode the old occupancy walk additionally caught — is now
349        // caught by the CRC check above instead: any bit that makes
350        // `slot_count` disagree with the actual slots, or with what was
351        // written, changes the body's hash.)
352        if entry.slot_count > entry.capacity {
353            return None;
354        }
355        let slots_len = (entry.capacity as usize).checked_mul(8)?;
356        let region = slot_base.checked_add(usize::try_from(entry.slots_offset).ok()?)?;
357        if region.checked_add(slots_len)? > buf.len() {
358            return None;
359        }
360        collections.push(LoadedCollection {
361            collection_id: entry.collection_id,
362            generation: entry.generation,
363            slots_offset: region,
364            capacity: entry.capacity,
365            slot_count: entry.slot_count,
366        });
367    }
368
369    Some(LoadedCheckpoint {
370        fingerprint: header.pack_fingerprint,
371        collections,
372        mmap,
373    })
374}

381    fn hash_for(seed: u16, i: usize) -> [u8; 16] {
382        let mut hash = [0u8; 16];
383        hash[..2].copy_from_slice(&seed.wrapping_add(u16::try_from(i).unwrap()).to_le_bytes());
384        hash
385    }

387    fn index_with_entries(seed: u16, count: usize) -> LossyIndex {
388        let index = LossyIndex::new(count.max(16).saturating_mul(2));
389        for i in 0..count {
390            let shard = u16::try_from(i % 64).unwrap();
391            let _ = index.insert(&hash_for(seed, i), shard, (i as u64).wrapping_mul(128));
392        }
393        index
394    }

419            vec![([1u8; 16], 1, 40), ([2u8; 16], 2, 200), ([3u8; 16], 3, 5)];

432        let path = dir.join(INDEX_CHECKPOINT_FILE);

439                .iter()
440                .map(|(id, blob)| (*id, 0, blob.as_slice()))
441                .collect::<Vec<_>>(),
442        )

458        for ((case, loaded), (_blob_id, expected_blob)) in

461            let (expected_id, seed, count) = case;
462            assert_eq!(loaded.collection_id, *expected_id);
463            let slots_len = (loaded.capacity as usize).saturating_mul(8);
464            assert_eq!(

466                &expected_blob[8..],
467                "slots must round-trip verbatim"
468            );
469
470            // The mmap-backed index must find every hash the live index had.
471            let loaded_index = LossyIndex::from_mmap_slots(
472                Arc::clone(&mmap),
473                loaded.slots_offset,
474                loaded.capacity,
475                loaded.slot_count,
476            );
477            let original_index = LossyIndex::deserialize(expected_blob).unwrap();
478            for i in 0..*count {
479                let hash = hash_for(*seed, i);
480                assert_eq!(
481                    loaded_index.lookup(&hash),
482                    original_index.lookup(&hash),
483                    "loaded index must agree with the source index on spot lookups"
484                );
485                assert!(
486                    loaded_index.lookup(&hash).is_some(),
487                    "round-tripped index lost a record"
488                );
489            }
490        }
491
492        std::fs::remove_dir_all(&dir).unwrap();
493    }

561    #[test]
562    fn content_crc32_catches_value_preserving_slot_corruption() {
563        // The pre-CRC occupancy walk only ever compared a *count* of
564        // non-empty slots against `slot_count` — a corruption that flips
565        // bits within an already-occupied slot (changing which shard/offset
566        // it names, without zeroing it) preserves that count and would have
567        // silently passed. The CRC covers the exact bytes, so this same
568        // corruption must now be rejected.
569        let dir = std::env::temp_dir().join(format!(
570            "mtxdb_checkpoint_slot_corruption_{}",
571            std::process::id()
572        ));
573        let _ = std::fs::remove_dir_all(&dir);
574        std::fs::create_dir_all(&dir).unwrap();
575        let path = dir.join(INDEX_CHECKPOINT_FILE);
576
577        let blobs = [([7u8; 16], index_with_entries(3, 40).serialize())];
578        write_checkpoint(
579            &path,
580            0,
581            &blobs
582                .iter()
583                .map(|(id, b)| (*id, 0, b.as_slice()))
584                .collect::<Vec<_>>(),
585        )
586        .unwrap();
587        assert!(
588            read_checkpoint(&path).is_some(),
589            "sanity check: the unmodified checkpoint must read back cleanly"
590        );
591
592        let mut bytes = std::fs::read(&path).unwrap();
593        // Flip one bit inside the slots section (well past the header +
594        // single directory entry) in a byte that is currently non-zero, so
595        // the slot stays non-empty — only its value changes.
596        let slots_start = CHECKPOINT_HEADER_LEN + COLLECTION_DIR_ENTRY_LEN;
597        let corrupt_at = bytes[slots_start..]
598            .iter()
599            .position(|&b| b != 0)
600            .map(|offset| slots_start + offset)
601            .expect("at least one non-zero slot byte exists");
602        bytes[corrupt_at] ^= 0x01;
603        std::fs::write(&path, &bytes).unwrap();
604
605        assert!(
606            read_checkpoint_with_policy(&path, CheckpointChecksumPolicy::Full).is_none(),
607            "a value-preserving bit flip inside an occupied slot must be rejected"
608        );
609        assert!(
610            read_checkpoint_with_policy(&path, CheckpointChecksumPolicy::WriteOnly).is_some(),
611            "write-only mode deliberately trusts a structurally valid checkpoint"
612        );
613
614        std::fs::remove_dir_all(&dir).unwrap();
615    }
616}
