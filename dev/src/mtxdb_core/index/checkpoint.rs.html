1//! On-disk serialization of the per-collection indexes (`index.checkpoint`).
2//!
3//! Packfiles stay authoritative: this file is a rebuildable acceleration
4//! structure. A full `rewrite` (see `PackfileStorage::persist_index_checkpoint`)
5//! is done atomically (temp file + rename) whenever the delta log can't
6//! continue — a structural change, a broken continuation, or the log's size
7//! cap. Between rewrites, incremental index mutations are persisted as
8//! batches in the `index.delta` log ([`crate::index::delta`]); the checkpoint
9//! pins the exact pack set those frames continue, and a crash at any point
10//! leaves a fingerprint mismatch the next open resolves with a rescan — never
11//! a wrong replay.
12//!
13//! Layout (all little-endian, fixed width — see [`crate::index::format`]):
14//!
15//! ```text
16//!   [CheckpointHeader 64B]
17//!   [CollectionDirEntry * count, 40B each]
18//!   [collection 0 raw slots: capacity * 8B]
19//!   [collection 1 raw slots: capacity * 8B]
20//!   ...
21//! ```
22//!
23//! The header's `pack_fingerprint` is a deterministic hash of the
24//! `(pack_id, length)` set the checkpoint describes. An opener only trusts
25//! the file when that fingerprint matches the packs currently on disk (optionally
26//! with a gated delta replay continuing it); any append, rotation, or repack
27//! changes it and forces a rescan.
28

41/// Magic identifying the persisted-index checkpoint format.
42pub const CHECKPOINT_MAGIC: [u8; 8] = *b"MTXIDX01";
43/// Current wire version (see [`CheckpointHeader::version`]).
44///
45/// Bumped to 5: the header now carries the journal `covered_lsn` the index
46/// snapshot incorporates, so a read-committed reader binds its overlay
47/// coverage to the exact index it loaded instead of a separately-read
48/// `journal.lsn`. A v4 checkpoint has no such field and is rebuilt.
49///
50/// Bumped to 4: homes and tails are now persisted alongside packed slots.
51/// A pre-v4 checkpoint has `homes_bytes`/`tails_bytes` of 0 and is loaded
52/// with empty identity side tables (cold-start tag-collision verification
53/// cost). A v4 checkpoint carries hydrated identity, eliminating packfile
54/// reads for tag collisions on cold start.
55pub const CHECKPOINT_VERSION: u32 = 5;
56/// File name of the persisted index checkpoint inside a store's base dir.
57pub const INDEX_CHECKPOINT_FILE: &str = "index.checkpoint";

59/// Whether an opener verifies the checkpoint body's CRC32.
60///
61/// This policy is intentionally independent from [`crate::packfile::ChecksumPolicy`]:
62/// frame checksums, delta-log checksums, and checkpoint integrity protect
63/// different on-disk structures and must not be disabled together by accident.
64/// `WriteOnly` retains the CRC when writing a checkpoint for offline/recovery
65/// tooling, but trusts it during normal opens.
66#[derive(Debug, Clone, Copy, PartialEq, Eq)]
67pub enum CheckpointChecksumPolicy {
68    /// Verify the persisted checkpoint CRC32 at every open (the default).
69    Full,
70    /// Retain the CRC32 on write but skip its read-time verification.
71    WriteOnly,
72}
73
74impl CheckpointChecksumPolicy {
75    /// Select the process-wide checkpoint policy from `MTXDB_CHECKPOINT_CHECKSUM`.
76    /// Unknown values intentionally retain the safe default.
77    #[must_use]
78    pub fn from_env() -> Self {
79        match std::env::var("MTXDB_CHECKPOINT_CHECKSUM").as_deref() {
80            Ok("writeonly") => Self::WriteOnly,
81            Ok("full" | _) | Err(_) => Self::Full,
82        }
83    }
84
85    #[must_use]
86    fn verifies_reads(self) -> bool {
87        matches!(self, Self::Full)
88    }
89}

136/// Deterministic FNV-1a hash of the `(pack_id, file_len)` set a checkpoint
137/// (or an open store's current packs) describes. The set is sorted before
138/// hashing so the result is order-independent.
139#[must_use]
140pub fn pack_fingerprint(packs: &[(u64, u64)]) -> u64 {
141    let mut sorted: Vec<(u64, u64)> = packs.to_vec();
142    sorted.sort_unstable();
143    let mut hash: u64 = 0xcbf2_9ce4_8422_2325;
144    for (pack_id, file_len) in sorted {
145        for byte in pack_id.to_le_bytes() {
146            hash = (hash ^ u64::from(byte)).wrapping_mul(0x0000_0100_0000_01b3);
147        }
148        for byte in file_len.to_le_bytes() {
149            hash = (hash ^ u64::from(byte)).wrapping_mul(0x0000_0100_0000_01b3);
150        }
151    }
152    hash
153}

155/// Decode the 8-byte capacity prefix of a `LossyIndex::serialize` blob.
156fn blob_capacity(blob: &[u8]) -> Option<u64> {
157    let head: [u8; 8] = blob.get(..8)?.try_into().ok()?;
158    Some(u64::from_le_bytes(head))
159}
160
161/// Number of occupied (non-`0u64`) slots in a `LossyIndex::serialize` blob's
162/// slot section. The blob format is `[capacity_u64][slots × cap][homes × cap][tails × cap]`.
163fn blob_slot_count(blob: &[u8]) -> u32 {
164    let Some(capacity) = blob_capacity(blob) else {
165        return 0;
166    };
167    let cap = usize::try_from(capacity).unwrap_or(usize::MAX);
168    let slots_end = 8_usize.saturating_add(cap.saturating_mul(8));
169    let slots = blob.get(8..slots_end).unwrap_or(&[]);
170    u32::try_from(
171        slots
172            .chunks_exact(8)
173            .filter(|slot| *slot != [0u8; 8])
174            .count(),
175    )
176    .unwrap_or(u32::MAX)
177}

179/// Write a complete checkpoint atomically: temp file, fsync, rename over the
180/// final path. `collections` holds each collection's `serialize` blob plus the
181/// index generation it was written at (recorded in its directory entry so a
182/// delta log's frames can be gated against the same generation on replay),
183/// the first collection listed becoming the first in the directory order.
184///
185/// # Contract
186/// The `fingerprint` and every `collections` blob must be *captured
187/// atomically*: derived from pack-file lengths and index slots that were both
188/// observed while every collection's writer lock was held. This function only
189/// makes that already-consistent (pack set, index snapshot) pair durable; it
190/// cannot reconcile the two if a concurrent `put` changed either side between
191/// capture and this call. The sole caller
192/// (`PackfileStorage::persist_index_checkpoint`) satisfies this by pinning the
193/// pack lengths and all collection generations under every `put_mutex` (and
194/// rotating the delta epoch there too) before handing them to this function.
195///
196/// # Errors
197/// Returns `io::Error` on any failure; the previous checkpoint (if any) is
198/// left intact in that case.
199#[allow(
200    clippy::too_many_lines,
201    reason = "the three phases (directory, slots/homes/tails offsets, body write) are one atomic construction; splitting them would scatter overflow-checked arithmetic that must stay in lockstep"
202)]
203pub fn write_checkpoint(
204    path: &Path,
205    fingerprint: u64,
206    covered_lsn: u64,
207    collections: &[([u8; 16], u64, &[u8])],
208) -> std::io::Result<()> {
209    let count = u32::try_from(collections.len())
210        .map_err(|_| std::io::Error::other("too many collections for checkpoint u32"))?;
211    let directory_bytes = u64::try_from(collections.len())
212        .ok()
213        .and_then(|n| n.checked_mul(COLLECTION_DIR_ENTRY_LEN as u64))
214        .ok_or_else(|| std::io::Error::other("checkpoint directory size overflow"))?;
215
216    // Compute per-section byte lengths from the serialize blobs.
217    // Blob format: [capacity_u64][slots × cap][homes × cap][tails × cap]
218    let mut slots_bytes: u64 = 0;
219    let mut homes_bytes: u64 = 0;
220    let mut tails_bytes: u64 = 0;
221    for (_, _, blob) in collections {
222        let capacity = blob_capacity(blob)
223            .ok_or_else(|| std::io::Error::other("collection slot blob missing capacity"))?;
224        let cap8 = capacity
225            .checked_mul(8)
226            .ok_or_else(|| std::io::Error::other("collection capacity × 8 overflow"))?;
227        slots_bytes = slots_bytes
228            .checked_add(cap8)
229            .ok_or_else(|| std::io::Error::other("checkpoint slots size overflow"))?;
230        // homes and tails are each capacity × 8 bytes when present in the blob
231        let blob_body = blob.len().saturating_sub(8);
232        let cap_usize = usize::try_from(capacity).unwrap_or(usize::MAX);
233        if blob_body >= cap_usize.saturating_mul(24) {
234            homes_bytes = homes_bytes
235                .checked_add(cap8)
236                .ok_or_else(|| std::io::Error::other("checkpoint homes size overflow"))?;
237            tails_bytes = tails_bytes
238                .checked_add(cap8)
239                .ok_or_else(|| std::io::Error::other("checkpoint tails size overflow"))?;
240        }
241    }
242
243    let body_bytes = slots_bytes
244        .checked_add(homes_bytes)
245        .and_then(|v| v.checked_add(tails_bytes))
246        .ok_or_else(|| std::io::Error::other("checkpoint body size overflow"))?;
247    let total_len = CHECKPOINT_HEADER_LEN
248        .saturating_add(usize::try_from(directory_bytes).unwrap_or(usize::MAX))
249        .saturating_add(usize::try_from(body_bytes).unwrap_or(usize::MAX));
250    // Reserve the header first, then build the body directly into the final
251    // buffer. This avoids retaining a second checkpoint-sized body merely to
252    // calculate the CRC carried by the header.
253    let mut buf = Vec::with_capacity(total_len);
254    buf.resize(CHECKPOINT_HEADER_LEN, 0);
255
256    // Phase 1: build directory entries, tracking per-collection offsets.
257    //
258    // Offsets must mirror Phase 3's actual write order exactly: each
259    // collection's slots/homes/tails are written back-to-back as one block
260    // (not as three separate all-slots/all-homes/all-tails sections), so a
261    // single running `body_offset` — not three independent per-kind
262    // counters — is what keeps a later collection's offsets from landing
263    // inside an earlier collection's bytes.
264    let mut body_offset: u64 = 0;
265    let mut dir_entries = Vec::with_capacity(collections.len());
266    for (collection_id, generation, blob) in collections {
267        let capacity = blob_capacity(blob)
268            .ok_or_else(|| std::io::Error::other("collection slot blob missing capacity"))?;
269        let capacity32 = u32::try_from(capacity)
270            .map_err(|_| std::io::Error::other("index capacity exceeds u32"))?;
271        let cap8 = capacity
272            .checked_mul(8)
273            .ok_or_else(|| std::io::Error::other("collection capacity × 8 overflow"))?;
274        let blob_body = blob.len().saturating_sub(8);
275        let cap_usize = usize::try_from(capacity).unwrap_or(usize::MAX);
276        let has_homes_tails = blob_body >= cap_usize.saturating_mul(24);
277
278        let slots_offset = body_offset;
279        let (homes_offset, tails_offset) = if has_homes_tails {
280            let homes_offset = body_offset
281                .checked_add(cap8)
282                .ok_or_else(|| std::io::Error::other("checkpoint homes offset overflow"))?;
283            let tails_offset = homes_offset
284                .checked_add(cap8)
285                .ok_or_else(|| std::io::Error::other("checkpoint tails offset overflow"))?;
286            (homes_offset, tails_offset)
287        } else {
288            (0, 0)
289        };
290        dir_entries.push(CollectionDirEntry {
291            collection_id: *collection_id,
292            generation: *generation,
293            slots_offset,
294            homes_offset,
295            tails_offset,
296            capacity: capacity32,
297            slot_count: blob_slot_count(blob),
298        });
299        let block_bytes = if has_homes_tails {
300            cap8.checked_mul(3)
301                .ok_or_else(|| std::io::Error::other("collection capacity × 24 overflow"))?
302        } else {
303            cap8
304        };
305        body_offset = body_offset
306            .checked_add(block_bytes)
307            .ok_or_else(|| std::io::Error::other("checkpoint body offset overflow"))?;
308    }
309
310    // Phase 2: write directory entries.
311    for entry in &dir_entries {
312        buf.extend_from_slice(&entry.encode());
313    }
314
315    // Phase 3: write slots, homes, tails as separate contiguous sections.
316    for (_, _, blob) in collections {
317        let capacity = blob_capacity(blob)
318            .ok_or_else(|| std::io::Error::other("collection slot blob missing capacity"))?;
319        let cap = usize::try_from(capacity).unwrap_or(usize::MAX);
320        let slots_end = 8_usize.saturating_add(cap.saturating_mul(8));
321        let homes_end = slots_end.saturating_add(cap.saturating_mul(8));
322        let tails_end = homes_end.saturating_add(cap.saturating_mul(8));
323        // Slots
324        if let Some(slots) = blob.get(8..slots_end) {
325            buf.extend_from_slice(slots);
326        }
327        // Homes
328        if let Some(homes) = blob.get(slots_end..homes_end) {
329            buf.extend_from_slice(homes);
330        }
331        // Tails
332        if let Some(tails) = blob.get(homes_end..tails_end) {
333            buf.extend_from_slice(tails);
334        }
335    }
336
337    let content_crc32 = crc32fast::hash(&buf[CHECKPOINT_HEADER_LEN..]);
338
339    let header = CheckpointHeader {
340        magic: CHECKPOINT_MAGIC,
341        version: CHECKPOINT_VERSION,
342        collection_count: count,
343        directory_bytes,
344        slots_bytes,
345        pack_fingerprint: fingerprint,
346        content_crc32,
347        homes_bytes,
348        tails_bytes,
349        covered_lsn,
350    };
351    buf[..CHECKPOINT_HEADER_LEN].copy_from_slice(&header.encode());
352
353    let unique = CHECKPOINT_TMP_COUNTER.fetch_add(1, Ordering::Relaxed);
354    let tmp_path = PathBuf::from(format!(
355        "{}.tmp.{}.{unique}",
356        path.display(),
357        std::process::id()
358    ));
359    let write_result = (|| -> std::io::Result<()> {
360        let mut tmp = fs::File::create(&tmp_path)?;
361        tmp.write_all(&buf)?;
362        // `sync_data` (fdatasync), not `sync_all` (fsync): per POSIX,
363        // fdatasync durably flushes the file's data plus whatever metadata
364        // is needed to retrieve it (the extended length included) — that's
365        // the whole content/length guarantee this write needs. It skips
366        // metadata that's *not* needed to read the data back (mtime/ctime).
367        // That's a real distinction, not a claim that all inode metadata is
368        // irrelevant here generally (permissions/ownership can still matter
369        // operationally) — it just happens that none of the metadata this
370        // skips is read back through this temp name: it's renamed over
371        // `path` immediately below and never opened by this name again. The
372        // caller (`persist_index_checkpoint`) separately fsyncs the base
373        // directory after the rename, which is what makes the rename itself
374        // durable; if that directory fsync fails, reopen falls back to
375        // whichever checkpoint (old or new) is actually intact rather than
376        // trusting the rename — this change doesn't alter that fallback.
377        tmp.sync_data()
378    })();
379    if let Err(e) = write_result {
380        let _ = fs::remove_file(&tmp_path);
381        return Err(e);
382    }
383    fs::rename(&tmp_path, path)
384}

386/// Read and validate a checkpoint. Returns `None` for any problem — a
387/// missing, truncated, or malformed file is treated identically to "no
388/// checkpoint", and the caller falls back to rebuilding from the packs.
389#[must_use]
390pub fn read_checkpoint(path: &Path) -> Option<LoadedCheckpoint> {
391    read_checkpoint_with_policy(path, CheckpointChecksumPolicy::from_env())
392}

394/// As [`read_checkpoint`], with an explicit integrity policy for callers that
395/// need deterministic configuration rather than the process environment.
396#[must_use]
397pub fn read_checkpoint_with_policy(
398    path: &Path,
399    checksum_policy: CheckpointChecksumPolicy,
400) -> Option<LoadedCheckpoint> {
401    let file = fs::File::open(path).ok()?;
402    let mmap = Arc::new(crate::packfile::map_pack(&file).ok()?);
403    let buf: &[u8] = &mmap;
404    let header_bytes: [u8; CHECKPOINT_HEADER_LEN] =
405        buf.get(..CHECKPOINT_HEADER_LEN)?.try_into().ok()?;
406    let header = CheckpointHeader::decode(&header_bytes)?;
407    if header.magic != CHECKPOINT_MAGIC || header.version != CHECKPOINT_VERSION {
408        return None;
409    }
410    let count = header.collection_count as usize;
411    if header.directory_bytes
412        != u64::try_from(count)
413            .ok()?
414            .checked_mul(COLLECTION_DIR_ENTRY_LEN as u64)?
415    {
416        return None;
417    }
418    let slot_base =
419        CHECKPOINT_HEADER_LEN.checked_add(usize::try_from(header.directory_bytes).ok()?)?;
420    let body_bytes = header
421        .slots_bytes
422        .checked_add(header.homes_bytes)
423        .and_then(|v| v.checked_add(header.tails_bytes))?;
424    if buf.len() != slot_base.checked_add(usize::try_from(body_bytes).ok()?)? {
425        return None;
426    }
427
428    // Verify the whole directory+slots body against the header's CRC in one
429    // pass, in place of re-deriving each collection's occupancy by walking
430    // its slots. This is strictly stronger than the old per-collection walk
431    // (it also covers the directory entries — `capacity`/`slot_count`
432    // themselves — which the walk never checked), and costs the same O(bytes)
433    // as a single SIMD-accelerated hash instead of a manual scan-and-branch
434    // loop over every slot of every collection.
435    if checksum_policy.verifies_reads() {
436        let content_crc32 = crc32fast::hash(buf.get(CHECKPOINT_HEADER_LEN..)?);
437        if content_crc32 != header.content_crc32 {
438            return None;
439        }
440    }
441
442    let mut collections = Vec::with_capacity(count);
443    for i in 0..count {
444        let entry_offset =
445            CHECKPOINT_HEADER_LEN.checked_add(i.checked_mul(COLLECTION_DIR_ENTRY_LEN)?)?;
446        let entry_bytes: [u8; COLLECTION_DIR_ENTRY_LEN] = buf
447            .get(entry_offset..entry_offset.saturating_add(COLLECTION_DIR_ENTRY_LEN))?
448            .try_into()
449            .ok()?;
450        let entry = CollectionDirEntry::decode(&entry_bytes)?;
451        if entry.capacity < 16 || !entry.capacity.is_power_of_two() {
452            return None;
453        }
454        // A corrupt or malicious directory entry could otherwise claim more
455        // occupied slots than the table has room for; a downstream consumer
456        // sizing an allocation off `slot_count` (e.g. a post-restart grow)
457        // would then trust a value with no upper bound instead of the
458        // checkpoint being rejected here. (An *understated* count — the
459        // failure mode the old occupancy walk additionally caught — is now
460        // caught by the CRC check above instead: any bit that makes
461        // `slot_count` disagree with the actual slots, or with what was
462        // written, changes the body's hash.)
463        if entry.slot_count > entry.capacity {
464            return None;
465        }
466        let slots_len = (entry.capacity as usize).checked_mul(8)?;
467        let region = slot_base.checked_add(usize::try_from(entry.slots_offset).ok()?)?;
468        if region.checked_add(slots_len)? > buf.len() {
469            return None;
470        }
471        let has_homes_tails = header.homes_bytes > 0 && header.tails_bytes > 0;
472        let homes_region = if has_homes_tails {
473            let r = slot_base.checked_add(usize::try_from(entry.homes_offset).ok()?)?;
474            if r.checked_add(slots_len)? > buf.len() {
475                return None;
476            }
477            r
478        } else {
479            0
480        };
481        let tails_region = if has_homes_tails {
482            let r = slot_base.checked_add(usize::try_from(entry.tails_offset).ok()?)?;
483            if r.checked_add(slots_len)? > buf.len() {
484                return None;
485            }
486            r
487        } else {
488            0
489        };
490        collections.push(LoadedCollection {
491            collection_id: entry.collection_id,
492            generation: entry.generation,
493            slots_offset: region,
494            homes_offset: homes_region,
495            tails_offset: tails_region,
496            capacity: entry.capacity,
497            slot_count: entry.slot_count,
498            has_homes_tails,
499        });
500    }
501
502    Some(LoadedCheckpoint {
503        fingerprint: header.pack_fingerprint,
504        covered_lsn: header.covered_lsn,
505        collections,
506        mmap,
507    })
508}

510/// Lightweight read of the persisted checkpoint's `pack_fingerprint`.
511///
512/// Returns `Ok(Some(fp))` for a valid checkpoint, `Ok(None)` if the file
513/// is missing (fresh store), and `Err` if the file exists but is truncated,
514/// has invalid magic/version, or an I/O error occurs.
515///
516/// # Errors
517///
518/// Returns a [`crate::index::delta::DeltaFingerprintError`] when an existing
519/// checkpoint cannot be read or validated.
520pub fn read_pack_fingerprint(
521    path: &Path,
522) -> Result<Option<u64>, crate::index::delta::DeltaFingerprintError> {
523    let file = match fs::File::open(path) {
524        Ok(f) => f,
525        Err(e) if e.kind() == io::ErrorKind::NotFound => return Ok(None),
526        Err(e) => return Err(crate::index::delta::io_error(path, "open checkpoint", e)),
527    };
528    let mut buf = [0u8; CHECKPOINT_HEADER_LEN];
529    let mut handle = file.take(CHECKPOINT_HEADER_LEN as u64);
530    handle
531        .read_exact(&mut buf)
532        .map_err(|e| crate::index::delta::io_error(path, "read checkpoint header", e))?;
533    let header = CheckpointHeader::decode(&buf)
534        .ok_or_else(|| crate::index::delta::invalid(path, "checkpoint header decode failed"))?;
535    if header.magic != CHECKPOINT_MAGIC || header.version != CHECKPOINT_VERSION {
536        return Err(crate::index::delta::invalid(
537            path,
538            "invalid checkpoint magic/version",
539        ));
540    }
541    Ok(Some(header.pack_fingerprint))
542}

554/// Read the durable pack fingerprint for a store directory.
555///
556/// The durable fingerprint is the checkpoint's `pack_fingerprint`, advanced
557/// by the validated delta log's `tail_fingerprint` when one exists and its
558/// `base_fingerprint` matches the checkpoint.  This detects synced appends
559/// that only wrote to the delta log without rewriting the checkpoint.
560///
561/// Returns `Ok(Some(fp))` for a validated durable fingerprint, `Ok(None)`
562/// when there is no checkpoint (fresh store), and `Err` when the checkpoint
563/// is corrupt or the delta log cannot be reliably read. On `Err` the caller
564/// must treat the state as unknown and force a refresh.
565///
566/// # Errors
567///
568/// Returns a [`crate::index::delta::DeltaFingerprintError`] when the delta
569/// log exists but cannot be validated or read reliably.
570pub fn read_durable_fingerprint(
571    base_dir: &Path,
572) -> Result<Option<DurableFingerprint>, crate::index::delta::DeltaFingerprintError> {
573    let ckpt_path = base_dir.join(INDEX_CHECKPOINT_FILE);
574    let Some(ckpt_fp) = read_pack_fingerprint(&ckpt_path)? else {
575        return Ok(None);
576    };
577    let delta_path = base_dir.join(format!(
578        "{}.{:016x}",
579        crate::index::delta::INDEX_DELTA_FILE,
580        ckpt_fp,
581    ));
582    // Use the lightweight forward parser to avoid decoding every DeltaFrame.
583    match crate::index::delta::read_delta_tail_fingerprint(&delta_path) {
584        Ok(Some(tail_fp)) if tail_fp.base_fingerprint == ckpt_fp => Ok(Some(DurableFingerprint {
585            fingerprint: tail_fp.tail_fingerprint,
586            torn_tail: tail_fp.torn_tail,
587        })),
588        Ok(Some(_) | None) => Ok(Some(DurableFingerprint {
589            fingerprint: ckpt_fp,
590            torn_tail: false,
591        })),
592        Err(e) => Err(e),
593    }
594}

602    fn hash_for(seed: u16, i: usize) -> [u8; 16] {
603        let mut hash = [0u8; 16];
604        hash[..2].copy_from_slice(&seed.wrapping_add(u16::try_from(i).unwrap()).to_le_bytes());
605        hash
606    }
607
608    fn index_with_entries(seed: u16, count: usize) -> LossyIndex {
609        let index = LossyIndex::new(count.max(16).saturating_mul(2));
610        for i in 0..count {
611            let shard = u16::try_from(i % 64).unwrap();
612            let _ = index.insert(&hash_for(seed, i), shard, (i as u64).wrapping_mul(128));
613        }
614        index
615    }

637    #[test]
638    fn checkpoint_round_trips_index_slots() {
639        let cases: Vec<([u8; 16], u16, usize)> =
640            vec![([1u8; 16], 1, 40), ([2u8; 16], 2, 200), ([3u8; 16], 3, 5)];
641        let indexes: Vec<LossyIndex> = cases
642            .iter()
643            .map(|(_, seed, count)| index_with_entries(*seed, *count))
644            .collect();
645        let blobs: Vec<([u8; 16], Vec<u8>)> = cases
646            .iter()
647            .zip(&indexes)
648            .map(|((id, _, _), index)| (*id, index.serialize()))
649            .collect();
650        let dir = std::env::temp_dir().join(format!("mtxdb_checkpoint_{}", std::process::id()));
651        let _ = std::fs::remove_dir_all(&dir);
652        std::fs::create_dir_all(&dir).unwrap();
653        let path = dir.join(INDEX_CHECKPOINT_FILE);
654
655        let fingerprint = pack_fingerprint(&[(7, 12345)]);
656        write_checkpoint(
657            &path,
658            fingerprint,
659            0,
660            &blobs
661                .iter()
662                .map(|(id, blob)| (*id, 0, blob.as_slice()))
663                .collect::<Vec<_>>(),
664        )
665        .unwrap();
666
667        let loaded = read_checkpoint(&path).unwrap_or_else(|| {
668            panic!(
669                "valid checkpoint should read; file has {} bytes",
670                std::fs::read(&path).map_or(0, |b| b.len())
671            )
672        });
673        assert_eq!(loaded.fingerprint, fingerprint);
674        assert_eq!(
675            loaded.collections.len(),
676            3,
677            "checkpoint must preserve directory order and count"
678        );
679        let mmap = Arc::clone(&loaded.mmap);
680        for ((case, loaded), (_blob_id, expected_blob)) in
681            cases.iter().zip(&loaded.collections).zip(&blobs)
682        {
683            let (expected_id, seed, count) = case;
684            assert_eq!(loaded.collection_id, *expected_id);
685            let slots_len = (loaded.capacity as usize).saturating_mul(8);
686            assert_eq!(
687                &mmap[loaded.slots_offset..loaded.slots_offset.saturating_add(slots_len)],
688                &expected_blob[8..8 + slots_len],
689                "slots must round-trip verbatim"
690            );
691
692            // The mmap-backed index must find every hash the live index had.
693            let loaded_index = if loaded.has_homes_tails {
694                LossyIndex::from_mmap_slots_with_homes_tails(
695                    Arc::clone(&mmap),
696                    loaded.slots_offset,
697                    loaded.homes_offset,
698                    loaded.tails_offset,
699                    loaded.capacity,
700                    loaded.slot_count,
701                    IndexConfig::default(),
702                )
703            } else {
704                LossyIndex::from_mmap_slots(
705                    Arc::clone(&mmap),
706                    loaded.slots_offset,
707                    loaded.capacity,
708                    loaded.slot_count,
709                )
710            };
711            let original_index = LossyIndex::deserialize(expected_blob).unwrap();
712            for i in 0..*count {
713                let hash = hash_for(*seed, i);
714                assert_eq!(
715                    loaded_index.lookup(&hash),
716                    original_index.lookup(&hash),
717                    "loaded index must agree with the source index on spot lookups"
718                );
719                assert!(
720                    loaded_index.lookup(&hash).is_some(),
721                    "round-tripped index lost a record"
722                );
723            }
724        }
725
726        std::fs::remove_dir_all(&dir).unwrap();
727    }

797    #[test]
798    fn content_crc32_catches_value_preserving_slot_corruption() {
799        // The pre-CRC occupancy walk only ever compared a *count* of
800        // non-empty slots against `slot_count` — a corruption that flips
801        // bits within an already-occupied slot (changing which shard/offset
802        // it names, without zeroing it) preserves that count and would have
803        // silently passed. The CRC covers the exact bytes, so this same
804        // corruption must now be rejected.
805        let dir = std::env::temp_dir().join(format!(
806            "mtxdb_checkpoint_slot_corruption_{}",
807            std::process::id()
808        ));
809        let _ = std::fs::remove_dir_all(&dir);
810        std::fs::create_dir_all(&dir).unwrap();
811        let path = dir.join(INDEX_CHECKPOINT_FILE);
812
813        let blobs = [([7u8; 16], index_with_entries(3, 40).serialize())];
814        write_checkpoint(
815            &path,
816            0,
817            0,
818            &blobs
819                .iter()
820                .map(|(id, b)| (*id, 0, b.as_slice()))
821                .collect::<Vec<_>>(),
822        )
823        .unwrap();
824        assert!(
825            read_checkpoint(&path).is_some(),
826            "sanity check: the unmodified checkpoint must read back cleanly"
827        );
828
829        let mut bytes = std::fs::read(&path).unwrap();
830        // Flip one bit inside the slots section (well past the header +
831        // single directory entry) in a byte that is currently non-zero, so
832        // the slot stays non-empty — only its value changes.
833        let slots_start = CHECKPOINT_HEADER_LEN + COLLECTION_DIR_ENTRY_LEN;
834        let corrupt_at = bytes[slots_start..]
835            .iter()
836            .position(|&b| b != 0)
837            .map(|offset| slots_start + offset)
838            .expect("at least one non-zero slot byte exists");
839        bytes[corrupt_at] ^= 0x01;
840        std::fs::write(&path, &bytes).unwrap();
841
842        assert!(
843            read_checkpoint_with_policy(&path, CheckpointChecksumPolicy::Full).is_none(),
844            "a value-preserving bit flip inside an occupied slot must be rejected"
845        );
846        assert!(
847            read_checkpoint_with_policy(&path, CheckpointChecksumPolicy::WriteOnly).is_some(),
848            "write-only mode deliberately trusts a structurally valid checkpoint"
849        );
850
851        std::fs::remove_dir_all(&dir).unwrap();
852    }
