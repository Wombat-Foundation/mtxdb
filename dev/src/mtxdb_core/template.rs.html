1//! Executable policy primitives for format-specific collection templates.
2//!
3//! Storage remains format-agnostic. Importers bind a collection to one of
4//! these policies and persist the resulting metadata alongside their index.
5
6/// Current generic collection-template format identifier.
7pub const COLLECTION_TEMPLATE_FORMAT_V1: &str = "mtxdb.collection-template/v1";
8
9/// How a record's stored payload is selected.
10#[derive(Debug, Clone, PartialEq, Eq)]
11pub enum PayloadPolicy {
12    /// Retain the complete received source record.
13    Source,
14    /// Retain only an explicit, derived projection of the source record.
15    Projection {
16        /// RFC 6901 pointers to fields that should be included in the projection.
17        include: Vec<String>,
18    },
19}
20
21/// Generic identity rule for an application record.
22#[derive(Debug, Clone, PartialEq, Eq)]
23pub struct RecordIdentityRule {
24    /// RFC 6901 pointer to a stable, source-level identity.
25    pub pointer: String,
26    /// Digest algorithm used to derive mtxdb's internal node key.
27    pub node_id_algorithm: String,
28}
29
30/// Generic rule used to assign a record to a collection.
31#[derive(Debug, Clone, PartialEq, Eq)]
32pub struct CollectionKeyRule {
33    /// RFC 6901 pointer to the source-level collection key.
34    pub pointer: String,
35    /// Digest algorithm used to derive mtxdb's internal collection key.
36    pub collection_id_algorithm: String,
37    /// RFC 6901 pointer to the user-facing collection identifier.
38    pub display_id_pointer: String,
39}
40
41/// Format-neutral, executable description of a collection template.
42///
43/// Protocol rules such as Matrix redaction and authorization deliberately do
44/// not belong here. They are consumers of this base model and are represented
45/// by a namespaced template extension.
46#[derive(Debug, Clone, PartialEq, Eq)]
47pub struct CollectionTemplate {
48    /// Application-defined template name.
49    pub name: String,
50    /// Application-defined collection kind; opaque to storage.
51    pub collection_kind: String,
52    /// Stable record identity and internal node-key derivation rule.
53    pub record_identity: RecordIdentityRule,
54    /// Source payload retention rule.
55    pub payload: PayloadPolicy,
56    /// Collection membership and internal collection-key derivation rule.
57    pub collection_key: CollectionKeyRule,
58}

60#[cfg(test)]
61mod tests {
62    use super::*;
63
64    #[test]
65    fn generic_template_does_not_require_a_protocol_extension() {
66        let template = CollectionTemplate {
67            name: "documents".into(),
68            collection_kind: "notebook".into(),
69            record_identity: RecordIdentityRule {
70                pointer: "/uuid".into(),
71                node_id_algorithm: "blake3-128".into(),
72            },
73            payload: PayloadPolicy::Source,
74            collection_key: CollectionKeyRule {
75                pointer: "/notebook".into(),
76                collection_id_algorithm: "blake3-128".into(),
77                display_id_pointer: "/notebook".into(),
78            },
79        };
80
81        assert_eq!(
82            COLLECTION_TEMPLATE_FORMAT_V1,
83            "mtxdb.collection-template/v1"
84        );
85        assert_eq!(template.payload, PayloadPolicy::Source);
86        assert_eq!(template.collection_key.display_id_pointer, "/notebook");
87    }
88}
