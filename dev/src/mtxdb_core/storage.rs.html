6/// A node ID is a 16-byte structural hash.
7pub type NodeId = [u8; 16];
8
9/// Opaque node data as raw bytes (the encoded HAMT node or PDU).
10#[derive(Debug, Clone)]
11pub struct NodeData {
12    /// The raw encoded node bytes.
13    pub bytes: bytes::Bytes,
14    /// Child references, if this node has been decoded/swizzled.
15    pub children: Vec<NodeRef>,
16}
17
18impl NodeData {
19    /// Wrap raw bytes as node data with no resolved children.
20    pub fn new(bytes: bytes::Bytes) -> Self {
21        Self {
22            bytes,
23            children: Vec::new(),
24        }
25    }
26}

28/// A reference to a node that may be resident in cache or need disk fetch.
29///
30/// This is the swizzling enum inspired by `LeanStore`:
31/// - `Lazy(id)`: the node is on disk, identified by its hash.
32/// - `Resolved(hash, data)`: the node is in memory, ready for use,
33///   with its structural hash stored alongside.
34#[derive(Debug, Clone)]
35pub enum NodeRef {
36    /// The node lives on disk, identified by its hash.
37    Lazy(NodeId),
38    /// The node is resolved in memory, alongside its structural hash.
39    Resolved(NodeId, Arc<NodeData>),
40}
41
42impl NodeRef {
43    /// The structural hash of the referenced node, regardless of residency.
44    #[must_use]
45    pub fn structural_hash(&self) -> &NodeId {
46        match self {
47            Self::Lazy(id) | Self::Resolved(id, _) => id,
48        }
49    }
50
51    // jscpd:ignore-start
52    // False-positive match against index/mod.rs's splitmix64 (a test-only
53    // hash mixer) — token-shape coincidence, not related logic. See the
54    // "why can't it be fixed" discussion: nothing to extract here.
55    /// The resolved node data, if this reference is already in memory.
56    #[must_use]
57    pub fn data(&self) -> Option<&Arc<NodeData>> {
58        match self {
59            Self::Lazy(_) => None,
60            Self::Resolved(_, data) => Some(data),
61        }
62    }
63    // jscpd:ignore-end
64
65    /// Returns `true` if this reference is already resolved in memory.
66    #[must_use]
67    pub fn is_resolved(&self) -> bool {
68        matches!(self, Self::Resolved(..))
69    }
70}

72/// The storage engine trait. Abstracts the backend so the packfile,
73/// index, cache, and frontier code don't depend on a specific engine.
74///
75/// Every operation is scoped to a single collection. The caller always knows
76/// which collection a node belongs to; the engine uses this to select the
77/// correct per-collection index and packfile, keeping each collection's active index
78/// at ~8KB (100 active collections < 1MB total).
79///
80/// Implementations:
81/// - `PackfileStorage`: the custom append-only packfile with lossy index.
82/// - `InMemoryStorage`: for tests.
83pub trait StorageEngine: Send + Sync {
84    /// Fetch a single node by its structural hash within a collection.
85    ///
86    /// # Errors
87    /// Returns `StorageError::Io` on I/O failure.
88    fn get(&self, collection_id: &[u8; 16], id: &NodeId) -> Result<Option<NodeData>, StorageError>;
89
90    /// Fetch multiple nodes by their structural hashes within a collection.
91    /// Returns results in the same order as the input keys.
92    ///
93    /// # Errors
94    /// Returns `StorageError::Io` on I/O failure.
95    fn get_many(
96        &self,
97        collection_id: &[u8; 16],
98        ids: &[NodeId],
99    ) -> Result<Vec<Option<NodeData>>, StorageError>;
100
101    /// Store a new node within a collection. The caller must ensure the node
102    /// is not already present (content-addressed: identical data produces
103    /// identical hash).
104    ///
105    /// # Errors
106    /// Returns `StorageError::Io` on I/O failure.
107    fn put(
108        &self,
109        collection_id: &[u8; 16],
110        id: &NodeId,
111        data: &NodeData,
112    ) -> Result<(), StorageError>;
113
114    /// Store multiple new nodes in a single batch within a collection.
115    ///
116    /// # Errors
117    /// Returns `StorageError::Io` on I/O failure.
118    fn put_many(
119        &self,
120        collection_id: &[u8; 16],
121        entries: &[(NodeId, NodeData)],
122    ) -> Result<(), StorageError>;
123
124    /// Delete all nodes for a given collection (range delete).
125    ///
126    /// # Errors
127    /// Returns `StorageError::Io` on I/O failure.
128    fn delete_collection(&self, collection_id: &[u8; 16]) -> Result<(), StorageError>;
129
130    /// Sync to disk (fsync).
131    ///
132    /// # Errors
133    /// Returns `StorageError::Io` on I/O failure.
134    fn sync(&self) -> Result<(), StorageError>;
135
136    /// Force a re-scan of a collection's shards from disk and atomically
137    /// swap in the freshly-built index, picking up records another process
138    /// wrote after this one last loaded (or never loaded) the collection.
139    ///
140    /// Every implementation's in-memory index is private to the process
141    /// that built it (see `PackfileStorage`'s per-process `LossyIndex`) --
142    /// there is no ambient cross-process invalidation, so a multi-worker
143    /// deployment must call this explicitly on a suspected miss to pull in
144    /// another worker's writes. Default no-op: engines with no on-disk
145    /// shard files of their own to rescan (e.g. `InMemoryStorage`, used
146    /// only in single-process tests) have nothing to refresh.
147    ///
148    /// # Errors
149    /// Returns `StorageError::Io` on I/O failure re-reading shard files.
150    fn refresh_collection(&self, _collection_id: &[u8; 16]) -> Result<(), StorageError> {
151        Ok(())
152    }
153}

155/// Errors returned by [`StorageEngine`] operations.
156#[derive(Debug)]
157pub enum StorageError {
158    /// An underlying I/O operation failed.
159    Io(std::io::Error),
160    /// The requested node was not found.
161    NotFound(NodeId),
162    /// The node's content did not match its requested hash.
163    VerificationFailed(NodeId),
164    /// The stored data is malformed or fails an integrity check.
165    Corrupt(String),
166    /// An internal invariant was violated.
167    Internal(String),
168}
169
170impl std::fmt::Display for StorageError {
171    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
172        match self {
173            Self::Io(e) => write!(f, "I/O error: {e}"),
174            Self::NotFound(id) => write!(f, "node not found: {id:?}"),
175            Self::VerificationFailed(id) => write!(f, "verification failed for node {id:?}"),
176            Self::Corrupt(msg) => write!(f, "corrupt data: {msg}"),
177            Self::Internal(msg) => write!(f, "internal error: {msg}"),
178        }
179    }
180}
181
182impl std::error::Error for StorageError {
183    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
184        match self {
185            Self::Io(e) => Some(e),
186            _ => None,
187        }
188    }
189}
190
191impl From<std::io::Error> for StorageError {
192    fn from(e: std::io::Error) -> Self {
193        Self::Io(e)
194    }
195}
196
197impl From<StorageError> for std::io::Error {
198    fn from(error: StorageError) -> Self {
199        match error {
200            StorageError::Io(error) => error,
201            StorageError::Corrupt(message) => {
202                std::io::Error::new(std::io::ErrorKind::InvalidData, message)
203            }
204            other => std::io::Error::other(other.to_string()),
205        }
206    }
207}

209/// In-memory storage engine for tests.
210///
211/// Partitions nodes by collection. Each collection's nodes are tracked in a
212/// per-collection `HashMap`, enabling correct `delete_collection` behavior.
213pub struct InMemoryStorage {
214    collections: RwLock<HashMap<[u8; 16], HashMap<NodeId, NodeData>>>,
215}
216
217impl InMemoryStorage {
218    /// Create an empty in-memory storage engine.
219    #[must_use]
220    pub fn new() -> Self {
221        Self {
222            collections: RwLock::new(HashMap::new()),
223        }
224    }
225}
226
227impl Default for InMemoryStorage {
228    fn default() -> Self {
229        Self::new()
230    }
231}
232
233impl StorageEngine for InMemoryStorage {
234    fn get(&self, collection_id: &[u8; 16], id: &NodeId) -> Result<Option<NodeData>, StorageError> {
235        let collections = self.collections.read();
236        Ok(collections
237            .get(collection_id)
238            .and_then(|r| r.get(id).cloned()))
239    }
240
241    fn get_many(
242        &self,
243        collection_id: &[u8; 16],
244        ids: &[NodeId],
245    ) -> Result<Vec<Option<NodeData>>, StorageError> {
246        let collections = self.collections.read();
247        let collection = collections.get(collection_id);
248        Ok(ids
249            .iter()
250            .map(|id| collection.and_then(|r| r.get(id).cloned()))
251            .collect())
252    }
253
254    fn put(
255        &self,
256        collection_id: &[u8; 16],
257        id: &NodeId,
258        data: &NodeData,
259    ) -> Result<(), StorageError> {
260        self.collections
261            .write()
262            .entry(*collection_id)
263            .or_default()
264            .insert(*id, data.clone());
265        Ok(())
266    }
267
268    fn put_many(
269        &self,
270        collection_id: &[u8; 16],
271        entries: &[(NodeId, NodeData)],
272    ) -> Result<(), StorageError> {
273        let mut collections = self.collections.write();
274        let collection = collections.entry(*collection_id).or_default();
275        for (id, data) in entries {
276            collection.insert(*id, data.clone());
277        }
278        Ok(())
279    }
280
281    fn delete_collection(&self, collection_id: &[u8; 16]) -> Result<(), StorageError> {
282        self.collections.write().remove(collection_id);
283        Ok(())
284    }
285
286    fn sync(&self) -> Result<(), StorageError> {
287        Ok(())
288    }
289}

318        let store = InMemoryStorage::new();
319        let entries: Vec<(NodeId, NodeData)> = (0..10)
320            .map(|i| {
321                let mut id = [0u8; 16];
322                id[0] = i;

325            .collect();

330        let results = store.get_many(&TEST_COLLECTION, &ids).unwrap();
331        assert_eq!(results.len(), 10);
332        for (i, result) in results.iter().enumerate() {
333            assert!(result.is_some());
