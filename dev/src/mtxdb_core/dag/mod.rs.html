3/// Dense local ID for a collection. No single collection exceeds 2^31 events.
4pub type LocalId = u32;
5
6/// A compressed sparse row (CSR) arena for the active event DAG.
7///
8/// This is the on-disk-compatible form of the in-memory DAG frontier.
9/// Events and their edges arrive sequentially, building contiguously.
10///
11/// Design from docs:
12/// - Two heap allocations per collection: `nodes` Vec and `edges` Vec.
13/// - Edges of a node are contiguous in the `edges` array (cache line).
14/// - O(1) drop: `nodes.clear()` + `edges.clear()`.
15/// - No self-referential structs (compiles in safe Rust).
16///
17/// Usage:
18/// - Resident edges use `GraphEdge::resident(slot)`.
19/// - Disk edges use `GraphEdge::disk(short_id)`.
20#[derive(Debug, Clone, Copy, PartialEq, Eq)]
21pub struct GraphEdge(u32);
22
23impl GraphEdge {
24    const RESIDENT_BIT: u32 = 1 << 31;
25    const PAYLOAD_MASK: u32 = !Self::RESIDENT_BIT;
26
27    #[inline]
28    #[must_use]
29    /// Returns whether this edge points to an arena-resident node.
30    pub fn is_resident(self) -> bool {
31        (self.0 & Self::RESIDENT_BIT) != 0
32    }
33
34    /// Returns the arena index for a resident edge.
35    ///
36    /// # Panics
37    /// Panics if this is a disk-resident edge.
38    #[inline]
39    #[must_use]
40    pub fn arena_index(self) -> usize {
41        assert!(
42            self.is_resident(),
43            "arena_index called on a disk-resident node"
44        );
45        (self.0 & Self::PAYLOAD_MASK) as usize
46    }
47
48    /// Returns the local ID for a disk-resident edge.
49    ///
50    /// # Panics
51    /// Panics if this is an arena-resident edge.
52    #[inline]
53    #[must_use]
54    pub fn local_id(self) -> LocalId {
55        assert!(
56            !self.is_resident(),
57            "local_id called on an arena-resident node"
58        );
59        self.0 & Self::PAYLOAD_MASK
60    }
61
62    /// Create a resident edge pointing to an arena index.
63    ///
64    /// # Panics
65    /// Panics if `index` exceeds the 31-bit payload capacity.
66    #[inline]
67    #[must_use]
68    pub fn resident(index: u32) -> Self {
69        assert!(
70            index <= Self::PAYLOAD_MASK,
71            "resident index exceeds payload capacity: {index} > {}",
72            Self::PAYLOAD_MASK
73        );
74        Self(Self::RESIDENT_BIT | (index & Self::PAYLOAD_MASK))
75    }
76
77    /// Create a disk-resident edge with a local ID.
78    ///
79    /// # Panics
80    /// Panics if `id` exceeds the 31-bit payload capacity.
81    #[inline]
82    #[must_use]
83    pub fn disk(id: LocalId) -> Self {
84        assert!(
85            id <= Self::PAYLOAD_MASK,
86            "disk id exceeds payload capacity: {id} > {}",
87            Self::PAYLOAD_MASK
88        );
89        Self(id & Self::PAYLOAD_MASK)
90    }
91}

93/// A node in the event DAG.
94#[derive(Debug, Clone)]
95pub struct EventNode {
96    /// Global short event ID.
97    pub short_id: u64,
98    /// Dense local ID within this collection.
99    pub local_id: LocalId,
100    /// Range into the edges array for `prev_events`: (start, len).
101    pub prev: (u32, u32),
102    /// Range into the edges array for `auth_events`: (start, len).
103    pub auth: (u32, u32),
104}
105
106/// CSR arena for an active collection's event DAG.
107///
108/// Contains all events loaded for an active collection, with edges stored
109/// contiguously in a single Vec. Graph algorithms like topological
110/// sort are linear scans over this array.
111#[derive(Debug)]
112pub struct ActiveRoomFrontier {
113    /// All event nodes in the collection.
114    pub nodes: Vec<EventNode>,
115    /// Contiguous edge storage. Edges for a node are at `edges[start..start+len]`.
116    pub edges: Vec<GraphEdge>,
117    /// Maps global `short_id` → index into nodes Vec.
118    pub resident: HashMap<u64, u32>,
119    /// Maps global `short_id` → dense `local_id` for this collection.
120    pub id_remap: HashMap<u64, LocalId>,
121    /// Next `local_id` to assign.
122    next_local_id: LocalId,
123}
124
125impl ActiveRoomFrontier {
126    /// Create an empty frontier arena.
127    #[must_use]
128    pub fn new() -> Self {
129        Self {
130            nodes: Vec::new(),
131            edges: Vec::new(),
132            resident: HashMap::new(),
133            id_remap: HashMap::new(),
134            next_local_id: 0,
135        }
136    }
137
138    /// Register a global `short_id` and assign it a dense `local_id`.
139    pub fn register_id(&mut self, short_id: u64) -> LocalId {
140        if let Some(&local) = self.id_remap.get(&short_id) {
141            return local;
142        }
143        let local = self.next_local_id;
144        self.next_local_id = self.next_local_id.saturating_add(1);
145        self.id_remap.insert(short_id, local);
146        local
147    }
148
149    /// Resolve a parent edge: if the parent is in the arena, use a resident
150    /// edge; if it has a `local_id`, use a disk edge; otherwise assign a new
151    /// `local_id` and use a disk edge.
152    fn resolve_edge(&mut self, parent_id: u64) -> GraphEdge {
153        let local = match self.id_remap.get(&parent_id) {
154            Some(&local) => local,
155            None => self.register_id(parent_id),
156        };
157        if let Some(&slot) = self.resident.get(&parent_id) {
158            GraphEdge::resident(slot)
159        } else {
160            GraphEdge::disk(local)
161        }
162    }
163
164    /// Insert an event with its raw prev and auth edges.
165    ///
166    /// For each parent in `raw_prevs` and `raw_auths`:
167    /// - If the parent is already in the arena, creates a `resident` edge.
168    /// - Otherwise, creates a `disk` edge with the parent's `local_id`.
169    ///
170    /// # Panics
171    /// Panics if the arena index exceeds `u32::MAX`.
172    pub fn insert_event(&mut self, short_id: u64, raw_prevs: &[u64], raw_auths: &[u64]) -> usize {
173        let local_id = self.register_id(short_id);
174
175        let node_idx: u32 =
176            u32::try_from(self.nodes.len()).expect("too many events for u32 arena index");
177
178        let prev_start: u32 =
179            u32::try_from(self.edges.len()).expect("too many edges for u32 arena index");
180        for &parent_id in raw_prevs {
181            let edge = self.resolve_edge(parent_id);
182            self.edges.push(edge);
183        }
184        let prev_end: u32 =
185            u32::try_from(self.edges.len()).expect("too many edges for u32 arena index");
186        let prev_len = prev_end
187            .checked_sub(prev_start)
188            .expect("prev_len underflow");
189
190        let auth_start: u32 =
191            u32::try_from(self.edges.len()).expect("too many edges for u32 arena index");
192        for &parent_id in raw_auths {
193            let edge = self.resolve_edge(parent_id);
194            self.edges.push(edge);
195        }
196        let auth_end: u32 =
197            u32::try_from(self.edges.len()).expect("too many edges for u32 arena index");
198        let auth_len = auth_end
199            .checked_sub(auth_start)
200            .expect("auth_len underflow");
201
202        self.nodes.push(EventNode {
203            short_id,
204            local_id,
205            prev: (prev_start, prev_len),
206            auth: (auth_start, auth_len),
207        });
208        self.resident.insert(short_id, node_idx);
209
210        node_idx as usize
211    }
212
213    /// Get the `prev_edges` for a node by its arena index.
214    ///
215    /// # Panics
216    /// Panics if `node_idx` is out of bounds.
217    #[must_use]
218    pub fn prev_edges(&self, node_idx: usize) -> &[GraphEdge] {
219        let node = &self.nodes[node_idx];
220        let start = node.prev.0 as usize;
221        let end = start.wrapping_add(node.prev.1 as usize);
222        &self.edges[start..end]
223    }
224
225    /// Get the `auth_edges` for a node by its arena index.
226    ///
227    /// # Panics
228    /// Panics if `node_idx` is out of bounds.
229    #[must_use]
230    pub fn auth_edges(&self, node_idx: usize) -> &[GraphEdge] {
231        let node = &self.nodes[node_idx];
232        let start = node.auth.0 as usize;
233        let end = start.wrapping_add(node.auth.1 as usize);
234        &self.edges[start..end]
235    }
236
237    /// Drop all data for this collection. O(1) — replaces backing stores
238    /// with empty containers; the old data is dropped when the returned
239    /// values go out of scope.
240    pub fn clear(&mut self) {
241        self.nodes = Vec::new();
242        self.edges = Vec::new();
243        self.resident = HashMap::new();
244        self.id_remap = HashMap::new();
245        self.next_local_id = 0;
246    }
247
248    /// Number of events in this collection's frontier.
249    #[must_use]
250    pub fn len(&self) -> usize {
251        self.nodes.len()
252    }
253
254    /// Returns `true` if this collection's frontier has no events.
255    #[must_use]
256    pub fn is_empty(&self) -> bool {
257        self.nodes.is_empty()
258    }
259}

272    #[test]
273    fn test_graph_edge_packing() {
274        let resident = GraphEdge::resident(42);
275        assert!(resident.is_resident());
276        assert_eq!(resident.arena_index(), 42);
277
278        let disk = GraphEdge::disk(12345);
279        assert!(!disk.is_resident());
280        assert_eq!(disk.local_id(), 12345);

285        let mut frontier = ActiveRoomFrontier::new();
286
287        let idx_a = frontier.insert_event(100, &[], &[]);
288        assert_eq!(idx_a, 0);
289        assert_eq!(frontier.len(), 1);
290
291        let idx_b = frontier.insert_event(101, &[100], &[100]);
292        assert_eq!(idx_b, 1);

295        let prev = frontier.prev_edges(idx_b);
296        assert_eq!(prev.len(), 1);
