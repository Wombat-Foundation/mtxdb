23/// Maximum shard size before rotation: `2^32 - 2` bytes (~4 GiB), the
24/// largest value for which every offset a shard can ever produce still
25/// fits `IndexSlot`'s 32-bit offset field (which reserves its all-zero
26/// encoding as the empty-slot sentinel, capping the max representable
27/// offset at `2^32 - 2` — see `index::IndexSlot`). Do not round this up
28/// to a clean `4 * 1024 * 1024 * 1024`: that's one byte over the ceiling and
29/// lets a shard produce an offset `IndexSlot::new` panics on.
30pub const MAX_SHARD_BYTES: u64 = (1u64 << 32) - 2;
31
32/// Default in-memory threshold before a shard's buffered frames are written
33/// to disk as one positioned write. Keeps bulk writes from paying a
34/// `pwrite` (plus the per-record `try_clone` and file-length store) for
35/// every single record — the whole remaining bulk-write gap vs the bench's
36/// mdbx (which writes a memory-mapped in-memory transaction and pays
37/// persistence once at commit). The default [`AppendPolicy`] is
38/// [`AppendPolicy::Eager`], so this threshold only applies to pools opened
39/// with [`AppendPolicy::Buffered`].
40pub(crate) const PENDING_FLUSH_BYTES: usize = 1 << 20;
41
42/// When a shard's appended frames are written to its underlying pack file.
43///
44/// The default is [`AppendPolicy::Eager`]: every `put_record` commits its
45/// own frame with one positioned write, matching the engine's historical
46/// behavior. An unsynced put is therefore immediately visible to other
47/// processes reading the pack files, and survives this process exiting (a
48/// crash or power loss still requires an explicit `sync_*` to be durable).
49///
50/// [`AppendPolicy::Buffered`] defers those writes into one positioned write
51/// per shard once `max_pending_bytes` of frames have accumulated (or at the
52/// rotation/sync/read/repack/checkpoint boundaries that flush first). This
53/// is far faster for bulk writes — one `pwrite` per ~1 MiB instead of one
54/// per record — but it changes the durability boundary:
55///
56/// **Records since the last flush are lost on process crash.**
57///
58/// Unflushed bytes live only in this process's memory, are invisible to
59/// other processes, and are gone if this process exits — there is
60/// deliberately no recovery path for them. (Recovery machinery here only
61/// covers bytes already committed to a pack file: checkpoint writes flush
62/// first, so the persisted fingerprint and index describe only committed
63/// lengths, never buffered ones.) A crash therefore leaves on-disk state
64/// self-consistent but drops everything still in the buffer. Buffered is a
65/// pure throughput option trading crash-loss of the unflushed tail; opt
66/// into it only where the caller already flushes or syncs on its own
67/// durability schedule (e.g. a batch import that ends in `sync_all`).
68#[derive(Clone, Copy, Debug, PartialEq, Eq)]
69pub enum AppendPolicy {
70    /// Commit each record's frame with its own positioned write, exactly as
71    /// mtxdb always did before buffering existed.
72    Eager,
73    /// Accumulate frames in memory and write them out once
74    /// `max_pending_bytes` have been buffered for a shard.
75    Buffered {
76        /// Per-shard in-memory flush threshold, in buffered frame bytes.
77        max_pending_bytes: usize,
78    },
79}
80
81impl AppendPolicy {
82    /// [`AppendPolicy::Buffered`] at the default flush threshold (~1 MiB).
83    #[must_use]
84    pub fn buffered() -> Self {
85        Self::Buffered {
86            max_pending_bytes: PENDING_FLUSH_BYTES,
87        }
88    }
89}

97    /// Ephemeral, recycled on retire. Never shown to operators.

100    /// Ensures a reused slot never collides on-disk with a still-referenced

103    /// The open file handle backing this shard.
104    pub file: File,
105    /// Filesystem path to this shard's file.

108    // Keep mappings reference-counted: a raw, uncompressed `Bytes` returned

110    // remap can therefore replace this slot without invalidating an already
111    // returned node.

113    /// Serializes appends to this shard.
114    pub(crate) append_lock: parking_lot::Mutex<()>,

117    /// fsync, then finds the dirty bit already cleared and skips its own —

126    /// Current file length, tracked atomically for rotation decisions
127    /// without a `metadata()` syscall on every put. This is the *committed*
128    /// on-disk length — buffered records occupy `[file_len, file_len +

131    /// Encoded frame bytes buffered since the last flush of this shard,
132    /// waiting to be written as one positioned `pwrite`. Guarded by

134    /// interior mutability everything else accesses via `Arc<Shard>`.
135    /// Index offsets returned by `put_record` point into the virtual
136    /// `[file_len, file_len + pending.len())` region.
137    pub(crate) pending: Mutex<Vec<u8>>,
138    /// Number of records represented by the buffered frames (every frame
139    /// carries a variable-length body, so this cannot be derived from
140    /// `pending.len()`). Guarded by `append_lock` and the `pending` mutex;
141    /// credited to `write_count` at flush time.
142    pub(crate) pending_records: AtomicU64,
143    /// Number of records appended to this shard.
144    write_count: AtomicU64,
145    /// Total payload bytes appended to this shard (serialized record length).
146    bytes_written: AtomicU64,
147    /// Number of times this shard's file has been fsynced.
148    sync_count: AtomicU64,
149    /// Set when a failed flush could not be rolled back (the tail-truncating
150    /// `set_len` itself failed), leaving torn, undeclared bytes past

152    /// shard is refused rather than risk writing on top of, or scanning
153    /// past, that corrupt physical tail.
154    poisoned: AtomicBool,

178/// Point-in-time snapshot of a shard's IO/sync counters.

180/// Loaded with `Ordering::Relaxed` — cheap to take, not synchronized

194    /// Total bytes written to this shard.
195    pub bytes_written: u64,
196    /// Number of fsync calls on this shard.
197    pub sync_count: u64,

255/// Filename for the one-time store-creation marker: records which
256/// `mtxdb-core` version created this store. Written once, when the very
257/// first shard is created, and never rewritten — unlike `pool.meta`, it
258/// has no in-place-updated field, so it needs no version-preservation
259/// dance across later opens.
260const STORE_META_FILENAME: &str = "store.meta";

262/// Store metadata format version.
263const STORE_META_VERSION: u8 = 1;
264
265/// Write the one-time store-creation marker, recording the `mtxdb-core`
266/// version (`CARGO_PKG_VERSION`) that created this store. Best-effort: a
267/// failure here doesn't fail store creation, since this is diagnostic
268/// metadata, not data the engine depends on to operate correctly.
269fn persist_store_meta(base_dir: &Path) {
270    let version = env!("CARGO_PKG_VERSION").as_bytes();
271    let Ok(version_len) = u8::try_from(version.len()) else {
272        return; // never true for a real semver string; just don't write garbage
273    };
274    let mut buf = Vec::with_capacity(6usize.saturating_add(version.len()));
275    buf.extend_from_slice(b"SMeta");
276    buf.push(STORE_META_VERSION);
277    buf.push(version_len);
278    buf.extend_from_slice(version);
279
280    let final_path = base_dir.join(STORE_META_FILENAME);
281    let tmp_path = final_path.with_extension(format!("meta.tmp.{}", std::process::id()));
282    let result = (|| -> io::Result<()> {
283        let mut tmp = File::create(&tmp_path)?;
284        tmp.write_all(&buf)?;
285        drop(tmp);
286        fs::rename(&tmp_path, &final_path)?;
287        Ok(())
288    })();
289    if result.is_err() {
290        let _ = fs::remove_file(&tmp_path);
291    }
292}

294/// Read back the `mtxdb-core` version that created this store, if the
295/// store was created by a build new enough to record it (`store.meta`
296/// predates this feature, so an older store — or one with an unreadable
297/// or corrupt marker — returns `None` rather than erroring: this is
298/// diagnostic-only information).
299#[must_use]
300pub fn store_created_by_version(base_dir: &Path) -> Option<String> {
301    let data = fs::read(base_dir.join(STORE_META_FILENAME)).ok()?;
302    if data.len() < 7 || &data[0..5] != b"SMeta" || data[5] != STORE_META_VERSION {
303        return None;
304    }
305    let version_len = usize::from(data[6]);
306    let version_bytes = data.get(7..7 + version_len)?;
307    String::from_utf8(version_bytes.to_vec()).ok()
308}

322        buf.extend_from_slice(&self.write_count.to_le_bytes());

330        let bytes_written = u64::from_le_bytes(rec[16..24].try_into().unwrap());
331        let sync_count = u64::from_le_bytes(rec[24..32].try_into().unwrap());
332        (
333            pack_id,
334            Self {
335                write_count,
336                bytes_written,
337                sync_count,
338            },
339        )
340    }

397impl Shard {
398    /// Get the memory-mapped view, creating it if absent.
399    ///
400    /// Remaps when the file grows past the current mapping.
401    ///
402    /// # Errors
403    /// Returns `io::Error` if the packfile cannot be mapped.
404    pub fn mmap(&self) -> io::Result<parking_lot::RwLockReadGuard<'_, Option<Arc<Mmap>>>> {
405        let guard = self.mmap.read();
406        if guard.is_some() {
407            return Ok(guard);
408        }
409        drop(guard);
410        let mut guard = self.mmap.write();
411        if guard.is_none() {
412            *guard = Some(Arc::new(packfile::map_pack(&self.file)?));
413        }
414        drop(guard);
415        Ok(self.mmap.read())
416    }
417}

444/// Pool of global shard files shared across all collections.
445///
446/// Only `MAX_SHARDS` files are open at any time, capping file descriptor
447/// usage regardless of collection count. The active write shard rotates when it
448/// exceeds `MAX_SHARD_BYTES`.
449pub struct ShardPool {
450    /// Fixed-size array of shard slots. `None` means unused.
451    shards: RwLock<Vec<Option<Arc<Shard>>>>,
452    /// Index of the shard currently accepting writes.
453    active_write: parking_lot::Mutex<u16>,
454    /// Serializes shard rotation (finding/creating the next shard).
455    rotation_lock: parking_lot::Mutex<()>,
456    base_dir: PathBuf,
457    /// Per-pool rotation threshold, in bytes. Defaults to `MAX_SHARD_BYTES`
458    /// but may be set lower (e.g. by a benchmark that wants many small
459    /// packs to exercise repack/locality behavior) — never higher, since
460    /// `MAX_SHARD_BYTES` is a hard ceiling imposed by `IndexSlot`'s 32-bit
461    /// offset field.
462    max_shard_bytes: u64,
463    /// Per-pool seed mixed into index bucket/tag derivation. Generated once
464    /// at first pool creation, persisted in `pool.meta`, immutable after
465    /// construction.
466    bucket_seed: u64,
467    /// Shard IDs written to since the last sync, for scoped fsync.
468    dirty: parking_lot::Mutex<HashSet<u16>>,
469    /// Monotonically increasing `pack_id` counter for pack filenames.
470    /// Each newly created shard file gets a unique `pack_id`, so a
471    /// reused slot never collides on-disk with a still-referenced old
472    /// shard at the same slot.
473    next_pack_id: AtomicU64,
474    /// Total number of shards retired (garbage-collected after a repack)
475    /// over the pool's lifetime.
476    retired_count: AtomicU64,
477    /// Number of `shard_stats.bin` snapshots renamed into place over the
478    /// pool's lifetime (counted only after the rename succeeds; the snapshot
479    /// is best-effort observability and is not fsynced). The snapshot is
480    /// observability data whose rewrites should be gated to dirty syncs;
481    /// exposing the count lets callers and tests confirm a steady-state
482    /// clean sync pays no stats metadata write.
483    stats_snapshots: AtomicU64,
484    /// Each collection's current "home" shard: writes for a collection are routed here
485    /// instead of always following the pool-wide active-write cursor, so a
486    /// collection's records stay contiguous within a shard rather than
487    /// interleaving with whichever other collections happen to write around the
488    /// same time. Shards remain shared — multiple collections' homes can and do
489    /// coincide, especially low-traffic collections — a collection just sticks to its
490    /// home until that shard fills, rather than following the global
491    /// cursor wherever unrelated collections have since moved it.
492    collection_home: RwLock<HashMap<[u8; 16], u16>>,
493    /// Unix-seconds timestamp of the currently-persisted stats snapshot,
494    /// if one has ever been restored or written by this pool — restored
495    /// at open time from an existing `shard_stats.bin`, and updated on
496    /// every successful `persist_stats`. Lets a caller (e.g. `mtxdb
497    /// shards`) show how stale the counters it's displaying are, since a
498    /// read-only pool only ever sees whatever the real writer last
499    /// flushed, not live in-process counters.
500    stats_persisted_at: RwLock<Option<u64>>,
501    /// Wall-clock instant of the last `maybe_persist_stats` flush (or
502    /// `None` if it's never been called), used to rate-limit that
503    /// timer-driven path — separate from `stats_persisted_at`, which is
504    /// the on-disk snapshot's own unix-seconds timestamp.
505    last_stats_flush: RwLock<Option<Instant>>,
506    /// (flush, fsync) wall-clock split of the last full `sync_all`, used to
507    /// attribute sync cost between buffered frame write-out and the fsync
508    /// calls themselves. `None` until the first sync.
509    last_sync_split: Mutex<Option<(Duration, Duration)>>,
510    /// Cumulative wall-clock time (microseconds) spent waiting to acquire
511    /// `dirty` inside `sync_dirty` and `sync_all`. Pure observability —
512    /// never resets, monotonically increasing. Lets operators confirm
513    /// whether the coarse dirty-set lock is actually a contention point
514    /// under concurrent load, before replacing it with a finer-grained
515    /// mechanism.
516    dirty_lock_wait: AtomicU64,
517    last_open_timings: Mutex<Option<ShardOpenTimings>>,
518    /// Whether this pool holds the writer lock on `base_dir` (see `open`
519    /// vs `open_read_only`). Gates `persist_stats`: a read-only pool
520    /// never writes anything, including its own (always-zero) stats
521    /// snapshot, so it can't race the real writer's.
522    writable: bool,
523    /// Whether records written through this pool are zstd-attempted (see
524    /// [`crate::packfile::write_record_with_options`]). `false` for a pool
525    /// whose payloads never shrink under compression (e.g. HAMT nodes),
526    /// to skip paying the compressor's cost on every write for no benefit.
527    compress: bool,
528    /// How much of the per-frame CRC32 this pool writes and verifies (see
529    /// [`packfile::ChecksumPolicy`]). Controls both what
530    /// [`Self::put_record`] hashes when appending and what
531    /// [`Self::read_at`] verifies on point lookups.
532    checksum_policy: packfile::ChecksumPolicy,
533    /// When a shard's appended frames are written to its pack file (see
534    /// [`AppendPolicy`]). Set at open; defaults to
535    /// [`AppendPolicy::Eager`], the historical per-put behavior. [`Self`]
536    /// owns it so `put_record`'s flush decision needs no extra indirection.
537    append_policy: AppendPolicy,
538    /// Present only for a writable pool — `open_read_only` takes no lock
539    /// at all, since it never writes or truncates anything (that risk
540    /// lives one layer up, in `PackfileStorage`'s collection-index rebuild, not
541    /// here), so there's nothing for a reader to need exclusivity from.
542    ///
543    /// Never read: this is an RAII guard held purely for its `Drop` side
544    /// effect (removing the `.mtxdb.lock` marker file when the pool
545    /// itself drops). Its value is deliberately never inspected — only
546    /// its lifetime matters — so `#[allow(dead_code)]` here is correct,
547    /// not a lint dodge: the field genuinely has no read access by
548    /// design, the same way a `MutexGuard` binding is never "used" either.
549    /// `None` on wasm32, where there's no cross-process model to guard.
550    #[cfg(not(target_arch = "wasm32"))]
551    #[allow(dead_code)]
552    writer_lock: Option<WriterLock>,
553}

689                ),
690            ));
691        }

736    /// Discovers packfiles in the pool directory.
737    ///
738    /// Only v4 filenames are accepted: `pack_XXXXXXXXXXXXXXXX.pack`
739    /// where the 16 lowercase hex digits encode the `pack_id`. Uppercase
740    /// hex is rejected to prevent case-insensitive collisions on
741    /// case-insensitive filesystems and to enforce a single canonical
742    /// spelling per ID.
743    ///
744    /// A `pack_*.pack` file that doesn't match the v4 format, or a legacy
745    /// `shard_*.pack` file, is rejected with `Unsupported` — this is a hard
746    /// cutover, not a silent skip. Other applications' `.pack` files are
747    /// unrelated and ignored. Duplicate `pack_id`s are also rejected.
748    fn discover_pack_files(base_dir: &Path) -> io::Result<Vec<(u64, PathBuf)>> {
749        let mut pack_files = Vec::new();
750        let mut seen = std::collections::HashSet::new();
751        for entry in fs::read_dir(base_dir)? {
752            let entry = entry?;
753            let path = entry.path();
754            if !path.extension().is_some_and(|e| e == "pack") {
755                continue;
756            }
757            let Some(stem) = path.file_stem().and_then(|s| s.to_str()) else {
758                continue;
759            };
760            let id_hex = match stem.strip_prefix("pack_") {
761                Some(id_hex) => id_hex,
762                None if stem.starts_with("shard_") => {
763                    return Err(io::Error::new(
764                        io::ErrorKind::Unsupported,
765                        format!(
766                            "found pre-v4 shard file {}; \
767                             use a fresh pool or explicitly migrate/reset it",
768                            path.display()
769                        ),
770                    ));
771                }
772                None => continue,
773            };
774
775            // v4 format: exactly 16 lowercase hex digits for pack_id.
776            if id_hex.len() != 16 {
777                return Err(io::Error::new(
778                    io::ErrorKind::Unsupported,
779                    format!(
780                        "found pre-v4 pack file {}; \
781                         use a fresh pool or explicitly migrate/reset it",
782                        path.display()
783                    ),
784                ));
785            }
786            if !id_hex.chars().all(|c| c.is_ascii_hexdigit()) {
787                return Err(io::Error::new(
788                    io::ErrorKind::InvalidData,
789                    format!(
790                        "pack filename {} contains non-hex characters",
791                        path.display()
792                    ),
793                ));
794            }
795            if id_hex != id_hex.to_ascii_lowercase() {
796                return Err(io::Error::new(
797                    io::ErrorKind::InvalidData,
798                    format!("pack filename {} must use lowercase hex", path.display()),
799                ));
800            }
801            let pack_id = u64::from_str_radix(id_hex, 16).map_err(|e| {
802                io::Error::new(io::ErrorKind::InvalidData, format!("{id_hex}: {e}"))
803            })?;
804            if !seen.insert(pack_id) {
805                return Err(io::Error::new(
806                    io::ErrorKind::InvalidData,
807                    format!(
808                        "duplicate pack_id {pack_id:#018x} found in {}",
809                        path.display()
810                    ),
811                ));
812            }
813            pack_files.push((pack_id, path));
814        }
815        Ok(pack_files)
816    }

818    /// Discover and sort pack files in `base_dir`, enforcing the
819    /// `MAX_SHARDS` capacity bound.
820    fn discover_pack_files_sorted(base_dir: &Path) -> io::Result<Vec<(u64, PathBuf)>> {
821        let mut pack_files = Self::discover_pack_files(base_dir)?;
822        if pack_files.len() > MAX_SHARDS {
823            return Err(io::Error::new(
824                io::ErrorKind::InvalidData,

826                    "too many pack files: {} found, but the pool can hold at most {}",

828                    MAX_SHARDS

836    /// Recover (if writable) and open every pack file, slotting them into
837    /// the shard vector. Consumes `pack_files` so each `PathBuf` moves into
838    /// its `Shard` rather than being cloned. Returns `(recovery_time,
839    /// recovery_calls, packfile_open_time, packfile_open_calls)`.
840    fn recover_and_open_packs(
841        writable: bool,
842        pack_files: Vec<(u64, PathBuf)>,
843        shards: &mut [Option<Arc<Shard>>],
844        next_slot: &mut u16,
845        max_pack_id: &mut u64,
846    ) -> io::Result<(Duration, u64, Duration, u64)> {
847        let mut recovery_time = Duration::ZERO;
848        let mut recovery_calls: u64 = 0;
849        let mut packfile_open_time = Duration::ZERO;
850        let mut packfile_open_calls: u64 = 0;
851        for (pack_id, path) in pack_files {
852            if writable {
853                let recovery_started = Instant::now();
854                let _ = packfile::scan_and_recover_packfile(&path).map_err(|error| {
855                    io::Error::new(
856                        io::ErrorKind::InvalidData,
857                        format!(
858                            "corrupt pack {}; failed to scan and recover: {error}",
859                            path.display()
860                        ),
861                    )
862                })?;
863                recovery_time = recovery_time.saturating_add(recovery_started.elapsed());
864                recovery_calls = recovery_calls.saturating_add(1);
865            }
866
867            let packfile_open_started = Instant::now();
868            let file = packfile::open_packfile(&path, writable, pack_id).map_err(|error| {
869                io::Error::new(
870                    io::ErrorKind::InvalidData,
871                    format!("corrupt pack {}; failed to open: {error}", path.display()),
872                )
873            })?;
874            packfile_open_time = packfile_open_time.saturating_add(packfile_open_started.elapsed());
875            packfile_open_calls = packfile_open_calls.saturating_add(1);
876
877            let file_len = file.metadata()?.len();
878            let slot = *next_slot;
879            *next_slot = next_slot.saturating_add(1);
880            let shard = Arc::new(Shard::new(slot, pack_id, file, path, file_len));
881            shards[slot as usize] = Some(shard);
882            if pack_id >= *max_pack_id {
883                *max_pack_id = pack_id.saturating_add(1);
884            }
885        }
886        Ok((
887            recovery_time,
888            recovery_calls,
889            packfile_open_time,
890            packfile_open_calls,
891        ))
892    }

894    /// Bootstrap a brand-new pool: create the first pack file, persist
895    /// `pool.meta` (with a fresh seed if needed), and write `store.meta`.
896    /// Returns `(next_pack_id, bucket_seed, store_meta_write_time,
897    /// pool_meta_persist_time, initial_pack_create_time)`.
898    ///
899    /// # Errors
900    /// Returns `io::Error` if the pool is empty and not writable.
901    #[allow(clippy::too_many_arguments)]
902    fn initialize_empty_pool(
903        base_dir: &Path,
904        shards: &mut [Option<Arc<Shard>>],
905        mut next_pack_id: u64,
906        mut bucket_seed: u64,
907        writable: bool,
908    ) -> io::Result<(u64, u64, Duration, Duration, Duration)> {
909        if !writable {
910            return Err(io::Error::new(
911                io::ErrorKind::NotFound,
912                format!(
913                    "no shards found in {} (nothing to read)",
914                    base_dir.display()
915                ),
916            ));
917        }
918        let pack_id = next_pack_id;
919
920        let t_store_meta = Instant::now();
921        if !base_dir.join(STORE_META_FILENAME).exists() {
922            persist_store_meta(base_dir);
923        }
924        let store_meta_write_time = t_store_meta.elapsed();
925
926        if bucket_seed == 0 {
927            let mut seed_bytes = [0u8; 8];
928            getrandom::fill(&mut seed_bytes).map_err(io::Error::other)?;
929            bucket_seed = u64::from_ne_bytes(seed_bytes);
930        }
931
932        let t_pool_meta_persist = Instant::now();
933        Self::persist_pool_meta_at_sync_dir(
934            base_dir,
935            pack_id.checked_add(1).expect("pack_id overflow"),
936            bucket_seed,
937            false,
938        )?;
939        let pool_meta_persist_time = t_pool_meta_persist.elapsed();
940
941        let t_initial_pack = Instant::now();
942        let (file, path) = Self::create_packfile_atomically(base_dir, pack_id)?;
943        let file_len = file.metadata()?.len();
944        shards[0] = Some(Arc::new(Shard::new(0, pack_id, file, path, file_len)));
945        next_pack_id = pack_id.checked_add(1).expect("pack_id overflow");
946
947        let dir = File::open(base_dir)?;
948        dir.sync_all()?;
949        let initial_pack_create_time = t_initial_pack.elapsed();
950
951        Ok((
952            next_pack_id,
953            bucket_seed,
954            store_meta_write_time,
955            pool_meta_persist_time,
956            initial_pack_create_time,
957        ))
958    }

986    /// Assemble a `ShardPool` from its discovered parts. Separated from
987    /// `open_internal` to keep the latter under the clippy line limit; the
988    /// raw timing values travel as one [`ShardOpenTimings`] rather than a
989    /// dozen positional `Duration`s, and the two derived fields
990    /// (`metadata_unattributed`, `unattributed`) are computed here.
991    #[allow(clippy::too_many_arguments)]
992    fn build_pool(
993        shards: Vec<Option<Arc<Shard>>>,
994        base_dir: PathBuf,
995        max_shard_bytes: u64,
996        bucket_seed: u64,
997        next_pack_id: u64,
998        highest_active: u16,
999        writable: bool,
1000        compress: bool,
1001        checksum_policy: packfile::ChecksumPolicy,
1002        stats_persisted_at: Option<u64>,
1003        mut timings: ShardOpenTimings,
1004        #[cfg(not(target_arch = "wasm32"))] writer_lock: Option<WriterLock>,
1005    ) -> Self {
1006        let metadata_subphases_sum = timings
1007            .pool_meta_restore
1008            .saturating_add(timings.store_meta_write)
1009            .saturating_add(timings.pool_meta_persist)
1010            .saturating_add(timings.initial_pack_create)
1011            .saturating_add(timings.persisted_stats_restore);
1012        timings.metadata_unattributed = timings
1013            .metadata_restore
1014            .saturating_sub(metadata_subphases_sum);
1015
1016        let named_phases = timings
1017            .discovery
1018            .saturating_add(timings.writer_lock)
1019            .saturating_add(timings.packfile_recovery)
1020            .saturating_add(timings.packfile_open)
1021            .saturating_add(timings.metadata_restore);
1022        timings.unattributed = timings.total.saturating_sub(named_phases);
1023
1024        Self {
1025            shards: RwLock::new(shards),
1026            active_write: parking_lot::Mutex::new(highest_active),
1027            rotation_lock: parking_lot::Mutex::new(()),
1028            base_dir,
1029            max_shard_bytes,
1030            bucket_seed,
1031            dirty: parking_lot::Mutex::new(HashSet::new()),
1032            next_pack_id: AtomicU64::new(next_pack_id),
1033            retired_count: AtomicU64::new(0),
1034            stats_snapshots: AtomicU64::new(0),
1035            collection_home: RwLock::new(HashMap::new()),
1036            stats_persisted_at: RwLock::new(stats_persisted_at),
1037            last_stats_flush: RwLock::new(None),
1038            last_sync_split: Mutex::new(None),
1039            dirty_lock_wait: AtomicU64::new(0),
1040            last_open_timings: Mutex::new(Some(timings)),
1041            writable,
1042            compress,
1043            checksum_policy,
1044            append_policy: AppendPolicy::Eager,
1045            #[cfg(not(target_arch = "wasm32"))]
1046            writer_lock,
1047        }
1048    }

1050    fn open_internal(
1051        base_dir: PathBuf,
1052        writable: bool,
1053        max_shard_bytes: u64,
1054        compress: bool,
1055        checksum_policy: packfile::ChecksumPolicy,
1056    ) -> io::Result<Self> {
1057        let open_started = Instant::now();
1058        if writable {
1059            fs::create_dir_all(&base_dir)?;
1060        } else if !base_dir.is_dir() {
1061            return Err(io::Error::new(
1062                io::ErrorKind::NotFound,
1063                format!(
1064                    "shard pool directory does not exist: {}",
1065                    base_dir.display()
1066                ),
1067            ));
1068        }
1069
1070        let writer_lock_started = Instant::now();
1071        #[cfg(not(target_arch = "wasm32"))]
1072        let writer_lock = writable
1073            .then(|| Self::acquire_writer_lock(&base_dir))
1074            .transpose()?;
1075        #[cfg(not(target_arch = "wasm32"))]
1076        let writer_lock_time = writer_lock_started.elapsed();
1077        #[cfg(target_arch = "wasm32")]
1078        let writer_lock_time = Duration::ZERO;
1079
1080        let mut shards: Vec<Option<Arc<Shard>>> = (0..MAX_SHARDS).map(|_| None).collect();
1081        let mut next_slot: u16 = 0;
1082        let mut max_pack_id: u64 = 0;
1083
1084        let discovery_started = Instant::now();
1085        let pack_files = Self::discover_pack_files_sorted(&base_dir)?;
1086        let discovery_time = discovery_started.elapsed();
1087
1088        let (recovery_time, recovery_calls, packfile_open_time, packfile_open_calls) =
1089            Self::recover_and_open_packs(
1090                writable,
1091                pack_files,
1092                &mut shards,
1093                &mut next_slot,
1094                &mut max_pack_id,
1095            )?;
1096
1097        let highest_active = next_slot.saturating_sub(1);
1098
1099        // Restore next_pack_id and bucket_seed from pool.meta if available,

1187    /// Claim the writer lock on `base_dir`: atomically create a
1188    /// `.mtxdb.lock` marker file, which fails with `AlreadyExists` if
1189    /// another writer already holds it. Pure `std::fs` — no OS-level
1190    /// advisory-lock API and no dependency, so it needs one thing real
1191    /// `flock` gives for free: recovery if the previous holder crashed
1192    /// without cleaning up (`WriterLock`'s `Drop` handles the clean-exit
1193    /// case). We store our `{pid, starttime}` in the file and, if creation
1194    /// fails because it already exists, check whether that exact process
1195    /// (not just that PID number) is still alive before concluding the
1196    /// lock is genuinely held — a stale file from a killed process is
1197    /// removed and retried once rather than wrongly blocking forever. The
1198    /// starttime is what makes this safe across PID reuse (see
1199    /// `lock_holder_is_dead`) — a bare PID is not enough on its own.
1200    #[cfg(not(target_arch = "wasm32"))]
1201    fn acquire_writer_lock(base_dir: &Path) -> io::Result<WriterLock> {
1202        let lock_path = base_dir.join(".mtxdb.lock");
1203        match Self::try_create_lock_file(&lock_path) {
1204            Ok(()) => Ok(WriterLock { path: lock_path }),
1205            Err(e) if e.kind() == io::ErrorKind::AlreadyExists => {
1206                if Self::lock_holder_is_dead(&lock_path) {
1207                    let _ = fs::remove_file(&lock_path);
1208                    Self::try_create_lock_file(&lock_path)?;
1209                    return Ok(WriterLock { path: lock_path });
1210                }
1211                Err(io::Error::new(
1212                    io::ErrorKind::WouldBlock,
1213                    format!(
1214                        "{} is already locked by another writer process",
1215                        base_dir.display()
1216                    ),
1217                ))
1218            }
1219            Err(e) => Err(e),
1220        }
1221    }

1223    /// Atomically create the lock file and write our `{pid, starttime}`
1224    /// into it (Linux) or just our PID (elsewhere, where starttime can't be
1225    /// read and `lock_holder_is_dead` never trusts a bare PID anyway). No
1226    /// `sync_all` here: this is advisory only, and `lock_holder_is_dead`
1227    /// already fails closed (treats an unparsable file as "might be
1228    /// alive") on a torn write from a crash mid-write — there's no
1229    /// correctness reason to pay an fsync on every lock acquisition to
1230    /// protect against that.
1231    #[cfg(not(target_arch = "wasm32"))]
1232    fn try_create_lock_file(lock_path: &Path) -> io::Result<()> {
1233        let mut file = File::options()
1234            .write(true)
1235            .create_new(true)
1236            .open(lock_path)?;
1237        let pid = std::process::id();
1238        #[cfg(target_os = "linux")]
1239        {
1240            if let Some(start_time) = Self::proc_start_time("self") {
1241                return write!(file, "{pid} {start_time}");
1242            }
1243        }
1244        write!(file, "{pid}")
1245    }

1247    /// Reads field 22 (`starttime`, clock ticks since boot) out of
1248    /// `/proc/<pid_or_self>/stat`. Parses from the *last* `)` rather than
1249    /// splitting on whitespace from the start: the second field (`comm`,
1250    /// the executable name in parens) can itself contain spaces and
1251    /// parens, which would otherwise misalign every field after it — a
1252    /// classic `/proc/stat` parsing bug. Field 22 is the 20th
1253    /// whitespace-separated token after that closing paren (field 3 is the
1254    /// first token after it).
1255    #[cfg(target_os = "linux")]
1256    fn proc_start_time(pid_or_self: &str) -> Option<u64> {
1257        let contents = fs::read_to_string(format!("/proc/{pid_or_self}/stat")).ok()?;
1258        let after_comm = contents.rsplit_once(')')?.1;
1259        after_comm.split_whitespace().nth(19)?.parse::<u64>().ok()
1260    }

1262    /// Liveness check for whoever wrote `lock_path`. Only verifies
1263    /// anything on Linux (`/proc/<pid>`); everywhere else this
1264    /// conservatively assumes the holder might still be alive.
1265    ///
1266    /// A time-based fallback ("reclaim if the lock file is older than N")
1267    /// was considered and rejected: a long-lived, mostly-idle writer (a
1268    /// Synapse process sitting quiet for hours is normal, not an edge
1269    /// case) looks indistinguishable from an abandoned lock once it
1270    /// crosses any fixed age threshold, since nothing refreshes the
1271    /// timestamp while the lock is held — that trades a rare stuck-lock
1272    /// annoyance (fixable by deleting the file) for an occasional silent
1273    /// second writer and packfile corruption, the wrong side of that
1274    /// trade.
1275    ///
1276    /// A bare `/proc/<pid>` existence check is *not* enough on its own,
1277    /// either: PIDs get recycled by the OS, and a container commonly has
1278    /// long-lived sibling processes (postgres, redis, nginx, ...) that
1279    /// outlive many respawns of a crash-looping writer. Once the OS
1280    /// happens to reuse a dead writer's old PID number for one of those
1281    /// unrelated long-lived processes, a plain PID check reports "alive"
1282    /// forever — not because the original holder is still running, but
1283    /// because something else now coincidentally has that PID — poisoning
1284    /// recovery permanently. Comparing the recorded `starttime` (written
1285    /// at lock-acquisition time) against the current holder of that PID
1286    /// closes this: a reused PID almost certainly has a different
1287    /// starttime, so a mismatch means the original writer is gone even
1288    /// though its old PID number is (again) in use. A lock file written by
1289    /// an older binary (bare PID, no starttime) has nothing to compare
1290    /// against and fails closed exactly as before, same as any other
1291    /// unparsable content.
1292    #[cfg(not(target_arch = "wasm32"))]
1293    fn lock_holder_is_dead(lock_path: &Path) -> bool {
1294        #[cfg(target_os = "linux")]
1295        {
1296            let Ok(contents) = fs::read_to_string(lock_path) else {
1297                return false;
1298            };
1299            let mut fields = contents.split_whitespace();
1300            let Some(pid) = fields.next().and_then(|s| s.parse::<u32>().ok()) else {
1301                return false;
1302            };
1303            if !Path::new(&format!("/proc/{pid}")).exists() {
1304                return true;
1305            }
1306            // The PID exists as a live process, but that alone doesn't mean
1307            // it's still our original writer (see doc comment above) —
1308            // disambiguate via starttime when we have one to compare.
1309            match fields.next().and_then(|s| s.parse::<u64>().ok()) {
1310                Some(recorded_start) => match Self::proc_start_time(&pid.to_string()) {
1311                    Some(current_start) => current_start != recorded_start,
1312                    // Couldn't read the current holder's stat (raced with
1313                    // its own exit, permissions, ...) — fail closed.
1314                    None => false,
1315                },
1316                // Old-format lock file, or starttime collection failed at
1317                // creation time — nothing to disambiguate a reuse with.
1318                None => false,
1319            }
1320        }
1321        #[cfg(not(target_os = "linux"))]
1322        {
1323            let _ = lock_path;
1324            false
1325        }
1326    }

1328    /// Discovers new pack files on disk and adds them to the pool.
1329    ///
1330    /// # Errors
1331    /// Returns `io::Error` if reading the directory, opening a discovered
1332    /// pack, reading its metadata, or allocating a shard slot fails.
1333    pub fn discover_shards(&self) -> io::Result<()> {
1334        let pack_files = Self::discover_pack_files(&self.base_dir)?;
1335        let mut shards = self.shards.write();
1336        let existing: std::collections::HashSet<u64> =
1337            shards.iter().flatten().map(|s| s.pack_id).collect();
1338        let mut sorted_files = pack_files;
1339        sorted_files.sort_unstable_by_key(|(id, _)| *id);
1340        for (pack_id, path) in sorted_files {
1341            if existing.contains(&pack_id) {
1342                continue;
1343            }
1344            let slot = shards.iter().position(Option::is_none).ok_or_else(|| {
1345                io::Error::new(
1346                    io::ErrorKind::InvalidData,
1347                    "shard pool full while discovering pack files",
1348                )
1349            })?;
1350            let file = crate::packfile::open_packfile(&path, false, pack_id)?;
1351            let file_len = file.metadata()?.len();
1352            let slot = u16::try_from(slot)
1353                .map_err(|_| io::Error::new(io::ErrorKind::InvalidData, "invalid shard slot"))?;
1354            shards[usize::from(slot)] = Some(std::sync::Arc::new(Shard::new(
1355                slot, pack_id, file, path, file_len,
1356            )));
1357        }
1358        Ok(())
1359    }

1422    /// Internal form of [`Self::persist_pool_meta_at`] allowing callers that
1423    /// create multiple files (such as fresh pool open) to defer directory
1424    /// synchronization until all renames have completed.
1425    fn persist_pool_meta_at_sync_dir(
1426        base_dir: &Path,
1427        next: u64,
1428        bucket_seed: u64,
1429        sync_dir: bool,
1430    ) -> io::Result<()> {
1431        let mut buf = Vec::with_capacity(22);
1432        buf.extend_from_slice(b"PMeta");
1433        buf.push(POOL_META_VERSION);
1434        buf.extend_from_slice(&next.to_le_bytes());
1435        buf.extend_from_slice(&bucket_seed.to_le_bytes());
1436
1437        let final_path = Self::pool_meta_path(base_dir);
1438        let tmp_path = final_path.with_extension(format!("meta.tmp.{}", std::process::id()));
1439        let write_result = (|| -> io::Result<()> {
1440            let mut tmp = File::create(&tmp_path)?;
1441            tmp.write_all(&buf)?;
1442            tmp.sync_all()?;
1443            drop(tmp);
1444            fs::rename(&tmp_path, &final_path)?;
1445            if sync_dir {
1446                // Fsync the containing directory so the rename is durable
1447                // across power loss — without this, a crash could leave the
1448                // old pool.meta (or no file) in place, allowing pack_id reuse.
1449                let dir = File::open(base_dir)?;
1450                dir.sync_all()?;
1451            }
1452            Ok(())
1453        })();
1454        if let Err(e) = write_result {
1455            let _ = fs::remove_file(&tmp_path);
1456            return Err(e);
1457        }
1458        Ok(())
1459    }

1505    /// Load a persisted stats snapshot, if one exists, and restore each
1506    /// shard's counters when its `pack_id` matches — a stale snapshot entry
1507    /// (from a shard since retired and deleted) is silently skipped
1508    /// rather than misapplied. Returns the snapshot's persisted-at
1509    /// unix timestamp, if the file was readable and current-format.
1510    ///
1511    /// Best-effort: a missing, truncated, or corrupt file just means no
1512    /// stats are restored — never a startup failure over stats alone.
1513    fn restore_persisted_stats(base_dir: &Path, shards: &[Option<Arc<Shard>>]) -> Option<u64> {
1514        let path = Self::stats_path(base_dir);
1515        let buf = fs::read(&path).ok()?;
1516        if buf.len() < STATS_HEADER_LEN || &buf[0..4] != STATS_MAGIC || buf[4] != STATS_VERSION {
1517            return None;
1518        }
1519        let persisted_at = u64::from_le_bytes(buf[5..13].try_into().ok()?);
1520        let body = &buf[STATS_HEADER_LEN..];
1521        for chunk in body.chunks(STATS_RECORD_LEN) {
1522            let Ok(rec) = <&[u8; STATS_RECORD_LEN]>::try_from(chunk) else {
1523                break;
1524            };
1525            let (pack_id, stats) = ShardStats::decode(rec);
1526            for shard in shards.iter().flatten() {
1527                if shard.pack_id == pack_id {
1528                    shard.restore_stats(stats);
1529                    break;
1530                }
1531            }
1532        }
1533        Some(persisted_at)
1534    }

1536    /// Persist every currently-open shard's IO/sync counters to disk,
1537    /// keyed by `pack_id` so a retired/reused slot's stale
1538    /// numbers are never mistakenly restored onto a new shard.
1539    ///
1540    /// Writes to a temp file and renames into place, so a crash mid-write
1541    /// leaves the previous snapshot (or none) rather than a torn file.
1542    ///
1543    /// # Errors
1544    /// Returns `io::Error` on write or rename failure.
1545    fn persist_stats(&self) -> io::Result<()> {
1546        let mut buf = Vec::new();
1547        buf.extend_from_slice(STATS_MAGIC);
1548        buf.push(STATS_VERSION);
1549        let persisted_at = SystemTime::now()
1550            .duration_since(UNIX_EPOCH)
1551            .map_or(0, |d| d.as_secs());
1552        buf.extend_from_slice(&persisted_at.to_le_bytes());
1553        for (_slot, shard) in self.all_shards() {
1554            shard.stats().encode(shard.pack_id, &mut buf);
1555        }
1556
1557        // Unique per (process, call) — the same base_dir can be opened by
1558        // more than one process at once (e.g. a long-running embedder plus
1559        // a short-lived `mtxdb shards` CLI invocation), and a fixed tmp
1560        // filename shared across them races: one process's rename() can
1561        // consume the shared tmp path out from under another's, which
1562        // then sees ENOENT on its own rename despite having written its
1563        // tmp file successfully moments earlier.
1564        let unique = STATS_TMP_COUNTER.fetch_add(1, Ordering::Relaxed);
1565        let tmp_path = Self::stats_path(&self.base_dir)
1566            .with_extension(format!("bin.tmp.{}.{unique}", std::process::id()));
1567        let final_path = Self::stats_path(&self.base_dir);
1568        // Deliberately no fsync here. The snapshot is observability data
1569        // (rebuilt from live counters; open tolerates a missing/old/short
1570        // file), and the rename below is never followed by a directory
1571        // fsync -- so the old `tmp.sync_all()` paid an fsync per persist for
1572        // a power-loss guarantee the rename didn't actually provide.
1573        let write_result = (|| -> io::Result<()> {
1574            let mut tmp = File::create(&tmp_path)?;
1575            tmp.write_all(&buf)?;
1576            Ok(())
1577        })();
1578        if let Err(e) = write_result {
1579            // Don't leave a half-written tmp file behind under its
1580            // now-unique name — best-effort, the write already failed.
1581            let _ = fs::remove_file(&tmp_path);
1582            return Err(e);
1583        }
1584        fs::rename(&tmp_path, &final_path)?;
1585        self.stats_snapshots.fetch_add(1, Ordering::Relaxed);
1586        *self.stats_persisted_at.write() = Some(persisted_at);
1587        Ok(())
1588    }

1734    /// Append a record to its collection's home shard (see `collection_home`).
1735    /// Returns `(slot, offset)`. Rotates the collection to a new home shard
1736    /// if its current one is full.
1737    ///
1738    /// Under the default [`AppendPolicy::Eager`] the frame is written to the
1739    /// page cache immediately — one positioned write per put, the historical
1740    /// behavior (unsynced bytes survive this process exiting; durability on
1741    /// crash/power loss still requires a `sync_*`). Under
1742    /// [`AppendPolicy::Buffered`] the frame is appended to the shard's
1743    /// in-memory pending region instead: offsets are handed out into the
1744    /// virtual `[file_len, file_len + pending)` range and committed by a
1745    /// later flush. Reads of a virtual offset first flush (see
1746    /// [`Self::read_at`]), so callers observing the returned offset always
1747    /// see the record either way.
1748    ///
1749    /// # Errors
1750    /// Returns `io::Error` on write, flush, or rotation failure.
1751    pub fn put_record(&self, record: &Record) -> io::Result<(u16, u64)> {
1752        let mut shard = self.shard_for_collection(&record.collection_id);
1753        loop {
1754            if shard.is_poisoned() {
1755                return Err(io::Error::other(format!(
1756                    "shard {} is poisoned after an unrecoverable rollback failure",
1757                    shard.slot
1758                )));
1759            }
1760            // Uncompressed upper bound, used only for the pre-append
1761            // capacity check below — `write_record` may compress the
1762            // payload and write fewer bytes than this, but never more,
1763            // so checking against this bound never lets a shard overflow
1764            // MAX_SHARD_BYTES; it may just rotate a little earlier than
1765            // strictly necessary when compression would have made it fit.
1766            let max_record_len = record.serialized_len() as u64;
1767
1768            // `append_lock` serializes writers *and* flushers, and is held
1769            // from the capacity check through the append, the automatic
1770            // flush, and (on failure) the rollback. That is what makes the
1771            // rollback safe: a shard is shared by many collections (its
1772            // `collection_home` maps several collections onto the same
1773            // slot), so per-collection put mutexes DON'T serialize the
1774            // appenders — only this lock does. While it is held, no other
1775            // put, of any collection, can append to `pending`, so our frame
1776            // is provably the tail when the failed flush needs to retract
1777            // it.
1778            let append_guard = shard.append_lock.lock();
1779            // Re-check poison after acquiring the lock: the check above is
1780            // only a fast-path skip for the common case. A concurrent
1781            // put/flush can poison the shard between that check and here —
1782            // `append_lock` is what actually excludes further appends, so
1783            // it's the only place this check is load-bearing.
1784            if shard.is_poisoned() {
1785                return Err(io::Error::other(format!(
1786                    "shard {} is poisoned after an unrecoverable rollback failure",
1787                    shard.slot
1788                )));
1789            }
1790            // `file_len` is updated at flush time, making the virtual end
1791            // (committed length plus already-buffered frames) the
1792            // authoritative next offset. No syscalls happen per append.
1793            let committed = shard.file_len.load(Ordering::Acquire);
1794            let buffered = shard.pending.lock().len() as u64;
1795            let virtual_end = committed.checked_add(buffered).ok_or_else(|| {
1796                io::Error::new(io::ErrorKind::InvalidData, "file_len + pending overflow")
1797            })?;
1798
1799            // Check capacity while holding the append lock and against
1800            // the virtual end — avoids a TOCTOU race where two threads
1801            // both pass the check then one exceeds the limit.
1802            let fits = virtual_end
1803                .checked_add(max_record_len)
1804                .is_some_and(|sum| sum <= self.max_shard_bytes);
1805            if !fits && committed > packfile::HEADER_LEN as u64 {
1806                drop(append_guard);
1807                // First make room: push the buffered bytes to disk so
1808                // the capacity check is measured against real length.
1809                self.flush_shard(&shard)?;
1810                shard = self.rotate_collection_full_home(&record.collection_id, shard.slot)?;
1811                continue;
1812            }
1813
1814            let frame = packfile::encode_record_with_options(
1815                record,
1816                self.compress,
1817                self.checksum_policy.computes_checksum(),
1818            )?;
1819            let frame_len = frame.len();
1820            let pending_start = {
1821                let mut pending = shard.pending.lock();
1822                let start = pending.len();
1823                pending.extend_from_slice(&frame);
1824                shard.pending_records.fetch_add(1, Ordering::Relaxed);
1825                start
1826            };
1827
1828            let flush_result = match self.append_policy {
1829                AppendPolicy::Eager => self.flush_shard_with_guard(&shard, &append_guard),
1830                AppendPolicy::Buffered { max_pending_bytes } => {
1831                    if shard.pending.lock().len() >= max_pending_bytes {
1832                        self.flush_shard_with_guard(&shard, &append_guard)
1833                    } else {
1834                        Ok(())
1835                    }
1836                }
1837            };
1838            if let Err(e) = flush_result {
1839                // The caller sees `Err` and has not indexed this record, yet
1840                // its frame is still in `pending` — a later flush would
1841                // silently make a "failed" put durable, surfacing it as
1842                // stored data on restart's full scan. Retract it: a failed
1843                // flush leaves `pending` untouched, and `append_guard` is
1844                // still held from the append so nothing appended after our
1845                // frame, making the tail truncation exact. A torn write's
1846                // partially-written prefix is also reined back (`set_len` to
1847                // the still-current committed frontier) so leftover bytes
1848                // never read back as phantom records on a rebuild.
1849                let mut pending = shard.pending.lock();
1850                let end = pending_start.saturating_add(frame_len);
1851                if pending.len() >= end {
1852                    // Only discard the frame once the tail truncation has
1853                    // actually succeeded — otherwise the torn bytes stay on
1854                    // disk past `committed` with no frame left to account
1855                    // for them, and a restart's full scan (which reads real
1856                    // file length, not our tracked `file_len`) can surface
1857                    // them as phantom data. If truncation itself fails,
1858                    // leave `pending` untouched and poison the shard so
1859                    // nothing else appends past, or scans over, the corrupt
1860                    // tail.
1861                    match shard.file.set_len(committed) {
1862                        Ok(()) => {
1863                            pending.truncate(pending_start);
1864                            shard.pending_records.fetch_sub(1, Ordering::Relaxed);
1865                        }
1866                        Err(rollback_err) => {
1867                            drop(pending);
1868                            shard.poisoned.store(true, Ordering::Release);
1869                            return Err(io::Error::other(format!(
1870                                "flush failed ({e}) and rollback truncation also failed \
1871                                 ({rollback_err}); shard {} poisoned",
1872                                shard.slot
1873                            )));
1874                        }
1875                    }
1876                }
1877                return Err(e);
1878            }
1879            return Ok((shard.slot, virtual_end));
1880        }
1881    }

1912    /// Commit one shard's buffered frames to disk as a single positioned
1913    /// write. No-op if the shard has nothing buffered. Updates `file_len`,
1914    /// IO/sync counters, and the dirty set only after a successful write,
1915    /// so a partial failure defers rather than loses the accounting.
1916    ///
1917    /// Safe to call concurrently with writers and other flushers: the
1918    /// additional buffered bytes are simply rolled into this (or the next)
1919    /// write, keeping `file_len` monotonic and offsets stable.
1920    ///
1921    /// Note: the shard→collection count bookkeeping is deliberately NOT
1922    /// here — it stays per-record in `PackfileStorage::put`/`put_many`
1923    /// (via `record_new_shard_collection`), entirely unchanged by
1924    /// buffering, so no flush-time aggregation or recompute can drift from
1925    /// its current semantics.
1926    ///
1927    /// # Errors
1928    /// Returns `io::Error` on flush failure.
1929    pub(crate) fn flush_shard(&self, shard: &Arc<Shard>) -> io::Result<()> {
1930        if shard.is_poisoned() {
1931            return Err(io::Error::other(format!(
1932                "shard {} is poisoned after an unrecoverable rollback failure",
1933                shard.slot
1934            )));
1935        }
1936        let guard = shard.append_lock.lock();
1937        self.flush_shard_with_guard(shard, &guard)
1938    }

1940    /// Body of `flush_shard` under a caller-supplied `append_lock` guard, so
1941    /// a writer that already holds the lock (see `put_record`, which must
1942    /// keep it across append → flush → rollback) can flush without
1943    /// re-acquiring it.
1944    fn flush_shard_with_guard(
1945        &self,
1946        shard: &Arc<Shard>,
1947        _guard: &parking_lot::MutexGuard<'_, ()>,
1948    ) -> io::Result<()> {
1949        // The guard proves `append_lock` is held, so this is the single
1950        // point every flush path — direct, via `flush_shard`, or inline
1951        // from `put_record`'s own automatic flush — actually funnels
1952        // through under lock. Enforcing the poison check here, not just at
1953        // each call site, means correctness never depends on every current
1954        // and future caller remembering to check first.
1955        if shard.is_poisoned() {
1956            return Err(io::Error::other(format!(
1957                "shard {} is poisoned after an unrecoverable rollback failure",
1958                shard.slot
1959            )));
1960        }
1961        {
1962            let mut pending_guard = shard.pending.lock();
1963            if pending_guard.is_empty() {
1964                // Nothing buffered: nothing to commit or dirty.
1965                return Ok(());
1966            }
1967            let committed = shard.file_len.load(Ordering::Acquire);
1968            // Write directly from the still-buffered bytes rather than
1969            // draining them first: `append_lock` is held for this whole
1970            // block, so nothing else can append to `pending` meanwhile,
1971            // and keeping the bytes in place until the write actually
1972            // succeeds means a failed write leaves them buffered for the
1973            // next flush attempt instead of silently discarding already
1974            // "successful" puts whose offsets the index has already
1975            // handed out.
1976            let file = shard.file.try_clone()?;
1977            #[cfg(unix)]
1978            file.write_all_at(&pending_guard, committed)?;
1979            #[cfg(not(unix))]
1980            {
1981                let mut file = file;
1982                file.seek(io::SeekFrom::Start(committed))?;
1983                file.write_all(&pending_guard)?;
1984            }
1985            let new_len = pending_guard.len() as u64;
1986            let file_len = committed.checked_add(new_len).ok_or_else(|| {
1987                io::Error::new(
1988                    io::ErrorKind::InvalidData,
1989                    "committed + buffered len overflow",
1990                )
1991            })?;
1992            pending_guard.clear();
1993            drop(pending_guard);
1994            shard.file_len.store(file_len, Ordering::Release);
1995            shard.write_count.fetch_add(
1996                shard.pending_records.swap(0, Ordering::Relaxed),
1997                Ordering::Relaxed,
1998            );
1999            shard.bytes_written.fetch_add(new_len, Ordering::Relaxed);
2000            // Hold `append_lock` for the whole commit (snapshot → write →
2001            // length/accounting update) so no put can observe a file_len
2002            // that hasn't caught up with bytes already handed out.
2003        }
2004        self.dirty.lock().insert(shard.slot);
2005        Ok(())
2006    }

2026    /// The actual on-disk byte length of the frame at `offset` — length
2027    /// prefix (4) + frame body + CRC (4). Unlike [`serialized_len`]
2028    /// (an uncompressed upper bound computed from a decoded `Record`),
2029    /// this reflects compression: it's read straight off the length
2030    /// prefix, without decompressing (or even reading) the node bytes.
2031    /// Used for disk-usage accounting (e.g. repack preflight), where an
2032    /// uncompressed estimate would overstate what a repack actually
2033    /// reclaims.
2034    ///
2035    /// [`serialized_len`]: crate::packfile::Record::serialized_len
2036    ///
2037    /// # Errors
2038    /// Returns `StorageError::Corrupt` on a truncated or invalid frame,
2039    /// `StorageError::Io` on I/O failure.
2040    ///
2041    /// # Panics
2042    /// Panics only on internal invariant violation (unreachable path).
2043    pub fn record_disk_len_at(
2044        shard: &Shard,
2045        offset: u64,
2046    ) -> Result<u64, crate::storage::StorageError> {
2047        use crate::storage::StorageError;
2048
2049        for attempt in 0..2 {
2050            let guard = shard.mmap().map_err(StorageError::Io)?;
2051            let Some(mapping) = guard.as_ref().cloned() else {
2052                return Err(StorageError::Io(io::Error::new(
2053                    io::ErrorKind::Unsupported,
2054                    "shard could not be mapped",
2055                )));
2056            };
2057            drop(guard);
2058            let mem = mapping.as_ref();
2059
2060            let offset_usize = usize::try_from(offset)
2061                .map_err(|_| StorageError::Corrupt(format!("offset too large: {offset}")))?;
2062
2063            let file_len_usize = usize::try_from(shard.file_len()).unwrap_or(usize::MAX);
2064
2065            if offset_usize
2066                .checked_add(4)
2067                .map_or(true, |end| end > file_len_usize || end > mem.len())
2068            {
2069                if attempt == 0 {
2070                    Self::remap_shard(shard)?;
2071                    continue;
2072                }
2073                return Err(StorageError::Corrupt("truncated length prefix".into()));
2074            }
2075
2076            let prefix_end = offset_usize.checked_add(4).expect("checked above");
2077            let frame_len_bytes: [u8; 4] = mem[offset_usize..prefix_end].try_into().unwrap();
2078            let frame_len = u32::from_le_bytes(frame_len_bytes);
2079
2080            if !(packfile::FRAME_FIXED_LEN..=packfile::MAX_RECORD_LEN).contains(&frame_len) {
2081                return Err(StorageError::Corrupt(format!(
2082                    "invalid record length: {frame_len}"
2083                )));
2084            }
2085
2086            let frame_end = prefix_end
2087                .checked_add(frame_len as usize)
2088                .and_then(|end| end.checked_add(4));
2089            if frame_end.map_or(true, |end| end > file_len_usize || end > mem.len()) {
2090                if attempt == 0 {
2091                    Self::remap_shard(shard)?;
2092                    continue;
2093                }
2094                return Err(StorageError::Corrupt("truncated frame body or CRC".into()));
2095            }
2096
2097            return Ok(u64::from(frame_len) + 8);
2098        }
2099        unreachable!("record_disk_len_at remap-retry is bounded to two iterations")
2100    }

2209                return Err(StorageError::Corrupt("truncated record metadata".into()));
2210            }

2221                .checked_add(16)
2222                .expect("validated record collection id");

2225            let mut hash = [0; 16];
2226            hash.copy_from_slice(&mem[collection_end..metadata_end]);

2233    /// `scan_packfile`, whose offsets are by construction within the file).
2234    /// Unlike [`Self::read_at`] this does not flush anything: a virtual
2235    /// (buffered) offset would fail the bounds checks. Intended for the
2236    /// offline inspection paths that only ever touch committed frames.
2237    ///
2238    /// # Errors
2239    /// Returns `StorageError::Corrupt` on CRC mismatch (when `verify` and the
2240    /// frame carries a checksum) or truncated frame, `StorageError::Io` on
2241    /// I/O failure.

2243    /// # Panics

2246        shard: &Arc<Shard>,
2247        offset: u64,
2248        verify: bool,
2249    ) -> Result<Record, crate::storage::StorageError> {
2250        use crate::storage::StorageError;

2254            let Some(mapping) = guard.as_ref().cloned() else {
2255                return Err(StorageError::Io(io::Error::new(
2256                    io::ErrorKind::Unsupported,
2257                    "shard could not be mapped",
2258                )));
2259            };

2264                .map_err(|_| StorageError::Corrupt(format!("offset too large: {offset}")))?;

2281            let frame_len = u32::from_le_bytes(frame_len_bytes);

2284                return Err(StorageError::Corrupt(format!(
2285                    "invalid record length: {frame_len}"
2286                )));
2287            }

2291                .checked_add(frame_len_usize)

2294                .checked_add(4)
2295                .ok_or_else(|| StorageError::Corrupt("crc_pos + 4 overflow".into()))?;

2301                }

2303            }

2311                    mmap: Arc::clone(&mapping),
2312                    // `frame_len >= FRAME_FIXED_LEN` was checked above, so
2313                    // the fixed 37-byte metadata prefix fits in the frame.
2314                    start: prefix_end
2315                        .checked_add(37)
2316                        .expect("validated frame metadata prefix fits in usize"),
2317                    end: crc_pos,
2318                },

2321        unreachable!("read_at remap-retry is bounded to two iterations")
2322    }

2327    /// compared; structural validation (lengths, flags, node bytes) still runs.

2333        node_bytes_owner: MmapRange,

2336

2338        if flags & !(packfile::FLAG_COMPRESSED | packfile::FLAG_CRC_DISABLED) != 0 {
2339            return Err(StorageError::Corrupt(format!(
2340                "unsupported record flags: {flags:#04x}"
2341            )));
2342        }

2438    /// Rotate while `rotation_lock` is held by the caller.
2439    fn rotate_locked(&self) -> io::Result<()> {
2440        let current = *self.active_write.lock();
2441
2442        let mut shards = self.shards.write();
2443
2444        // Find the next available slot (skip current, prefer retired/empty)
2445        for offset in 1..=MAX_SHARDS_U16 {
2446            let candidate = current.wrapping_add(offset).wrapping_rem(MAX_SHARDS_U16);
2447            if shards[candidate as usize].is_none() {
2448                let pack_id = self.next_pack_id.load(Ordering::Relaxed);
2449                let next = pack_id.checked_add(1).expect("pack_id overflow");
2450
2451                // Persist the high-water mark BEFORE creating the pack file.
2452                // A crash after pack creation but before the next persist
2453                // would leave a pack_id in use with no pool.meta reservation.
2454                Self::persist_pool_meta_at(&self.base_dir, next, self.bucket_seed)?;
2455
2456                let (file, path) = Self::create_packfile_atomically(&self.base_dir, pack_id)?;
2457                let file_len = file.metadata()?.len();
2458                shards[candidate as usize] = Some(Arc::new(Shard::new(
2459                    candidate, pack_id, file, path, file_len,
2460                )));
2461                // Advance the in-memory counter only after successful creation.
2462                self.next_pack_id.store(next, Ordering::Relaxed);
2463                drop(shards);
2464                *self.active_write.lock() = candidate;
2465                return Ok(());
2466            }
2467        }
2468
2469        // All slots occupied and none retired — fail rather than silently
2470        // overwriting a shard that may still be referenced by collection indexes.
2471        // The caller must repack to reclaim retired shard slots before
2472        // rotating again.
2473        Err(io::Error::other(
2474            "shard pool full: all slots occupied by active shards; repack to reclaim",
2475        ))
2476    }

2562            if last.is_some_and(|prev| now.duration_since(prev) < min_interval) {
2563                return;
2564            }

2566        }

2619    /// Sync all shards to disk.
2620    ///
2621    /// Commits any buffered frames first, so fsync covers everything a
2622    /// caller believes it has put.
2623    ///
2624    /// **Deliberately does not skip-based-coalesce.** Unlike [`Self::sync_dirty`],
2625    /// which may skip a shard once another caller has fsynced it, this fsyncs
2626    /// every shard unconditionally: its contract is that everything put before
2627    /// the call is durable, and "someone else fsynced this shard at some point"
2628    /// does not prove that. The per-shard `sync_lock` here therefore provides
2629    /// mutual exclusion only, not the coalescing win `sync_dirty` gets — two
2630    /// concurrent `sync_all` callers can each issue their own fsync, which is
2631    /// correct, just not deduplicated.
2632    ///
2633    /// # Errors
2634    /// Returns `io::Error` on sync failure.
2635    pub fn sync_all(&self) -> io::Result<()> {
2636        let flush_started = Instant::now();
2637        self.flush_all()?;
2638        let flush_elapsed = flush_started.elapsed();
2639        let fsync_started = Instant::now();
2640        let mut had_dirty = false;
2641        for (id, shard) in self.all_shards() {
2642            // Full-pool sync fsyncs every shard unconditionally (that is its
2643            // contract), so unlike `sync_dirty` it can't skip a clean shard —
2644            // but the per-shard lock still serializes it against a concurrent
2645            // `sync_dirty` fsyncing the same file, and the dirty bit is
2646            // claimed before the fsync rather than under the pool lock
2647            // across it.
2648            let _sync_guard = shard.sync_lock.lock();
2649            if self.take_dirty(id) {
2650                had_dirty = true;
2651            }
2652            if let Err(error) = shard.file.sync_all() {
2653                self.restore_dirty(id);
2654                return Err(error);
2655            }
2656            shard.sync_count.fetch_add(1, Ordering::Relaxed);
2657        }
2658        let fsync_elapsed = fsync_started.elapsed();
2659        *self.last_sync_split.lock() = Some((flush_elapsed, fsync_elapsed));
2660        // The stats snapshot is observability, not state: only rewrite it when
2661        // a sync actually moved or committed data (parallel to `sync_dirty`),
2662        // so a steady-state writer that syncs between writes pays no metadata
2663        // write+rename for a file nothing changed. Periodic freshness is
2664        // `maybe_persist_stats`' job.
2665        if had_dirty {
2666            self.persist_stats_best_effort();
2667        }
2668        Ok(())
2669    }

2687    /// Sync only shards written to since the last sync.
2688    ///
2689    /// Each shard's dirty bit is claimed before its fsync and restored if the
2690    /// fsync fails, so a partial failure leaves the untried shards marked
2691    /// dirty for the next call.
2692    ///
2693    /// Concurrent callers coalesce per shard: the first to claim a shard's
2694    /// dirty bit fsyncs it; any other caller wanting the same shard waits on
2695    /// that shard's `sync_lock`, then finds the bit already cleared and skips
2696    /// its own fsync. The pool-wide `dirty` lock is never held across an
2697    /// fsync.
2698    ///
2699    /// Skipping is safe because a shard's dirty bit is only cleared when an
2700    /// fsync that *covers that shard's current bytes* has completed: every
2701    /// flush writes its bytes (`flush_shard_with_guard`) before marking the
2702    /// shard dirty, and each caller flushes before it claims — so any byte
2703    /// whose dirty bit this caller observes was written before the fsync that
2704    /// clears it, and an fsync covers all writes that completed before it.
2705    /// A write that lands after the claim re-marks the shard dirty and is
2706    /// caught by the next sync.
2707    ///
2708    /// Commits any buffered frames first, so the fsync covers everything a
2709    /// caller believes it has put.
2710    ///
2711    /// # Errors
2712    /// Returns `io::Error` on sync failure.
2713    pub fn sync_dirty(&self) -> io::Result<()> {
2714        let flush_started = Instant::now();
2715        self.flush_all()?;
2716        let flush_elapsed = flush_started.elapsed();
2717        let fsync_started = Instant::now();
2718        let mut synced_any = false;
2719        for id in self.dirty_snapshot() {
2720            let Some(shard) = self.get_shard(id) else {
2721                // Slot was retired between snapshot and lookup; leave any
2722                // remaining dirty bit for the next pass to resolve.
2723                continue;
2724            };
2725            let _sync_guard = shard.sync_lock.lock();
2726            if !self.take_dirty(id) {
2727                // A concurrent syncer we just waited for already fsynced
2728                // this shard — nothing left to do.
2729                continue;
2730            }
2731            // `sync_data` (fdatasync): the packfile is append-only and its
2732            // frame framing/size live in the data stream, so flushing data
2733            // plus the size metadata makes appended bytes readable after a
2734            // crash; the mode/timestamp metadata `sync_all` would also write
2735            // is not needed. The full-barrier `sync_all` above keeps
2736            // `sync_all`.
2737            if let Err(error) = shard.file.sync_data() {
2738                self.restore_dirty(id);
2739                return Err(error);
2740            }
2741            shard.sync_count.fetch_add(1, Ordering::Relaxed);
2742            synced_any = true;
2743        }
2744        let fsync_elapsed = fsync_started.elapsed();
2745        *self.last_sync_split.lock() = Some((flush_elapsed, fsync_elapsed));
2746        if synced_any {
2747            // Rate-limit the snapshot on the hot dirty path: it is
2748            // observability data, and `sync_all` (on dirty shards) / `Drop`
2749            // persist it.
2750            self.maybe_persist_stats(STATS_FLUSH_MIN_INTERVAL);
2751        }
2752        Ok(())
2753    }

2755    /// Retire a shard: mark it for deletion and free its pool slot.
2756    ///
2757    /// The shard file is deleted when the last `Arc<Shard>` reference drops
2758    /// (via `Shard::drop` when `is_current == false`). Setting the slot to
2759    /// `None` allows `rotate()` to reuse it.
2760    ///
2761    /// Does nothing if the slot is already empty or is the active write shard.
2762    ///
2763    /// Also does nothing if the shard has buffered frames that fail to
2764    /// flush: retiring anyway would mark the shard for deletion (via
2765    /// `Shard::drop`) while data an index still references is only in
2766    /// memory, destroying it. Leaving the slot occupied means the next
2767    /// flush attempt (or another retire) gets another chance.
2768    pub fn retire_slot(&self, slot: u16) {
2769        // Commit any buffered frames first: records are only ever retired
2770        // after their bytes are on disk, so a retire can never strand data
2771        // that an index still references.
2772        if let Some(shard) = self.get_shard(slot) {
2773            if self.flush_shard(&shard).is_err() {
2774                return;
2775            }
2776        }
2777        let mut shards = self.shards.write();
2778        let mut dirty = self.dirty.lock();
2779
2780        if *self.active_write.lock() == slot {
2781            return; // leave dirty unchanged — sync_dirty must fsync later
2782        }
2783        if let Some(slot_entry) = shards.get_mut(slot as usize) {
2784            if let Some(shard) = slot_entry.take() {
2785                dirty.remove(&slot);
2786                shard.is_current.store(false, Ordering::Release);
2787                self.retired_count.fetch_add(1, Ordering::Relaxed);
2788                drop(shards);
2789                self.collection_home.write().retain(|_, home| *home != slot);
2790            }
2791        }
2792    }
2793
2794    /// Total number of shards retired over the pool's lifetime.
2795    #[must_use]
2796    pub fn retired_count(&self) -> u64 {
2797        self.retired_count.load(Ordering::Relaxed)
2798    }

2800    /// Scan a shard file and return `(collection_id, hash, offset)` entries.
2801    /// Used during startup to rebuild per-collection indexes.
2802    ///
2803    /// # Errors
2804    /// Returns `io::Error` on file open or header read failure.
2805    pub fn scan_shard(path: &Path) -> io::Result<Vec<ShardEntry>> {
2806        let file = File::open(path)?;
2807        let mut reader = BufReader::new(file);
2808        let mut entries = Vec::new();
2809
2810        if packfile::read_header(&mut reader)?.is_none() {
2811            return Ok(entries);
2812        }
2813
2814        loop {
2815            let offset = reader.stream_position()?;
2816            match packfile::read_record(&mut reader) {
2817                Ok(Some(record)) => {
2818                    entries.push((record.collection_id, record.hash, offset));
2819                }
2820                Ok(None) => break,
2821                Err(e) if e.kind() == io::ErrorKind::UnexpectedEof => break,
2822                Err(e) => return Err(e),
2823            }
2824        }
2825
2826        Ok(entries)
2827    }
