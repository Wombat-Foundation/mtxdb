1//! Named auxiliary indexes for application-owned lookup tables.
2//!
3//! Auxiliary indexes are logical stores, not additional packfile pools. They
4//! use the normal [`StorageEngine`] so callers do not create one directory or
5//! file per room, namespace, or index name.
6//!
7//! Identity note: auxiliary key digests and collection ids are intentionally
8//! BLAKE3 (via [`DigestAlgorithm::Blake3`] and [`derive_collection_id`]). These
9//! indexes are core-internal and unreleased, so there is no persisted SHA-256
10//! data to remain compatible with and no versioned dual-read or migration is
11//! required. Reverting these derivations to SHA-256 would be a regression, not
12//! a compatibility fix.
13

22const VALUE_MAGIC: &[u8; 4] = b"AUX1";
23const DIGEST_LEN: usize = 32;
24
25/// The canonical identity of an auxiliary-index key.
26pub type AuxiliaryKeyDigest = [u8; DIGEST_LEN];
27
28/// Derive the canonical full digest for an auxiliary-index key.
29#[must_use]
30pub fn auxiliary_key_digest(key: &[u8]) -> AuxiliaryKeyDigest {
31    DigestAlgorithm::Blake3.digest(key)
32}
33
34/// Derive the logical collection identity for a named auxiliary index.
35///
36/// Auxiliary indexes are core-internal collections, so they use the shared
37/// [`derive_collection_id`] with the core-internal pool namespace rather than a
38/// private domain string. The returned 16-byte value is the routing id; key
39/// identities remain the full 32-byte digests stored in each envelope.
40#[must_use]
41pub fn auxiliary_collection_id(name: &str) -> [u8; 16] {
42    derive_collection_id(Some(MEMBER_NAMESPACE_INTL), name.as_bytes())
43}
44
45fn physical_id(digest: &AuxiliaryKeyDigest) -> NodeId {
46    let mut id = [0u8; 16];
47    id.copy_from_slice(&digest[..16]);
48    id
49}

51/// A named logical auxiliary index backed by an existing storage engine.
52pub struct AuxiliaryIndex<'a, S: StorageEngine + ?Sized> {
53    engine: &'a S,
54    name: String,
55    collection_id: [u8; 16],
56}
57
58impl<'a, S: StorageEngine + ?Sized> AuxiliaryIndex<'a, S> {
59    /// Open a named auxiliary index in an existing storage engine.
60    #[must_use]
61    pub fn open(engine: &'a S, name: &str) -> Self {
62        Self {
63            engine,
64            name: name.to_owned(),
65            collection_id: auxiliary_collection_id(name),
66        }
67    }

69    /// Return this index's canonical name.
70    #[must_use]
71    pub fn name(&self) -> &str {
72        &self.name
73    }

75    /// Return this index's collection metadata.
76    #[must_use]
77    pub fn metadata(&self) -> CollectionMetadata {
78        CollectionMetadata {
79            member_namespace: Some(MEMBER_NAMESPACE_INTL),
80            collection_canonical_id: self.name.as_bytes().to_vec(),
81            record_id_rule: RecordIdentityRule {
82                policy: FrameIdPolicy::Key,
83                digest_algorithm: DigestAlgorithm::Blake3,
84            },
85            payload: PayloadPolicy::Source,
86            extension: None,
87            role: Some("system_auxiliary".to_owned()),
88            schema: None,
89        }
90    }

92    /// Ensure that collection metadata is established for this auxiliary index.
93    ///
94    /// # Errors
95    /// Returns [`StorageError::Internal`] if existing metadata conflicts, or propagates backend error.
96    pub fn ensure_metadata(&self) -> Result<(), StorageError> {
97        self.engine
98            .create_or_put_established(&self.collection_id, &self.metadata(), &[])
99    }

101    /// Return this index's compatibility collection ID.
102    #[must_use]
103    pub fn collection_id(&self) -> [u8; 16] {
104        self.collection_id
105    }

107    /// Read a value by its logical key.
108    ///
109    /// Legacy unwrapped values are returned unchanged. New values verify the
110    /// full 32-byte digest before returning the application payload.
111    ///
112    /// # Errors
113    /// Returns [`StorageError::Corrupt`] for a malformed or mismatched
114    /// envelope, or propagates a backend error.
115    pub fn get(&self, key: &[u8]) -> Result<Option<Vec<u8>>, StorageError> {
116        let digest = auxiliary_key_digest(key);
117        let node_id = physical_id(&digest);
118        let data = self.engine.get(&self.collection_id, &node_id)?;
119        decode_stored_value(&digest, data.as_ref())
120    }
121
122    /// Read several logical keys with one backend lookup.
123    ///
124    /// The returned values are in the same order as `keys`. Missing keys are
125    /// represented by `None`; stored envelopes are validated just like
126    /// [`Self::get`].
127    ///
128    /// # Errors
129    /// Returns a storage or auxiliary-envelope error.
130    pub fn get_many(&self, keys: &[&[u8]]) -> Result<Vec<Option<Vec<u8>>>, StorageError> {
131        let digests: Vec<AuxiliaryKeyDigest> =
132            keys.iter().map(|key| auxiliary_key_digest(key)).collect();
133        let ids: Vec<NodeId> = digests.iter().map(physical_id).collect();
134        let existing = self.engine.get_many(&self.collection_id, &ids)?;
135        existing
136            .iter()
137            .zip(digests.iter())
138            .map(|(data, digest)| decode_stored_value(digest, data.as_ref()))
139            .collect()
140    }
141
142    /// Insert or update a value by its logical key.
143    ///
144    /// Atomically establishes the collection metadata on the first write
145    /// and appends the key-value envelope. Retains idempotency (same key and value succeeds),
146    /// permits same-key replacement with updated values, and rejects truncated-ID collisions
147    /// (distinct key with the same 16-byte physical node ID) with [`StorageError::Collision`].
148    ///
149    /// # Errors
150    /// Returns [`StorageError::Collision`] on physical ID key collisions,
151    /// [`StorageError::Corrupt`] on corrupted existing records, or propagates backend error.
152    #[allow(clippy::arithmetic_side_effects)]
153    pub fn put(&self, key: &[u8], value: &[u8]) -> Result<(), StorageError> {
154        let digest = auxiliary_key_digest(key);
155        let node_id = physical_id(&digest);
156        if let Some(existing) = self.engine.get(&self.collection_id, &node_id)? {
157            if !existing.bytes.is_empty() {
158                decode_value(&digest, &existing.bytes)?;
159            }
160        }
161        let capacity = VALUE_MAGIC
162            .len()
163            .checked_add(DIGEST_LEN)
164            .and_then(|length| length.checked_add(value.len()))
165            .ok_or_else(|| StorageError::Corrupt("auxiliary value length overflow".to_owned()))?;
166        let mut encoded = Vec::with_capacity(capacity);
167        encoded.extend_from_slice(VALUE_MAGIC);
168        encoded.extend_from_slice(&digest);
169        encoded.extend_from_slice(value);
170        let data = NodeData::new(bytes::Bytes::from(encoded));
171
172        let mut validate = |existing: Option<&NodeData>| -> Result<(), StorageError> {
173            if let Some(existing) = existing {
174                if !existing.bytes.is_empty() {
175                    decode_value(&digest, &existing.bytes)?;
176                }
177            }
178            Ok(())
179        };
180
181        self.engine.create_or_upsert_established_validated(
182            &self.collection_id,
183            &self.metadata(),
184            &node_id,
185            &data,
186            &mut validate,
187        )
188    }
189
190    /// Insert or replace several values with one read and one write batch.
191    ///
192    /// Existing records are still checked for full-digest collisions, but
193    /// callers avoid one storage round trip per auxiliary entry.
194    ///
195    /// Writes are last-writer-wins per record and are not serialized against
196    /// other writers: if two `put_many` calls race on the same key, the value
197    /// from the batch the backing store commits last is retained. Callers must
198    /// not rely on any ordering between concurrent writers.
199    ///
200    /// # Errors
201    /// Returns [`StorageError::Collision`] for distinct keys with the same
202    /// physical ID, [`StorageError::Corrupt`] for malformed existing envelopes,
203    /// and propagates storage errors.
204    pub fn put_many(&self, entries: &[(&[u8], &[u8])]) -> Result<usize, StorageError> {
205        if entries.is_empty() {
206            self.ensure_metadata()?;
207            return Ok(0);
208        }
209        let mut physical: Vec<(NodeId, NodeData)> = Vec::with_capacity(entries.len());
210        let mut seen: HashMap<NodeId, AuxiliaryKeyDigest> = HashMap::with_capacity(entries.len());
211        for (key, value) in entries {
212            let digest = auxiliary_key_digest(key);
213            let node_id = physical_id(&digest);
214            if let Some(previous) = seen.insert(node_id, digest) {
215                if previous != digest {
216                    return Err(StorageError::Collision(
217                        "auxiliary-index key digest collision".to_owned(),
218                    ));
219                }
220            }
221            let capacity = VALUE_MAGIC
222                .len()
223                .checked_add(DIGEST_LEN)
224                .and_then(|length| length.checked_add(value.len()))
225                .ok_or_else(|| {
226                    StorageError::Corrupt("auxiliary value length overflow".to_owned())
227                })?;
228            let mut encoded = Vec::with_capacity(capacity);
229            encoded.extend_from_slice(VALUE_MAGIC);
230            encoded.extend_from_slice(&digest);
231            encoded.extend_from_slice(value);
232            physical.push((node_id, NodeData::new(bytes::Bytes::from(encoded))));
233        }
234
235        self.ensure_metadata()?;
236        let ids: Vec<NodeId> = physical.iter().map(|(id, _)| *id).collect();
237        let existing = self.engine.get_many(&self.collection_id, &ids)?;
238        for ((_, data), existing) in physical.iter().zip(existing) {
239            if let Some(existing) = existing {
240                // The full digest is retained in the envelope. Validate it
241                // before allowing the batch to replace the record.
242                let digest_start = VALUE_MAGIC.len();
243                let digest_end = digest_start.checked_add(DIGEST_LEN).ok_or_else(|| {
244                    StorageError::Corrupt("auxiliary digest length overflow".to_owned())
245                })?;
246                let digest: AuxiliaryKeyDigest = data
247                    .bytes
248                    .get(digest_start..digest_end)
249                    .and_then(|bytes| bytes.try_into().ok())
250                    .ok_or_else(|| {
251                        StorageError::Corrupt(
252                            "new auxiliary value has an incomplete digest envelope".to_owned(),
253                        )
254                    })?;
255                decode_value(&digest, &existing.bytes)?;
256            }
257        }
258        self.engine.put_many(&self.collection_id, &physical)
259    }
260}
261
262fn decode_value(
263    expected_digest: &AuxiliaryKeyDigest,
264    encoded: &[u8],
265) -> Result<Vec<u8>, StorageError> {
266    if !encoded.starts_with(VALUE_MAGIC) {
267        return Ok(encoded.to_vec());
268    }
269    let digest_start = VALUE_MAGIC.len();
270    let value_start = digest_start.checked_add(DIGEST_LEN).ok_or_else(|| {
271        StorageError::Corrupt("auxiliary-index envelope length overflow".to_owned())
272    })?;
273    if encoded.len() < value_start {
274        return Err(StorageError::Corrupt(
275            "truncated auxiliary-index value envelope".to_owned(),
276        ));
277    }
278    if encoded[digest_start..value_start] != expected_digest[..] {
279        return Err(StorageError::Collision(
280            "auxiliary-index key digest collision".to_owned(),
281        ));
282    }
283    Ok(encoded[value_start..].to_vec())
284}
285
286fn decode_stored_value(
287    digest: &AuxiliaryKeyDigest,
288    data: Option<&NodeData>,
289) -> Result<Option<Vec<u8>>, StorageError> {
290    let Some(data) = data else {
291        return Ok(None);
292    };
293    if data.bytes.is_empty() {
294        return Ok(None);
295    }
296    decode_value(digest, &data.bytes).map(Some)
297}

308        let first = AuxiliaryIndex::open(&engine, "first");

310        assert_ne!(first.collection_id(), second.collection_id());
311        first.put(b"key", b"value").unwrap();
312        assert_eq!(first.get(b"key").unwrap(), Some(b"value".to_vec()));

422    #[test]
423    fn auxiliary_index_rejects_truncated_id_collision_without_corrupting() {
424        let engine = InMemoryStorage::new();
425        let index = AuxiliaryIndex::open(&engine, "sys:test-collisions");
426        index.put(b"key_1", b"val_1").unwrap();
427        assert_eq!(index.get(b"key_1").unwrap(), Some(b"val_1".to_vec()));
428
429        let digest1 = auxiliary_key_digest(b"key_1");
430        let node_id1 = physical_id(&digest1);
431
432        // Fabricate a colliding key that has a different full digest but targets the same node_id
433        let mut colliding_digest = digest1;
434        colliding_digest[31] ^= 0xFF; // Different 32-byte digest, same first 16 bytes!
435        assert_eq!(&colliding_digest[..16], &node_id1[..]);
436
437        let mut encoded = Vec::new();
438        encoded.extend_from_slice(VALUE_MAGIC);
439        encoded.extend_from_slice(&colliding_digest);
440        encoded.extend_from_slice(b"colliding_val");
441        let colliding_data = NodeData::new(bytes::Bytes::from(encoded));
442
443        let mut validate = |existing: Option<&NodeData>| -> Result<(), StorageError> {
444            if let Some(existing) = existing {
445                if !existing.bytes.is_empty() {
446                    decode_value(&colliding_digest, &existing.bytes)?;
447                }
448            }
449            Ok(())
450        };
451
452        let err = engine
453            .create_or_upsert_established_validated(
454                &index.collection_id(),
455                &index.metadata(),
456                &node_id1,
457                &colliding_data,
458                &mut validate,
459            )
460            .unwrap_err();
461
462        assert!(matches!(err, StorageError::Collision(_)));
463
464        // Original key remains intact and uncorrupted:
465        assert_eq!(index.get(b"key_1").unwrap(), Some(b"val_1".to_vec()));
466    }
