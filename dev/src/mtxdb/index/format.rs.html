1//! Fixed-width, little-endian wire records for the persisted index cache.
2//!
3//! The cache is deliberately a separate, rebuildable acceleration structure:
4//! packfiles remain authoritative.  Keeping these records fixed-width makes a
5//! torn delta tail unambiguous and lets loaders bounds-check before allocating.
6
7#[cfg(not(target_endian = "little"))]
8compile_error!("the persisted index cache is currently supported only on little-endian targets");
9
10/// Bytes in one [`DeltaFrame`].
11pub const DELTA_FRAME_LEN: usize = 36;
12/// Bytes in the checkpoint header.
13pub const CHECKPOINT_HEADER_LEN: usize = 80;
14/// Bytes in one collection directory entry.
15pub const COLLECTION_DIR_ENTRY_LEN: usize = 56;
16/// Bytes in one [`PackTableEntry`].
17pub const PACK_TABLE_ENTRY_LEN: usize = 12;
18
19/// One slot overwrite after a checkpoint.
20///
21/// `bucket` is the frame's landing bucket — the actual table position the
22/// slot was written at, after `insert_tracked` advanced past any occupied
23/// slots during linear probing — not the hash's home bucket. `replay_frames`
24/// writes directly to it, so a producer logging a frame at the home bucket
25/// instead would place the delta at the wrong slot. `slot` is the packed
26/// [`crate::index::IndexSlot`] representation stored there. `generation`
27/// prevents a delta for a pre-resize table being applied to a resized or
28/// repacked collection.
29#[derive(Debug, Clone, Copy, PartialEq, Eq)]
30pub struct DeltaFrame {
31    /// The collection whose index changed.
32    pub collection_id: [u8; 16],
33    /// The affected table bucket.
34    pub bucket: u32,
35    /// The collection-index generation this frame applies to.
36    pub generation: u64,
37    /// The packed `IndexSlot` value to store.
38    pub slot: u64,
39}
40
41impl DeltaFrame {
42    #[must_use]
43    /// Encodes this frame in its fixed-width little-endian representation.
44    pub fn encode(self) -> [u8; DELTA_FRAME_LEN] {
45        let mut bytes = [0; DELTA_FRAME_LEN];
46        bytes[..16].copy_from_slice(&self.collection_id);
47        bytes[16..20].copy_from_slice(&self.bucket.to_le_bytes());
48        bytes[20..28].copy_from_slice(&self.generation.to_le_bytes());
49        bytes[28..36].copy_from_slice(&self.slot.to_le_bytes());
50        bytes
51    }
52
53    #[must_use]
54    /// Decodes exactly one fixed-width frame, rejecting any other length.
55    pub fn decode(bytes: &[u8]) -> Option<Self> {
56        let bytes: &[u8; DELTA_FRAME_LEN] = bytes.try_into().ok()?;
57        Some(Self {
58            collection_id: bytes[..16].try_into().ok()?,
59            bucket: u32::from_le_bytes(bytes[16..20].try_into().ok()?),
60            generation: u64::from_le_bytes(bytes[20..28].try_into().ok()?),
61            slot: u64::from_le_bytes(bytes[28..36].try_into().ok()?),
62        })
63    }
64}

66/// Checkpoint envelope.  The directory immediately follows this header,
67/// followed by the concatenated raw index slot arrays, then homes and tails.
68#[derive(Debug, Clone, Copy, PartialEq, Eq)]
69pub struct CheckpointHeader {
70    /// Format-identifying magic bytes.
71    pub magic: [u8; 8],
72    /// Wire-format version.
73    pub version: u32,
74    /// Number of collection directory entries.
75    pub collection_count: u32,
76    /// Total byte length of the directory section.
77    pub directory_bytes: u64,
78    /// Total byte length of the raw-slot section.
79    pub slots_bytes: u64,
80    /// Fingerprint of the packfile set the checkpoint describes.
81    pub pack_fingerprint: u64,
82    /// CRC-32/IEEE (`crc32fast`, the same crate/polynomial already used for
83    /// pack-frame checksums) of every byte after this header — the
84    /// directory, raw slot, homes, and tails sections combined. Verified
85    /// once, in full, at read time.
86    pub content_crc32: u32,
87    /// Total byte length of the homes section (all collections concatenated).
88    /// Zero for pre-v4 checkpoints.
89    pub homes_bytes: u64,
90    /// Total byte length of the tails section (all collections concatenated).
91    /// Zero for pre-v4 checkpoints.
92    pub tails_bytes: u64,
93    /// Journal LSN whose mutations this checkpoint's index snapshot already
94    /// incorporates. Written atomically with the checkpoint, so a reader can
95    /// bind its overlay coverage to exactly the index it loaded rather than to
96    /// a separately-read `journal.lsn` that may have advanced past it. Zero
97    /// when the checkpoint was written without a journal.
98    pub covered_lsn: u64,
99    /// Number of [`PackTableEntry`] records in the pack table.
100    pub pack_table_count: u32,
101    /// Total byte length of the pack table section
102    /// (`pack_table_count * PACK_TABLE_ENTRY_LEN`).
103    pub pack_table_bytes: u64,
104}
105
106impl CheckpointHeader {
107    #[must_use]
108    /// Encodes this header in its fixed-width little-endian representation.
109    pub fn encode(self) -> [u8; CHECKPOINT_HEADER_LEN] {
110        let mut bytes = [0; CHECKPOINT_HEADER_LEN];
111        bytes[..8].copy_from_slice(&self.magic);
112        bytes[8..12].copy_from_slice(&self.version.to_le_bytes());
113        bytes[12..16].copy_from_slice(&self.collection_count.to_le_bytes());
114        bytes[16..24].copy_from_slice(&self.directory_bytes.to_le_bytes());
115        bytes[24..32].copy_from_slice(&self.slots_bytes.to_le_bytes());
116        bytes[32..40].copy_from_slice(&self.pack_fingerprint.to_le_bytes());
117        bytes[40..44].copy_from_slice(&self.content_crc32.to_le_bytes());
118        bytes[44..52].copy_from_slice(&self.homes_bytes.to_le_bytes());
119        bytes[52..60].copy_from_slice(&self.tails_bytes.to_le_bytes());
120        bytes[60..68].copy_from_slice(&self.covered_lsn.to_le_bytes());
121        bytes[68..72].copy_from_slice(&self.pack_table_count.to_le_bytes());
122        bytes[72..80].copy_from_slice(&self.pack_table_bytes.to_le_bytes());
123        bytes
124    }
125
126    #[must_use]
127    /// Decodes exactly one fixed-width header, rejecting any other length.
128    pub fn decode(bytes: &[u8]) -> Option<Self> {
129        let bytes: &[u8; CHECKPOINT_HEADER_LEN] = bytes.try_into().ok()?;
130        Some(Self {
131            magic: bytes[..8].try_into().ok()?,
132            version: u32::from_le_bytes(bytes[8..12].try_into().ok()?),
133            collection_count: u32::from_le_bytes(bytes[12..16].try_into().ok()?),
134            directory_bytes: u64::from_le_bytes(bytes[16..24].try_into().ok()?),
135            slots_bytes: u64::from_le_bytes(bytes[24..32].try_into().ok()?),
136            pack_fingerprint: u64::from_le_bytes(bytes[32..40].try_into().ok()?),
137            content_crc32: u32::from_le_bytes(bytes[40..44].try_into().ok()?),
138            homes_bytes: u64::from_le_bytes(bytes[44..52].try_into().ok()?),
139            tails_bytes: u64::from_le_bytes(bytes[52..60].try_into().ok()?),
140            covered_lsn: u64::from_le_bytes(bytes[60..68].try_into().ok()?),
141            pack_table_count: u32::from_le_bytes(bytes[68..72].try_into().ok()?),
142            pack_table_bytes: u64::from_le_bytes(bytes[72..80].try_into().ok()?),
143        })
144    }
145}

147/// One `(slot, pack_id)` binding in a checkpoint's pack table: the writer's
148/// local `ShardPool` slot for a live pack at checkpoint-write time, and that
149/// pack's stable, monotonically-allocated identity (`ShardPool::next_pack_id`
150/// is persisted and never decremented or reused within a pool — see
151/// `shard.rs`). A reader translates every checkpoint-encoded `shard_id`
152/// through this table into its own local slot for the same `pack_id`,
153/// rather than trusting the writer's raw slot number directly — slot
154/// numbers are a process-local, ephemeral handle, not identity, and drift
155/// after any shard retirement leaves a hole in the writer's slot table that
156/// a fresh reader's `discover_shards` (first-free-slot in `pack_id` order)
157/// does not reproduce.
158#[derive(Debug, Clone, Copy, PartialEq, Eq)]
159pub struct PackTableEntry {
160    /// The writer's local shard slot at checkpoint-write time.
161    pub slot: u16,
162    /// The pack's stable identity.
163    pub pack_id: u64,
164}
165
166impl PackTableEntry {
167    #[must_use]
168    /// Encodes this entry in its fixed-width little-endian representation.
169    pub fn encode(self) -> [u8; PACK_TABLE_ENTRY_LEN] {
170        let mut bytes = [0; PACK_TABLE_ENTRY_LEN];
171        bytes[..2].copy_from_slice(&self.slot.to_le_bytes());
172        bytes[4..12].copy_from_slice(&self.pack_id.to_le_bytes());
173        bytes
174    }
175
176    #[must_use]
177    /// Decodes exactly one fixed-width pack table entry, rejecting other lengths.
178    pub fn decode(bytes: &[u8]) -> Option<Self> {
179        let bytes: &[u8; PACK_TABLE_ENTRY_LEN] = bytes.try_into().ok()?;
180        Some(Self {
181            slot: u16::from_le_bytes(bytes[..2].try_into().ok()?),
182            pack_id: u64::from_le_bytes(bytes[4..12].try_into().ok()?),
183        })
184    }
185}

187/// Locates one collection's raw slot, homes, and tails arrays in a checkpoint.
188#[derive(Debug, Clone, Copy, PartialEq, Eq)]
189pub struct CollectionDirEntry {
190    /// The collection whose raw slots this entry locates.
191    pub collection_id: [u8; 16],
192    /// The checkpoint generation for this collection.
193    pub generation: u64,
194    /// Byte offset of the collection's slot data in the checkpoint.
195    pub slots_offset: u64,
196    /// Byte offset of the collection's homes data in the checkpoint.
197    /// Zero for pre-v4 checkpoints (homes not persisted).
198    pub homes_offset: u64,
199    /// Byte offset of the collection's tails data in the checkpoint.
200    /// Zero for pre-v4 checkpoints (tails not persisted).
201    pub tails_offset: u64,
202    /// Number of slots allocated by the collection's index.
203    pub capacity: u32,
204    /// Number of occupied slots.
205    pub slot_count: u32,
206}
207
208impl CollectionDirEntry {
209    #[must_use]
210    /// Encodes this entry in its fixed-width little-endian representation.
211    pub fn encode(self) -> [u8; COLLECTION_DIR_ENTRY_LEN] {
212        let mut bytes = [0; COLLECTION_DIR_ENTRY_LEN];
213        bytes[..16].copy_from_slice(&self.collection_id);
214        bytes[16..24].copy_from_slice(&self.generation.to_le_bytes());
215        bytes[24..32].copy_from_slice(&self.slots_offset.to_le_bytes());
216        bytes[32..40].copy_from_slice(&self.homes_offset.to_le_bytes());
217        bytes[40..48].copy_from_slice(&self.tails_offset.to_le_bytes());
218        bytes[48..52].copy_from_slice(&self.capacity.to_le_bytes());
219        bytes[52..56].copy_from_slice(&self.slot_count.to_le_bytes());
220        bytes
221    }
222
223    #[must_use]
224    /// Decodes exactly one fixed-width directory entry, rejecting other lengths.
225    pub fn decode(bytes: &[u8]) -> Option<Self> {
226        let bytes: &[u8; COLLECTION_DIR_ENTRY_LEN] = bytes.try_into().ok()?;
227        Some(Self {
228            collection_id: bytes[..16].try_into().ok()?,
229            generation: u64::from_le_bytes(bytes[16..24].try_into().ok()?),
230            slots_offset: u64::from_le_bytes(bytes[24..32].try_into().ok()?),
231            homes_offset: u64::from_le_bytes(bytes[32..40].try_into().ok()?),
232            tails_offset: u64::from_le_bytes(bytes[40..48].try_into().ok()?),
233            capacity: u32::from_le_bytes(bytes[48..52].try_into().ok()?),
234            slot_count: u32::from_le_bytes(bytes[52..56].try_into().ok()?),
235        })
236    }
237}

239#[cfg(test)]
240mod tests {
241    use super::*;
242
243    #[test]
244    fn fixed_width_records_round_trip() {
245        let delta = DeltaFrame {
246            collection_id: [1; 16],
247            bucket: 2,
248            generation: 3,
249            slot: 4,
250        };
251        assert_eq!(DeltaFrame::decode(&delta.encode()), Some(delta));
252
253        let header = CheckpointHeader {
254            magic: *b"MTXIDX01",
255            version: 4,
256            collection_count: 2,
257            directory_bytes: 112,
258            slots_bytes: 128,
259            pack_fingerprint: 9,
260            content_crc32: 0xDEAD_BEEF,
261            homes_bytes: 0,
262            tails_bytes: 0,
263            covered_lsn: 11,
264            pack_table_count: 2,
265            pack_table_bytes: 24,
266        };
267        assert_eq!(CheckpointHeader::decode(&header.encode()), Some(header));
268
269        let entry = CollectionDirEntry {
270            collection_id: [5; 16],
271            generation: 6,
272            slots_offset: 7,
273            homes_offset: 100,
274            tails_offset: 200,
275            capacity: 8,
276            slot_count: 9,
277        };
278        assert_eq!(CollectionDirEntry::decode(&entry.encode()), Some(entry));
279
280        let pack_entry = PackTableEntry {
281            slot: 3,
282            pack_id: 0xDEAD_BEEF_0000_0001,
283        };
284        assert_eq!(
285            PackTableEntry::decode(&pack_entry.encode()),
286            Some(pack_entry)
287        );
288    }
