13/// Per-database-instance tuning for [`LossyIndex`] construction.
14///
15/// `seed` is mixed into bucket *and* tag derivation to prevent adversarial
16/// precomputation against a fixed, unsalted content hash. Every real
17/// `PackfileStorage`-sourced config must carry the pool's actual persisted
18/// seed; the `Default` impl uses `seed: 0` only for tests and transient
19/// construction that never touches disk.
20#[derive(Debug, Clone, Copy)]
21pub struct IndexConfig {
22    /// Mixed into bucket and tag derivation to prevent adversarial
23    /// precomputation. See `index/mod.rs` seed docs.
24    pub seed: u64,
25    /// Minimum initial capacity for a newly created collection's index.
26    /// Composes with the hard 16-slot minimum: `effective_min = floor.max(16)`.
27    pub floor: u32,
28    /// Load factor (percent, 1..=90) at which the index grows.
29    pub load_factor_percent: u8,
30}
31
32impl Default for IndexConfig {
33    fn default() -> Self {
34        Self {
35            seed: 0,
36            floor: 64,
37            load_factor_percent: 75,
38        }
39    }
40}

42impl IndexConfig {
43    /// Validate the configuration.
44    ///
45    /// # Errors
46    /// Returns `Err(message)` if `load_factor_percent` is 0 or greater than
47    /// 90, or if `floor` is 0.
48    pub fn validate(&self) -> Result<(), &'static str> {
49        if self.load_factor_percent == 0 || self.load_factor_percent > 90 {
50            return Err("load_factor_percent must be in 1..=90");
51        }
52        if self.floor == 0 {
53            return Err("floor must be > 0");
54        }
55        Ok(())
56    }

59/// Splitmix64-style avalanche finalizer. Cheap, well-distributed, no
60/// dependencies. Used to mix the per-pool seed into bucket and tag
61/// derivation so an attacker cannot precompute hash → bucket/tag mappings
62/// without knowing the seed.
63#[inline]
64fn mix(x: u64) -> u64 {
65    let mut v = x;
66    v = v.wrapping_add(0x9E37_79B9_7F4A_7C15);
67    v = (v ^ (v >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
68    v = (v ^ (v >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
69    v ^ (v >> 31)
70}

72/// Per-slot entry in the lossy fanout index.
73///
74/// Layout: `[16-bit tag | 16-bit shard_id | 32-bit offset]` packed into a `u64`.
75///
76/// - **tag** (high 16 bits): truncated fingerprint for fast rejection.
77/// - **`shard_id`** (next 16 bits): which shard file this record lives in.
78/// - **offset** (low 32 bits): byte offset within the shard, stored as
79///   `offset + 1` so that the all-zeros encoding is reserved as the empty
80///   sentinel. Actual offset 0 is stored as 1, and `offset()` subtracts 1
81///   to recover the real value.
82///
83/// Empty slots are all-zeros. This is safe regardless of tag value: the
84/// `offset + 1` encoding guarantees a live slot's low 32 bits are never zero,
85/// so the all-zeros pattern can never be produced by a real insert — the tag
86/// bits play no role in reserving the sentinel.
87#[derive(Debug, Clone, Copy, PartialEq, Eq)]
88pub struct IndexSlot(u64);
89
90impl IndexSlot {
91    const EMPTY: Self = Self(0);
92
93    /// Largest byte offset representable by the 32-bit offset field while
94    /// reserving the all-zeros encoding for the empty-slot sentinel.
95    pub const MAX_OFFSET: u64 = (1u64 << 32) - 2;
96
97    const TAG_SHIFT: u64 = 48; // SHARD_BITS + OFFSET_BITS
98    const SHARD_SHIFT: u64 = 32; // OFFSET_BITS
99    const OFFSET_MASK: u64 = 0xFFFF_FFFF;
100
101    /// Create a new slot from its components.
102    ///
103    /// The offset is stored as `offset + 1` so that the all-zeros `u64`
104    /// encoding is reserved as the empty sentinel. Actual offset 0 is
105    /// stored as 1 in the slot.
106    ///
107    /// # Panics
108    /// Panics if `tag` exceeds 16 bits or `offset` exceeds [`Self::MAX_OFFSET`].
109    #[must_use]
110    pub fn new(tag: u32, shard_id: u16, offset: u64) -> Self {
111        assert!(tag <= 0xFFFF, "tag must fit in 16 bits");
112        assert!(
113            offset <= Self::MAX_OFFSET,
114            "offset must fit in 32 bits minus 1 (reserved for empty sentinel)"
115        );
116        Self(
117            (u64::from(tag) << Self::TAG_SHIFT)
118                | (u64::from(shard_id) << Self::SHARD_SHIFT)
119                | ((offset.wrapping_add(1)) & Self::OFFSET_MASK),
120        )
121    }
122
123    /// The sentinel value representing an empty slot.
124    #[must_use]
125    pub fn empty() -> Self {
126        Self::EMPTY
127    }
128
129    /// Returns `true` if this slot is the empty sentinel.
130    #[must_use]
131    pub fn is_empty(self) -> bool {
132        self.0 == 0
133    }
134
135    /// The 16-bit tag stored in this slot.
136    #[must_use]
137    pub fn tag(self) -> u32 {
138        ((self.0 >> Self::TAG_SHIFT) & 0xFFFF) as u32
139    }
140
141    /// The 16-bit shard ID stored in this slot.
142    #[must_use]
143    pub fn shard_id(self) -> u16 {
144        ((self.0 >> Self::SHARD_SHIFT) & 0xFFFF) as u16
145    }
146
147    /// The byte offset within the shard stored in this slot.
148    #[must_use]
149    pub fn offset(self) -> u64 {
150        (self.0 & Self::OFFSET_MASK).wrapping_sub(1)
151    }
152}

154/// A per-collection lossy fanout index for content-addressed records.
155///
156/// Uses open addressing with linear probing on a power-of-two sized table.
157/// The index is mmap-able (flat `u64` array) and small enough to stay
158/// resident in RAM for active collections.
159///
160/// Design decisions (from docs):
161/// - Partition by collection: ~8KB per 1000-node collection, 100 active collections < 1MB.
162/// - Empty slot terminates probe (write-once, no tombstones needed).
163/// - Tag collisions coexist in the probe chain; a tag is only a candidate
164///   filter and never an overwrite/equality proof.
165/// - Owned indexes keep `homes` and `tails` identity side tables: two
166///   anonymous `u64`s (16 bytes) per slot beyond the packed slot array.
167///   Checkpoints retain only packed slots; identities are hydrated lazily
168///   from authoritative frame headers when a writer needs them.
169/// - Power-of-two capacity: shift-and-mask bucket selection, cache-aligned probes.
170#[derive(Debug)]
171pub struct LossyIndex {
172    /// Power-of-two capacity.
173    capacity: u32,
174    /// Bitmask for bucket selection: capacity - 1.
175    mask: u32,
176    /// Shift to extract top bits from hash for bucket index.
177    shift: u32,
178    /// Per-instance config: seed, floor, load factor.
179    config: IndexConfig,
180    /// The flat slot array. Checkpoint-loaded indexes borrow the immutable
181    /// raw slots from their mmap until their first write, which clones them
182    /// into the owned atomic representation.
183    slots: SlotStorage,
184    /// The first 64 bits of each slot's hash. Index slots intentionally pack
185    /// only a tag, shard, and offset; retaining the home hash separately lets
186    /// a live writer grow the table without rescanning the packfiles.
187    homes: Mutex<Vec<u64>>,
188    /// The remaining 40 bits of each live slot's hash, plus a high-bit
189    /// marker that says the identity is known. Together with `homes` and the
190    /// packed tag this makes overwrite equality exact without putting
191    /// full hashes in the checkpoint.
192    tails: Mutex<Vec<u64>>,
193    /// `deserialize` restores slots without their source hashes. Such an
194    /// index remains readable, but must be rebuilt from packfiles if it ever
195    /// needs to grow.
196    can_grow: bool,
197    /// Number of occupied slots.
198    len: AtomicU32,
199    /// Longest linear-probe chain walked by any insert or lookup against
200    /// this index instance so far (never reset, monotonically
201    /// non-decreasing). Pure observability — insert has no probe-length
202    /// cap (see the doc on `InsertError::TableFull`: growth is what keeps
203    /// chains short, not a cap), so this exists to let an operator notice
204    /// a probe chain growing unexpectedly (e.g. an unseeded/misconfigured
205    /// deployment under adversarial content) without changing behavior.
206    max_probe_len: AtomicU32,
207}

209#[derive(Debug)]
210enum SlotStorage {
211    Owned(Vec<AtomicU64>),
212    Mmap { mmap: Arc<Mmap>, offset: usize },
213}
214
215impl SlotStorage {
216    #[inline]
217    fn get(&self, index: usize) -> u64 {
218        match self {
219            Self::Owned(slots) => slots[index].load(Ordering::Acquire),
220            Self::Mmap { mmap, offset } => {
221                let start = offset.saturating_add(index.saturating_mul(8));
222                let bytes: [u8; 8] = mmap[start..start.saturating_add(8)]
223                    .try_into()
224                    .expect("validated checkpoint slot range");
225                u64::from_le_bytes(bytes)
226            }
227        }
228    }
229
230    fn materialize(&self, capacity: usize) -> Vec<AtomicU64> {
231        (0..capacity)
232            .map(|index| AtomicU64::new(self.get(index)))
233            .collect()
234    }
235}

237impl Clone for LossyIndex {
238    fn clone(&self) -> Self {
239        Self {
240            capacity: self.capacity,
241            mask: self.mask,
242            shift: self.shift,
243            config: self.config,
244            slots: SlotStorage::Owned(self.slots.materialize(self.capacity as usize)),
245            // Checkpoint-backed indexes deliberately omit source hashes. On
246            // their first write, clone their slots into owned storage but
247            // keep an empty home table; `can_grow` remains false, so a later
248            // capacity boundary correctly falls back to a pack rebuild.
249            homes: Mutex::new(if self.is_mmap_backed() {
250                vec![0; self.capacity as usize]
251            } else {
252                self.homes.lock().clone()
253            }),
254            tails: Mutex::new(if self.is_mmap_backed() {
255                vec![0; self.capacity as usize]
256            } else {
257                self.tails.lock().clone()
258            }),
259            can_grow: self.can_grow,
260            len: AtomicU32::new(self.len.load(Ordering::Acquire)),
261            max_probe_len: AtomicU32::new(self.max_probe_len.load(Ordering::Relaxed)),
262        }
263    }
264}

288    /// Create a new index with at least `min_capacity` slots using the given config.
289    /// Capacity is rounded up to the next power of two. The configured `floor`
290    /// composes with the hard 16-slot minimum: `effective_min = floor.max(16)`.
291    ///
292    /// # Panics
293    /// Panics if `min_capacity` exceeds `u32::MAX` when rounded to a power of two,
294    /// or if `config` fails validation.
295    #[must_use]
296    pub fn with_config(min_capacity: usize, config: IndexConfig) -> Self {
297        config.validate().expect("invalid IndexConfig");
298        let effective_min = (config.floor.max(16)) as usize;
299        let capacity_usize = min_capacity.max(effective_min).next_power_of_two();
300        let capacity = u32::try_from(capacity_usize).expect("index capacity exceeds u32::MAX");
301        let shift = 64_u32.wrapping_sub(capacity.trailing_zeros());
302        Self {
303            mask: capacity.wrapping_sub(1),
304            capacity,
305            shift,
306            config,
307            slots: SlotStorage::Owned((0..capacity_usize).map(|_| AtomicU64::new(0)).collect()),
308            homes: Mutex::new(vec![0; capacity_usize]),
309            tails: Mutex::new(vec![0; capacity_usize]),
310            can_grow: true,
311            len: AtomicU32::new(0),
312            max_probe_len: AtomicU32::new(0),
313        }

340    /// Extract the bucket index from a 16-byte hash.
341    ///
342    /// The seed is `XORed` into the raw home value before the avalanche mix,
343    /// so the same content hash lands in different buckets across pools
344    /// with different seeds. The mixed value is what gets stored in `homes`,
345    /// so `grow()` (which rehashes from stored homes) sees the already-mixed
346    /// value and doesn't need to know the seed at all.
347    #[inline]
348    fn bucket(&self, hash: &[u8; 16]) -> usize {
349        let raw = u64::from_be_bytes(hash[..8].try_into().unwrap());
350        let home = mix(raw ^ self.config.seed);
351        self.bucket_for_home(home)
352    }
353
354    /// Mix a raw home value with this index's seed. Used by insert to store
355    /// the post-mix value in `homes`, and by `hydrate_slot_identity` to
356    /// reproduce the same home from a recovered hash.
357    #[inline]
358    fn mix_home(&self, raw_home: u64) -> u64 {
359        mix(raw_home ^ self.config.seed)
360    }
361
362    #[inline]
363    fn bucket_for_home(&self, home: u64) -> usize {
364        let masked = (home >> self.shift) & u64::from(self.mask);
365        // Safety: masked is always <= mask < capacity which fits in usize on all platforms
366        usize::try_from(masked).unwrap_or(usize::MAX)
367    }

369    /// Extract the 16-bit tag from a 16-byte hash, seeded with this index's seed.
370    ///
371    /// Uses bytes 8..12, disjoint from the bytes `bucket()` reads (0..8).
372    /// Seeding the tag prevents an attacker from grinding offline for
373    /// tag collisions against a known target bucket — without the seed,
374    /// a narrower 16-bit tag would make I/O amplification attacks feasible.
375    #[inline]
376    fn tag(&self, hash: &[u8; 16]) -> u32 {
377        let raw = u32::from_be_bytes(hash[8..12].try_into().unwrap());
378        let mixed = mix(u64::from(raw) ^ self.config.seed);
379        (mixed >> 48) as u32
380    }

382    /// The packed-slot tag for `hash`, exposed to crate-local recovery code
383    /// that must validate a persisted slot against its authoritative frame.
384    ///
385    /// Must use this index's own seed: every stored slot's tag was computed
386    /// via `self.tag(hash)` at insert time, so a recovery comparison against
387    /// an unseeded (or differently-seeded) tag would fail closed on every
388    /// occupied slot, not just colliding ones.
389    #[inline]
390    pub(crate) fn tag_for_hash(&self, hash: &[u8; 16]) -> u32 {
391        self.tag(hash)
392    }

394    #[inline]
395    fn tail(hash: &[u8; 16]) -> u64 {
396        const KNOWN: u64 = 1_u64 << 63;
397        let mut bytes = [0; 8];
398        bytes[3..].copy_from_slice(&hash[11..]);
399        KNOWN | u64::from_be_bytes(bytes)
400    }

411    /// Like [`Self::insert`], but also reports the write's exact landing spot —
412    /// the affected bucket and the packed slot value stored there — so the
413    /// delta persistence layer can append a replay frame without re-probing.
414    ///
415    /// # Errors
416    /// Returns `InsertError::TableFull` under the same conditions as
417    /// [`Self::insert`] (the table is left unmodified).
418    // `bucket` is always below `self.capacity`, which is capped at `u32::MAX`
419    // by construction, so the checked `u32::try_from` in `insert_undoable`
420    // below can never fail.
421    pub fn insert_tracked(
422        &self,
423        hash: &[u8; 16],
424        shard_id: u16,
425        offset: u64,
426    ) -> Result<(u32, u64), InsertError> {
427        self.insert_undoable(hash, shard_id, offset)
428            .map(|(bucket, value, _undo)| (bucket, value))

431    /// Like [`Self::insert_tracked`], but also returns a [`SlotUndo`]
432    /// capturing the slot's prior contents, so a caller that needs to
433    /// mutate the live (not cloned) index for a multi-record batch can
434    /// undo this one write if a later entry in the batch fails.
435    ///
436    /// # Errors
437    /// Returns the same errors as [`Self::insert_tracked`] and leaves the
438    /// table unmodified in every error case.
439    ///
440    /// # Panics
441    /// Never panics in practice: `hash[..8]` is always exactly 8 bytes for a
442    /// `&[u8; 16]` input, so the `try_into` this performs cannot fail.
443    pub fn insert_undoable(
444        &self,
445        hash: &[u8; 16],
446        shard_id: u16,
447        offset: u64,
448    ) -> Result<(u32, u64, SlotUndo), InsertError> {
449        let tag = self.tag(hash);
450        let raw_home = u64::from_be_bytes(hash[..8].try_into().unwrap());
451        let home = self.mix_home(raw_home);
452        let mut bucket = self.bucket_for_home(home);
453        let mut probe_len: u32 = 0;
454
455        loop {
456            // `bucket` is masked by `self.mask` (capacity - 1) and capacity is
457            // a u32, so the conversion is lossless; it is done once here so the
458            // return/undo structs below don't each need a checked cast.
459            let bucket_wire = u32::try_from(bucket)
460                .expect("bucket is masked by the u32 capacity, so it always fits");
461            let SlotStorage::Owned(slots) = &self.slots else {
462                self.bump_max_probe_len(probe_len);
463                return Err(InsertError::TableFull);
464            };
465            let slot = IndexSlot(slots[bucket].load(Ordering::Acquire));
466            if slot.is_empty() {
467                // Only reject when actually inserting into a new slot
468                let threshold = self
469                    .capacity
470                    .wrapping_mul(u32::from(self.config.load_factor_percent))
471                    / 100;
472                if self.len.load(Ordering::Relaxed) >= threshold {
473                    self.bump_max_probe_len(probe_len);
474                    return Err(InsertError::TableFull);
475                }
476                let old_home = self.homes.lock()[bucket];
477                let old_tail = self.tails.lock()[bucket];
478                // Writers are serialized by the collection put lock. Publish
479                // the slot last so concurrent readers see either the prior
480                // empty slot or a complete record location.
481                self.homes.lock()[bucket] = home;
482                self.tails.lock()[bucket] = Self::tail(hash);
483                let value = IndexSlot::new(tag, shard_id, offset).0;
484                slots[bucket].store(value, Ordering::Release);
485                self.len.fetch_add(1, Ordering::Relaxed);
486                let undo = SlotUndo {
487                    bucket: bucket_wire,
488                    old_slot: 0,
489                    old_home,
490                    old_tail,
491                    was_empty: true,
492                };
493                self.bump_max_probe_len(probe_len);
494                return Ok((bucket_wire, value, undo));
495            }
496            // A tag only filters candidates; it never proves key equality.
497            // Checkpoint-derived slots initially lack their 40-bit tail and
498            // ask storage to hydrate it from the frame before deciding.
499            if slot.tag() == tag {
500                let tail = self.tails.lock()[bucket];
501                if tail == 0 {
502                    self.bump_max_probe_len(probe_len);
503                    return Err(InsertError::NeedsIdentity {
504                        bucket: bucket_wire,
505                        shard_id: slot.shard_id(),
506                        offset: slot.offset(),
507                    });
508                }
509                if self.homes.lock()[bucket] == home && tail == Self::tail(hash) {
510                    let old_slot = slot.0;
511                    let old_home = self.homes.lock()[bucket];
512                    let value = IndexSlot::new(tag, shard_id, offset).0;
513                    slots[bucket].store(value, Ordering::Release);
514                    let undo = SlotUndo {
515                        bucket: bucket_wire,
516                        old_slot,
517                        old_home,
518                        old_tail: tail,
519                        was_empty: false,
520                    };
521                    self.bump_max_probe_len(probe_len);
522                    return Ok((bucket_wire, value, undo));
523                }
524            }
525            probe_len = probe_len.saturating_add(1);
526            bucket = bucket.wrapping_add(1) & self.mask as usize;
527        }
528    }

530    /// Reverse a write reported by [`Self::insert_undoable`]. Callers must
531    /// replay a batch's undo entries in the reverse order they were
532    /// produced, and only against the same live index that produced them
533    /// (never a cloned/owned index materialized after the fact).
534    ///
535    /// # Panics
536    /// Never panics; a no-op if called against a checkpoint-mmap-backed
537    /// index, since [`Self::insert_undoable`] cannot have produced an undo
538    /// entry against one (it errors with `TableFull` first).
539    pub fn rollback_slot(&self, undo: &SlotUndo) {
540        let SlotStorage::Owned(slots) = &self.slots else {
541            return;
542        };
543        let bucket = undo.bucket as usize;
544        self.homes.lock()[bucket] = undo.old_home;
545        self.tails.lock()[bucket] = undo.old_tail;
546        slots[bucket].store(undo.old_slot, Ordering::Release);
547        if undo.was_empty {
548            self.len.fetch_sub(1, Ordering::Relaxed);
549        }
550    }

552    /// Mark a checkpoint-derived slot with the full identity recovered from
553    /// its authoritative frame. Returns `false` if the slot changed or the
554    /// supplied hash cannot describe that slot.
555    pub(crate) fn hydrate_slot_identity(&self, bucket: u32, hash: &[u8; 16]) -> bool {
556        let bucket = bucket as usize;
557        if bucket >= self.capacity as usize {
558            return false;
559        }
560        let slot = IndexSlot(self.slot_at(bucket));
561        if slot.is_empty() || slot.tag() != self.tag(hash) {
562            return false;
563        }
564        let raw_home = u64::from_be_bytes(hash[..8].try_into().expect("16-byte hash"));
565        let home = self.mix_home(raw_home);
566        self.homes.lock()[bucket] = home;
567        self.tails.lock()[bucket] = Self::tail(hash);
568        true
569    }

571    /// Apply recorded delta frames on top of a checkpoint-backed index,
572    /// replicating the live insert/overwrite sequence that produced them.
573    ///
574    /// Only valid on an owned (materialized) index — callers must clone an
575    /// mmap-backed index first. Frames are applied exactly as the live
576    /// session wrote them: an overwrite of a non-empty slot leaves the length
577    /// unchanged, a write into an empty slot increments it. The index's `len`
578    /// after replay therefore equals the checkpoint's occupancy plus the
579    /// number of fresh-slot writes, which is what the sidecar integrity sum
580    /// checks against.
581    ///
582    /// # Errors
583    /// Returns `DeltaReplayError` if a frame targets a bucket outside this
584    /// index's capacity, or carries the empty-slot sentinel (either is a
585    /// structurally inconsistent log, rejected wholesale), or the index is
586    /// still mmap-backed.
587    pub fn replay_frames(&self, frames: &[DeltaFrame]) -> Result<(), DeltaReplayError> {
588        let SlotStorage::Owned(slots) = &self.slots else {
589            return Err(DeltaReplayError::RequiresOwnedIndex);
590        };
591        // Validate every frame before mutating any slot: the contract is
592        // wholesale rejection of a structurally inconsistent log, so a
593        // caller must never observe a partially replayed index.
594        for frame in frames {
595            let bucket = frame.bucket as usize;
596            if bucket >= self.capacity as usize {
597                return Err(DeltaReplayError::FrameOutOfBounds {
598                    bucket: frame.bucket,
599                    capacity: self.capacity,
600                });
601            }
602            if frame.slot == 0 {
603                // A legitimate insert never logs the empty sentinel (see
604                // `record_delta`'s only caller). Storing it here would erase
605                // whatever live entry currently occupies this bucket and cut
606                // the probe chain short for anything beyond it — reject the
607                // whole log instead of silently losing data.
608                return Err(DeltaReplayError::EmptySlot {
609                    bucket: frame.bucket,
610                });
611            }
612        }
613        for frame in frames {
614            let bucket = frame.bucket as usize;
615            let previous = slots[bucket].load(Ordering::Acquire);
616            if previous == 0 {
617                self.len.fetch_add(1, Ordering::Relaxed);
618            }
619            slots[bucket].store(frame.slot, Ordering::Release);
620        }
621        Ok(())
622    }

624    /// Double this live index's capacity without reading packfiles.
625    ///
626    /// Returns `false` for an index deserialized from the compact on-disk
627    /// representation, whose original hashes are unavailable for rehashing.
628    /// Callers should retain their existing packfile rebuild fallback in that
629    /// case.
630    pub fn grow(&self) -> Option<Self> {
631        if !self.can_grow || self.capacity > u32::MAX / 2 {
632            return None;
633        }
634        let grown = Self::with_config(self.capacity as usize * 2, self.config);
635        let homes = self.homes.lock();
636        let tails = self.tails.lock();
637        for (index, home) in homes.iter().copied().enumerate() {
638            let slot = IndexSlot(self.slots.get(index));
639            if slot.is_empty() {
640                continue;
641            }
642            let mut bucket = grown.bucket_for_home(home);
643            while !IndexSlot(grown.slot_at(bucket)).is_empty() {
644                bucket = bucket.wrapping_add(1) & grown.mask as usize;
645            }
646            grown.homes.lock()[bucket] = home;
647            grown.tails.lock()[bucket] = tails[index];
648            let SlotStorage::Owned(slots) = &grown.slots else {
649                unreachable!("new index is owned")
650            };
651            slots[bucket].store(slot.0, Ordering::Relaxed);
652            grown.len.fetch_add(1, Ordering::Relaxed);
653        }
654        Some(grown)
655    }

657    /// Double the table by recovering each occupied entry's full hash from
658    /// its stored location.
659    ///
660    /// Checkpoint-backed indexes deliberately omit `homes`, so their first
661    /// post-restart resize cannot use [`Self::grow`].  The packed slots still
662    /// retain every `(shard_id, offset)`, however.  A caller can therefore
663    /// recover only the `len` source hashes from the authoritative records
664    /// and rehash without rescanning every pack in the store. Locations are
665    /// visited in `(shard, offset)` order rather than hash-table order, so a
666    /// cold mmap walk follows pack order and lets the kernel coalesce page
667    /// faults/read-ahead. The callback must return the hash for precisely the
668    /// supplied location.
669    ///
670    /// `Ok(None)` means the table cannot grow further; callers retain their
671    /// full-rebuild fallback for that terminal case.
672    pub(crate) fn grow_by_recovering_hashes<E>(
673        &self,
674        mut hash_at: impl FnMut(u16, u64, u32) -> Result<[u8; 16], E>,
675    ) -> Result<Option<Self>, E> {
676        if self.capacity > u32::MAX / 2 {
677            return Ok(None);
678        }
679
680        let mut locations = Vec::with_capacity(self.len());
681        for index in 0..self.capacity as usize {
682            let slot = IndexSlot(self.slot_at(index));
683            if slot.is_empty() {
684                continue;
685            }
686            locations.push((slot.shard_id(), slot.offset(), slot.tag()));
687        }
688        locations.sort_unstable_by_key(|(shard_id, offset, _)| (*shard_id, *offset));
689
690        let grown = Self::with_config(self.capacity as usize * 2, self.config);
691        for (shard_id, offset, slot_tag) in locations {
692            let hash = hash_at(shard_id, offset, slot_tag)?;
693            // A doubled table is at most 37.5% full because insertion only
694            // requests growth at 75%, so this cannot hit TableFull.
695            let _ = grown.insert(&hash, shard_id, offset);
696        }
697        Ok(Some(grown))
698    }

712    /// Look up all candidate offsets for a hash, yielding tag collisions.
713    ///
714    /// Candidates with hydrated in-memory identity (home + tail) are verified
715    /// before yielding, so tag collisions in a live/warm index are filtered
716    /// out here rather than forcing the caller to disk. For a slot whose
717    /// identity is not yet hydrated (e.g. a fresh checkpoint restore), the
718    /// caller **must** verify the candidate against the caller-requested
719    /// hash — a raw tag collision (~1/65536 at 16-bit tags) can still surface
720    /// unverified.
721    ///
722    /// Yields `(shard_id, offset)` for each slot whose tag matches and isn't
723    /// ruled out, then terminates at the first empty slot or after
724    /// `capacity` probes.
725    ///
726    /// # Panics
727    /// Never panics in practice: `hash[..8]` is always exactly 8 bytes for a
728    /// `&[u8; 16]` input, so the `try_into` this performs cannot fail.
729    #[inline]
730    #[must_use]
731    pub fn lookup_all(&self, hash: &[u8; 16]) -> LookupIter<'_> {
732        let tag = self.tag(hash);
733        let bucket = self.bucket(hash);
734        let raw_home = u64::from_be_bytes(hash[..8].try_into().unwrap());
735        let mixed_home = self.mix_home(raw_home);
736        let expected_tail = Self::tail(hash);
737        LookupIter {
738            index: self,
739            tag,
740            mixed_home,
741            expected_tail,
742            bucket,
743            mask: self.mask as usize,
744            remaining: self.capacity as usize,
745            steps: 0,
746        }
747    }

854    /// Rewrites every occupied slot's `shard_id` through `remap`, in place.
855    ///
856    /// A checkpoint-loaded index's `shard_id`s were encoded by the writer's
857    /// own local `ShardPool` slots at checkpoint-write time — a process-local
858    /// handle, not a stable identity. `remap` (built by the caller from the
859    /// checkpoint's persisted pack table, translated through this reader's
860    /// own currently-open packs) reconciles that against this reader's own
861    /// slot numbering, which can differ once any shard has ever been retired
862    /// (see `CHECKPOINT_VERSION`'s v6 doc comment).
863    ///
864    /// Checks first without touching storage: the common case (no shard has
865    /// ever been retired, so a fresh reader's slot numbering already agrees
866    /// with the checkpoint's) needs no rewrite at all, and mmap-backed
867    /// indexes must stay mmap-backed when nothing actually changes — callers
868    /// and tests depend on an unmodified checkpoint-loaded index staying
869    /// zero-copy until its first real write. Only materializes into an owned
870    /// copy (writes are never applied to the mmap) when at least one
871    /// occupied slot's `shard_id` actually needs to change.
872    ///
873    /// Returns `false`, leaving the index unmodified, if any occupied slot's
874    /// `shard_id` has no entry in `remap` — the checkpoint's pack table
875    /// should cover every `shard_id` any of its own index entries use, so a
876    /// miss means the checkpoint is internally inconsistent and the caller
877    /// should fall back to a full rescan rather than serve unresolvable
878    /// slots.
879    #[must_use]
880    pub fn remap_shard_ids(&mut self, remap: &std::collections::HashMap<u16, u16>) -> bool {
881        let capacity = self.capacity as usize;
882        let mut needs_rewrite = false;
883        for index in 0..capacity {
884            let slot = IndexSlot(self.slots.get(index));
885            if slot.is_empty() {
886                continue;
887            }
888            let Some(&new_shard_id) = remap.get(&slot.shard_id()) else {
889                return false;
890            };
891            if new_shard_id != slot.shard_id() {
892                needs_rewrite = true;
893            }
894        }
895        if !needs_rewrite {
896            return true;
897        }
898        if !matches!(self.slots, SlotStorage::Owned(_)) {
899            self.slots = SlotStorage::Owned(self.slots.materialize(capacity));
900        }
901        let SlotStorage::Owned(slots) = &self.slots else {
902            unreachable!("just materialized to Owned above");
903        };
904        for cell in slots {
905            let raw = cell.load(Ordering::Acquire);
906            let slot = IndexSlot(raw);
907            if slot.is_empty() {
908                continue;
909            }
910            let Some(&new_shard_id) = remap.get(&slot.shard_id()) else {
911                return false;
912            };
913            if new_shard_id == slot.shard_id() {
914                continue;
915            }
916            let remapped = IndexSlot::new(slot.tag(), new_shard_id, slot.offset()).0;
917            cell.store(remapped, Ordering::Relaxed);
918        }
919        true
920    }

922    /// Serialize the index to bytes for persistence.
923    ///
924    /// Format (all little-endian):
925    /// ```text
926    ///   [capacity: u64]
927    ///   [slot_0: u64] ... [slot_{cap-1}: u64]     ← capacity × 8 B
928    ///   [home_0: u64] ... [home_{cap-1}: u64]     ← capacity × 8 B (0 if unhydrated)
929    ///   [tail_0: u64] ... [tail_{cap-1}: u64]     ← capacity × 8 B (0 if unhydrated)
930    /// ```
931    /// Total: `8 + capacity * 24` bytes.
932    #[must_use]
933    pub fn serialize(&self) -> Vec<u8> {
934        let cap = self.capacity as usize;
935        let byte_len = 8_usize.wrapping_add(cap.wrapping_mul(24));
936        let mut buf = Vec::with_capacity(byte_len);
937        buf.extend_from_slice(&u64::from(self.capacity).to_le_bytes());
938        for index in 0..cap {
939            let slot = IndexSlot(self.slot_at(index));
940            buf.extend_from_slice(&slot.0.to_le_bytes());
941        }
942        let homes = self.homes.lock();
943        let tails = self.tails.lock();
944        for index in 0..cap {
945            buf.extend_from_slice(&homes[index].to_le_bytes());
946        }
947        for index in 0..cap {
948            buf.extend_from_slice(&tails[index].to_le_bytes());
949        }
950        buf
951    }

966    /// Deserialize an index from bytes with a given config (providing seed).
967    ///
968    /// Accepts both the legacy slots-only format (`8 + cap*8` bytes) and the
969    /// v4 slots+homes+tails format (`8 + cap*24` bytes). Homes and tails are
970    /// populated from the blob when present; zeroed otherwise.
971    ///
972    /// # Errors
973    /// Returns `DeserializationError::TooShort` if the data is too short,
974    /// or `DeserializationError::InvalidCapacity` if the capacity is not
975    /// a power of two or is less than 16.
976    ///
977    /// # Panics
978    /// Panics if the 8-byte capacity header cannot be read (guaranteed by the length check).
979    pub fn deserialize_with_config(
980        data: &[u8],
981        config: IndexConfig,
982    ) -> Result<Self, DeserializationError> {
983        if data.len() < 8 {
984            return Err(DeserializationError::TooShort);
985        }
986        let capacity_wire = u64::from_le_bytes(data[..8].try_into().unwrap());
987        let capacity = u32::try_from(capacity_wire).map_err(|_| DeserializationError::TooShort)?;
988        let capacity_usize = capacity as usize;
989
990        // Reject unsupported capacities: must be power of two and >= 16
991        if capacity < 16 || !capacity.is_power_of_two() {
992            return Err(DeserializationError::InvalidCapacity);
993        }
994
995        let slots_len = 8_usize.wrapping_add(capacity_usize.wrapping_mul(8));
996        let full_len = 8_usize.wrapping_add(capacity_usize.wrapping_mul(24));
997        if data.len() < slots_len {
998            return Err(DeserializationError::TooShort);
999        }
1000        let has_homes_tails = data.len() >= full_len;
1001
1002        let mut slots = Vec::with_capacity(capacity_usize);
1003        let mut len: u32 = 0;
1004        for i in 0..capacity_usize {
1005            let offset = 8_usize.wrapping_add(i.wrapping_mul(8));
1006            let val = u64::from_le_bytes(data[offset..offset.wrapping_add(8)].try_into().unwrap());
1007            let slot = IndexSlot(val);
1008            if !slot.is_empty() {
1009                len = len.wrapping_add(1);
1010            }
1011            slots.push(AtomicU64::new(slot.0));
1012        }
1013
1014        let shift = 64_u32.wrapping_sub(capacity.trailing_zeros());
1015
1016        let homes = if has_homes_tails {
1017            let base = slots_len;
1018            let mut h = Vec::with_capacity(capacity_usize);
1019            for i in 0..capacity_usize {
1020                let offset = base.wrapping_add(i.wrapping_mul(8));
1021                h.push(u64::from_le_bytes(
1022                    data[offset..offset.wrapping_add(8)].try_into().unwrap(),
1023                ));
1024            }
1025            h
1026        } else {
1027            vec![0; capacity_usize]
1028        };
1029
1030        let tails = if has_homes_tails {
1031            let base = slots_len.wrapping_add(capacity_usize.wrapping_mul(8));
1032            let mut t = Vec::with_capacity(capacity_usize);
1033            for i in 0..capacity_usize {
1034                let offset = base.wrapping_add(i.wrapping_mul(8));
1035                t.push(u64::from_le_bytes(
1036                    data[offset..offset.wrapping_add(8)].try_into().unwrap(),
1037                ));
1038            }
1039            t
1040        } else {
1041            vec![0; capacity_usize]
1042        };
1043
1044        Ok(Self {
1045            mask: capacity.wrapping_sub(1),
1046            capacity,
1047            shift,
1048            config,
1049            slots: SlotStorage::Owned(slots),
1050            homes: Mutex::new(homes),
1051            tails: Mutex::new(tails),
1052            can_grow: false,
1053            len: AtomicU32::new(len),
1054            max_probe_len: AtomicU32::new(0),
1055        })
1056    }

1220/// Iterator over candidate offsets for a hash lookup.
1221///
1222/// Yields `(shard_id, offset)` for each slot whose tag matches and whose
1223/// in-memory identity (home + tail) matches the query hash, terminating at
1224/// the first empty slot or after the full capacity is probed.
1225///
1226/// When the index has hydrated identity (live/warm index), candidates are
1227/// verified in-memory before yielding — this eliminates disk I/O for tag
1228/// collisions on the normal path. For checkpoint-backed indexes without
1229/// hydrated identity (tail == 0), all tag-matching candidates are yielded
1230/// and the caller must verify against the packfile.
1231pub struct LookupIter<'a> {
1232    index: &'a LossyIndex,
1233    tag: u32,
1234    /// The mixed home for the query hash, for in-memory verification.
1235    mixed_home: u64,
1236    /// The expected tail for the query hash.
1237    expected_tail: u64,
1238    bucket: usize,
1239    mask: usize,
1240    remaining: usize,
1241    /// Steps walked so far, for `LossyIndex::max_probe_len` observability.
1242    steps: u32,
1243}
1244
1245impl Iterator for LookupIter<'_> {
1246    type Item = (u16, u64);
1247
1248    #[inline]
1249    fn next(&mut self) -> Option<Self::Item> {
1250        while self.remaining > 0 {
1251            self.remaining = self.remaining.wrapping_sub(1);
1252            let slot = IndexSlot(self.index.slot_at(self.bucket));
1253            if slot.is_empty() {
1254                self.index.bump_max_probe_len(self.steps);
1255                return None;
1256            }
1257            self.steps = self.steps.saturating_add(1);
1258            let current = self.bucket;
1259            self.bucket = self.bucket.wrapping_add(1) & self.mask;
1260            if slot.tag() != self.tag {
1261                continue;
1262            }
1263            // Tag matches. For a live index with hydrated identity, verify
1264            // home + tail in-memory to avoid yielding false positives that
1265            // would force the caller into an expensive packfile read.
1266            let tails = self.index.tails.lock();
1267            let tail = if tails.len() > current {
1268                tails[current]
1269            } else {
1270                0
1271            };
1272            drop(tails);
1273            if tail != 0 {
1274                // Identity is hydrated — check home and tail.
1275                let homes = self.index.homes.lock();
1276                let home = if homes.len() > current {
1277                    homes[current]
1278                } else {
1279                    0
1280                };
1281                drop(homes);
1282                if home == self.mixed_home && tail == self.expected_tail {
1283                    self.index.bump_max_probe_len(self.steps);
1284                    return Some((slot.shard_id(), slot.offset()));
1285                }
1286                // Tag matched but home/tail didn't — false positive, skip.
1287                continue;
1288            }
1289            // Checkpoint-backed without hydrated identity — yield and let
1290            // the caller verify against the packfile.
1291            self.index.bump_max_probe_len(self.steps);
1292            return Some((slot.shard_id(), slot.offset()));
1293        }
1294        self.index.bump_max_probe_len(self.steps);
1295        None
1296    }
1297}
