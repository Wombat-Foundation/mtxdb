37/// A durable mutation represented in a journal group.
38#[derive(Clone, Debug, PartialEq, Eq)]
39pub enum Mutation {
40    /// Insert a content-addressed record.
41    Put {
42        /// Collection containing the record.

65pub enum TxnStageState {

83/// Transaction-local **journal** publication buffer -- not a storage
84/// transaction.
85///
86/// Journal entries for a SQL transaction are buffered here and published from
87/// its post-commit callback. Call [`Self::discard`] from the transaction's
88/// error callback; that is required because Synapse retains after-callbacks
89/// across retry attempts.
90///
91/// # Not a rollback
92///
93/// [`Self::discard`] drops only the buffered journal mutations. It does not
94/// touch packs or the live index, because this buffer never writes them; the
95/// deferred-publication redesign (prepare/commit) will route the pack/index
96/// mutation through this buffer's commit path. Until that lands, no
97/// production caller stages through here.
98pub struct TxnStage {
99    state: std::sync::atomic::AtomicU8,
100    data: Mutex<TxnStageData>,
101}

114    /// Create an empty stage for one transaction attempt.

127    /// Current lifecycle state.
128    #[must_use]
129    pub fn state(&self) -> TxnStageState {
130        match self.state.load(Ordering::Acquire) {
131            Self::DISCARDED => TxnStageState::Discarded,
132            Self::PUBLISHED => TxnStageState::Published,
133            _ => TxnStageState::Active,
134        }

137    /// Discard an active attempt. Safe to call more than once.
138    ///
139    /// See the "Not a rollback" section on [`TxnStage`]: this drops only the
140    /// buffered journal mutations and never touches packs or the live index.
141    pub fn discard(&self) {
142        let mut data = self.data.lock();
143        if self.state.load(Ordering::Acquire) == Self::ACTIVE {
144            data.pools.iter_mut().for_each(Vec::clear);
145            data.bytes = 0;
146            self.state.store(Self::DISCARDED, Ordering::Release);
147        }
148    }

150    /// Ensure an estimated batch fits before buffering its journal mutations.
151    ///
152    /// # Errors
153    /// Returns `InvalidInput` if the stage is not active or the estimated
154    /// total exceeds [`MAX_TXN_STAGE_BYTES`].
155    pub fn ensure_capacity(&self, additional: usize) -> io::Result<()> {
156        if self.state.load(Ordering::Acquire) != Self::ACTIVE {
157            return Err(io::Error::other("transaction stage is not active"));
158        }
159        let data = self.data.lock();
160        let total = data.bytes.checked_add(additional).ok_or_else(|| {
161            io::Error::new(
162                io::ErrorKind::InvalidInput,
163                "transaction stage size overflow",
164            )
165        })?;
166        if total > MAX_TXN_STAGE_BYTES {
167            return Err(io::Error::new(
168                io::ErrorKind::InvalidInput,
169                "transaction journal stage exceeds its 64 MiB limit",
170            ));
171        }
172        Ok(())
173    }

175    /// Add an immutable put to this attempt's pool-ordered journal batch.
176    ///
177    /// # Errors
178    /// Returns `InvalidInput` if adding the payload would exceed the stage
179    /// limit, or another error if the stage has already been discarded or
180    /// published.
181    pub fn stage_put(
182        &self,
183        pool: ShardType,
184        collection_id: [u8; 16],
185        node_id: [u8; 16],
186        payload: Vec<u8>,
187    ) -> io::Result<()> {
188        let charge = payload.len().saturating_add(64);
189        let mut data = self.data.lock();
190        if self.state.load(Ordering::Acquire) != Self::ACTIVE {
191            return Err(io::Error::other("transaction stage is not active"));
192        }
193        let total = data.bytes.checked_add(charge).ok_or_else(|| {
194            io::Error::new(
195                io::ErrorKind::InvalidInput,
196                "transaction stage size overflow",
197            )
198        })?;
199        if total > MAX_TXN_STAGE_BYTES {
200            return Err(io::Error::new(
201                io::ErrorKind::InvalidInput,
202                "transaction journal stage exceeds its 64 MiB limit",
203            ));
204        }
205        data.pools[pool_index(pool)].push(Mutation::Put {
206            collection_id,
207            node_id,
208            payload,
209        });
210        data.bytes = total;
211        Ok(())
212    }

214    /// Add a batch of immutable puts atomically.
215    ///
216    /// # Errors
217    /// Returns `InvalidInput` if the batch would exceed the stage limit, or
218    /// another error if the stage has already been discarded or published.
219    pub fn stage_puts(
220        &self,
221        pool: ShardType,
222        collection_id: [u8; 16],
223        entries: &[([u8; 16], Vec<u8>)],
224    ) -> io::Result<()> {
225        let charge = entries
226            .iter()
227            .try_fold(0usize, |total, (_, payload)| {
228                total.checked_add(payload.len().saturating_add(64))
229            })
230            .ok_or_else(|| {
231                io::Error::new(
232                    io::ErrorKind::InvalidInput,
233                    "transaction stage size overflow",
234                )
235            })?;
236        let mut data = self.data.lock();
237        if self.state.load(Ordering::Acquire) != Self::ACTIVE {
238            return Err(io::Error::other("transaction stage is not active"));
239        }
240        let total = data.bytes.checked_add(charge).ok_or_else(|| {
241            io::Error::new(
242                io::ErrorKind::InvalidInput,
243                "transaction stage size overflow",
244            )
245        })?;
246        if total > MAX_TXN_STAGE_BYTES {
247            return Err(io::Error::new(
248                io::ErrorKind::InvalidInput,
249                "transaction journal stage exceeds its 64 MiB limit",
250            ));
251        }
252        let pool_mutations = &mut data.pools[pool_index(pool)];
253        pool_mutations.extend(entries.iter().map(|(node_id, payload)| Mutation::Put {
254            collection_id,
255            node_id: *node_id,
256            payload: payload.clone(),
257        }));
258        data.bytes = total;
259        Ok(())
260    }

295    /// Append staged pool groups in dependency order: auth-chain, event-DAG,
296    /// then state. Does not fsync; the ordinary coalesced sync remains the
297    /// durability boundary. Repeated calls are safe, including after a
298    /// partial error.
299    ///
300    /// # Errors
301    /// Returns an error if a staged pool has no coordinator or its journal
302    /// cannot append the group's complete framing and trailer.
303    pub fn publish(
304        &self,
305        auth_chain: Option<&JournalCoordinator>,
306        event_dag: Option<&JournalCoordinator>,
307        state: Option<&JournalCoordinator>,
308    ) -> io::Result<()> {
309        let mut data = self.data.lock();
310        match self.state.load(Ordering::Acquire) {
311            Self::DISCARDED | Self::PUBLISHED => return Ok(()),
312            _ => {}
313        }
314        let coordinators = [auth_chain, event_dag, state];
315        if coordinators.iter().all(Option::is_none) {
316            // Journaling is disabled process-wide, so there is no journal to
317            // publish into.
318            self.state.store(Self::PUBLISHED, Ordering::Release);
319            return Ok(());
320        }
321        let pools = [ShardType::AuthChain, ShardType::EventDag, ShardType::State];
322        for (ordered_index, pool) in pools.into_iter().enumerate() {
323            let index = pool_index(pool);
324            if data.appended[index] || data.pools[index].is_empty() {
325                data.appended[index] = true;
326                continue;
327            }
328            let coordinator = coordinators[ordered_index].ok_or_else(|| {
329                io::Error::new(io::ErrorKind::NotConnected, "staged pool has no journal")
330            })?;
331            coordinator.append_pending(&data.pools[index])?;
332            data.appended[index] = true;
333        }
334        self.state.store(Self::PUBLISHED, Ordering::Release);
335        Ok(())
336    }
337}

343        ShardType::AuthChain => 2,
344    }

441/// Serializes mutation publication and durable commits for one journal.
442///
443/// Mutations are assigned LSNs under a short queue lock. A sync caller captures
444/// the published LSN, detaches the covered mutations, then commits them while
445/// holding the journal lock. Publishers can queue later mutations during the
446/// fsync, and concurrent sync callers recheck the committed LSN after taking
447/// the journal lock.
448pub struct JournalCoordinator {
449    journal: Mutex<Journal>,
450    pending: Mutex<Vec<(u64, Mutation)>>,
451    /// Next LSN to assign. Advanced under `pending`, independently of journal
452    /// I/O, so a publish never blocks behind a sync's fsync.
453    next_lsn: AtomicU64,
454    published_lsn: AtomicU64,
455    /// Highest LSN whose group is complete (trailer appended) but not
456    /// necessarily fsynced. Advanced between [`Journal::append_group`] and
457    /// [`Journal::make_durable`] so a read-only overlay can observe a
458    /// committed-but-unflushed group.
459    visible_lsn: AtomicU64,
460    committed_lsn: AtomicU64,
461    /// Optional shared, cross-pool group-sequence allocator. When present,
462    /// each committed group draws its sequence from here instead of the
463    /// segment's own counter, so groups across several pool segments share one
464    /// global order. See [`Self::with_shared_sequence`].
465    sequence: Option<Arc<AtomicU64>>,
466    /// Mirrors the journal's poison bit, so `publish` can reject without
467    /// taking the `journal` mutex (which a sync holds across its fsync).
468    poisoned: AtomicBool,
469}

476        let next_lsn = journal.next_lsn;

489    /// Build a coordinator whose groups draw their sequence from a shared,
490    /// cross-pool counter instead of this segment's own numbering.
491    ///
492    /// The caller must initialize `sequence` above every participating

527    /// Assign an LSN, publish the mutation to the caller's live overlay, and
528    /// queue it for the next covering durable group.
529    ///
530    /// The callback runs while publication is serialized and before the LSN
531    /// becomes visible to sync callers. It must not call back into this
532    /// coordinator. Keeping overlay publication in this critical section
533    /// prevents a successful sync from racing ahead of a not-yet-visible put.
534    ///
535    /// # Errors
536    /// Returns an error if the journal has been poisoned by a failed commit or
537    /// if its LSN space is exhausted.
538    pub fn publish(
539        &self,
540        mutation: Mutation,
541        publish_overlay: impl FnOnce(u64),
542    ) -> io::Result<u64> {
543        if self.poisoned.load(Ordering::Acquire) {
544            return Err(io::Error::other(
545                "journal is poisoned after a failed commit",
546            ));
547        }
548        let mut pending = self.pending.lock();
549        let lsn = self.next_lsn.load(Ordering::Relaxed);
550        let next_lsn = lsn
551            .checked_add(1)
552            .ok_or_else(|| io::Error::other("journal LSN exhausted"))?;
553        publish_overlay(lsn);
554        pending.push((lsn, mutation));
555        self.next_lsn.store(next_lsn, Ordering::Relaxed);
556        self.published_lsn.store(lsn, Ordering::Release);
557        Ok(lsn)
558    }

567    /// Durably commit pending mutations through `target_lsn`.
568    ///
569    /// Calls with the same or an earlier target are covered by an already
570    /// completed group. Later published mutations are left for a later group.
571    /// The journal mutex serializes the disk commit; concurrent sync callers
572    /// wait for it and then recheck the committed LSN before deciding whether
573    /// to write.
574    ///
575    /// # Errors
576    /// Returns an error if the target was never published, the journal is
577    /// poisoned, or writing/syncing the covering group fails.
578    pub fn sync_through(&self, target_lsn: u64) -> io::Result<Option<CommitReceipt>> {
579        if target_lsn == 0 {
580            return Ok(None);
581        }
582        if target_lsn > self.published_lsn.load(Ordering::Acquire) {
583            return Err(io::Error::new(
584                io::ErrorKind::InvalidInput,
585                "sync target has not been published",
586            ));
587        }
588        if target_lsn <= self.committed_lsn.load(Ordering::Acquire) {
589            return Ok(None);
590        }
591        if self.poisoned.load(Ordering::Acquire) {
592            return Err(io::Error::other(
593                "journal is poisoned after a failed commit",
594            ));
595        }
596
597        let mut journal = self.journal.lock();
598        if self.poisoned.load(Ordering::Acquire) {
599            return Err(io::Error::other(
600                "journal is poisoned after a failed commit",
601            ));
602        }
603        if target_lsn <= self.committed_lsn.load(Ordering::Acquire) {
604            return Ok(None);
605        }
606        // Transaction-staged groups are already complete in the segment, so
607        // the coalesced sync only needs to make their bytes durable.
608        if target_lsn <= self.visible_lsn.load(Ordering::Acquire) {
609            if let Err(error) = journal.make_durable() {
610                self.poisoned.store(true, Ordering::Release);
611                return Err(error);
612            }
613            self.committed_lsn.store(target_lsn, Ordering::Release);
614            return Ok(None);
615        }
616        let batch = {
617            let mut pending = self.pending.lock();
618            let covered_count = pending
619                .iter()
620                .take_while(|(lsn, _)| *lsn <= target_lsn)
621                .count();
622            if covered_count == 0 {
623                // Another sync may have committed and drained this target
624                // after our first check but before we acquired the journal
625                // lock. In that case its durable group covers this caller.
626                if target_lsn <= self.committed_lsn.load(Ordering::Acquire) {
627                    return Ok(None);
628                }
629                return Err(io::Error::other(
630                    "published sync target has no pending journal mutations",
631                ));
632            }
633            let first_lsn = pending[0].0;
634            let last_lsn = covered_count
635                .checked_sub(1)
636                .and_then(|last_index| pending.get(last_index))
637                .map(|(lsn, _)| *lsn)
638                .ok_or_else(|| io::Error::other("pending journal batch is incomplete"))?;
639            if first_lsn != journal.next_lsn || last_lsn != target_lsn {
640                return Err(io::Error::other(
641                    "journal pending LSN sequence does not cover sync target",
642                ));
643            }
644            pending.drain(..covered_count).collect::<Vec<_>>()
645        };
646        self.append_and_sync_batch(&mut journal, batch, target_lsn)
647    }

649    fn append_and_sync_batch(
650        &self,
651        journal: &mut Journal,
652        batch: Vec<(u64, Mutation)>,
653        target_lsn: u64,
654    ) -> io::Result<Option<CommitReceipt>> {
655        let mutations: Vec<Mutation> = batch.iter().map(|(_, mutation)| mutation.clone()).collect();
656        let sequence = self
657            .sequence
658            .as_ref()
659            .map(|counter| counter.fetch_add(1, Ordering::Relaxed));
660        let receipt = match journal.append_group_with_sequence(&mutations, sequence) {
661            Ok(receipt) => receipt,
662            Err(error) => {
663                if journal.poisoned {
664                    self.poisoned.store(true, Ordering::Release);
665                } else {
666                    let mut pending = self.pending.lock();
667                    pending.splice(0..0, batch);
668                }
669                return Err(error);
670            }
671        };
672        // The group is complete and readable by a read-only overlay, but not
673        // yet durable. Publish the visibility boundary before the fsync so
674        // workers can observe it. A crash before `make_durable` may lose it,
675        // which is safe: an unfsynced group is never acknowledged.
676        self.visible_lsn
677            .fetch_max(receipt.last_lsn, Ordering::Release);
678        if let Err(error) = journal.make_durable() {
679            self.poisoned.store(true, Ordering::Release);
680            return Err(error);
681        }
682        // The group is durable, so record its extent before checking the
683        // receipt. This prevents a retry from re-appending committed entries.
684        self.committed_lsn
685            .store(receipt.last_lsn, Ordering::Release);
686        let committed_count = usize::try_from(
687            receipt
688                .last_lsn
689                .saturating_sub(receipt.first_lsn)
690                .saturating_add(1),
691        )
692        .unwrap_or(batch.len())
693        .min(batch.len());
694        if committed_count < batch.len() {
695            let mut pending = self.pending.lock();
696            pending.splice(0..0, batch.into_iter().skip(committed_count));
697        }
698        if receipt.last_lsn != target_lsn || committed_count != mutations.len() {
699            return Err(io::Error::other(
700                "durable journal group did not cover requested sync target",
701            ));
702        }
703        Ok(Some(receipt))
704    }

706    /// Append a complete transaction group without fsyncing it.
707    ///
708    /// Unlike [`Self::publish`], this does not add mutations to the legacy
709    /// queue: it appends queued legacy mutations first, followed by the staged
710    /// transaction mutations, in one complete group and then advances the
711    /// visible boundary. This preserves assigned LSN order and avoids making a
712    /// post-commit callback fail merely because background/legacy writes are
713    /// queued in the same pool.
714    ///
715    /// # Errors
716    /// Returns an error if queued LSNs are inconsistent, the journal is
717    /// poisoned, or the append fails. A partial append poisons the underlying
718    /// journal; subsequent publication is rejected until reopen/recovery.
719    pub fn append_pending(&self, mutations: &[Mutation]) -> io::Result<CommitReceipt> {
720        if mutations.is_empty() {
721            return Err(io::Error::new(
722                io::ErrorKind::InvalidInput,
723                "cannot append an empty transaction group",
724            ));
725        }
726        if self.poisoned.load(Ordering::Acquire) {
727            return Err(io::Error::other(
728                "journal is poisoned after a failed append",
729            ));
730        }
731        let mut journal = self.journal.lock();
732        if self.poisoned.load(Ordering::Acquire) {
733            return Err(io::Error::other(
734                "journal is poisoned after a failed append",
735            ));
736        }
737        // Hold the queue lock through the append so a concurrent legacy
738        // publisher cannot assign an LSN between the queued and staged parts.
739        let mut pending = self.pending.lock();
740        let mut group_mutations = Vec::with_capacity(pending.len().saturating_add(mutations.len()));
741        for (offset, (lsn, mutation)) in pending.iter().enumerate() {
742            let expected = journal
743                .next_lsn
744                .checked_add(u64::try_from(offset).unwrap_or(u64::MAX))
745                .ok_or_else(|| io::Error::other("journal LSN exhausted"))?;
746            if *lsn != expected {
747                return Err(io::Error::new(
748                    io::ErrorKind::InvalidData,
749                    "legacy journal queue is not contiguous with the journal tail",
750                ));
751            }
752            group_mutations.push(mutation.clone());
753        }
754        group_mutations.extend_from_slice(mutations);
755        let expected_first_lsn = journal.next_lsn;
756        let expected_count = u64::try_from(group_mutations.len()).unwrap_or(u64::MAX);
757        let sequence = self
758            .sequence
759            .as_ref()
760            .map(|counter| counter.fetch_add(1, Ordering::Relaxed));
761        let receipt = match journal.append_group_with_sequence(&group_mutations, sequence) {
762            Ok(receipt) => receipt,
763            Err(error) => {
764                if journal.poisoned {
765                    self.poisoned.store(true, Ordering::Release);
766                }
767                return Err(error);
768            }
769        };
770        debug_assert_eq!(
771            receipt.first_lsn, expected_first_lsn,
772            "appended group must start at the journal tail"
773        );
774        debug_assert_eq!(
775            receipt
776                .last_lsn
777                .saturating_sub(receipt.first_lsn)
778                .saturating_add(1),
779            expected_count,
780            "appended group must cover every queued and staged mutation"
781        );
782        let next_lsn = receipt
783            .last_lsn
784            .checked_add(1)
785            .ok_or_else(|| io::Error::other("journal LSN exhausted"))?;
786        pending.clear();
787        self.next_lsn.store(next_lsn, Ordering::Relaxed);
788        self.published_lsn
789            .fetch_max(receipt.last_lsn, Ordering::Release);
790        self.visible_lsn
791            .fetch_max(receipt.last_lsn, Ordering::Release);
792        drop(pending);
793        Ok(receipt)
794    }

825    /// torn tail. A read-only worker uses this to observe only complete,
826    /// committed groups while the writer may still be appending or reclaiming
827    /// the segment.
828    ///
829    /// An incomplete final group is reported in [`Scan::truncated_tail`] and
830    /// excluded from `groups`; the file is left byte-for-byte unchanged. A
831    /// malformed committed group remains an error.
832    ///
833    /// A missing file or one shorter than the file header is treated as an
834    /// empty segment. This permits a read-only worker to start before the
835    /// writer has created the journal, without creating or repairing it.
836    ///
837    /// # Errors
838    /// Returns `io::Error` if the file is unreadable, a complete header is
839    /// invalid, the segment is oversized, or a committed group fails
840    /// validation.
841    pub fn scan_read_only(path: impl AsRef<Path>) -> io::Result<Scan> {
842        let bytes = match fs::read(path) {
843            Ok(bytes) => bytes,
844            Err(error) if error.kind() == io::ErrorKind::NotFound => return Ok(Scan::empty()),
845            Err(error) => return Err(error),
846        };
847        if bytes.len() < FILE_HEADER_LEN {
848            return Ok(Scan::empty());
849        }
850        if u64::try_from(bytes.len()).unwrap_or(u64::MAX) > MAX_SEGMENT_LEN {
851            return Err(invalid_data("journal segment exceeds the 256 MiB limit"));
852        }
853        let (base_sequence, base_lsn) = validate_file_header(&bytes)?;
854        scan_bytes(&bytes, base_sequence, base_lsn)
855    }

857    /// Scan only the committed groups appended at or after `start`, a group
858    /// boundary returned by an earlier scan's `valid_len`.
859    ///
860    /// Reads only the tail bytes, so a reader that already applied the earlier
861    /// groups does not re-decode them. `expected_lsn` is the next LSN the
862    /// caller expects (its highest applied LSN plus one); a mismatch means the
863    /// segment was replaced or rotated, and the caller should rebuild from a
864    /// full [`Self::scan_read_only`]. Never repairs or creates the file.
865    ///
866    /// # Errors
867    /// Returns `io::Error` if the tail is unreadable or a group fails
868    /// validation. A missing file yields an empty scan.
869    // This intentionally seeks to `start` and reads only the appended tail;
870    // `fs::read` would decode the whole segment on every incremental refresh.
871    #[allow(clippy::verbose_file_reads)]
872    pub fn scan_read_only_from(
873        path: impl AsRef<Path>,
874        start: u64,
875        expected_lsn: u64,
876    ) -> io::Result<Scan> {
877        let mut file = match File::open(path) {
878            Ok(file) => file,
879            Err(error) if error.kind() == io::ErrorKind::NotFound => return Ok(Scan::empty()),
880            Err(error) => return Err(error),
881        };
882        let len = file.metadata()?.len();
883        if len < u64::try_from(FILE_HEADER_LEN).unwrap_or(u64::MAX) {
884            return Ok(Scan::empty());
885        }
886        file.seek(SeekFrom::Start(0))?;
887        let mut header = vec![0; FILE_HEADER_LEN];
888        file.read_exact(&mut header)?;
889        let (_, base_lsn) = validate_file_header(&header)?;
890        if start >= len {
891            return Ok(Scan {
892                groups: Vec::new(),
893                valid_len: start.min(len),
894                truncated_tail: false,
895                base_lsn,
896            });
897        }
898        file.seek(SeekFrom::Start(start))?;
899        let mut tail = Vec::new();
900        file.read_to_end(&mut tail)?;
901        scan_groups_from(&tail, start, 0, expected_lsn, base_lsn)
902    }

904    /// Open a journal and recover its committed groups.
905    ///
906    /// A segment shorter than the file header (a fresh file, or a crash while
907    /// the header was being written) is reset and re-initialised: no group can
908    /// exist without a fully synced header, so there is no acknowledged data
909    /// to lose. An incomplete final group is truncated. A malformed
910    /// *committed* group fails open and is never silently skipped.
911    ///
912    /// # Errors
913    /// Returns `io::Error` if the segment cannot be created or read, if a
914    /// data-bearing segment's header is invalid, if any committed group fails
915    /// validation, or if the segment exceeds `MAX_SEGMENT_LEN`.
916    pub fn open(path: impl AsRef<Path>) -> io::Result<(Self, Scan)> {
917        let path = path.as_ref().to_path_buf();
918        if let Some(parent) = path.parent() {
919            if !parent.as_os_str().is_empty() {
920                fs::create_dir_all(parent)?;
921            }
922        }
923        let mut file = OpenOptions::new()
924            .create(true)
925            .read(true)
926            .write(true)
927            .truncate(false)
928            .open(&path)?;
929
930        let len = file.metadata()?.len();
931        if len < FILE_HEADER_LEN as u64 {
932            // Empty (fresh) or a crash while writing the header: no committed
933            // group can exist yet, so reset and re-initialise rather than
934            // failing open forever on a partial header.
935            file.set_len(0)?;
936            file.seek(SeekFrom::Start(0))?;
937            write_file_header(&mut file, 1, 1)?;
938            file.sync_all()?;
939            sync_parent_dir(&path)?;
940        }
941
942        let bytes = fs::read(&path)?;
943        let (base_sequence, base_lsn) = validate_file_header(&bytes)?;
944        if bytes.len() as u64 > MAX_SEGMENT_LEN {
945            return Err(invalid_data("journal segment exceeds the 256 MiB limit"));
946        }
947        let scan = scan_bytes(&bytes, base_sequence, base_lsn)?;
948        if scan.truncated_tail {
949            file.set_len(scan.valid_len)?;
950        }
951        file.seek(SeekFrom::Start(scan.valid_len))?;
952        let next_sequence = scan
953            .groups
954            .last()
955            .map_or(base_sequence, |group| group.sequence.saturating_add(1));
956        let next_lsn = scan
957            .groups
958            .last()
959            .map_or(base_lsn, |group| group.last_lsn.saturating_add(1));
960
961        Ok((
962            Self {
963                path,
964                file,
965                next_sequence,
966                next_lsn,
967                poisoned: false,
968            },
969            scan,
970        ))
971    }

974    ///
975    /// Each mutation receives its own monotonic LSN. The group is complete and
976    /// readable by [`Self::scan_read_only`] as soon as this returns, so a
977    /// caller can publish a visibility boundary before [`Self::make_durable`]

983    /// oversized, the LSN/sequence space is exhausted, the segment is full
984    /// (`WouldBlock`; the caller must drain/rotate), or the write fails.

989    /// Append a group using `sequence` when given, or this segment's own next
990    /// sequence otherwise.
991    ///
992    /// A shared sequence lets several pools' segments be ordered by one global
993    /// group sequence; a gap between a segment's consecutive groups is then
994    /// expected, and recovery permits it (integrity still comes from the group
995    /// header CRC and commit trailer, which both cover the sequence).
996    ///
997    /// # Errors
998    /// Same as [`Self::append_group`], plus `InvalidInput` if `sequence` is
999    /// behind this segment's next sequence.
1000    pub fn append_group_with_sequence(
1001        &mut self,
1002        mutations: &[Mutation],
1003        sequence: Option<u64>,
1004    ) -> io::Result<CommitReceipt> {
1005        if self.poisoned {
1006            return Err(io::Error::other(
1007                "journal handle is poisoned after an earlier failed commit",
1008            ));
1009        }
1010        if mutations.is_empty() {
1011            return Err(io::Error::new(
1012                io::ErrorKind::InvalidInput,
1013                "cannot commit an empty journal group",
1014            ));
1015        }
1016
1017        let record_count = u32::try_from(mutations.len()).map_err(|_| {
1018            io::Error::new(io::ErrorKind::InvalidInput, "too many journal mutations")
1019        })?;
1020        let first_lsn = self.next_lsn;
1021        let last_lsn = first_lsn
1022            .checked_add(u64::from(record_count).saturating_sub(1))
1023            .ok_or_else(|| io::Error::other("journal LSN exhausted"))?;
1024        let sequence = sequence.unwrap_or(self.next_sequence);
1025        if sequence < self.next_sequence {
1026            return Err(io::Error::new(
1027                io::ErrorKind::InvalidInput,
1028                "journal group sequence would move backwards",
1029            ));
1030        }
1031
1032        let mut payload = Vec::new();
1033        for (index, mutation) in mutations.iter().enumerate() {
1034            let lsn = first_lsn
1035                .checked_add(u64::try_from(index).map_err(|_| {
1036                    io::Error::new(io::ErrorKind::InvalidInput, "mutation index exceeds u64")
1037                })?)
1038                .ok_or_else(|| io::Error::other("journal LSN exhausted"))?;
1039            encode_mutation(lsn, mutation, &mut payload)?;
1040            if u64::try_from(payload.len()).unwrap_or(u64::MAX) > MAX_GROUP_LEN {
1041                return Err(io::Error::new(
1042                    io::ErrorKind::InvalidInput,
1043                    "journal group exceeds the 256 MiB format limit",
1044                ));
1045            }
1046        }
1047        let payload_len = u64::try_from(payload.len())
1048            .map_err(|_| io::Error::new(io::ErrorKind::InvalidInput, "group too large"))?;
1049        if payload_len > MAX_GROUP_LEN {
1050            return Err(io::Error::new(
1051                io::ErrorKind::InvalidInput,
1052                "journal group exceeds the 256 MiB segment limit",
1053            ));
1054        }
1055
1056        let header = encode_group_header(sequence, first_lsn, last_lsn, payload_len, record_count);
1057        let mut group_checksum = Hasher::new();
1058        group_checksum.update(&header);
1059        group_checksum.update(&payload);
1060        let trailer = encode_group_trailer(sequence, group_checksum.finalize());
1061
1062        let group_size = u64::try_from(
1063            header
1064                .len()
1065                .saturating_add(payload.len())
1066                .saturating_add(trailer.len()),
1067        )
1068        .map_err(|_| io::Error::new(io::ErrorKind::InvalidInput, "group too large"))?;
1069        let segment_size = self.file.metadata()?.len();
1070        if segment_size.saturating_add(group_size) > MAX_SEGMENT_LEN {
1071            return Err(io::Error::new(
1072                io::ErrorKind::WouldBlock,
1073                "journal segment is full; drain/rotate before accepting more writes",
1074            ));
1075        }
1076        let next_sequence = sequence
1077            .checked_add(1)
1078            .ok_or_else(|| io::Error::other("journal group sequence exhausted"))?;
1079        let next_lsn = last_lsn
1080            .checked_add(1)
1081            .ok_or_else(|| io::Error::other("journal LSN exhausted"))?;
1082
1083        let write_result = (|| -> io::Result<()> {
1084            self.file.write_all(&header)?;
1085            self.file.write_all(&payload)?;
1086            self.file.write_all(&trailer)?;
1087            Ok(())
1088        })();
1089        if let Err(error) = write_result {
1090            self.poisoned = true;
1091            return Err(error);
1092        }
1093
1094        self.next_sequence = next_sequence;
1095        self.next_lsn = next_lsn;
1096
1097        Ok(CommitReceipt {
1098            sequence,
1099            first_lsn,
1100            last_lsn,
1101        })
1102    }

1156    /// Compact this segment, dropping committed groups the checkpoint has
1157    /// already materialized.
1158    ///
1159    /// Every group with `last_lsn <= covered_lsn` is removed; the survivors are
1160    /// re-encoded with their original sequence and LSNs into a temporary file,
1161    /// durably synced, and atomically renamed over the active path, after which
1162    /// the parent directory is synced. A crash before the rename leaves the old
1163    /// segment intact and the temp file inert. Because the header records the
1164    /// surviving base numbering, the next [`Self::commit_group`] continues past
1165    /// the last committed LSN rather than restarting at 1.
1166    ///
1167    /// A checkpoint that has not advanced past the current segment (or has
1168    /// already reclaimed it) is a no-op.
1169    ///
1170    /// # Errors
1171    /// Returns `io::Error` if the handle is poisoned, the segment cannot be
1172    /// read or re-encoded, or the replacement cannot be synced or renamed.
1173    pub fn reclaim_through(&mut self, covered_lsn: u64) -> io::Result<Reclaim> {
1174        if self.poisoned {
1175            return Err(io::Error::other(
1176                "journal handle is poisoned after an earlier failed commit",
1177            ));
1178        }
1179        if covered_lsn >= self.next_lsn {
1180            return Err(io::Error::new(
1181                io::ErrorKind::InvalidInput,
1182                "cannot reclaim an LSN that has not been committed",
1183            ));
1184        }
1185        let bytes = fs::read(&self.path)?;
1186        let (base_sequence, base_lsn) = validate_file_header(&bytes)?;
1187        let scan = scan_bytes(&bytes, base_sequence, base_lsn)?;
1188        let retained = scan
1189            .groups
1190            .iter()
1191            .filter(|group| group.last_lsn > covered_lsn)
1192            .collect::<Vec<_>>();
1193        if retained.len() == scan.groups.len() {
1194            return Ok(Reclaim {
1195                retained_groups: u64::try_from(retained.len()).unwrap_or(u64::MAX),
1196                reclaimed_bytes: 0,
1197            });
1198        }
1199
1200        let (new_base_sequence, new_base_lsn) = retained
1201            .first()
1202            .map_or((self.next_sequence, self.next_lsn), |group| {
1203                (group.sequence, group.first_lsn)
1204            });
1205        let mut rebuilt = file_header_bytes(new_base_sequence, new_base_lsn);
1206        for group in &retained {
1207            encode_group(group, &mut rebuilt)?;
1208        }
1209
1210        let temp_path = self.path.with_extension("rotate");
1211        let write_result = (|| -> io::Result<()> {
1212            let mut temp = OpenOptions::new()
1213                .create(true)
1214                .write(true)
1215                .truncate(true)
1216                .open(&temp_path)?;
1217            temp.write_all(&rebuilt)?;
1218            temp.sync_all()?;
1219            drop(temp);
1220            fs::rename(&temp_path, &self.path)
1221        })();
1222        if let Err(error) = write_result {
1223            let _ = fs::remove_file(&temp_path);
1224            return Err(error);
1225        }
1226        // The rename has replaced the path, so from here onward any failure
1227        // must poison the handle: continuing to append through the old file
1228        // descriptor would write an unlinked inode and could acknowledge data
1229        // that is absent from the journal path.
1230        let replacement = match OpenOptions::new().read(true).write(true).open(&self.path) {
1231            Ok(file) => file,
1232            Err(error) => {
1233                self.poisoned = true;
1234                return Err(error);
1235            }
1236        };
1237        self.file = replacement;
1238        if let Err(error) = sync_parent_dir(&self.path) {
1239            self.poisoned = true;
1240            return Err(error);
1241        }
1242        if let Err(error) = self.file.seek(SeekFrom::End(0)) {
1243            self.poisoned = true;
1244            return Err(error);
1245        }
1246        Ok(Reclaim {
1247            retained_groups: u64::try_from(retained.len()).unwrap_or(u64::MAX),
1248            reclaimed_bytes: u64::try_from(bytes.len().saturating_sub(rebuilt.len()))
1249                .unwrap_or(u64::MAX),
1250        })
1251    }
1252}

1270/// Validate the file header, returning the base `(sequence, lsn)` this segment
1271/// starts numbering at. A rotated segment records the first surviving group's
1272/// numbers here so scanning never has to assume the sequence begins at 1.
1273fn validate_file_header(bytes: &[u8]) -> io::Result<(u64, u64)> {
1274    let Some(header) = bytes.get(..FILE_HEADER_LEN) else {
1275        return Err(invalid_data("truncated journal file header"));
1276    };
1277    if &header[..8] != FILE_MAGIC {
1278        return Err(invalid_data("invalid journal magic"));
1279    }
1280    if u32::from_le_bytes(header[8..12].try_into().expect("fixed slice")) != FILE_VERSION {
1281        return Err(invalid_data("unsupported journal version"));
1282    }
1283    let mut crc = Hasher::new();
1284    crc.update(&header[..28]);
1285    if u32::from_le_bytes(header[28..32].try_into().expect("fixed slice")) != crc.finalize() {
1286        return Err(invalid_data("journal header checksum mismatch"));
1287    }
1288    Ok((
1289        u64::from_le_bytes(header[12..20].try_into().expect("fixed slice")),
1290        u64::from_le_bytes(header[20..28].try_into().expect("fixed slice")),
1291    ))
1292}

1294fn encode_group_header(
1295    sequence: u64,
1296    first_lsn: u64,
1297    last_lsn: u64,
1298    payload_len: u64,
1299    record_count: u32,
1300) -> [u8; GROUP_HEADER_LEN] {
1301    let mut header = [0_u8; GROUP_HEADER_LEN];
1302    let header_len = u32::try_from(GROUP_HEADER_LEN).expect("GROUP_HEADER_LEN must fit in a u32");
1303    header[..4].copy_from_slice(GROUP_MAGIC);
1304    header[4..8].copy_from_slice(&header_len.to_le_bytes());
1305    header[8..16].copy_from_slice(&sequence.to_le_bytes());
1306    header[16..24].copy_from_slice(&first_lsn.to_le_bytes());
1307    header[24..32].copy_from_slice(&last_lsn.to_le_bytes());
1308    header[32..40].copy_from_slice(&payload_len.to_le_bytes());
1309    header[40..44].copy_from_slice(&record_count.to_le_bytes());
1310    let mut crc = Hasher::new();
1311    crc.update(&header[..44]);
1312    header[44..48].copy_from_slice(&crc.finalize().to_le_bytes());
1313    header
1314}

1316fn encode_group_trailer(sequence: u64, group_crc: u32) -> [u8; GROUP_TRAILER_LEN] {
1317    let mut trailer = [0_u8; GROUP_TRAILER_LEN];
1318    trailer[..4].copy_from_slice(GROUP_COMMIT_MAGIC);
1319    trailer[4..12].copy_from_slice(&sequence.to_le_bytes());
1320    trailer[12..16].copy_from_slice(&group_crc.to_le_bytes());
1321    trailer
1322}

1324/// Re-encode a committed group with its original sequence and LSNs, for a
1325/// segment rewrite. Mirrors exactly the framing `commit_group` writes.
1326fn encode_group(group: &CommittedGroup, into: &mut Vec<u8>) -> io::Result<()> {
1327    let mut payload = Vec::new();
1328    for entry in &group.entries {
1329        encode_mutation(entry.lsn, &entry.mutation, &mut payload)?;
1330    }
1331    let record_count = u32::try_from(group.entries.len())
1332        .map_err(|_| io::Error::new(io::ErrorKind::InvalidData, "too many journal mutations"))?;
1333    let payload_len = u64::try_from(payload.len())
1334        .map_err(|_| io::Error::new(io::ErrorKind::InvalidData, "journal group too large"))?;
1335    let header = encode_group_header(
1336        group.sequence,
1337        group.first_lsn,
1338        group.last_lsn,
1339        payload_len,
1340        record_count,
1341    );
1342    let mut group_crc = Hasher::new();
1343    group_crc.update(&header);
1344    group_crc.update(&payload);
1345    let trailer = encode_group_trailer(group.sequence, group_crc.finalize());
1346    into.extend_from_slice(&header);
1347    into.extend_from_slice(&payload);
1348    into.extend_from_slice(&trailer);
1349    Ok(())
1350}

1352fn encode_mutation(lsn: u64, mutation: &Mutation, into: &mut Vec<u8>) -> io::Result<()> {
1353    let start = into.len();
1354    match mutation {
1355        Mutation::Put {
1356            collection_id,
1357            node_id,
1358            payload,
1359        } => {
1360            into.push(1);
1361            into.push(0);
1362            into.extend_from_slice(&0_u16.to_le_bytes());
1363            into.extend_from_slice(&lsn.to_le_bytes());
1364            into.extend_from_slice(collection_id);
1365            into.extend_from_slice(node_id);
1366            let payload_len = u32::try_from(payload.len()).map_err(|_| {
1367                io::Error::new(io::ErrorKind::InvalidInput, "mutation payload exceeds u32")
1368            })?;
1369            into.extend_from_slice(&payload_len.to_le_bytes());
1370            into.extend_from_slice(payload);
1371        }
1372        Mutation::DeleteCollection { collection_id } => {
1373            into.push(2);
1374            into.push(0);
1375            into.extend_from_slice(&0_u16.to_le_bytes());
1376            into.extend_from_slice(&lsn.to_le_bytes());
1377            into.extend_from_slice(collection_id);
1378            into.extend_from_slice(&[0_u8; 16]);
1379            into.extend_from_slice(&0_u32.to_le_bytes());
1380        }
1381    }
1382    let mut crc = Hasher::new();
1383    crc.update(&into[start..]);
1384    into.extend_from_slice(&crc.finalize().to_le_bytes());
1385    Ok(())
1386}

1388/// Validated fixed fields of one committed group header.
1389struct GroupHeader {
1390    sequence: u64,
1391    first_lsn: u64,
1392    last_lsn: u64,
1393    payload_len: u64,
1394    record_count: u32,
1395}

1397/// Validate a group header's magic, length, and CRC, returning its fields.
1398fn parse_group_header(header: &[u8]) -> io::Result<GroupHeader> {
1399    if &header[..4] != GROUP_MAGIC
1400        || u32::from_le_bytes(header[4..8].try_into().expect("fixed slice"))
1401            != u32::try_from(GROUP_HEADER_LEN).expect("GROUP_HEADER_LEN fits in u32")
1402    {
1403        return Err(invalid_data("invalid journal group header"));
1404    }
1405    let mut header_crc = Hasher::new();
1406    header_crc.update(&header[..44]);
1407    if u32::from_le_bytes(header[44..48].try_into().expect("fixed slice")) != header_crc.finalize()
1408    {
1409        return Err(invalid_data("journal group header checksum mismatch"));
1410    }
1411    Ok(GroupHeader {
1412        sequence: u64::from_le_bytes(header[8..16].try_into().expect("fixed slice")),
1413        first_lsn: u64::from_le_bytes(header[16..24].try_into().expect("fixed slice")),
1414        last_lsn: u64::from_le_bytes(header[24..32].try_into().expect("fixed slice")),
1415        payload_len: u64::from_le_bytes(header[32..40].try_into().expect("fixed slice")),
1416        record_count: u32::from_le_bytes(header[40..44].try_into().expect("fixed slice")),
1417    })
1418}

1420/// Verify a group's commit trailer and whole-group CRC, returning its payload.
1421fn verify_group_payload<'a>(
1422    bytes: &'a [u8],
1423    header: &[u8],
1424    group: &GroupHeader,
1425    payload_start: usize,
1426    payload_len: usize,
1427) -> io::Result<&'a [u8]> {
1428    let payload_end = payload_start
1429        .checked_add(payload_len)
1430        .ok_or_else(|| invalid_data("journal group length overflow"))?;
1431    let trailer_end = payload_end
1432        .checked_add(GROUP_TRAILER_LEN)
1433        .ok_or_else(|| invalid_data("journal group length overflow"))?;
1434    let trailer = bytes
1435        .get(payload_end..trailer_end)
1436        .ok_or_else(|| invalid_data("truncated journal commit trailer"))?;
1437    if &trailer[..4] != GROUP_COMMIT_MAGIC
1438        || u64::from_le_bytes(trailer[4..12].try_into().expect("fixed slice")) != group.sequence
1439    {
1440        return Err(invalid_data("invalid journal commit trailer"));
1441    }
1442    let payload = bytes
1443        .get(payload_start..payload_end)
1444        .ok_or_else(|| invalid_data("truncated journal group payload"))?;
1445    let mut group_crc = Hasher::new();
1446    group_crc.update(header);
1447    group_crc.update(payload);
1448    if u32::from_le_bytes(trailer[12..16].try_into().expect("fixed slice")) != group_crc.finalize()
1449    {
1450        return Err(invalid_data("journal committed group checksum mismatch"));
1451    }
1452    Ok(payload)
1453}

1468/// Scan complete groups from `bytes`, which begins at absolute file offset
1469/// `base_offset`.
1470///
1471/// `expected_sequence` is a floor, not an exact match: a segment sharing a
1472/// global sequence with other pools may skip values. `expected_lsn` must match
1473/// the first group's `first_lsn` exactly, since LSNs stay contiguous within a
1474/// segment. `Scan::valid_len` is absolute in the file, not relative to `bytes`.
1475fn scan_groups_from(
1476    bytes: &[u8],
1477    base_offset: u64,
1478    mut expected_sequence: u64,
1479    mut expected_lsn: u64,
1480    segment_base_lsn: u64,
1481) -> io::Result<Scan> {
1482    let mut cursor = 0usize;
1483    let mut valid_len = base_offset;
1484    let mut groups = Vec::new();
1485    let mut truncated_tail = false;
1486
1487    while cursor < bytes.len() {
1488        let remaining = bytes.len().saturating_sub(cursor);
1489        if remaining < GROUP_HEADER_LEN {
1490            truncated_tail = true;
1491            break;
1492        }
1493        let header = bytes
1494            .get(cursor..cursor.saturating_add(GROUP_HEADER_LEN))
1495            .ok_or_else(|| invalid_data("truncated journal group header"))?;
1496        let group = parse_group_header(header)?;
1497        if group.sequence < expected_sequence
1498            || group.first_lsn != expected_lsn
1499            || group.last_lsn < group.first_lsn
1500            || group
1501                .last_lsn
1502                .saturating_sub(group.first_lsn)
1503                .saturating_add(1)
1504                != u64::from(group.record_count)
1505            || group.record_count == 0
1506            || group.payload_len > MAX_GROUP_LEN
1507            || u64::from(group.record_count).saturating_mul(MIN_FRAME_LEN as u64)
1508                > group.payload_len
1509        {
1510            return Err(invalid_data("invalid journal sequence or group bounds"));
1511        }
1512        let payload_len = usize::try_from(group.payload_len)
1513            .map_err(|_| invalid_data("journal group length exceeds address space"))?;
1514        let total_len = GROUP_HEADER_LEN
1515            .checked_add(payload_len)
1516            .and_then(|len| len.checked_add(GROUP_TRAILER_LEN))
1517            .ok_or_else(|| invalid_data("journal group length overflow"))?;
1518        if remaining < total_len {
1519            truncated_tail = true;
1520            break;
1521        }
1522        let payload_index = cursor.saturating_add(GROUP_HEADER_LEN);
1523        let payload = verify_group_payload(bytes, header, &group, payload_index, payload_len)?;
1524        let payload_offset = base_offset
1525            .saturating_add(u64::try_from(payload_index).unwrap_or(u64::MAX))
1526            .try_into()
1527            .unwrap_or(usize::MAX);
1528        let entries =
1529            decode_mutations(payload, group.first_lsn, group.record_count, payload_offset)?;
1530        groups.push(CommittedGroup {
1531            sequence: group.sequence,
1532            first_lsn: group.first_lsn,
1533            last_lsn: group.last_lsn,
1534            entries,
1535        });
1536        cursor = cursor.saturating_add(total_len);
1537        valid_len = base_offset.saturating_add(u64::try_from(cursor).unwrap_or(u64::MAX));
1538        expected_sequence = group
1539            .sequence
1540            .checked_add(1)
1541            .ok_or_else(|| invalid_data("journal group sequence overflow"))?;
1542        expected_lsn = group
1543            .last_lsn
1544            .checked_add(1)
1545            .ok_or_else(|| invalid_data("journal LSN overflow"))?;
1546    }
1547
1548    Ok(Scan {
1549        groups,
1550        valid_len,
1551        truncated_tail,
1552        base_lsn: segment_base_lsn,
1553    })
1554}

1556fn decode_mutations(
1557    payload: &[u8],
1558    first_lsn: u64,
1559    record_count: u32,
1560    payload_offset: usize,
1561) -> io::Result<Vec<JournalEntry>> {
1562    let mut cursor = 0_usize;
1563    let mut entries = Vec::with_capacity(usize::try_from(record_count).unwrap_or(0));
1564    for index in 0..record_count {
1565        let start = cursor;
1566        if payload.len().saturating_sub(cursor) < MIN_FRAME_LEN {
1567            return Err(invalid_data("truncated journal mutation frame"));
1568        }
1569        let fixed_end = cursor
1570            .checked_add(FRAME_FIXED_LEN)
1571            .ok_or_else(|| invalid_data("journal mutation length overflow"))?;
1572        let frame = payload
1573            .get(cursor..fixed_end)
1574            .ok_or_else(|| invalid_data("truncated journal mutation frame"))?;
1575        let kind = frame[0];
1576        let flags = frame[1];
1577        if flags != 0 || frame[2..4] != [0, 0] {
1578            return Err(invalid_data("unsupported journal mutation flags"));
1579        }
1580        let lsn = u64::from_le_bytes(frame[4..12].try_into().expect("fixed slice"));
1581        let expected = first_lsn
1582            .checked_add(u64::from(index))
1583            .ok_or_else(|| invalid_data("journal mutation LSN overflow"))?;
1584        if lsn != expected {
1585            return Err(invalid_data("journal mutation LSN out of order"));
1586        }
1587        let collection_id: [u8; 16] = frame[12..28].try_into().expect("fixed slice");
1588        let node_id: [u8; 16] = frame[28..44].try_into().expect("fixed slice");
1589        let payload_len = usize::try_from(u32::from_le_bytes(
1590            frame[44..48].try_into().expect("fixed slice"),
1591        ))
1592        .map_err(|_| invalid_data("journal mutation payload length exceeds address space"))?;
1593        let frame_end = fixed_end
1594            .checked_add(payload_len)
1595            .ok_or_else(|| invalid_data("journal mutation length overflow"))?;
1596        let crc_end = frame_end
1597            .checked_add(FRAME_TRAILER_LEN)
1598            .ok_or_else(|| invalid_data("journal mutation length overflow"))?;
1599        let data = payload
1600            .get(fixed_end..frame_end)
1601            .ok_or_else(|| invalid_data("journal mutation payload exceeds group"))?;
1602        let crc_bytes = payload
1603            .get(frame_end..crc_end)
1604            .ok_or_else(|| invalid_data("journal mutation checksum exceeds group"))?;
1605        let mut crc = Hasher::new();
1606        crc.update(
1607            payload
1608                .get(start..frame_end)
1609                .ok_or_else(|| invalid_data("journal mutation frame exceeds group"))?,
1610        );
1611        if u32::from_le_bytes(crc_bytes.try_into().expect("fixed slice")) != crc.finalize() {
1612            return Err(invalid_data("journal mutation checksum mismatch"));
1613        }
1614        let mutation = match kind {
1615            1 => Mutation::Put {
1616                collection_id,
1617                node_id,
1618                payload: data.to_vec(),
1619            },
1620            2 if payload_len == 0 && node_id == [0; 16] => {
1621                Mutation::DeleteCollection { collection_id }
1622            }
1623            _ => return Err(invalid_data("unknown or malformed journal mutation kind")),
1624        };
1625        entries.push(JournalEntry {
1626            lsn,
1627            offset: u64::try_from(payload_offset.saturating_add(start)).unwrap_or(u64::MAX),
1628            frame_len: u64::try_from(crc_end.saturating_sub(start)).unwrap_or(u64::MAX),
1629            mutation,
1630        });
1631        cursor = crc_end;
1632    }
1633    if cursor != payload.len() {
1634        return Err(invalid_data("extra bytes after journal mutation frames"));
1635    }
1636    Ok(entries)
1637}

1639fn sync_parent_dir(path: &Path) -> io::Result<()> {
1640    let parent = path
1641        .parent()
1642        .filter(|parent| !parent.as_os_str().is_empty())
1643        .unwrap_or_else(|| Path::new("."));
1644    #[cfg(unix)]
1645    {
1646        File::open(parent)?.sync_all()
1647    }
1648    #[cfg(not(unix))]
1649    {
1650        // Rust's portable std API has no directory-handle durability
1651        // operation on non-Unix platforms. Refuse to enable the journal until
1652        // a platform-specific implementation can uphold create/rotation
1653        // durability; silently succeeding would weaken the contract.
1654        let _ = parent;
1655        Err(io::Error::new(
1656            io::ErrorKind::Unsupported,
1657            "durable journal creation requires platform directory sync support",
1658        ))
1659    }
1660}
